import java.util.ArrayList;
import java.util.Collection;
import java.util.LinkedHashMap;
import java.util.Map;

public class OccurrenceFilter {
    private final Map<Integer, Criterion> criteria = new LinkedHashMap<>();
    private int nextId = 1;

    public int addCriterion(CriterionType type, String value) {
        int id = nextId++;
        criteria.put(id, new Criterion(id, type, value));
        return id;
    }

    public boolean removeCriterion(int id) {
        return criteria.remove(id) != null;
    }

    public Collection<Criterion> getCriteria() {
        return new ArrayList<>(criteria.values());
    }

    public OccurrenceFilter copy() {
        OccurrenceFilter copy = new OccurrenceFilter();
        for (Criterion criterion : criteria.values()) {
            copy.addCriterion(criterion.getType(), criterion.getValue());
        }
        return copy;
    }
}

        roles.add(role);

        roles.remove(role);

public class Resource {
    private final String id;
    private final ResourceType type;
    private String title;
    private Map<String, String> metadata;

    public Resource(String id, ResourceType type, String title, Map<String, String> metadata) {
        this.id = id;
        this.type = type;
        this.title = title;
        this.metadata = new HashMap<>(metadata);
    }

    public String getId() {
        return id;
    }

    public ResourceType getType() {
        return type;
    }

    public String getTitle() {
        return title;
    }

    public void setTitle(String title) {
        this.title = title;
    }

    public Map<String, String> getMetadata() {
        return new HashMap<>(metadata);
    }

    public void setMetadata(Map<String, String> metadata) {
        this.metadata = new HashMap<>(metadata);
    }

    public boolean matchesKeyword(String keyword) {
        String q = keyword.toLowerCase(Locale.ROOT);
        if (title.toLowerCase(Locale.ROOT).contains(q)) {
            return true;
        }
        for (String value : metadata.values()) {
            if (value.toLowerCase(Locale.ROOT).contains(q)) {
                return true;
            }
        }
        return false;
    }
}

import java.util.ArrayList;
import java.util.List;

public class OccurrenceSelection {
    private final List<OccurrenceRecord> records;
    private final OccurrenceFilter filter;

    public OccurrenceSelection(List<OccurrenceRecord> records, OccurrenceFilter filter) {
        this.records = new ArrayList<>(records);
        this.filter = filter;
    }

    public List<OccurrenceRecord> getRecords() {
        return new ArrayList<>(records);
    }

    public OccurrenceFilter getFilter() {
        return filter;
    }

    public String toTable() {
        StringBuilder builder = new StringBuilder();
        builder.append("ID | Scientific name | Dataset | Latitude | Longitude | Date\n");
        for (OccurrenceRecord record : records) {
            builder.append(record.getId()).append(" | ")
                    .append(record.getScientificName()).append(" | ")
                    .append(record.getDatasetId()).append(" | ")
                    .append(record.getLatitude()).append(" | ")
                    .append(record.getLongitude()).append(" | ")
                    .append(record.getDate()).append("\n");
        }
        return builder.toString();
    }

    public String export(String format) {
        if ("CSV".equalsIgnoreCase(format)) {
            StringBuilder builder = new StringBuilder();
            builder.append("id,scientificName,datasetId,latitude,longitude,date\n");
            for (OccurrenceRecord record : records) {
                builder.append(record.getId()).append(",")
                        .append(record.getScientificName()).append(",")
                        .append(record.getDatasetId()).append(",")
                        .append(record.getLatitude()).append(",")
                        .append(record.getLongitude()).append(",")
                        .append(record.getDate()).append("\n");
            }
            return builder.toString();
        }
        if ("JSON".equalsIgnoreCase(format)) {
            StringBuilder builder = new StringBuilder();
            builder.append("[\n");
            for (int i = 0; i < records.size(); i++) {
                OccurrenceRecord record = records.get(i);
                builder.append("  {\"id\":\"").append(record.getId())
                        .append("\",\"scientificName\":\"").append(record.getScientificName())
                        .append("\",\"datasetId\":\"").append(record.getDatasetId())
                        .append("\",\"latitude\":").append(record.getLatitude())
                        .append(",\"longitude\":").append(record.getLongitude())
                        .append(",\"date\":\"").append(record.getDate()).append("\"}");
                if (i < records.size() - 1) {
                    builder.append(",");
                }
                builder.append("\n");
            }
            builder.append("]\n");
            return builder.toString();
        }
        return toTable();
    }
}

        if (user == null) {
            throw new IllegalArgumentException("Unknown user: " + username);
        }
        return user;
    }

    private void requireRole(String username, Role role) {
        if (!requireUser(username).hasRole(role)) {
            throw new SecurityException("User " + username + " does not have role " + role);
        }
    }

    private void requireDataManager(String actor) {
        requireRole(actor, Role.DATA_MANAGER);
    }

        requireRole(actor, Role.PORTAL_MANAGER);

    private void audit(String actor, String action) {
        auditLogs.add(new LogEntry(LogType.AUDIT, actor + " " + action));

    private Resource requireResource(String id) {
        Resource resource = resources.get(id);
        if (resource == null) {
            throw new IllegalArgumentException("Unknown resource: " + id);
        }
        return resource;
    }

    private Resource requireResourceType(String id, ResourceType type) {
        Resource resource = requireResource(id);
        if (resource.getType() != type) {
            throw new IllegalArgumentException("Resource " + id + " is not of type " + type);
        }
        return resource;
    }

    private String nextResourceId(ResourceType type) {
        String prefix = type.name().length() >= 3 ? type.name().substring(0, 3) : type.name();
        return prefix + "-" + nextResourceId++;
    }

        for (Resource resource : resources.values()) {
            String sourceId = resource.getMetadata().get("sourceId");
            if (sourceId != null && !dataSources.containsKey(sourceId)) {

    public OccurrenceSelection selectOccurrenceRecords(OccurrenceFilter filter) {
        List<OccurrenceRecord> selected = new ArrayList<>();
        for (OccurrenceRecord record : occurrenceRecords) {
            boolean matchesAll = true;
            for (Criterion criterion : filter.getCriteria()) {
                if (!matches(record, criterion)) {
                    matchesAll = false;
                    break;
                }
            }
            if (matchesAll) {
                selected.add(record);
            }
        }
        return new OccurrenceSelection(selected, filter);
    }

    private boolean matches(OccurrenceRecord record, Criterion criterion) {
        try {
            switch (criterion.getType()) {
                case TAXONOMIC:
                    return record.getScientificName().toLowerCase(Locale.ROOT)
                            .contains(criterion.getValue().toLowerCase(Locale.ROOT));
                case DATASET:
                    return record.getDatasetId().equalsIgnoreCase(criterion.getValue());
                case DEPTH:
                    return inRange(record.getDepth(), criterion.getValue());
                case ALTITUDE:
                    return inRange(record.getAltitude(), criterion.getValue());
                case BASIS_OF_RECORD:
                    return record.getBasisOfRecord().toLowerCase(Locale.ROOT)
                            .contains(criterion.getValue().toLowerCase(Locale.ROOT));
                case GEOGRAPHIC:
                    return inGeographicRange(record, criterion.getValue());
                case DATE:
                    return inDateRange(record.getDate(), criterion.getValue());
                default:
                    return true;
            }
        } catch (RuntimeException e) {
            return false;
        }
    }

    private boolean inRange(double value, String specification) {
        if (specification.contains(",")) {
            String[] parts = specification.split(",");
            double min = Double.parseDouble(parts[0].trim());
            double max = Double.parseDouble(parts[1].trim());
            return value >= min && value <= max;
        }
        double expected = Double.parseDouble(specification.trim());
        return Math.abs(value - expected) < 0.000001;
    }

    private boolean inGeographicRange(OccurrenceRecord record, String specification) {
        String[] parts = specification.split(",");
        if (parts.length != 4) {
            return false;
        }
        double minLat = Double.parseDouble(parts[0].trim());
        double minLon = Double.parseDouble(parts[1].trim());
        double maxLat = Double.parseDouble(parts[2].trim());
        double maxLon = Double.parseDouble(parts[3].trim());
        return record.getLatitude() >= minLat && record.getLatitude() <= maxLat
                && record.getLongitude() >= minLon && record.getLongitude() <= maxLon;
    }

    private boolean inDateRange(LocalDate date, String specification) {
        if (specification.contains(",")) {
            String[] parts = specification.split(",");
            LocalDate start = LocalDate.parse(parts[0].trim());
            LocalDate end = LocalDate.parse(parts[1].trim());
            return !date.isBefore(start) && !date.isAfter(end);
        }
        return date.toString().contains(specification.trim());
    }

    public List<String> fullTextSearch(String query) {
        String q = query.toLowerCase(Locale.ROOT);
        List<String> results = new ArrayList<>();
        for (Resource resource : resources.values()) {
            if (resource.matchesKeyword(q)) {
                results.add("RESOURCE " + resource.getId() + ": " + resource.getTitle());
            }
        }
        for (NewsEntry news : newsEntries.values()) {
            if (news.getTitle().toLowerCase(Locale.ROOT).contains(q)
                    || news.getContent().toLowerCase(Locale.ROOT).contains(q)) {
                results.add("NEWS " + news.getId() + ": " + news.getTitle());
            }
        }
        for (BlogPost post : blogPosts.values()) {
            if (post.getTitle().toLowerCase(Locale.ROOT).contains(q)
                    || post.getContent().toLowerCase(Locale.ROOT).contains(q)) {
                results.add("BLOG " + post.getId() + ": " + post.getTitle());
            }
        }
        for (PortalPage page : portalPages.values()) {
            if (page.getTitle().toLowerCase(Locale.ROOT).contains(q)
                    || page.getContent().toLowerCase(Locale.ROOT).contains(q)) {
                results.add("PAGE " + page.getId() + ": " + page.getTitle());
            }
        }
        for (Map.Entry<String, String> entry : taxonDetails.entrySet()) {
            if (entry.getKey().toLowerCase(Locale.ROOT).contains(q)
                    || entry.getValue().toLowerCase(Locale.ROOT).contains(q)) {
                results.add("TAXON: " + entry.getKey());
            }
        }
        return results;
    }

        users.put(username, account);
        return account;
    }
