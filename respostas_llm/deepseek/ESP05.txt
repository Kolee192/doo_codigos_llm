        return activities.removeIf(a -> a.getId().equals(activityId));
    }

        for (CampActivity activity : activities) {
            if (ignoreActivityId != null && activity.getId().equals(ignoreActivityId)) {
                continue;
            }
            if (activity.overlaps(start, end)) {
                return true;
            }
        }
        return false;
    }

    @Override
    public String toString() {
        return name + " (" + year + ")";
    }

// File: Payment.java
public class Payment {
    private String id;
    private String camperId;
    private String parentId;
    private double amountDue;
    private double amountPaid;
    private PaymentStatus status;

    public Payment(String id, String camperId, String parentId, double amountDue) {
        this.id = id;
        this.camperId = camperId;
        this.parentId = parentId;
        this.amountDue = amountDue;
        this.amountPaid = 0.0;
        this.status = PaymentStatus.UNPAID;
        updateStatus();
    }

    public String getId() {
        return id;
    }

    public String getCamperId() {
        return camperId;
    }

    public String getParentId() {
        return parentId;
    }

    public double getAmountDue() {
        return amountDue;
    }

    public double getAmountPaid() {
        return amountPaid;
    }

    public double getRemaining() {
        return Math.max(0.0, amountDue - amountPaid);
    }

    public PaymentStatus getStatus() {
        return status;
    }

    public void pay(double amount) {
        if (amount < 0) {
            throw new IllegalArgumentException("Payment amount cannot be negative");
        }
        amountPaid += amount;
        if (amountPaid > amountDue) {
            amountPaid = amountDue;
        }
        updateStatus();
    }

    private void updateStatus() {
        if (amountPaid <= 0.0) {
            status = PaymentStatus.UNPAID;
        } else if (amountPaid < amountDue) {
            status = PaymentStatus.PARTIAL;
        } else {
            status = PaymentStatus.PAID;
        }
    }

    @Override
    public String toString() {
        return "Payment " + id + " due=" + amountDue + " paid=" + amountPaid + " status=" + status;
    }
}

    public CampActivity scheduleActivity(String groupId, String name, String description,
                                         LocalDateTime start, LocalDateTime end, String facilityId) {
        validateTimeRange(start, end);
        CampGroup group = requireGroup(groupId);
        if (group.hasActivityConflict(start, end, null)) {
            throw new IllegalArgumentException("Group already has an activity during that time");
        }
        String facilityKey = facilityId == null ? "" : facilityId;
        if (!facilityKey.isBlank()) {
            Facility facility = requireFacility(facilityKey);
            if (facility.hasConflict(start, end, null)) {
                throw new IllegalArgumentException("Facility is already in use during that time");
            }
        }
        CampActivity activity = new CampActivity(id("A"), name, description, start, end, groupId, facilityKey);
        group.addActivity(activity);
        if (!facilityKey.isBlank()) {
            Facility facility = facilities.get(facilityKey);
            facility.addUsage(new FacilityUsage(id("FU"), facilityKey, groupId,
                    activity.getId(), start, end));
        }
        return activity;
    }

    public void deleteActivity(String activityId) {
        CampActivity activity = findActivity(activityId);
        CampGroup group = requireGroup(activity.getGroupId());
        group.removeActivity(activityId);
        if (activity.getFacilityId() != null && !activity.getFacilityId().isBlank()) {
            Facility facility = facilities.get(activity.getFacilityId());
            if (facility != null) {
                facility.removeUsageByActivity(activityId);
            }
        }
    }

    public void modifyActivity(String activityId, String name, String description,
                               LocalDateTime start, LocalDateTime end, String facilityId) {
        validateTimeRange(start, end);
        CampActivity activity = findActivity(activityId);
        CampGroup group = requireGroup(activity.getGroupId());
        if (group.hasActivityConflict(start, end, activityId)) {
            throw new IllegalArgumentException("Group already has an activity during that time");
        }
        String newFacilityId = facilityId == null ? "" : facilityId;
        if (!newFacilityId.isBlank()) {
            Facility facility = requireFacility(newFacilityId);
            if (facility.hasConflict(start, end, activityId)) {
                throw new IllegalArgumentException("Facility is already in use during that time");
            }
        }
        if (activity.getFacilityId() != null && !activity.getFacilityId().isBlank()) {
            Facility oldFacility = facilities.get(activity.getFacilityId());
            if (oldFacility != null) {
                oldFacility.removeUsageByActivity(activityId);
            }
        }
        if (name != null) {
            activity.setName(name);
        }
        if (description != null) {
            activity.setDescription(description);
        }
        activity.setStart(start);
        activity.setEnd(end);
        activity.setFacilityId(newFacilityId);
        if (!newFacilityId.isBlank()) {
            Facility facility = facilities.get(newFacilityId);
            facility.addUsage(new FacilityUsage(id("FU"), newFacilityId, group.getId(),
                    activityId, start, end));
        }
    }
