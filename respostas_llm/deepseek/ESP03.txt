import java.util.Collections;
import java.util.EnumSet;
import java.util.Set;

public class User {
    private final String id;
    private String username;
    private final Set<Role> roles = EnumSet.noneOf(Role.class);

    public User(String id, String username, Set<Role> roles) {
        this.id = id;
        setUsername(username);
        if (roles != null) {
            this.roles.addAll(roles);
        }
    }

    public String getId() {
        return id;
    }

    public String getUsername() {
        return username;
    }

    public void setUsername(String username) {
        if (username == null || username.trim().isEmpty()) {
            throw new IllegalArgumentException("Username is required");
        }
        this.username = username.trim();
    }

    public Set<Role> getRoles() {
        return Collections.unmodifiableSet(roles);
    }

    public boolean hasRole(Role role) {
        return roles.contains(role);
    }

    public void addRole(Role role) {
        roles.add(role);
    }

    public boolean isPublisher() {
        return hasRole(Role.DATA_PUBLISHER) || isAdministrator();
    }

    public boolean isAdministrator() {
        return hasRole(Role.PLATFORM_ADMIN);
    }

    @Override
    public String toString() {
        return username + roles;
    }
}

import java.util.LinkedHashMap;
import java.util.Map;
import java.util.OptionalInt;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class DataRow {
    private final Map<String, Object> values;

    public DataRow() {
        this.values = new LinkedHashMap<>();
    }

    public DataRow(Map<String, Object> values) {
        this.values = new LinkedHashMap<>(values);
    }

    public void put(String key, Object value) {
        values.put(key, value);
    }

    public Object get(String key) {
        if (values.containsKey(key)) {
            return values.get(key);
        }
        for (Map.Entry<String, Object> entry : values.entrySet()) {
            if (entry.getKey().equalsIgnoreCase(key)) {
                return entry.getValue();
            }
        }
        return null;
    }

    public String getString(String key) {
        Object value = get(key);
        return value == null ? null : String.valueOf(value);
    }

    public double getDouble(String key) {
        Object value = get(key);
        if (value == null) {
            return Double.NaN;
        }
        if (value instanceof Number) {
            return ((Number) value).doubleValue();
        }
        try {
            return Double.parseDouble(String.valueOf(value).replace(",", "").trim());
        } catch (NumberFormatException ex) {
            return Double.NaN;
        }
    }

    public OptionalInt getYear() {
        for (Map.Entry<String, Object> entry : values.entrySet()) {
            String key = entry.getKey().toLowerCase();
            if (key.equals("year") || key.equals("date") || key.equals("period")) {
                OptionalInt year = parseYear(entry.getValue());
                if (year.isPresent()) {
                    return year;
                }
            }
        }
        return OptionalInt.empty();
    }

    private OptionalInt parseYear(Object value) {
        if (value instanceof Number) {
            return OptionalInt.of(((Number) value).intValue());
        }
        Matcher matcher = Pattern.compile("(\\d{4})").matcher(String.valueOf(value));
        if (matcher.find()) {
            return OptionalInt.of(Integer.parseInt(matcher.group(1)));
        }
        return OptionalInt.empty();
    }

    public Map<String, Object> asMap() {
        return new LinkedHashMap<>(values);
    }
}

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;

public class CsvDataSource implements DataSource {
    private final Path path;

    public CsvDataSource(Path path) {
        this.path = Objects.requireNonNull(path, "path");
    }

    @Override
    public String getSourceId() {
        return path.toString();
    }

    @Override
    public List<DataRow> loadData() throws DataSourceException {
        try {
            List<String> lines = Files.readAllLines(path);
            if (lines.isEmpty()) {
                return new ArrayList<>();
            }
            String[] headers = splitCsvLine(lines.get(0));
            List<DataRow> rows = new ArrayList<>();
            for (int i = 1; i < lines.size(); i++) {
                if (lines.get(i).trim().isEmpty()) {
                    continue;
                }
                String[] cells = splitCsvLine(lines.get(i));
                Map<String, Object> values = new LinkedHashMap<>();
                for (int j = 0; j < headers.length; j++) {
                    values.put(headers[j], j < cells.length ? cells[j] : "");
                }
                rows.add(new DataRow(values));
            }
            return rows;
        } catch (IOException ex) {
            throw new DataSourceException("Could not read CSV source " + path, ex);
        }
    }

    @Override
    public boolean supportsAutoUpdate() {
        return true;
    }

    public Path getPath() {
        return path;
    }

    private static String[] splitCsvLine(String line) {
        List<String> fields = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        boolean inQuotes = false;
        for (int i = 0; i < line.length(); i++) {
            char c = line.charAt(i);
            if (c == '"') {
                if (inQuotes && i + 1 < line.length() && line.charAt(i + 1) == '"') {
                    current.append('"');
                    i++;
                } else {
                    inQuotes = !inQuotes;
                }
            } else if (c == ',' && !inQuotes) {
                fields.add(current.toString().trim());
                current.setLength(0);
            } else {
                current.append(c);
            }
        }
        fields.add(current.toString().trim());
        return fields.toArray(new String[0]);
    }
}

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public final class JsonParser {
    private final String input;
    private int position;

    private JsonParser(String input) {
        this.input = input;
    }

    public static List<Map<String, Object>> parseObjectArray(String json) {
        return new JsonParser(json).parseObjectArrayInternal();
    }

    private List<Map<String, Object>> parseObjectArrayInternal() {
        skipWhitespace();
        expect('[');
        List<Map<String, Object>> result = new ArrayList<>();
        skipWhitespace();
        if (peek(']')) {
            position++;
            return result;
        }
        while (true) {
            skipWhitespace();
            result.add(parseObject());
            skipWhitespace();
            char c = input.charAt(position);
            if (c == ',') {
                position++;
            } else if (c == ']') {
                position++;
                break;
            } else {
                throw error("Expected ',' or ']'");
            }
        }
        return result;
    }

    private Map<String, Object> parseObject() {
        Map<String, Object> map = new LinkedHashMap<>();
        skipWhitespace();
        expect('{');
        skipWhitespace();
        if (peek('}')) {
            position++;
            return map;
        }
        while (true) {
            skipWhitespace();
            String key = readString();
            skipWhitespace();
            expect(':');
            skipWhitespace();
            Object value = parseValue();
            map.put(key, value);
            skipWhitespace();
            char c = input.charAt(position);
            if (c == ',') {
                position++;
            } else if (c == '}') {
                position++;
                break;
            } else {
                throw error("Expected ',' or '}'");
            }
        }
        return map;
    }

    private List<Object> parseArray() {
        List<Object> list = new ArrayList<>();
        skipWhitespace();
        expect('[');
        skipWhitespace();
        if (peek(']')) {
            position++;
            return list;
        }
        while (true) {
            skipWhitespace();
            list.add(parseValue());
            skipWhitespace();
            char c = input.charAt(position);
            if (c == ',') {
                position++;
            } else if (c == ']') {
                position++;
                break;
            } else {
                throw error("Expected ',' or ']'");
            }
        }
        return list;
    }

    private Object parseValue() {
        skipWhitespace();
        if (peek('"')) {
            return readString();
        }
        if (peek('{')) {
            return parseObject();
        }
        if (peek('[')) {
            return parseArray();
        }
        if (startsWith("true")) {
            position += 4;
            return Boolean.TRUE;
        }
        if (startsWith("false")) {
            position += 5;
            return Boolean.FALSE;
        }
        if (startsWith("null")) {
            position += 4;
            return null;
        }
        return readNumber();
    }

    private String readString() {
        skipWhitespace();
        expect('"');
        StringBuilder builder = new StringBuilder();
        while (position < input.length()) {
            char c = input.charAt(position++);
            if (c == '"') {
                return builder.toString();
            }
            if (c == '\\') {
                char escaped = input.charAt(position++);
                switch (escaped) {
                    case '"':
                        builder.append('"');
                        break;
                    case '\\':
                        builder.append('\\');
                        break;
                    case '/':
                        builder.append('/');
                        break;
                    case 'b':
                        builder.append('\b');
                        break;
                    case 'f':
                        builder.append('\f');
                        break;
                    case 'n':
                        builder.append('\n');
                        break;
                    case 'r':
                        builder.append('\r');
                        break;
                    case 't':
                        builder.append('\t');
                        break;
                    case 'u':
                        String hex = input.substring(position, position + 4);
                        builder.append((char) Integer.parseInt(hex, 16));
                        position += 4;
                        break;
                    default:
                        throw error("Invalid escape sequence");
                }
            } else {
                builder.append(c);
            }
        }
        throw error("Unterminated string");
    }

    private Object readNumber() {
        int start = position;
        while (position < input.length()) {
            char c = input.charAt(position);
            if ((c >= '0' && c <= '9') || c == '-' || c == '+' || c == '.' || c == 'e' || c == 'E') {
                position++;
            } else {
                break;
            }
        }
        String number = input.substring(start, position);
        if (number.contains(".") || number.contains("e") || number.contains("E")) {
            return Double.parseDouble(number);
        }
        try {
            return Long.parseLong(number);
        } catch (NumberFormatException ex) {
            return Double.parseDouble(number);
        }
    }

    private void skipWhitespace() {
        while (position < input.length() && Character.isWhitespace(input.charAt(position))) {
            position++;
        }
    }

    private boolean peek(char c) {
        return position < input.length() && input.charAt(position) == c;
    }

    private boolean startsWith(String text) {
        return input.startsWith(text, position);
    }

    private void expect(char c) {
        if (position >= input.length() || input.charAt(position) != c) {
            throw error("Expected '" + c + "'");
        }
        position++;
    }

    private IllegalArgumentException error(String message) {
        return new IllegalArgumentException(message + " at position " + position);
    }
}

    }

    public List<VisualizationType> defaultVisualizationTypes() {
        return List.of(VisualizationType.TREEMAP, VisualizationType.BUBBLE_TREE, VisualizationType.MAP, VisualizationType.PIVOT_TABLE);
    }

    public boolean matchesSearch(String query) {
        if (query == null || query.trim().isEmpty()) {
            return true;
        }
        String lower = query.toLowerCase();
        if (title.toLowerCase().contains(lower)) {
            return true;
        }
        for (String value : metadata.values()) {
            if (value != null && value.toLowerCase().contains(lower)) {
                return true;
            }
        }
        return false;
    }

    public boolean matchesMetadata(Map<String, String> query) {
        for (Map.Entry<String, String> entry : query.entrySet()) {
            String value = metadata.get(entry.getKey());
            if (value == null || !value.equalsIgnoreCase(entry.getValue())) {
                return false;
            }
        }
        return true;
    }

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Arrays;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public final class CsvValidator {
    private CsvValidator() {
    }

    public static CsvValidationResult validate(Path path) {
        CsvValidationResult result = new CsvValidationResult();
        if (path == null || !Files.exists(path)) {
            result.addError("CSV file does not exist");
            return result;
        }
        try {
            List<String> lines = Files.readAllLines(path);
            if (lines.isEmpty()) {
                result.addError("CSV file is empty");
                return result;
            }
            String[] headers = lines.get(0).split(",", -1);
            Set<String> seen = new HashSet<>();
            for (String header : headers) {
                String clean = header.trim();
                if (clean.isEmpty()) {
                    result.addError("Blank header found");
                } else if (!seen.add(clean)) {
                    result.addError("Duplicate header: " + clean);
                }
            }
            for (int i = 1; i < lines.size(); i++) {
                if (lines.get(i).trim().isEmpty()) {
                    result.addWarning("Blank row at line " + (i + 1));
                    continue;
                }
                String[] cells = lines.get(i).split(",", -1);
                if (cells.length != headers.length) {
                    result.addError("Row " + (i + 1) + " has " + cells.length + " cells; expected " + headers.length);
                }
            }
        } catch (IOException ex) {
            result.addError("Could not read CSV: " + ex.getMessage());
        }
        return result;
    }
}

    }

    }

public class Normalizer {
    private final Map<NormalizationType, Map<String, Double>> contexts = new EnumMap<>(NormalizationType.class);

    public Normalizer() {
        contexts.put(NormalizationType.POPULATION, new java.util.HashMap<>());
        contexts.put(NormalizationType.AREA, new java.util.HashMap<>());
        contexts.put(NormalizationType.GDP, new java.util.HashMap<>());
        contexts.put(NormalizationType.GINI, new java.util.HashMap<>());
    }

    public void addContext(String regionCode, NormalizationType type, double value) {
        Map<String, Double> map = contexts.get(type);
        if (map == null) {
            throw new IllegalArgumentException("Unsupported normalization type: " + type);
        }
        map.put(regionCode, value);
    }

    public double normalize(double value, String regionCode, NormalizationType type) {
        if (type == null || type == NormalizationType.NONE) {
            return value;
        }
        Map<String, Double> map = contexts.get(type);
        if (map == null) {
            return value;
        }
        Double denominator = map.get(regionCode);
        if (denominator == null || denominator <= 0) {
            return value;
        }
        return value / denominator;
    }
}

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

public class QueryResult {
    private final List<String> columns;
    private final List<Map<String, Object>> rows;

    public QueryResult(List<String> columns, List<Map<String, Object>> rows) {
        this.columns = new ArrayList<>(columns);
        this.rows = new ArrayList<>();
        for (Map<String, Object> row : rows) {
            this.rows.add(new java.util.LinkedHashMap<>(row));
        }
    }

    public List<String> getColumns() {
        return List.copyOf(columns);
    }

    public List<Map<String, Object>> getRows() {
        return List.copyOf(rows);
    }

    public String toCsv() {
        StringBuilder builder = new StringBuilder();
        for (int i = 0; i < columns.size(); i++) {
            if (i > 0) {
                builder.append(',');
            }
            builder.append(escape(columns.get(i)));
        }
        builder.append('\n');
        for (Map<String, Object> row : rows) {
            for (int i = 0; i < columns.size(); i++) {
                if (i > 0) {
                    builder.append(',');
                }
                builder.append(escape(row.get(columns.get(i))));
            }
            builder.append('\n');
        }
        return builder.toString();
    }

    private static String escape(Object value) {
        if (value == null) {
            return "";
        }
        String text = String.valueOf(value);
        if (text.contains(",") || text.contains("\"") || text.contains("\n")) {
            return "\"" + text.replace("\"", "\"\"") + "\"";
        }
        return text;
    }
}

import java.util.Collection;
import java.util.Comparator;

import java.util.LinkedHashSet;

public class CubeQuery {
    private final List<String> dimensions = new ArrayList<>();
    private final List<String> measures = new ArrayList<>();
    private final List<AggregationSpec> aggregations = new ArrayList<>();
    private final Map<String, Set<String>> filters = new LinkedHashMap<>();
    private final List<SortSpec> sorts = new ArrayList<>();
    private String targetCurrency;
    private NormalizationType normalization = NormalizationType.NONE;
    private boolean percentageOfTotal;
    private String year;

    public CubeQuery addDimension(String dimension) {
        dimensions.add(dimension);

    }

    public CubeQuery addMeasure(String measure) {

    }

    public CubeQuery addAggregation(AggregationSpec aggregation) {
        aggregations.add(aggregation);
        return this;
    }

    public CubeQuery addFilter(String field, String value) {
        filters.computeIfAbsent(field, key -> new LinkedHashSet<>()).add(value);
        return this;
    }

    public CubeQuery addSort(SortSpec sort) {
        sorts.add(sort);
        return this;
    }

    public void setTargetCurrency(String targetCurrency) {
        this.targetCurrency = targetCurrency;
    }

    public void setNormalization(NormalizationType normalization) {
        this.normalization = normalization;
    }

    public void setPercentageOfTotal(boolean percentageOfTotal) {
        this.percentageOfTotal = percentageOfTotal;
    }

    public void setYear(String year) {
        this.year = year;
    }

    public QueryResult execute(Collection<Dataset> datasets, CurrencyConverter converter, Normalizer normalizer) {
        List<DataRow> matchingRows = new ArrayList<>();
        for (Dataset dataset : datasets) {
            for (DataRow row : dataset.getDataRows()) {
                if (matches(row)) {
                    matchingRows.add(row);
                }
            }
        }

        Map<List<String>, List<DataRow>> groups = new LinkedHashMap<>();
        for (DataRow row : matchingRows) {
            List<String> key = new ArrayList<>();
            for (String dimension : dimensions) {
                key.add(row.getString(dimension));
            }
            groups.computeIfAbsent(key, k -> new ArrayList<>()).add(row);
        }

        List<AggregationSpec> effective = new ArrayList<>(aggregations);
        if (effective.isEmpty()) {
            for (String measure : measures) {
                effective.add(new AggregationSpec(measure, Aggregation.SUM));
            }
        }
        if (effective.isEmpty()) {
            effective.add(new AggregationSpec("count", Aggregation.COUNT));
        }

        List<String> columns = new ArrayList<>(dimensions);
        Set<String> measureColumns = new LinkedHashSet<>();
        for (AggregationSpec spec : effective) {
            measureColumns.add(spec.getMeasure());
            if (!columns.contains(spec.getMeasure())) {
                columns.add(spec.getMeasure());
            }
        }

        List<Map<String, Object>> resultRows = new ArrayList<>();
        for (Map.Entry<List<String>, List<DataRow>> entry : groups.entrySet()) {
            Map<String, Object> out = new LinkedHashMap<>();
            for (int i = 0; i < dimensions.size(); i++) {
                out.put(dimensions.get(i), entry.getKey().get(i));
            }
            for (AggregationSpec spec : effective) {
                out.put(spec.getMeasure(), aggregate(entry.getValue(), spec));
            }
            resultRows.add(out);
        }

        if (targetCurrency != null && converter != null) {
            for (Map<String, Object> row : resultRows) {
                for (String measure : measureColumns) {
                    Object value = row.get(measure);
                    if (value instanceof Number) {
                        row.put(measure, converter.convert(((Number) value).doubleValue(), "USD", targetCurrency));
                    }
                }
            }
        }

        if (normalization != NormalizationType.NONE && normalizer != null) {
            for (Map<String, Object> row : resultRows) {
                String region = dimensions.isEmpty() ? "" : String.valueOf(row.get(dimensions.get(0)));
                for (String measure : measureColumns) {
                    Object value = row.get(measure);
                    if (value instanceof Number) {
                        row.put(measure, normalizer.normalize(((Number) value).doubleValue(), region, normalization));
                    }
                }
            }
        }

        if (percentageOfTotal) {
            for (String measure : measureColumns) {
                double total = 0.0;
                for (Map<String, Object> row : resultRows) {
                    total += asDouble(row.get(measure));
                }
                if (total != 0.0) {
                    for (Map<String, Object> row : resultRows) {
                        row.put(measure, asDouble(row.get(measure)) / total * 100.0);
                    }
                }
            }
        }

        Comparator<Map<String, Object>> comparator = null;
        for (SortSpec sort : sorts) {
            Comparator<Map<String, Object>> current = (a, b) -> compareValues(a.get(sort.getField()), b.get(sort.getField()));
            if (sort.getDirection() == SortDirection.DESC) {
                current = current.reversed();
            }
            comparator = comparator == null ? current : comparator.thenComparing(current);
        }
        if (comparator != null) {
            resultRows.sort(comparator);
        }

        return new QueryResult(columns, resultRows);
    }

    private boolean matches(DataRow row) {
        if (year != null) {
            OptionalInt rowYear = row.getYear();
            if (!rowYear.isPresent() || rowYear.getAsInt() != Integer.parseInt(year)) {
                return false;
            }
        }
        for (Map.Entry<String, Set<String>> entry : filters.entrySet()) {
            String value = row.getString(entry.getKey());
            if (value == null || !entry.getValue().contains(value)) {
                return false;
            }
        }
        return true;
    }

    private double aggregate(List<DataRow> rows, AggregationSpec spec) {
        if (spec.getAggregation() == Aggregation.COUNT) {
            return rows.size();
        }
        double sum = 0.0;
        double count = 0.0;
        double min = Double.POSITIVE_INFINITY;
        double max = Double.NEGATIVE_INFINITY;
        for (DataRow row : rows) {
            Object raw = row.get(spec.getMeasure());
            double value = asDouble(raw);
            if (Double.isNaN(value)) {
                continue;
            }
            sum += value;
            count++;
            min = Math.min(min, value);
            max = Math.max(max, value);
        }
        switch (spec.getAggregation()) {
            case SUM:
                return sum;
            case AVG:
                return count == 0.0 ? 0.0 : sum / count;
            case MIN:
                return count == 0.0 ? 0.0 : min;
            case MAX:
                return count == 0.0 ? 0.0 : max;
            case NONE:
            case COUNT:
            default:
                return sum;
        }
    }

    private static double asDouble(Object value) {
        if (value instanceof Number) {
            return ((Number) value).doubleValue();
        }
        if (value == null) {
            return Double.NaN;
        }
        try {
            return Double.parseDouble(String.valueOf(value).replace(",", "").trim());
        } catch (NumberFormatException ex) {
            return Double.NaN;
        }
    }

    private static int compareValues(Object a, Object b) {
        if (a == null && b == null) {
            return 0;
        }
        if (a == null) {
            return -1;
        }
        if (b == null) {
            return 1;
        }
        if (a instanceof Number && b instanceof Number) {
            return Double.compare(((Number) a).doubleValue(), ((Number) b).doubleValue());
        }
        return String.valueOf(a).compareTo(String.valueOf(b));
    }
}

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public class MapService {
    private final Map<String, GeoArea> areas = new LinkedHashMap<>();

    public void addArea(GeoArea area) {
        areas.put(area.getHascCode(), area);
    }

    public GeoArea getArea(String hascCode) {
        return areas.get(hascCode);
    }

    public List<GeoArea> getBorderingRegions(String hascCode) {
        List<GeoArea> result = new ArrayList<>();
        GeoArea area = areas.get(hascCode);
        if (area != null) {
            for (String bordering : area.getBorderingHascCodes()) {
                GeoArea borderingArea = areas.get(bordering);
                if (borderingArea != null) {
                    result.add(borderingArea);
                }
            }
        }
        return result;
    }

    public List<String> getDynamicPolygons(List<String> hascCodes) {
        List<String> polygons = new ArrayList<>();
        for (String code : hascCodes) {
            GeoArea area = areas.get(code);
            if (area != null) {
                polygons.add(area.getPolygonGeoJson());
            }
        }
        return polygons;
    }

    public List<Integer> getSteppedZoomLevels() {
        return List.of(1, 2, 4, 8, 16);
    }

    public String getConsistentColor(String hascCode) {
        int hash = hascCode == null ? 0 : Math.abs(hascCode.hashCode());
        return String.format("#%06X", hash & 0xFFFFFF);
    }

    public boolean canVisualizeOnMap(Dataset dataset) {
        if (dataset == null || dataset.getModel() == null) {
            return false;
        }
        boolean hasGeo = false;
        for (Field field : dataset.getModel().getDimensions()) {
            if (field.getType() == FieldType.GEO) {
                hasGeo = true;
                break;
            }
        }
        boolean hasMeasure = !dataset.getModel().getMeasures().isEmpty();
        return hasGeo && hasMeasure;
    }

    public List<String> getMapRequirements() {
        return List.of(
                "A geographic dimension using HASC codes",
                "At least one numeric measure",
                "Optional GeoJSON boundary sources for improved map visualisations"
        );
    }

    public List<String> validateDatasetForMap(Dataset dataset) {
        List<String> missing = new ArrayList<>();
        if (dataset == null || dataset.getModel() == null) {
            missing.add("Dataset model");
            return missing;
        }
        boolean hasGeo = false;
        for (Field field : dataset.getModel().getDimensions()) {
            if (field.getType() == FieldType.GEO) {
                hasGeo = true;
                break;
            }
        }
        if (!hasGeo) {
            missing.add("Geographic dimension using HASC codes");
        }
        if (dataset.getModel().getMeasures().isEmpty()) {
            missing.add("At least one numeric measure");
        }
        return missing;
    }
}
