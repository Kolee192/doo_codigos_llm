import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import java.util.concurrent.atomic.AtomicInteger;

public class User {
    private static final AtomicInteger IDS = new AtomicInteger(1);

    private final int id;
    private final String username;
    private String email;
    private final String password;
    private String bio;
    private String company;
    private final Set<Role> roles;
    private Sponsor sponsorCompany;
    private final List<Registration> registrations;
    private final List<ConferenceSession> submittedSessions;
    private final Schedule schedule;
    private boolean newsletterSubscribed;

    public User(String username, String email, String password, Role... roles) {
        this.id = IDS.getAndIncrement();
        this.username = username;
        this.email = email;
        this.password = password;
        this.bio = "";
        this.company = "";
        this.roles = new HashSet<>();
        if (roles == null || roles.length == 0) {
            this.roles.add(Role.AUTHENTICATED);
        } else {
            for (Role role : roles) {
                this.roles.add(role);
            }
        }
        this.registrations = new ArrayList<>();
        this.submittedSessions = new ArrayList<>();
        this.schedule = new Schedule();
    }

    public int getId() {
        return id;
    }

    public String getUsername() {
        return username;
    }

    public String getEmail() {
        return email;
    }

    public String getPassword() {
        return password;
    }

    public String getBio() {
        return bio;
    }

    public String getCompany() {
        return company;
    }

    public Set<Role> getRoles() {
        return roles;
    }

    public void addRole(Role role) {
        roles.add(role);
    }

    public boolean hasRole(Role role) {
        return roles.contains(role);
    }

    public boolean isAuthenticated() {
        return !(roles.size() == 1 && roles.contains(Role.ANONYMOUS));
    }

    public Sponsor getSponsorCompany() {
        return sponsorCompany;
    }

    public void setSponsorCompany(Sponsor sponsorCompany) {
        this.sponsorCompany = sponsorCompany;
    }

    public void updateProfile(String bio, String company, String email) {
        if (bio != null) {
            this.bio = bio;
        }
        if (company != null) {
            this.company = company;
        }
        if (email != null) {
            this.email = email;
        }
    }

    public List<Registration> getRegistrations() {
        return registrations;
    }

    public void addRegistration(Registration registration) {
        registrations.add(registration);
        if (registration.getStatus() == RegistrationStatus.CONFIRMED
                || registration.getStatus() == RegistrationStatus.WAITLISTED) {
            schedule.add(registration.getEvent());
        }
    }

    public void removeRegistration(Registration registration) {
        registrations.remove(registration);
        schedule.remove(registration.getEvent());
    }

    public List<ConferenceSession> getSubmittedSessions() {
        return submittedSessions;
    }

    public void addSubmittedSession(ConferenceSession session) {
        submittedSessions.add(session);
    }

    public Schedule getSchedule() {
        return schedule;
    }

    public boolean isNewsletterSubscribed() {
        return newsletterSubscribed;
    }

    public void setNewsletterSubscribed(boolean newsletterSubscribed) {
        this.newsletterSubscribed = newsletterSubscribed;
    }

    @Override
    public String toString() {
        return username + " <" + email + "> " + roles;
    }
}

        this.openingHours = openingHours;
    }

    public List<Room> getRooms() {
        return rooms;
    }

        rooms.add(room);
        room.setVenue(this);
    }

        StringBuilder builder = new StringBuilder();
        builder.append(name).append(" - ").append(address).append("\n");
        builder.append(description).append("\n");
        builder.append("Map: ").append(mapUrl).append("\n");
        builder.append("Hours: ").append(openingHours).append("\n");
        builder.append("Rooms: ");
        for (Room room : rooms) {
            builder.append(room.getName()).append(" ");
        }
        return builder.toString().trim();
    }

import java.time.LocalDateTime;

public class TimeSlot {
    private final LocalDateTime start;
    private final LocalDateTime end;

    public TimeSlot(LocalDateTime start, LocalDateTime end) {
        if (start == null || end == null || end.isBefore(start)) {
            throw new IllegalArgumentException("Invalid time slot");
        }
        this.start = start;
        this.end = end;
    }

    public LocalDateTime getStart() {
        return start;
    }

    public LocalDateTime getEnd() {
        return end;
    }

    public boolean overlaps(TimeSlot other) {
        return start.isBefore(other.end) && other.start.isBefore(end);
    }

    @Override
    public String toString() {
        return start + " - " + end;
    }
}

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;

public class Training implements Registrable {
    private static int nextId = 1;

    private final int id;
    private String title;
    private String summary;
    private String description;
    private String logoUrl;
    private String slidesLink;
    private final List<User> trainers;
    private Venue venue;
    private Room room;
    private final LocalDateTime start;
    private final LocalDateTime end;
    private final SkillLevel skillLevel;
    private final int capacity;
    private final List<Registration> registrations;
    private final List<Registration> waitlist;
    private final List<Sponsor> sponsors;
    private final LocalDateTime cancellationDeadline;

    public Training(String title, String summary, String description,
                    LocalDateTime start, LocalDateTime end, int capacity,
                    SkillLevel skillLevel, LocalDateTime cancellationDeadline) {
        this.id = nextId++;
        this.title = title;
        this.summary = summary;
        this.description = description;
        this.logoUrl = "/images/default-training.png";
        this.slidesLink = "";
        this.trainers = new ArrayList<>();
        this.start = start;
        this.end = end;
        this.skillLevel = skillLevel;
        this.capacity = capacity;
        this.registrations = new ArrayList<>();
        this.waitlist = new ArrayList<>();
        this.sponsors = new ArrayList<>();
        this.cancellationDeadline = cancellationDeadline;
    }

    public int getId() {
        return id;
    }

    @Override
    public String getTitle() {
        return title;
    }

    public String getSummary() {
        return summary;
    }

    public String getDescription() {
        return description;
    }

    public String getLogoUrl() {
        return logoUrl;
    }

    public void setLogoUrl(String logoUrl) {
        this.logoUrl = logoUrl;
    }

    public String getSlidesLink() {
        return slidesLink;
    }

    public List<User> getTrainers() {
        return trainers;
    }

    public void addTrainer(User trainer) {
        if (!trainers.contains(trainer)) {
            trainers.add(trainer);
            trainer.addRole(Role.TRAINER);
        }
    }

    public Venue getVenue() {
        return venue;
    }

    public Room getRoom() {
        return room;
    }

    public void setLocation(Venue venue, Room room) {
        this.venue = venue;
        this.room = room;
        if (venue != null && room != null && !venue.getRooms().contains(room)) {
            venue.addRoom(room);
        }
    }

    @Override
    public LocalDateTime getStartDateTime() {
        return start;
    }

    @Override
    public LocalDateTime getEndDateTime() {
        return end;
    }

    public SkillLevel getSkillLevel() {
        return skillLevel;
    }

    public int getCapacity() {
        return capacity;
    }

    public List<Registration> getAllRegistrations() {
        return registrations;
    }

    public List<Registration> getWaitlist() {
        return waitlist;
    }

    public List<Sponsor> getSponsors() {
        return sponsors;
    }

    public void addSponsor(Sponsor sponsor) {
        if (!sponsors.contains(sponsor)) {
            sponsors.add(sponsor);
        }
    }

    public LocalDateTime getCancellationDeadline() {
        return cancellationDeadline;
    }

    @Override
    public int getRemainingRegistrations() {
        int confirmed = 0;
        for (Registration registration : registrations) {
            if (registration.getStatus() == RegistrationStatus.CONFIRMED
                    || registration.getStatus() == RegistrationStatus.ATTENDED) {
                confirmed++;
            }
        }
        return Math.max(0, capacity - confirmed);
    }

    @Override
    public String getLocationDetails() {
        if (venue == null || room == null) {
            return "Location to be announced";
        }
        return venue.getName() + ", " + room.getName() + " (" + venue.getAddress() + ")";
    }

    @Override
    public String getSeoPath() {
        return "/training/" + slug(title);
    }

    @Override
    public boolean isCancellationAllowed(LocalDateTime now) {
        return !now.isAfter(cancellationDeadline);
    }

    @Override
    public boolean acceptsRegistration(User user, boolean acceptedCancellationTerms, LocalDateTime now) {
        if (user == null || !user.isAuthenticated()) {
            return false;
        }
        if (!acceptedCancellationTerms) {
            return false;
        }
        if (now.isAfter(start)) {
            return false;
        }
        if (getRegistrationFor(user) != null) {
            return false;
        }
        for (Registration registration : user.getRegistrations()) {
            if (registration.getStatus() == RegistrationStatus.CANCELLED
                    || registration.getStatus() == RegistrationStatus.REFUNDED) {
                continue;
            }
            LocalDate otherDay = registration.getEvent().getStartDateTime().toLocalDate();
            if (otherDay.equals(start.toLocalDate())) {
                return false;
            }
        }
        return true;
    }

    @Override
    public Registration register(User user, boolean acceptedCancellationTerms, LocalDateTime now) {
        if (!acceptedCancellationTerms) {
            throw new IllegalArgumentException("Cancellation terms must be accepted.");
        }
        if (!acceptsRegistration(user, acceptedCancellationTerms, now)) {
            throw new IllegalStateException("Registration is not allowed. Check authentication, date, and one-registration-per-day rule.");
        }
        boolean full = getRemainingRegistrations() <= 0;
        Registration registration = new Registration(user, this, now, true, null);
        if (full) {
            registration.setStatus(RegistrationStatus.WAITLISTED);
            waitlist.add(registration);
        } else {
            registration.setStatus(RegistrationStatus.CONFIRMED);
            registrations.add(registration);
        }
        user.addRegistration(registration);
        return registration;
    }

    public Registration getRegistrationFor(User user) {
        for (Registration registration : registrations) {
            if (registration.getUser().equals(user)) {
                return registration;
            }
        }
        for (Registration registration : waitlist) {
            if (registration.getUser().equals(user)) {
                return registration;
            }
        }
        return null;
    }

    public List<Registration> getConfirmedAttendees() {
        List<Registration> result = new ArrayList<>();
        for (Registration registration : registrations) {
            if (registration.getStatus() == RegistrationStatus.CONFIRMED
                    || registration.getStatus() == RegistrationStatus.ATTENDED) {
                result.add(registration);
            }
        }
        return result;
    }

    public void cancel(User user, LocalDateTime now) {
        Registration registration = getRegistrationFor(user);
        if (registration == null) {
            throw new IllegalArgumentException("No registration found for this user.");
        }
        if (!registration.isAcceptedCancellationTerms()) {
            throw new IllegalStateException("Cancellation terms were not accepted.");
        }
        if (!isCancellationAllowed(now)) {
            throw new IllegalStateException("Cancellation deadline has passed.");
        }
        registration.cancel();
        registrations.remove(registration);
        waitlist.remove(registration);
        user.removeRegistration(registration);
        if (!waitlist.isEmpty()) {
            Registration promoted = waitlist.remove(0);
            promoted.setStatus(RegistrationStatus.CONFIRMED);
            registrations.add(promoted);
            promoted.getUser().getSchedule().add(this);
        }
    }

    public void editTrainingNode(User editor, String title, String summary, String description, String slidesLink) {
        if (!trainers.contains(editor)
                && !editor.hasRole(Role.TRAINING_COORDINATOR)
                && !editor.hasRole(Role.ADMINISTRATOR)) {
            throw new SecurityException("Only assigned trainers or training coordinators can edit this training.");
        }
        if (title != null && !title.isBlank()) {
            this.title = title;
        }
        if (summary != null) {
            this.summary = summary;
        }
        if (description != null) {
            this.description = description;
        }
        if (slidesLink != null) {
            this.slidesLink = slidesLink;
        }
    }

    public String getRegisterLink() {
        return getSeoPath() + "/register";
    }

    public String getDetailsLink() {
        return getSeoPath();
    }

    private String slug(String input) {
        return input.toLowerCase().replaceAll("[^a-z0-9]+", "-").replaceAll("(^-|-$)", "");
    }

    @Override
    public String toString() {
        return title + " [" + skillLevel + "] " + start + " remaining=" + getRemainingRegistrations();
    }
}

import java.util.ArrayList;
import java.util.List;

public class Schedule {
    private final List<Registrable> items;

    public Schedule() {
        this.items = new ArrayList<>();
    }

    public boolean add(Registrable item) {
        if (item == null) {
            return false;
        }
        for (Registrable existing : items) {
            if (existing.getTitle().equals(item.getTitle())
                    && existing.getStartDateTime().equals(item.getStartDateTime())) {
                return false;
            }
        }
        items.add(item);
        return true;
    }

    public boolean remove(Registrable item) {
        return items.remove(item);
    }

    public List<Registrable> getItems() {
        return items;
    }

    public boolean hasConflict(Registrable candidate) {
        for (Registrable item : items) {
            if (overlaps(item, candidate)) {
                return true;
            }
        }
        return false;
    }

    private boolean overlaps(Registrable a, Registrable b) {
        return a.getStartDateTime().isBefore(b.getEndDateTime())
                && b.getStartDateTime().isBefore(a.getEndDateTime());
    }

    public String view() {
        StringBuilder builder = new StringBuilder("My Schedule\n");
        for (Registrable item : items) {
            builder.append("- ").append(item.getTitle()).append(" at ")
                    .append(item.getLocationDetails()).append(" ")
                    .append(item.getStartDateTime()).append("\n");
        }
        return builder.toString();
    }
}
