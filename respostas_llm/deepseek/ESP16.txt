public class GeoPoint {
    private final double latitude;
    private final double longitude;

    public GeoPoint(double latitude, double longitude) {
        this.latitude = latitude;
        this.longitude = longitude;
    }

    public double getLatitude() {
        return latitude;
    }

    public double getLongitude() {
        return longitude;
    }

    public double distanceKmTo(GeoPoint other) {
        if (other == null) {
            return Double.MAX_VALUE;
        }
        double earthRadiusKm = 6371.0;
        double latDistance = Math.toRadians(other.latitude - latitude);
        double lonDistance = Math.toRadians(other.longitude - longitude);
        double a = Math.sin(latDistance / 2) * Math.sin(latDistance / 2)
                + Math.cos(Math.toRadians(latitude)) * Math.cos(Math.toRadians(other.latitude))
                * Math.sin(lonDistance / 2) * Math.sin(lonDistance / 2);
        double c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
        return earthRadiusKm * c;
    }

    @Override
    public String toString() {
        return "(" + latitude + ", " + longitude + ")";
    }
}

        for (Dish dish : menu) {

                return restaurant;
            }

    public List<Restaurant> findRestaurantsByName(String name) {
        List<Restaurant> result = new ArrayList<>();
        if (name != null) {
            for (Restaurant restaurant : restaurants) {
                if (restaurant.getName() != null && restaurant.getName().equalsIgnoreCase(name)) {
                    result.add(restaurant);
                }
            }
        }
        return result;
    }

        return new ArrayList<>(restaurants);
    }

    public void addRestaurantType(RestaurantType type) {
        if (type != null) {
            restaurantTypes.add(type);
        }
    }

    public RestaurantType findRestaurantTypeByName(String name) {
        for (RestaurantType type : restaurantTypes) {
            if (type.getName() != null && type.getName().equalsIgnoreCase(name)) {
                return type;
            }
        }
        return null;
    }

    public boolean deleteRestaurantType(RestaurantType type) {
        if (type == null) {
            return false;
        }
        boolean removed = restaurantTypes.remove(type);
        if (removed) {
            for (Restaurant restaurant : restaurants) {
                if (type.equals(restaurant.getType())) {
                    restaurant.setType(null);
                }
            }
        }
        return removed;
    }

    public List<RestaurantType> getRestaurantTypes() {
        return new ArrayList<>(restaurantTypes);
    }

    public void addDish(Dish dish) {
        if (dish != null) {
            dishes.add(dish);
        }
    }

    public Dish findDishByName(String name) {

            if (dish.getName() != null && dish.getName().equalsIgnoreCase(name)) {
                return dish;
            }
        }
        return null;
    }

    public boolean deleteDish(Dish dish) {
        if (dish == null) {
            return false;
        }

import java.util.EnumSet;
import java.util.Set;

public class SearchCriteria {
    private final Set<SearchOption> selectedOptions;
    private int minPrice;
    private int maxPrice;
    private int minDistanceKm;
    private int maxDistanceKm;

        this.minDistanceKm = 0;
        this.maxDistanceKm = 10;
        this.freeText = "";
    }

    public boolean hasOption(SearchOption option) {
        return selectedOptions.contains(option);
    }

    public Set<SearchOption> getSelectedOptions() {
        return Collections.unmodifiableSet(selectedOptions);
    }

        return minPrice;
    }

    public int getMaxPrice() {
        return maxPrice;
    }

    public int getMinDistanceKm() {
        return minDistanceKm;
    }

    public int getMaxDistanceKm() {
        return maxDistanceKm;
    }

    public RestaurantType getRestaurantType() {
        return restaurantType;
    }

    public Dish getSpecificDish() {
        return specificDish;
    }

    public String getFreeText() {
        return freeText;
    }

    public void setPriceRange(int minPrice, int maxPrice) {
        if (minPrice < 0 || maxPrice < minPrice) {
            throw new IllegalArgumentException("Invalid price range");
        }
        this.minPrice = minPrice;
        this.maxPrice = maxPrice;
        selectedOptions.add(SearchOption.PRICE);
    }

    public void setDestinationRange(int minDistanceKm, int maxDistanceKm) {
        if (minDistanceKm < 0 || maxDistanceKm < minDistanceKm) {
            throw new IllegalArgumentException("Invalid distance range");
        }
        this.minDistanceKm = minDistanceKm;
        this.maxDistanceKm = maxDistanceKm;
        selectedOptions.add(SearchOption.DESTINATION);
    }

    public void setRestaurantType(RestaurantType restaurantType) {
        this.restaurantType = restaurantType;
        if (restaurantType != null) {
            selectedOptions.add(SearchOption.RESTAURANT_TYPE);
        }
    }

    public void setSpecificDish(Dish specificDish) {
        this.specificDish = specificDish;
        if (specificDish != null) {
            selectedOptions.add(SearchOption.SPECIFIC_DISH);
        }
    }

    public void setFreeText(String freeText) {
        this.freeText = freeText == null ? "" : freeText;
        selectedOptions.add(SearchOption.FREE_TEXT);
    }

    public ViewMode getDefaultView() {
        if (selectedOptions.contains(SearchOption.DESTINATION)
                || selectedOptions.contains(SearchOption.RESTAURANT_TYPE)
                || selectedOptions.contains(SearchOption.SPECIFIC_DISH)
                || selectedOptions.contains(SearchOption.FREE_TEXT)) {
            return ViewMode.MAP;
        }
        if (selectedOptions.contains(SearchOption.PRICE)) {
            return ViewMode.LIST;
        }
        return ViewMode.MAP;
    }

import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

public class SearchResult {
    public static final int MAX_RESULTS = 100;

    private final SearchCriteria criteria;
    private final GeoPoint userLocation;
    private final List<RestaurantResult> baseResults;
    private final FilterCriteria activeFilter;
    private List<RestaurantResult> visibleResults;
    private SortField currentSort;
    private SortDirection currentSortDirection;

    public SearchResult(SearchCriteria criteria, List<RestaurantResult> results, GeoPoint userLocation) {
        this.criteria = criteria;
        this.userLocation = userLocation;
        this.baseResults = new ArrayList<>(results);
        this.activeFilter = new FilterCriteria();
        if (criteria != null && criteria.hasOption(SearchOption.PRICE)) {
            this.currentSort = SortField.PRICE;
        } else {
            this.currentSort = SortField.DISTANCE;
        }
        this.currentSortDirection = SortDirection.ASCENDING;
        recomputeVisible();
    }

    public ViewMode getDefaultView() {
        return criteria == null ? ViewMode.MAP : criteria.getDefaultView();
    }

    public boolean hasResults() {
        return !baseResults.isEmpty();
    }

    public List<RestaurantResult> getVisibleResults() {
        return new ArrayList<>(visibleResults);
    }

    public List<RestaurantResult> getPins() {
        return getVisibleResults();
    }

    public List<RestaurantResult> getBaseResults() {
        return new ArrayList<>(baseResults);
    }

    public FilterCriteria getActiveFilter() {
        return activeFilter;
    }

    public void applyFilter(FilterCriteria filter) {
        if (filter == null) {
            return;
        }
        activeFilter.setMaxDistanceKm(filter.getMaxDistanceKm());
        activeFilter.setMaxPrice(filter.getMaxPrice());
        activeFilter.setRestaurantType(filter.getRestaurantType());
        activeFilter.setDish(filter.getDish());
        recomputeVisible();
    }

    public void sort(SortField field) {
        if (field == null) {
            return;
        }
        if (currentSort == field) {
            currentSortDirection = currentSortDirection == SortDirection.ASCENDING
                    ? SortDirection.DESCENDING : SortDirection.ASCENDING;
        } else {
            currentSort = field;
            currentSortDirection = SortDirection.ASCENDING;
        }
        recomputeVisible();
    }

    private void recomputeVisible() {
        List<RestaurantResult> filtered = new ArrayList<>();
        for (RestaurantResult result : baseResults) {
            if (matchesFilter(result)) {
                filtered.add(result);
            }
        }
        filtered.sort(buildComparator(currentSort));
        int limit = Math.min(MAX_RESULTS, filtered.size());
        visibleResults = new ArrayList<>(filtered.subList(0, limit));
    }

    private boolean matchesFilter(RestaurantResult result) {
        Restaurant restaurant = result.getRestaurant();
        if (activeFilter.getMaxDistanceKm() != null
                && result.getDistanceKm() > activeFilter.getMaxDistanceKm()) {
            return false;
        }
        if (activeFilter.getMaxPrice() != null
                && restaurant.getAveragePrice() > activeFilter.getMaxPrice()) {
            return false;
        }
        if (activeFilter.getRestaurantType() != null) {
            if (restaurant.getType() == null
                    || !restaurant.getType().equals(activeFilter.getRestaurantType())) {
                return false;
            }
        }
        if (activeFilter.getDish() != null && !restaurant.hasDish(activeFilter.getDish().getName())) {
            return false;
        }
        return true;
    }

    private Comparator<RestaurantResult> buildComparator(SortField field) {
        Comparator<RestaurantResult> byDistance = Comparator.comparingDouble(RestaurantResult::getDistanceKm);
        Comparator<RestaurantResult> byPrice = Comparator.comparingDouble(r -> r.getRestaurant().getAveragePrice());
        Comparator<RestaurantResult> byType = Comparator.comparing(
                r -> lower(r.getRestaurant().getType() == null ? "" : r.getRestaurant().getType().getName()));
        Comparator<RestaurantResult> byDish = Comparator.comparing(r -> lower(dishName(r)));
        Comparator<RestaurantResult> byName = Comparator.comparing(r -> lower(r.getRestaurant().getName()));

        Comparator<RestaurantResult> comparator;
        switch (field) {
            case NAME:
                comparator = byName.thenComparing(byPrice).thenComparing(byDistance);
                break;
            case PRICE:
                comparator = byPrice.thenComparing(byDistance).thenComparing(byType).thenComparing(byDish);
                break;
            case RESTAURANT_TYPE:
                comparator = byType.thenComparing(byDistance).thenComparing(byPrice).thenComparing(byDish);
                break;
            case DISH:
                comparator = byDish.thenComparing(byDistance).thenComparing(byPrice).thenComparing(byType);
                break;
            case DISTANCE:
            default:
                comparator = byDistance.thenComparing(byPrice).thenComparing(byType).thenComparing(byDish);
                break;
        }

        if (currentSortDirection == SortDirection.DESCENDING) {
            comparator = comparator.reversed();
        }
        return comparator;
    }

    private static String lower(String value) {
        return value == null ? "" : value.toLowerCase();
    }

    private String dishName(RestaurantResult result) {
        if (criteria != null && criteria.getSpecificDish() != null) {
            return criteria.getSpecificDish().getName();
        }
        if (result.getRestaurant().getMenu().isEmpty()) {
            return "";
        }
        return result.getRestaurant().getMenu().get(0).getName();
    }

import java.util.ArrayList;
import java.util.List;

public class RestaurantSearchService {
    private final RestaurantRepository restaurantRepository;

    public RestaurantSearchService(RestaurantRepository restaurantRepository) {
        this.restaurantRepository = restaurantRepository;
    }

    public SearchResult search(SearchCriteria criteria, GeoPoint userLocation) {
        if (criteria == null) {
            criteria = new SearchCriteria();
        }
        GeoPoint location = userLocation == null ? new GeoPoint(0, 0) : userLocation;
        List<RestaurantResult> matches = new ArrayList<>();

        for (Restaurant restaurant : restaurantRepository.getRestaurants()) {
            double distance = restaurant.getLocation() == null
                    ? 0.0 : location.distanceKmTo(restaurant.getLocation());

            if (criteria.hasOption(SearchOption.PRICE)
                    && (restaurant.getAveragePrice() < criteria.getMinPrice()
                    || restaurant.getAveragePrice() > criteria.getMaxPrice())) {
                continue;
            }

            if (criteria.hasOption(SearchOption.DESTINATION)
                    && (distance < criteria.getMinDistanceKm()
                    || distance > criteria.getMaxDistanceKm())) {
                continue;
            }

            if (criteria.hasOption(SearchOption.RESTAURANT_TYPE)
                    && criteria.getRestaurantType() != null) {
                if (restaurant.getType() == null
                        || !restaurant.getType().equals(criteria.getRestaurantType())) {
                    continue;
                }
            }

            if (criteria.hasOption(SearchOption.SPECIFIC_DISH)
                    && criteria.getSpecificDish() != null) {
                if (!restaurant.hasDish(criteria.getSpecificDish().getName())) {
                    continue;
                }
            }

            if (criteria.hasOption(SearchOption.FREE_TEXT)) {
                String text = criteria.getFreeText();
                if (text != null && !text.trim().isEmpty() && !restaurant.matchesFreeText(text)) {
                    continue;
                }
            }

            matches.add(new RestaurantResult(restaurant, distance));
        }

        return new SearchResult(criteria, matches, userLocation);
    }
}
