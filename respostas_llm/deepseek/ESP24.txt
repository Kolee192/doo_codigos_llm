import java.time.LocalDateTime;
import java.time.YearMonth;

public class Main {
    public static void main(String[] args) {
        GatewayDevice gateway = new GatewayDevice("GW-001");
        gateway.connectInternet();

        DigitalHomeSystem home = new DigitalHomeSystem("Demo Home", gateway);

        Thermostat thermostat = new Thermostat("T1", "Living Room Thermostat", "Living Room", 72);
        Humidistat humidistat = new Humidistat("H1", "Basement Humidistat", "Basement", 45);
        ContactSensor contactSensor = new ContactSensor("C1", "Front Door", "Entry");
        PowerSwitch powerSwitch = new PowerSwitch("P1", "Lamp", "Living Room");

        home.addThermostat(thermostat);
        home.addHumidistat(humidistat);
        home.addContactSensor(contactSensor);
        home.addPowerSwitch(powerSwitch);

        LocalDateTime now = LocalDateTime.now()
                .withHour(10)
                .withMinute(0)
                .withSecond(0)
                .withNano(0);

        for (int i = 0; i < 3; i++) {
            LocalDateTime time = now.minusHours(2 - i);
            thermostat.recordReading(70 + i, time);
            humidistat.recordHumidity(44 + i, time);
        }

        thermostat.scheduleSetpoint(now.getDayOfWeek(), now.getHour(), 74);
        home.applySchedules(now);

        HomeWebServer server = new HomeWebServer(home);
        server.setThermostatSetpoint("T1", 76, now.plusMinutes(5));
        server.setHumidistatSetpoint("H1", 50, now.plusMinutes(5));
        server.setPowerSwitch("P1", true, now.plusMinutes(5));

        server.openContactSensor("C1", now.plusMinutes(10));

        YearMonth month = YearMonth.from(now);
        home.createMonthlyPlan(month);

        ParameterSettings settings = new ParameterSettings();
        settings.putThermostatSetpoint("T1", 75);
        settings.putHumidistatSetpoint("H1", 48);
        settings.putPowerSwitchState("P1", false);
        settings.putContactArmed("C1", true);

        PeriodPlan periodPlan = new PeriodPlan(
                now.toLocalTime().minusMinutes(1),
                now.toLocalTime().plusHours(1),
                settings
        );

        home.addPlanPeriod(month, now.toLocalDate(), periodPlan);
        home.applyMonthlyPlan(month, now.plusMinutes(15));

        home.setOperating(false, now.plusMinutes(20));
        home.setOperating(true, now.plusMinutes(30));

        MonthlyReport report = home.generateMonthlyReport(month);

        System.out.println("DigitalHome demo");
        System.out.println("Thermostat T1 temperature: "
                + server.readThermostatTemperature("T1", TemperatureUnit.CELSIUS) + " C");
        System.out.println("Alarm active: " + home.getAlarmSystem().isActive());
        System.out.println(report.toSummaryString());

        home.resetAlarm(now.plusHours(1));
        System.out.println("Alarm reset. Active: " + home.getAlarmSystem().isActive());
    }
}

// File: TemperatureUnit.java
public enum TemperatureUnit {
    FAHRENHEIT,
    CELSIUS;

    public static double fromFahrenheit(double fahrenheit, TemperatureUnit unit) {
        if (unit == CELSIUS) {
            return (fahrenheit - 32.0) * 5.0 / 9.0;
        }
        return fahrenheit;
    }

    public static double toFahrenheit(double value, TemperatureUnit unit) {
        if (unit == CELSIUS) {
            return value * 9.0 / 5.0 + 32.0;
        }
        return value;
    }
}

import java.time.DayOfWeek;
import java.util.EnumMap;
import java.util.Map;

public class WeeklySchedule {
    private final Map<DayOfWeek, HourlySchedule> days = new EnumMap<>(DayOfWeek.class);

    public WeeklySchedule() {
        for (DayOfWeek day : DayOfWeek.values()) {
            days.put(day, new HourlySchedule());
        }
    }

    public boolean set(DayOfWeek day, int hour, int value) {
        return days.get(day).set(hour, value);
    }

    public Integer get(DayOfWeek day, int hour) {
        return days.get(day).get(hour);
    }

    public HourlySchedule getDay(DayOfWeek day) {
        return days.get(day);
    }
}

import java.time.DayOfWeek;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.temporal.ChronoUnit;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Thermostat extends EnvironmentalDevice {
    public static final int MIN_FAHRENHEIT = 60;
    public static final int MAX_FAHRENHEIT = 80;

    private double currentTemperatureF;
    private int setpointF;
    private final WeeklySchedule schedule;
    private LocalDateTime manualOverrideUntil;
    private final List<SensorReading> readings;

    public Thermostat(String id, String name, String room, int initialSetpointF) {
        super(id, name, room);
        validateSetpoint(initialSetpointF);
        this.setpointF = initialSetpointF;
        this.currentTemperatureF = initialSetpointF;
        this.schedule = new WeeklySchedule();
        this.readings = new ArrayList<>();
    }

    private static void validateSetpoint(int fahrenheit) {
        if (fahrenheit < MIN_FAHRENHEIT || fahrenheit > MAX_FAHRENHEIT) {
            throw new IllegalArgumentException("Thermostat setpoint must be between 60 F and 80 F inclusive");
        }
    }

    public double readTemperature(TemperatureUnit unit) {
        return TemperatureUnit.fromFahrenheit(currentTemperatureF, unit);
    }

    public int getSetpointF() {
        return setpointF;
    }

    public void setManualSetpointF(int fahrenheit, LocalDateTime overrideUntil) {
        validateSetpoint(fahrenheit);
        this.setpointF = fahrenheit;
        this.manualOverrideUntil = overrideUntil;
    }

    public void applyScheduledSetpointF(int fahrenheit, LocalDateTime now) {
        validateSetpoint(fahrenheit);
        if (manualOverrideUntil != null && now.isBefore(manualOverrideUntil)) {
            return;
        }
        this.setpointF = fahrenheit;
        this.manualOverrideUntil = null;
    }

    public boolean scheduleSetpoint(DayOfWeek day, int hour, int fahrenheit) {
        validateSetpoint(fahrenheit);
        return schedule.set(day, hour, fahrenheit);
    }

    public void applyWeeklySchedule(LocalDateTime now) {
        Integer scheduled = schedule.get(now.getDayOfWeek(), now.getHour());
        if (scheduled != null) {
            applyScheduledSetpointF(scheduled, now);
        }
    }

    public void recordReading(double temperatureF, LocalDateTime time) {
        currentTemperatureF = temperatureF;
        readings.add(new SensorReading(time, temperatureF));
    }

    public List<SensorReading> getReadings() {
        return Collections.unmodifiableList(readings);
    }

    public DailyStats dailyStats(LocalDate date) {
        double sum = 0.0;
        double minimum = Double.POSITIVE_INFINITY;
        double maximum = Double.NEGATIVE_INFINITY;
        LocalDateTime minimumTime = null;
        LocalDateTime maximumTime = null;
        int count = 0;

        for (SensorReading reading : readings) {
            if (reading.getTimestamp().toLocalDate().equals(date)) {
                double value = reading.getValue();
                sum += value;
                count++;
                if (value < minimum) {
                    minimum = value;
                    minimumTime = reading.getTimestamp();
                }
                if (value > maximum) {
                    maximum = value;
                    maximumTime = reading.getTimestamp();
                }
            }
        }

        if (count == 0) {
            return DailyStats.empty(date);
        }

        return new DailyStats(date, sum / count, maximum, maximumTime, minimum, minimumTime, count);
    }

    public LocalDateTime getManualOverrideUntil() {
        return manualOverrideUntil;
    }

    public WeeklySchedule getSchedule() {
        return schedule;
    }

    public LocalDateTime defaultOverrideEnd(LocalDateTime now) {
        return now.truncatedTo(ChronoUnit.HOURS).plusHours(1);
    }
}

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Optional;

public class ApplianceManager {
    public static final int MAX_SWITCHES = 100;

    private final List<PowerSwitch> switches;

    public ApplianceManager() {
        this.switches = new ArrayList<>();
    }

    public boolean addPowerSwitch(PowerSwitch powerSwitch) {
        if (powerSwitch == null || switches.size() >= MAX_SWITCHES) {
            return false;
        }
        switches.add(powerSwitch);
        return true;
    }

    public Optional<PowerSwitch> findSwitch(String id) {
        for (PowerSwitch powerSwitch : switches) {
            if (powerSwitch.getId().equals(id)) {
                return Optional.of(powerSwitch);
            }
        }
        return Optional.empty();
    }

    public boolean getState(String id) {
        return findSwitch(id).map(PowerSwitch::getState).orElse(false);
    }

    public boolean setState(String id, boolean state, LocalDateTime now, boolean manual) {
        Optional<PowerSwitch> optional = findSwitch(id);
        if (optional.isEmpty()) {
            return false;
        }
        PowerSwitch powerSwitch = optional.get();
        if (manual) {
            powerSwitch.setManualState(state, now);
        } else {
            powerSwitch.applyScheduledState(state, now);
        }
        return true;
    }

    public List<PowerSwitch> getSwitches() {
        return Collections.unmodifiableList(switches);
    }
}

import java.time.LocalTime;

public class PeriodPlan {
    private final LocalTime start;
    private final LocalTime end;
    private final ParameterSettings settings;

    public PeriodPlan(LocalTime start, LocalTime end, ParameterSettings settings) {
        if (start == null || end == null || settings == null) {
            throw new IllegalArgumentException("Period start, end and settings are required");
        }
        if (!start.isBefore(end)) {
            throw new IllegalArgumentException("Period start must be before end");
        }
        this.start = start;
        this.end = end;
        this.settings = settings;
    }

    public LocalTime getStart() {
        return start;
    }

    public LocalTime getEnd() {
        return end;
    }

    public ParameterSettings getSettings() {
        return settings;
    }

    public boolean contains(LocalTime time) {
        return !time.isBefore(start) && time.isBefore(end);
    }
}

import java.time.LocalDate;
import java.time.YearMonth;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;

public class MonthlyPlan {
    private final YearMonth month;
    private final Map<LocalDate, DayPlan> days;

    public MonthlyPlan(YearMonth month) {
        if (month == null) {
            throw new IllegalArgumentException("Month is required");
        }
        this.month = month;
        this.days = new TreeMap<>();
    }

    public boolean addPeriod(LocalDate date, PeriodPlan periodPlan) {
        if (date == null || periodPlan == null) {
            return false;
        }
        if (!YearMonth.from(date).equals(month)) {
            throw new IllegalArgumentException("Date does not belong to plan month");
        }
        DayPlan dayPlan = days.computeIfAbsent(date, DayPlan::new);
        return dayPlan.addPeriod(periodPlan);
    }

    public DayPlan getDayPlan(LocalDate date) {
        return days.get(date);
    }

    public List<DayPlan> getDayPlans() {
        return new ArrayList<>(days.values());
    }

    public YearMonth getMonth() {
        return month;
    }
}

    public MonthlyPlan getPlan(YearMonth month) {
        return plans.get(month);
    }

    public boolean addPeriod(YearMonth month, LocalDate date, PeriodPlan periodPlan) {
        if (month == null || date == null || periodPlan == null) {
            return false;
        }
        MonthlyPlan monthlyPlan = createOrGetPlan(month);
        return monthlyPlan.addPeriod(date, periodPlan);
    }

    public PeriodPlan getPeriodFor(YearMonth month, LocalDate date, LocalTime time) {
        MonthlyPlan monthlyPlan = plans.get(month);
        if (monthlyPlan == null) {
            return null;
        }
        DayPlan dayPlan = monthlyPlan.getDayPlan(date);
        if (dayPlan == null) {
            return null;
        }
        return dayPlan.getPeriodAt(time);
    }

    public ParameterSettings getSettingsFor(YearMonth month, LocalDate date, LocalTime time) {
        PeriodPlan periodPlan = getPeriodFor(month, date, time);
        return periodPlan == null ? null : periodPlan.getSettings();
    }

import java.time.YearMonth;
import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public class MonthlyReport {
    private final YearMonth month;
    private final Map<String, List<DailyStats>> thermostatStats;
    private final Map<String, List<DailyStats>> humidistatStats;
    private final List<SecurityBreach> securityBreaches;
    private final List<DowntimePeriod> downtimePeriods;

    public MonthlyReport(
            YearMonth month,
            Map<String, List<DailyStats>> thermostatStats,
            Map<String, List<DailyStats>> humidistatStats,
            List<SecurityBreach> securityBreaches,
            List<DowntimePeriod> downtimePeriods
    ) {
        this.month = month;
        this.thermostatStats = new LinkedHashMap<>();
        for (Map.Entry<String, List<DailyStats>> entry : thermostatStats.entrySet()) {
            this.thermostatStats.put(entry.getKey(), new ArrayList<>(entry.getValue()));
        }
        this.humidistatStats = new LinkedHashMap<>();
        for (Map.Entry<String, List<DailyStats>> entry : humidistatStats.entrySet()) {
            this.humidistatStats.put(entry.getKey(), new ArrayList<>(entry.getValue()));
        }
        this.securityBreaches = new ArrayList<>(securityBreaches);
        this.downtimePeriods = new ArrayList<>(downtimePeriods);
    }

    public YearMonth getMonth() {
        return month;
    }

    public Map<String, List<DailyStats>> getThermostatStats() {
        return Collections.unmodifiableMap(thermostatStats);
    }

    public Map<String, List<DailyStats>> getHumidistatStats() {
        return Collections.unmodifiableMap(humidistatStats);
    }

    public List<SecurityBreach> getSecurityBreaches() {
        return Collections.unmodifiableList(securityBreaches);
    }

    public List<DowntimePeriod> getDowntimePeriods() {
        return Collections.unmodifiableList(downtimePeriods);
    }

    public String toSummaryString() {
        StringBuilder builder = new StringBuilder();
        builder.append("Monthly Report for ").append(month).append("\n");

        for (Map.Entry<String, List<DailyStats>> entry : thermostatStats.entrySet()) {
            builder.append("Thermostat ").append(entry.getKey()).append(":\n");
            appendStats(builder, entry.getValue());
        }

        for (Map.Entry<String, List<DailyStats>> entry : humidistatStats.entrySet()) {
            builder.append("Humidistat ").append(entry.getKey()).append(":\n");
            appendStats(builder, entry.getValue());
        }

        builder.append("Security breaches:\n");
        if (securityBreaches.isEmpty()) {
            builder.append("  none\n");
        } else {
            for (SecurityBreach breach : securityBreaches) {
                builder.append("  sensor ")
                        .append(breach.getSensorId())
                        .append(" at ")
                        .append(breach.getTime())
                        .append("\n");
            }
        }

        builder.append("Downtime periods:\n");
        if (downtimePeriods.isEmpty()) {
            builder.append("  none\n");
        } else {
            for (DowntimePeriod period : downtimePeriods) {
                builder.append("  ")
                        .append(period.getStart())
                        .append(" to ")
                        .append(period.getEnd())
                        .append("\n");
            }
        }

        return builder.toString();
    }

    private void appendStats(StringBuilder builder, List<DailyStats> statsList) {
        for (DailyStats stats : statsList) {
            if (stats.getSampleCount() > 0) {
                builder.append("  ").append(stats.getDate())
                        .append(" avg=").append(String.format("%.1f", stats.getAverage()))
                        .append(" max=").append(String.format("%.1f", stats.getMaximum()))
                        .append(" at ").append(stats.getMaximumTime())
                        .append(" min=").append(String.format("%.1f", stats.getMinimum()))
                        .append(" at ").append(stats.getMinimumTime())
                        .append("\n");
            }
        }
    }
}

import java.time.LocalDate;
import java.time.LocalDateTime;

public class DigitalHomeSystem {
    public static final int MAX_THERMOSTATS = 8;
    public static final int MAX_HUMIDISTATS = 8;

    public DigitalHomeSystem(String name, GatewayDevice gateway) {

    public void applySchedules(LocalDateTime now) {
        if (!operating) {
            return;
        }
        for (Thermostat thermostat : thermostats) {
            thermostat.applyWeeklySchedule(now);
        }
        for (Humidistat humidistat : humidistats) {
            humidistat.applyWeeklySchedule(now);
        }

    public void applyMonthlyPlan(YearMonth month, LocalDateTime now) {
        if (!operating) {
            return;
        }

        ParameterSettings settings = planner.getSettingsFor(month, now.toLocalDate(), now.toLocalTime());
        if (settings == null) {
            return;
        }

        for (Map.Entry<String, Integer> entry : settings.getThermostatSetpointsF().entrySet()) {
            findThermostat(entry.getKey()).ifPresent(thermostat ->
                    thermostat.applyScheduledSetpointF(entry.getValue(), now)
            );
        }

        for (Map.Entry<String, Integer> entry : settings.getHumidistatSetpoints().entrySet()) {
            findHumidistat(entry.getKey()).ifPresent(humidistat ->
                    humidistat.applyScheduledHumidity(entry.getValue(), now)
            );
        }

        for (Map.Entry<String, Boolean> entry : settings.getContactArmed().entrySet()) {
            Optional<ContactSensor> optional = findContactSensor(entry.getKey());
            if (optional.isPresent()) {
                ContactSensor contactSensor = optional.get();
                contactSensor.setArmed(entry.getValue());
                if (contactSensor.isBreach()) {
                    recordBreach(contactSensor, now);
                }
            }
        }

        for (Map.Entry<String, Boolean> entry : settings.getPowerSwitchStates().entrySet()) {
            findPowerSwitch(entry.getKey()).ifPresent(powerSwitch ->
                    powerSwitch.applyScheduledState(entry.getValue(), now)
            );
        }
    }

    private LocalDateTime endOfCurrentPeriod(LocalDateTime now) {
        PeriodPlan periodPlan = planner.getPeriodFor(
                YearMonth.from(now),
                now.toLocalDate(),
                now.toLocalTime()
        );
        if (periodPlan != null) {
            return LocalDateTime.of(now.toLocalDate(), periodPlan.getEnd());
        }
        return now.truncatedTo(ChronoUnit.HOURS).plusHours(1);
    }

    public void setOperating(boolean operating, LocalDateTime time) {
        if (time == null) {
            throw new IllegalArgumentException("Time is required");
        }
        if (this.operating == operating) {
            return;
        }
        if (!operating) {
            this.operationDownSince = time;
        } else {
            if (operationDownSince != null) {
                downtimePeriods.add(new DowntimePeriod(operationDownSince, time));
                operationDownSince = null;
            }
        }
        this.operating = operating;
    }

    public boolean isOperating() {
        return operating;
    }

    public MonthlyReport generateMonthlyReport(YearMonth month) {
        YearMonth currentMonth = YearMonth.now();
        if (month.isAfter(currentMonth) || month.isBefore(currentMonth.minusYears(2))) {
            throw new IllegalArgumentException("Reports are available for the current month and previous two years");
        }

        Map<String, List<DailyStats>> thermostatStats = new LinkedHashMap<>();
        for (Thermostat thermostat : thermostats) {
            List<DailyStats> stats = new ArrayList<>();
            for (int day = 1; day <= month.lengthOfMonth(); day++) {
                stats.add(thermostat.dailyStats(month.atDay(day)));
            }
            thermostatStats.put(thermostat.getId(), stats);
        }

        Map<String, List<DailyStats>> humidistatStats = new LinkedHashMap<>();
        for (Humidistat humidistat : humidistats) {
            List<DailyStats> stats = new ArrayList<>();
            for (int day = 1; day <= month.lengthOfMonth(); day++) {
                stats.add(humidistat.dailyStats(month.atDay(day)));
            }
            humidistatStats.put(humidistat.getId(), stats);
        }

        List<SecurityBreach> monthBreaches = new ArrayList<>();
        for (SecurityBreach breach : securityBreaches) {
            if (YearMonth.from(breach.getTime()).equals(month)) {
                monthBreaches.add(breach);
            }
        }

        List<DowntimePeriod> monthDowntime = new ArrayList<>();
        LocalDateTime monthStart = month.atDay(1).atStartOfDay();
        LocalDateTime monthEnd = month.atEndOfMonth().atTime(23, 59, 59);
        for (DowntimePeriod period : downtimePeriods) {
            if (!period.getEnd().isBefore(monthStart) && !period.getStart().isAfter(monthEnd)) {
                monthDowntime.add(period);
            }
        }

        return new MonthlyReport(month, thermostatStats, humidistatStats, monthBreaches, monthDowntime);
    }

    public List<Thermostat> getThermostats() {
        return Collections.unmodifiableList(thermostats);
    }

    public List<Humidistat> getHumidistats() {
        return Collections.unmodifiableList(humidistats);
    }

    public List<SecurityBreach> getSecurityBreaches() {
        return Collections.unmodifiableList(securityBreaches);
    }

    public List<DowntimePeriod> getDowntimePeriods() {
        return Collections.unmodifiableList(downtimePeriods);
    }
