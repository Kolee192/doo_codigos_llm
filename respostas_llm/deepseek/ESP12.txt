import java.util.*;

public class MetadataService {
    public Map<String, String> autoFill(User user, Dataset previousDataset) {
        Map<String, String> metadata = new HashMap<>();
        metadata.put("creator", user.getName());
        metadata.put("creatorEmail", user.getEmail());
        metadata.put("institution", user.getInstitution());
        metadata.put("language", user.getLanguage());
        if (previousDataset != null && previousDataset.getCurrentVersion() != null) {
            metadata.putAll(previousDataset.getCurrentVersion().getMetadata());
        }
        metadata.putAll(user.getPreviousMetadata());
        return metadata;
    }

    public void rememberMetadata(User user, Map<String, String> metadata) {
        user.getPreviousMetadata().putAll(metadata);
    }

    public void validateMinimumMetadata(Map<String, String> metadata) {
        String[] required = {"title", "creator", "description", "license"};
        for (String req : required) {
            if (metadata == null || !metadata.containsKey(req) || metadata.get(req) == null || metadata.get(req).isBlank()) {
                throw new IllegalArgumentException("Missing required metadata: " + req);
            }
        }
    }

    public void applySubjectMetadata(SubjectMetadata sm, String discipline, List<String> keywords, Map<String, String> fields) {
        sm.setDiscipline(discipline);
        sm.getKeywords().clear();
        sm.getKeywords().addAll(keywords);
        sm.getFields().putAll(fields);
    }
}

// File: AccessControlService.java
import java.time.LocalDate;

public class AccessControlService {
    public boolean checkAccess(User user, Dataset dataset, DatasetVersion version) {
        if (user == null || version == null) return false;
        if (user.getId().equals(dataset.getOwnerId())) return true;
        if (user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) return true;
        if (dataset.getCollaboratorIds().contains(user.getId())) return true;
        if (dataset.getDelegatedDepositorIds().contains(user.getId())) return true;
        if (version.getStatus() == DatasetStatus.PUBLISHED && version.getAccessLevel() == AccessLevel.PUBLIC) return true;
        if (version.getAccessLevel() == AccessLevel.EMBARGOED) {
            if (version.getEmbargoUntil() != null && LocalDate.now().isAfter(version.getEmbargoUntil())) return true;
            return false;
        }
        if (version.getAccessLevel() == AccessLevel.RESTRICTED || version.getAccessLevel() == AccessLevel.PRIVILEGED) {
            return dataset.getCollaboratorIds().contains(user.getId()) || user.hasRole(Role.ARCHIVE_ADMINISTRATOR);
        }
        return false;
    }

    public void grantPrivilegedAccess(Dataset dataset, User collaborator, User grantor) {
        if (grantor == null || (!grantor.getId().equals(dataset.getOwnerId()) && !grantor.hasRole(Role.ARCHIVE_ADMINISTRATOR))) {
            throw new SecurityException("Only owner or administrator can grant access");
        }
        dataset.addCollaborator(collaborator.getId());
    }
}

import java.util.*;

public class SearchService {
    public List<Dataset> search(List<Dataset> datasets, String query, String language) {
        String q = query == null ? "" : query.toLowerCase();
        List<Dataset> results = new ArrayList<>();
        for (Dataset d : datasets) {
            if (matches(d, q)) results.add(d);
        }
        return results;
    }

    private boolean matches(Dataset d, String q) {
        if (d.getTitle() != null && d.getTitle().toLowerCase().contains(q)) return true;
        DatasetVersion v = d.getCurrentVersion();
        if (v != null) {
            if (v.getDescription() != null && v.getDescription().toLowerCase().contains(q)) return true;
            for (String val : v.getMetadata().values()) {
                if (val != null && val.toLowerCase().contains(q)) return true;
            }
            for (DatasetFile f : v.getFiles()) {
                if (f.getFileName().toLowerCase().contains(q)) return true;
            }
        }
        SubjectMetadata sm = d.getSubjectMetadata();
        if (sm != null) {
            if (sm.getDiscipline() != null && sm.getDiscipline().toLowerCase().contains(q)) return true;
            for (String k : sm.getKeywords()) {
                if (k.toLowerCase().contains(q)) return true;
            }
            for (String val : sm.getFields().values()) {
                if (val != null && val.toLowerCase().contains(q)) return true;
            }
        }
        return false;
    }

    public List<String> examineFiles(Dataset dataset) {
        List<String> result = new ArrayList<>();
        DatasetVersion v = dataset.getCurrentVersion();
        if (v == null) return result;
        for (DatasetFile f : v.getFiles()) {
            result.add(f.getFileName() + " (" + f.getSizeBytes() + " bytes, " + f.getFormat() + ")");
        }
        return result;
    }

    public String exampleCitation(Dataset dataset) {
        DatasetVersion v = dataset.getCurrentVersion();
        if (v == null) return "No version available";
        String doi = v.getDoi() == null ? "DOI not minted" : v.getDoi();
        return dataset.getOwnerId() + " (" + v.getCreatedAt().getYear() + "). " + dataset.getTitle()
                + " (Version " + v.getVersionNumber() + "). University of Bath Data Archive. " + doi;
    }
}

    public Map<String, Long> downloadsByCountry(Dataset dataset) {
        Map<String, Long> map = new HashMap<>();
        for (DatasetVersion v : dataset.getVersions()) {
            for (DownloadRecord r : v.getDownloadRecords()) {
                map.merge(r.getCountry(), 1L, Long::sum);
            }
        }
        return map;
    }

        Map<String, Long> map = new HashMap<>();
        for (DatasetVersion v : dataset.getVersions()) {
            for (DownloadRecord r : v.getDownloadRecords()) {
                map.merge(r.getSector(), 1L, Long::sum);
            }
        }
        return map;
    }

    public int citationCount(Dataset dataset) {
        int count = 0;
        for (DatasetVersion v : dataset.getVersions()) {
            count += v.getCitations().size();
        }
        return count;
    }
}

// File: IntegrityService.java
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;

public class IntegrityService {
    public String computeChecksum(DatasetFile file) {
        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            String data = file.getFileName() + file.getSizeBytes() + file.getFormat() + file.getContentPreview();
            byte[] hash = md.digest(data.getBytes(StandardCharsets.UTF_8));
            String checksum = toHex(hash);
            file.setChecksum(checksum);
            return checksum;
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    public String computeDatasetChecksum(DatasetVersion version) {
        StringBuilder sb = new StringBuilder();
        for (DatasetFile f : version.getFiles()) {
            if (f.getChecksum() == null) computeChecksum(f);
            sb.append(f.getChecksum());
        }
        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            byte[] hash = md.digest(sb.toString().getBytes(StandardCharsets.UTF_8));
            String checksum = toHex(hash);
            version.setIntegrityChecksum(checksum);
            return checksum;
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    public boolean verify(DatasetVersion version) {
        String old = version.getIntegrityChecksum();
        if (old == null) return false;
        String computed = computeDatasetChecksum(version);
        return old.equals(computed);
    }

    private String toHex(byte[] hash) {
        StringBuilder sb = new StringBuilder();
        for (byte b : hash) sb.append(String.format("%02x", b));
        return sb.toString();
    }
}

import java.util.HashMap;

public class VersionService {
    public DatasetVersion createNewVersion(Dataset dataset, User user, String description, String versionNote) {
        if (user == null || (!user.getId().equals(dataset.getOwnerId())
                && !dataset.getDelegatedDepositorIds().contains(user.getId())
                && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR))) {
            throw new SecurityException("Not allowed to create version");
        }
        DatasetVersion current = dataset.getCurrentVersion();
        DatasetVersion next = new DatasetVersion(current.getVersionNumber() + 1, description, DatasetStatus.DRAFT,
                current.getAccessLevel(), current.getLicense(), current.getEmbargoUntil());
        next.setMetadata(new HashMap<>(current.getMetadata()));
        next.setVersionNote(versionNote);
        dataset.addVersion(next);
        return next;
    }
}

public class DisposalService {
    public void requestDisposal(Dataset dataset, DisposalPolicy policy, User user) {
        if (user == null || (!user.getId().equals(dataset.getOwnerId()) && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR))) {
            throw new SecurityException("Not allowed to request disposal");
        }
        dataset.setDisposalPolicy(policy);
        dataset.getCurrentVersion().setStatus(DatasetStatus.DISPOSAL_REQUESTED);
    }

    public void approveDisposal(Dataset dataset, User admin) {
        if (admin == null || !admin.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Only archive administrator can approve disposal");
        }
        if (dataset.getDisposalPolicy() == null) {
            throw new IllegalStateException("No disposal policy");
        }
        dataset.getDisposalPolicy().setApproved(true);
        dataset.getDisposalPolicy().setApprovedBy(admin.getName());
        dataset.getCurrentVersion().setStatus(DatasetStatus.DISPOSAL_APPROVED);
    }
}

import java.util.*;
import java.util.function.Predicate;

public class AdminService {
    public void reviewDataset(Dataset dataset, User admin, boolean approve, MetadataService metadataService) {
        if (admin == null || !admin.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Only archive administrator can review");
        }
        DatasetVersion v = dataset.getCurrentVersion();
        if (approve) {
            metadataService.validateMinimumMetadata(v.getMetadata());
            v.setStatus(DatasetStatus.PUBLISHED);
        } else {
            v.setStatus(DatasetStatus.UNDER_REVIEW);
        }
    }

    public List<Dataset> queryArchive(List<Dataset> datasets, Predicate<Dataset> predicate) {
        List<Dataset> result = new ArrayList<>();
        for (Dataset d : datasets) {
            if (predicate.test(d)) result.add(d);
        }
        return result;
    }

    public Dataset importExternalDataset(String id, String title, String ownerId, String externalUrl) {
        Dataset d = new Dataset(id, title, ownerId);
        DatasetVersion v = new DatasetVersion(1, "Imported external dataset", DatasetStatus.EXTERNAL,
                AccessLevel.PUBLIC, LicenseType.CUSTOM, null);
        Map<String, String> metadata = new HashMap<>();
        metadata.put("title", title);
        metadata.put("creator", ownerId);
        metadata.put("description", "External dataset");
        metadata.put("license", LicenseType.CUSTOM.name());
        v.setMetadata(metadata);
        d.addVersion(v);
        d.addExternalLink(new ExternalRepositoryLink("External", externalUrl, id));
        return d;
    }
}

import java.util.List;

public class ExportService {
    public String exportAll(List<Dataset> datasets) {
        StringBuilder sb = new StringBuilder();
        for (Dataset d : datasets) {
            sb.append("DATASET|").append(d.getId()).append("|").append(d.getTitle()).append("|").append(d.getOwnerId()).append("\n");
            for (DatasetVersion v : d.getVersions()) {
                sb.append("VERSION|").append(d.getId()).append("|").append(v.getVersionNumber()).append("|")
                        .append(v.getStatus()).append("|").append(v.getLicense()).append("\n");
                for (DatasetFile f : v.getFiles()) {
                    sb.append("FILE|").append(f.getFileName()).append("|").append(f.getSizeBytes()).append("|")
                            .append(f.getChecksum()).append("\n");
                }
            }
        }
        return sb.toString();
    }

// File: CrisIntegration.java
import java.util.Collection;
import java.util.List;

public class CrisIntegration {
    public String harvestMetadata(List<Dataset> datasets, Collection<Project> projects, String funder) {
        StringBuilder sb = new StringBuilder();
        for (Dataset d : datasets) {
            Project project = null;
            for (Project p : projects) {
                if (p.getId().equals(d.getProjectId())) {
                    project = p;
                    break;
                }
            }
            if (project != null && funder.equalsIgnoreCase(project.getFunder())) {
                sb.append(d.getId()).append("|").append(d.getTitle()).append("|")
                        .append(project.getId()).append("|").append(project.getFunder()).append("\n");
            }
        }
        return sb.toString();
    }
}

import java.time.LocalDate;
import java.util.*;
import java.util.function.Predicate;

public class ArchiveSystem {
    private final Map<String, User> users = new LinkedHashMap<>();
    private final Map<String, Dataset> datasets = new LinkedHashMap<>();
    private final Map<String, Project> projects = new LinkedHashMap<>();
    private final MetadataService metadataService = new MetadataService();
    private final DoiService doiService = new DoiService();
    private final AccessControlService accessControlService = new AccessControlService();
    private final SearchService searchService = new SearchService();
    private final StatisticsService statisticsService = new StatisticsService();
    private final IntegrityService integrityService = new IntegrityService();
    private final VersionService versionService = new VersionService();
    private final DisposalService disposalService = new DisposalService();
    private final AdminService adminService = new AdminService();
    private final ExportService exportService = new ExportService();
    private final PureIntegration pureIntegration = new PureIntegration();
    private final CrisIntegration crisIntegration = new CrisIntegration();
    private final PrimoIntegration primoIntegration = new PrimoIntegration();
    private final LdapIntegration ldapIntegration = new LdapIntegration();
    private final HcpObjectStore hcpObjectStore = new HcpObjectStore();
    private final Sword2Api sword2Api = new Sword2Api();
    private int datasetCounter = 0;

    public User registerUser(String name, String email, Role... roles) {
        String id = "user-" + (users.size() + 1);
        User u = new User(id, name, email, roles);
        users.put(id, u);
        return u;
    }

    public Project createProject(String title, String funder, String dmpLink) {
        String id = "proj-" + (projects.size() + 1);
        Project p = new Project(id, title, funder, dmpLink);
        projects.put(id, p);
        return p;
    }

    public Dataset depositDataset(User user, String title, String description, LicenseType license, AccessLevel access, String projectId) {
        if (!user.hasRole(Role.DEPOSITOR)
                && !user.hasRole(Role.RESEARCH_FACILITY_MANAGER)
                && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)
                && !user.hasRole(Role.DEVELOPER)) {
            throw new SecurityException("User cannot deposit");
        }
        String id = "ds-" + (++datasetCounter);
        Dataset dataset = new Dataset(id, title, user.getId());
        dataset.setProjectId(projectId);
        DatasetVersion version = new DatasetVersion(1, description, DatasetStatus.DRAFT, access, license, null);
        Map<String, String> metadata = metadataService.autoFill(user, null);
        metadata.put("title", title);
        metadata.put("description", description);
        metadata.put("license", license.name());
        version.setMetadata(metadata);
        dataset.addVersion(version);
        datasets.put(id, dataset);
        if (projectId != null && projects.containsKey(projectId)) {
            projects.get(projectId).addDatasetId(id);
        }
        return dataset;
    }

    public void addFile(Dataset dataset, DatasetFile file) {
        DatasetVersion v = dataset.getCurrentVersion();
        String key = hcpObjectStore.store(file);
        file.setObjectStoreKey(key);
        integrityService.computeChecksum(file);
        v.addFile(file);
        integrityService.computeDatasetChecksum(v);
    }

    public void submitDataset(Dataset dataset, User user) {
        if (!user.getId().equals(dataset.getOwnerId())
                && !dataset.getDelegatedDepositorIds().contains(user.getId())
                && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Not allowed to submit");
        }
        dataset.getCurrentVersion().setStatus(DatasetStatus.SUBMITTED);
    }

    public void reviewDataset(Dataset dataset, User admin, boolean approve) {
        adminService.reviewDataset(dataset, admin, approve, metadataService);
    }

    public String mintDoi(Dataset dataset, User user) {
        if (!user.getId().equals(dataset.getOwnerId()) && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Not allowed to mint DOI");
        }
        return doiService.mintDoi(dataset.getCurrentVersion());
    }

    public void publishDataset(Dataset dataset, User admin) {
        if (!admin.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Only archive administrator can publish");
        }
        metadataService.validateMinimumMetadata(dataset.getCurrentVersion().getMetadata());
        dataset.getCurrentVersion().setStatus(DatasetStatus.PUBLISHED);
    }

    public List<Dataset> search(String query, String language) {
        return searchService.search(new ArrayList<>(datasets.values()), query, language);
    }

    public List<Dataset> searchPrimo(String query) {
        return primoIntegration.search(new ArrayList<>(datasets.values()), query);
    }

    public boolean canAccess(User user, Dataset dataset) {
        return accessControlService.checkAccess(user, dataset, dataset.getCurrentVersion());
    }

    public void grantPrivilegedAccess(Dataset dataset, User grantor, User collaborator) {
        accessControlService.grantPrivilegedAccess(dataset, collaborator, grantor);
    }

    public void addCollaborator(Dataset dataset, User collaborator, User grantor) {
        grantPrivilegedAccess(dataset, grantor, collaborator);
    }

    public void addDelegatedDepositor(Dataset dataset, User delegated, User owner) {
        if (!owner.getId().equals(dataset.getOwnerId()) && !owner.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Not allowed to delegate depositor");
        }
        dataset.addDelegatedDepositor(delegated.getId());
    }

    public void addPublicationLink(Dataset dataset, PublicationLink link, User user) {
        if (!user.getId().equals(dataset.getOwnerId()) && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Not allowed to add publication link");
        }
        dataset.addPublicationLink(link);
    }

    public void addExternalRepositoryLink(Dataset dataset, ExternalRepositoryLink link, User user) {
        if (!user.getId().equals(dataset.getOwnerId()) && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Not allowed to add external repository link");
        }
        dataset.addExternalLink(link);
    }

    public void applySubjectMetadata(Dataset dataset, User user, String discipline, List<String> keywords, Map<String, String> fields) {
        if (!user.getId().equals(dataset.getOwnerId()) && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Not allowed to apply subject metadata");
        }
        SubjectMetadata sm = dataset.getSubjectMetadata();
        if (sm == null) {
            sm = new SubjectMetadata();
            dataset.setSubjectMetadata(sm);
        }
        metadataService.applySubjectMetadata(sm, discipline, keywords, fields);
    }

    public void setDmpLink(Dataset dataset, User user, String dmpLink) {
        if (!user.getId().equals(dataset.getOwnerId()) && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Not allowed to set DMP link");
        }
        dataset.setDmpLink(dmpLink);
    }

    public void setEmbargo(Dataset dataset, User user, LocalDate embargoUntil) {
        if (!user.getId().equals(dataset.getOwnerId()) && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Not allowed to set embargo");
        }
        DatasetVersion v = dataset.getCurrentVersion();
        v.setEmbargoUntil(embargoUntil);
        v.setAccessLevel(AccessLevel.EMBARGOED);
        v.setStatus(DatasetStatus.EMBARGOED);
    }

    public void recordDownload(Dataset dataset, User user, String country, String sector) {
        if (!canAccess(user, dataset)) {
            throw new SecurityException("No access to dataset");
        }
        statisticsService.recordDownload(dataset.getCurrentVersion(), country, sector);
    }

    public void recordCitation(Dataset dataset, String text, String source) {
        statisticsService.recordCitation(dataset.getCurrentVersion(), text, source);
    }

    public Map<String, Long> getDownloadStatsByCountry(Dataset dataset) {

        disposalService.approveDisposal(dataset, admin);

    public List<Dataset> queryArchive(User admin, Predicate<Dataset> predicate) {
        if (!admin.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Only archive administrator can query archive");
        }
        return adminService.queryArchive(new ArrayList<>(datasets.values()), predicate);
    }

    public Dataset importExternalDataset(User admin, String title, String ownerId, String externalUrl) {
        if (!admin.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Only archive administrator can import external data");
        }
        String id = "ext-" + (++datasetCounter);
        Dataset d = adminService.importExternalDataset(id, title, ownerId, externalUrl);
        datasets.put(id, d);
        return d;
    }

    public String exportAll(User user) {
        if (!user.hasRole(Role.UNIVERSITY_IT_SERVICE) && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {
            throw new SecurityException("Not allowed to export archive");
        }
        return exportService.exportAll(new ArrayList<>(datasets.values()));
    }

    public List<String> authenticateViaLdap(String username, String password) {
        return ldapIntegration.authenticate(username, password);
    }

    public String harvestMetadataForFunder(User user, String funder) {
        if (!user.hasRole(Role.FUNDING_BODY)
                && !user.hasRole(Role.RESEARCH_INFORMATION_MANAGER)
                && !user.hasRole(Role.ARCHIVE_ADMINISTRATOR)) {

        return crisIntegration.harvestMetadata(new ArrayList<>(datasets.values()), projects.values(), funder);
