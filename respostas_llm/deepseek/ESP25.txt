// File: PermissionManager.java
import java.util.EnumMap;
import java.util.EnumSet;
import java.util.HashMap;
import java.util.Map;
import java.util.Set;

public class PermissionManager {
    private final Map<Role, Set<Permission>> rolePermissions;
    private final Map<String, Set<Permission>> userPermissions;

    public PermissionManager() {
        this.rolePermissions = new EnumMap<>(Role.class);
        for (Role role : Role.values()) {
            rolePermissions.put(role, EnumSet.noneOf(Permission.class));
        }
        this.userPermissions = new HashMap<>();
    }

    public void grant(Role role, Permission permission) {
        rolePermissions.get(role).add(permission);
    }

    public void grantAll(Role role, Permission... permissions) {
        for (Permission permission : permissions) {
            rolePermissions.get(role).add(permission);
        }
    }

    public void revoke(Role role, Permission permission) {
        rolePermissions.get(role).remove(permission);
    }

    public void grant(String userId, Permission permission) {
        userPermissions.computeIfAbsent(userId, key -> EnumSet.noneOf(Permission.class)).add(permission);
    }

    public void revoke(String userId, Permission permission) {
        Set<Permission> permissions = userPermissions.get(userId);
        if (permissions != null) {
            permissions.remove(permission);
        }
    }

    public boolean can(Staff staff, Permission permission) {
        Set<Permission> permissions = userPermissions.get(staff.getId());
        if (permissions != null && permissions.contains(permission)) {
            return true;
        }
        return rolePermissions.getOrDefault(staff.getRole(), Set.of()).contains(permission);
    }
}

    }

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Set;

public class ReportTemplate {
    private final String name;
    private final List<String> baseFields;
    private final Map<String, String> baseFilters;
    private final Set<String> lockedFields;
    private final Set<String> lockedFilters;
    private final Set<String> allowedFields;
    private final Set<String> allowedFilters;
    private final boolean baseIncludeDeleted;

    public ReportTemplate(String name, List<String> baseFields, Map<String, String> baseFilters,
                          Set<String> lockedFields, Set<String> lockedFilters,
                          Set<String> allowedFields, Set<String> allowedFilters,
                          boolean baseIncludeDeleted) {
        this.name = name;
        this.baseFields = new ArrayList<>(baseFields);
        this.baseFilters = new LinkedHashMap<>(baseFilters);
        this.lockedFields = Set.copyOf(lockedFields);
        this.lockedFilters = Set.copyOf(lockedFilters);
        this.allowedFields = Set.copyOf(allowedFields);
        this.allowedFilters = Set.copyOf(allowedFilters);
        this.baseIncludeDeleted = baseIncludeDeleted;
    }

    public String getName() {
        return name;
    }

    public ReportConfig createConfig(List<String> requestedFields, Map<String, String> requestedFilters,
                                     boolean requestedIncludeDeleted) {
        List<String> fields = new ArrayList<>(baseFields);
        for (String field : requestedFields) {
            if (allowedFields.contains(field) && !lockedFields.contains(field) && !fields.contains(field)) {
                fields.add(field);
            }
        }

        Map<String, String> filters = new LinkedHashMap<>(baseFilters);
        for (Map.Entry<String, String> entry : requestedFilters.entrySet()) {
            String key = entry.getKey();
            if (lockedFilters.contains(key)) {
                continue;
            }
            if (allowedFilters.contains(key)) {
                filters.put(key, entry.getValue());
            }
        }

        boolean includeDeleted = baseIncludeDeleted;
        if (!lockedFilters.contains("deletedItems")) {
            includeDeleted = requestedIncludeDeleted;
        }

        return new ReportConfig(fields, filters, includeDeleted, "template:" + name);
    }

    public ReportTemplate cloneAs(String newName) {
        return new ReportTemplate(newName, baseFields, baseFilters, lockedFields, lockedFilters,
                allowedFields, allowedFilters, baseIncludeDeleted);
    }

    public Set<String> getLockedFilters() {
        return lockedFilters;
    }

    public Set<String> getLockedFields() {
        return lockedFields;
    }
}

import java.util.Collection;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public class ReportTemplateManager {
    private final Map<String, ReportTemplate> templates;

    public ReportTemplateManager() {
        this.templates = new LinkedHashMap<>();
    }

    public void createTemplate(Staff staff, PermissionManager permissionManager, ReportTemplate template) {
        if (!permissionManager.can(staff, Permission.CREATE_REPORT_TEMPLATE)) {
            throw new SecurityException("Staff cannot create report templates");
        }
        templates.put(template.getName(), template);
    }

    public void cloneTemplate(Staff staff, PermissionManager permissionManager, String name, String newName) {
        if (!permissionManager.can(staff, Permission.CLONE_REPORT_TEMPLATE)) {
            throw new SecurityException("Staff cannot clone report templates");
        }
        ReportTemplate template = templates.get(name);
        if (template == null) {
            throw new IllegalArgumentException("Unknown template");
        }
        templates.put(newName, template.cloneAs(newName));
    }

    public ReportTemplate get(String name) {
        return templates.get(name);
    }

    public Collection<ReportTemplate> listTemplates() {
        return templates.values();
    }

    public ReportConfig createConfig(String templateName, List<String> requestedFields,
                                     Map<String, String> requestedFilters, boolean requestedIncludeDeleted) {
        ReportTemplate template = templates.get(templateName);
        if (template == null) {
            throw new IllegalArgumentException("Unknown template");
        }
        return template.createConfig(requestedFields, requestedFilters, requestedIncludeDeleted);
    }
}

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;

public class ReportQueue {
    private final List<ReportJob> jobs;

    public ReportQueue() {
        this.jobs = new ArrayList<>();
    }

    public void submit(ReportJob job) {
        jobs.add(job);
    }

    public int getPosition(ReportJob job) {
        int position = 1;
        for (ReportJob current : jobs) {
            if (current.getStatus() == ReportStatus.QUEUED) {
                if (current == job) {
                    return position;
                }
                position++;
            }
        }
        return -1;
    }

    public ReportJob next() {
        for (ReportJob job : jobs) {
            if (job.getStatus() == ReportStatus.QUEUED) {
                job.setStatus(ReportStatus.RUNNING);
                job.setStartedAt(LocalDateTime.now());
                return job;
            }
        }
        return null;
    }

    public void complete(ReportJob job, ReportResult result) {
        job.setResult(result);
        job.setStatus(ReportStatus.COMPLETED);
        job.setCompletedAt(LocalDateTime.now());
    }

    public List<ReportJob> getJobs() {
        return new ArrayList<>(jobs);
    }
}

    }

    }

    public void addRow(String... values) {
        List<String> row = new ArrayList<>();
        for (String value : values) {
            row.add(value);
        }
        rows.add(row);
    }

    @Override
    public String toString() {
        StringBuilder builder = new StringBuilder();
        builder.append(title).append('\n');
        builder.append("Query selections: ").append(querySelections).append('\n');

        }
        return builder.toString();

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.regex.Pattern;

public class QueryBuilder {
    private final List<String> selectedFields;
    private final List<QueryCriterion> criteria;
    private final Map<String, List<String>> picklists;

    public QueryBuilder() {
        this.selectedFields = new ArrayList<>();
        this.criteria = new ArrayList<>();
        this.picklists = new LinkedHashMap<>();
    }

    public void selectField(String field) {
        selectedFields.add(field);
    }

    public void addCriterion(String field, QueryOperator operator, String value) {
        criteria.add(new QueryCriterion(field, operator, value));
    }

    public void addCriterion(String field, QueryOperator operator, String value, BooleanOperator booleanOperator) {
        criteria.add(new QueryCriterion(field, operator, value, booleanOperator));
    }

    public void addPicklistValue(String field, String value) {
        picklists.computeIfAbsent(field, key -> new ArrayList<>()).add(value);
    }

    public List<String> getPicklistValues(String field) {
        return new ArrayList<>(picklists.getOrDefault(field, List.of()));
    }

    public QueryResult execute(List<Map<String, String>> records) {
        List<Map<String, String>> results = new ArrayList<>();
        for (Map<String, String> record : records) {
            if (matches(record)) {
                results.add(new LinkedHashMap<>(record));
            }
        }
        return new QueryResult(selectedFields, criteria, results);
    }

    private boolean matches(Map<String, String> record) {
        boolean overall = true;
        boolean first = true;
        for (QueryCriterion criterion : criteria) {
            boolean current = matchesCriterion(record, criterion);
            if (first) {
                overall = current;
                first = false;
            } else if (criterion.getBooleanOperator() == BooleanOperator.AND) {
                overall = overall && current;
            } else {
                overall = overall || current;
            }
        }
        return overall;
    }

    private boolean matchesCriterion(Map<String, String> record, QueryCriterion criterion) {
        String actual = record.getOrDefault(criterion.getField(), "");
        String expected = criterion.getValue();
        switch (criterion.getOperator()) {
            case EQUALS:
                return actual.equals(expected);
            case NOT_EQUALS:
                return !actual.equals(expected);
            case CONTAINS:
                return actual.contains(expected);
            case STARTS_WITH:
                return actual.startsWith(expected);
            case REGEX:
                return Pattern.compile(expected).matcher(actual).find();
            case GREATER_THAN:
                return compareValues(actual, expected) > 0;
            case LESS_THAN:
                return compareValues(actual, expected) < 0;
            case GREATER_OR_EQUAL:
                return compareValues(actual, expected) >= 0;
            case LESS_OR_EQUAL:
                return compareValues(actual, expected) <= 0;
            default:
                return false;
        }
    }

    private int compareValues(String actual, String expected) {
        try {
            return Double.compare(Double.parseDouble(actual), Double.parseDouble(expected));
        } catch (NumberFormatException exception) {
            return actual.compareTo(expected);
        }
    }
}

        this.items = new ArrayList<>();
        this.originalLocations = new HashMap<>();
    }

    public void addItem(Item item) {
        items.add(item);
    }

    public List<Item> getItems() {
        return new ArrayList<>(items);
    }

    public List<Item> findByBranch(LibraryBranch branch) {
        List<Item> result = new ArrayList<>();
        for (Item item : items) {
            if (item.getBranch() != null && item.getBranch().getId().equals(branch.getId())) {
                result.add(item);
            }
        }
        return result;
    }

    public List<Item> query(ItemStatus status, String genre, LibraryBranch branch) {
        List<Item> result = new ArrayList<>();
        for (Item item : items) {
            if (status != null && item.getStatus() != status) {
                continue;
            }
            if (genre != null && !genre.equals(item.getGenre())) {
                continue;
            }
            if (branch != null && (item.getBranch() == null || !item.getBranch().getId().equals(branch.getId()))) {
                continue;
            }
            result.add(item);
        }
        return result;
    }

    public List<Item> queryCandidates(ItemStatus status, String genre) {
        return query(status, genre, null);
    }

    public void transferBatch(List<Item> batch, LibraryBranch newBranch) {
        for (Item item : batch) {
            originalLocations.putIfAbsent(item.getId(), item.getBranch());
            item.setBranch(newBranch);
        }
    }

    public void revertItem(Item item) {
        LibraryBranch original = originalLocations.get(item.getId());
        if (original != null) {
            item.setBranch(original);
        }
    }

    public List<Item> purgeCandidates(int missingMonths, LocalDate now) {
        List<Item> result = new ArrayList<>();
        LocalDate cutoff = now.minusMonths(missingMonths);
        for (Item item : items) {
            if (item.getStatus() == ItemStatus.MISSING || item.getStatus() == ItemStatus.DISCARD_WEED) {
                if (item.getStatusChangedDate() == null || item.getStatusChangedDate().isBefore(cutoff)) {
                    result.add(item);
                }
            }
        }
        return result;
    }

    public int deleteItems(List<Item> batch) {
        int count = 0;
        for (Item item : batch) {
            if (items.remove(item)) {
                count++;
            }
        }
        return count;
    }

    public Map<String, Integer> volumeReport(LibraryBranch branch, boolean includeOut) {
        Map<String, Integer> counts = new LinkedHashMap<>();
        for (Item item : items) {
            if (branch != null && (item.getBranch() == null || !item.getBranch().getId().equals(branch.getId()))) {
                continue;
            }
            if (!includeOut && item.getStatus() == ItemStatus.CHECKED_OUT) {
                continue;
            }
            String category = item.getGenre() + " / " + item.getFormat() + " / " + item.getStatus();
            counts.merge(category, 1, Integer::sum);

    private final List<LedgerEntry> ledgerEntries;
    private final Map<String, BigDecimal> waivers;
    private final Map<String, BigDecimal> waiversByBranch;
    private int nextLedgerId;

    public FinancialService() {
        this.charges = new ArrayList<>();
        this.payments = new ArrayList<>();
        this.ledgerEntries = new ArrayList<>();
        this.waivers = new LinkedHashMap<>();
        this.waiversByBranch = new LinkedHashMap<>();
        this.nextLedgerId = 1;
    }

    public void addCharge(Charge charge) {
        charges.add(charge);
    }

    public void addPayment(Payment payment, Map<String, BigDecimal> allocations) {
        payments.add(payment);
        for (Map.Entry<String, BigDecimal> allocation : allocations.entrySet()) {
            ledgerEntries.add(new LedgerEntry("L" + nextLedgerId++, payment.getPatronId(),
                    allocation.getKey(), payment.getId(), allocation.getValue(),
                    "Payment allocation", payment.getDate()));
        }
    }

    public void waive(String patronId, BigDecimal amount, LibraryBranch branch) {
        waivers.merge(patronId, amount, BigDecimal::add);
        waiversByBranch.merge(branch.getName(), amount, BigDecimal::add);
        ledgerEntries.add(new LedgerEntry("L" + nextLedgerId++, patronId, null, null,
                amount.negate(), "Waiver", LocalDate.now()));
    }

    public BigDecimal getBalance(String patronId) {
        BigDecimal balance = BigDecimal.ZERO;
        for (Charge charge : charges) {
            if (charge.getPatronId().equals(patronId)) {
                balance = balance.add(charge.getAmount());
            }
        }
        for (Payment payment : payments) {
            if (payment.getPatronId().equals(patronId)) {
                balance = balance.subtract(payment.getAmount());
            }
        }
        balance = balance.subtract(waivers.getOrDefault(patronId, BigDecimal.ZERO));
        return balance;
    }

    public List<String> getPatronsWithBalanceGreaterThan(BigDecimal threshold, PatronService patronService) {
        List<String> result = new ArrayList<>();
        for (Patron patron : patronService.getAllPatrons()) {
            if (getBalance(patron.getId()).compareTo(threshold) > 0) {
                result.add(patron.getName() + " (" + patron.getId() + ")");
            }
        }
        return result;
    }

    public Map<String, BigDecimal> balancesBy(PatronService patronService, Function<Patron, String> classifier) {
        Map<String, BigDecimal> result = new LinkedHashMap<>();

            if (key == null) {

        }
        return result;
    }

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.function.Function;

public class PatronService {
    private final Map<String, Patron> patrons;

    public PatronService() {
        this.patrons = new LinkedHashMap<>();
    }

    public void addPatron(Patron patron) {
        patrons.put(patron.getId(), patron);
    }

    public Patron getPatron(String id) {
        return patrons.get(id);
    }

    public List<Patron> getAllPatrons() {
        return new ArrayList<>(patrons.values());
    }

    public List<Patron> inactivePatrons(int days, LocalDate now, LibraryBranch branch) {
        List<Patron> result = new ArrayList<>();
        LocalDate cutoff = now.minusDays(days);
        for (Patron patron : patrons.values()) {
            if (branch != null && (patron.getHomeBranch() == null
                    || !patron.getHomeBranch().getId().equals(branch.getId()))) {
                continue;
            }
            if (patron.getLastActivityDate() == null || patron.getLastActivityDate().isBefore(cutoff)) {
                result.add(patron);
            }
        }
        return result;
    }

    public Map<String, Long> groupBy(Function<Patron, String> classifier) {
        Map<String, Long> result = new LinkedHashMap<>();
        for (Patron patron : patrons.values()) {
            String key = classifier.apply(patron);
            if (key == null) {
                key = "Unknown";
            }
            result.merge(key, 1L, Long::sum);
        }
        return result;
    }

    public boolean deletePatron(String id) {
        return patrons.remove(id) != null;
    }
}

import java.math.BigDecimal;

    public ReportResult boardReport() {
        ReportResult result = new ReportResult("Board Report",
                List.of("Metric", "Count"), Map.of("period", "current"));
        result.addRow("Check-outs", String.valueOf(transactionService.count(TransactionType.CHECK_OUT)));
        result.addRow("Check-ins", String.valueOf(transactionService.count(TransactionType.CHECK_IN)));
        result.addRow("Holds placed", String.valueOf(transactionService.count(TransactionType.HOLD_PLACED)));
        result.addRow("Holds filled", String.valueOf(transactionService.count(TransactionType.HOLD_FILLED)));
        result.addRow("Patron registrations", String.valueOf(patronService.getAllPatrons().size()));
        return result;
    }

    public ReportResult boardReportBy(String dimension) {
        Function<Transaction, String> classifier = transaction -> {
            if ("terminal".equals(dimension)) {
                return transaction.getTerminalId();
            }
            if ("branch".equals(dimension)) {
                return transaction.getBranch() == null ? "Unknown" : transaction.getBranch().getName();
            }
            if ("system".equals(dimension)) {
                return transaction.getBranch() == null ? "Unknown" : transaction.getBranch().getSystemName();
            }
            if ("hour".equals(dimension)) {
                return String.valueOf(transaction.getTimestamp().getHour());
            }
            if ("patronAge".equals(dimension)) {
                return transaction.getPatronAgeRange();
            }
            if ("patronType".equals(dimension)) {
                return transaction.getPatronType() == null ? "Unknown" : transaction.getPatronType().name();
            }
            if ("patronLocation".equals(dimension)) {
                return transaction.getPatronLocation();
            }
            return "Unknown";
        };
        Map<String, Long> counts = transactionService.countBy(classifier);
        ReportResult result = new ReportResult("Board Report by " + dimension,
                List.of(dimension, "Count"), Map.of("dimension", dimension));
        for (Map.Entry<String, Long> entry : counts.entrySet()) {
            result.addRow(entry.getKey(), String.valueOf(entry.getValue()));
        }
        return result;
    }

    public ReportResult periodicReport(PredefinedReportType type, int xDays) {
        if (type == PredefinedReportType.PATRONS_WITH_INVALID_HOME_LIBRARY) {
            ReportResult result = new ReportResult("Periodic Report: " + type,
                    List.of("Patron ID", "Name", "Home Branch"), Map.of("type", type.name()));
            for (Patron patron : patronService.getAllPatrons()) {
                if (patron.getHomeBranch() == null) {
                    result.addRow(patron.getId(), patron.getName(), "Invalid");
                }
            }
            return result;
        }

        List<Item> selected = new ArrayList<>();
        LocalDate now = LocalDate.now();
        switch (type) {
            case ITEMS_NOT_CHECKED_OUT_IN_X_DAYS:
                for (Item item : itemService.getItems()) {
                    if (item.getLastCheckedOut() == null || item.getLastCheckedOut().isBefore(now.minusDays(xDays))) {
                        selected.add(item);
                    }
                }
                break;
            case IN_TRANSIT_ITEMS_MORE_THAN_X_DAYS:
                selected.addAll(itemService.inTransitMoreThan(xDays, now));
                break;
            case LAST_COPY_IN_SYSTEM:
                selected.addAll(itemService.lastCopies());
                break;
            case ITEMS_WITH_LONG_CALL_NUMBER:
                for (Item item : itemService.getItems()) {
                    if (item.getCallNumber() != null && item.getCallNumber().length() > 20) {
                        selected.add(item);
                    }
                }
                break;
            case MISSING_ITEMS:
                selected.addAll(itemService.query(ItemStatus.MISSING, null, null));
                break;
            default:
                selected.addAll(itemService.getItems());
                break;
        }

        ReportResult result = new ReportResult("Periodic Report: " + type,
                List.of("Item ID", "Title", "Status", "Branch"), Map.of("type", type.name()));
        for (Item item : selected) {
            result.addRow(item.getId(), item.getTitle(), item.getStatus().name(),
                    item.getBranch() == null ? "Unknown" : item.getBranch().getName());
        }
        return result;
    }

    public ReportResult inventoryVolumeReport(LibraryBranch branch, boolean includeOut) {

    public ReportResult capacityReport(List<LibraryBranch> branches) {
        ReportResult result = new ReportResult("System Capacity Report",
                List.of("Branch", "Total Items", "Capacity", "Usage %"), Map.of());
        for (LibraryBranch branch : branches) {
            int total = itemService.totalItems(branch);
            double usage = branch.getCapacity() == 0 ? 0.0 : (100.0 * total / branch.getCapacity());
            result.addRow(branch.getName(), String.valueOf(total),
                    String.valueOf(branch.getCapacity()),
                    String.format(Locale.US, "%.2f", usage));
        }
        return result;
    }

    public ReportResult shelfSpaceReport(LibraryBranch branch) {
        Map<String, Integer> collection = new LinkedHashMap<>();
        Map<String, Integer> circulation = new LinkedHashMap<>();
        Map<String, Double> shelfSpace = new LinkedHashMap<>();
        int totalCollection = 0;
        int totalCirculation = 0;
        double totalShelfSpace = 0.0;

        for (Item item : itemService.getItems()) {
            if (branch != null && (item.getBranch() == null
                    || !item.getBranch().getId().equals(branch.getId()))) {
                continue;
            }
            String key = item.getGenre() + " / " + item.getFormat();
            collection.merge(key, 1, Integer::sum);
            circulation.merge(key, item.getCheckOuts(), Integer::sum);
            shelfSpace.merge(key, item.getShelfSpaceUnits(), Double::sum);
            totalCollection++;
            totalCirculation += item.getCheckOuts();
            totalShelfSpace += item.getShelfSpaceUnits();
        }

        ReportResult result = new ReportResult("Shelf Space Report",
                List.of("Genre / Format", "Collection %", "Circulation %", "Shelf Space %"), Map.of());
        for (String key : collection.keySet()) {
            double collectionPercent = totalCollection == 0 ? 0.0
                    : 100.0 * collection.get(key) / totalCollection;
            double circulationPercent = totalCirculation == 0 ? 0.0
                    : 100.0 * circulation.getOrDefault(key, 0) / totalCirculation;
            double shelfPercent = totalShelfSpace == 0.0 ? 0.0
                    : 100.0 * shelfSpace.getOrDefault(key, 0.0) / totalShelfSpace;
            result.addRow(key,
                    String.format(Locale.US, "%.2f", collectionPercent),
                    String.format(Locale.US, "%.2f", circulationPercent),
                    String.format(Locale.US, "%.2f", shelfPercent));
        }
        return result;
    }

    public ReportResult inTransitReport(int days) {
        List<Item> items = itemService.inTransitMoreThan(days, LocalDate.now());
        ReportResult result = new ReportResult("In-Transit Items Report",
                List.of("Item ID", "Title", "From", "To", "Days In Transit"), Map.of("days", String.valueOf(days)));
        for (Item item : items) {
            long daysInTransit = item.getInTransitSince() == null ? 0
                    : java.time.temporal.ChronoUnit.DAYS.between(item.getInTransitSince(), LocalDate.now());
            result.addRow(item.getId(), item.getTitle(),
                    item.getTransitFrom() == null ? "Unknown" : item.getTransitFrom().getName(),
                    item.getTransitTo() == null ? "Unknown" : item.getTransitTo().getName(),
                    String.valueOf(daysInTransit));
        }
        return result;
    }

    public ReportResult itemValueReport(ItemStatus status, String genre) {
        BigDecimal total = BigDecimal.ZERO;
        ReportResult result = new ReportResult("Value of Items Report",
                List.of("Item ID", "Title", "Status", "Genre", "Value"), Map.of());
        for (Item item : itemService.getItems()) {
            if (status != null && item.getStatus() != status) {
                continue;
            }
            if (genre != null && !genre.equals(item.getGenre())) {
                continue;
            }
            total = total.add(item.getValue());
            result.addRow(item.getId(), item.getTitle(), item.getStatus().name(),
                    item.getGenre(), item.getValue().toPlainString());
        }
        result.addRow("TOTAL", "", "", "", total.toPlainString());
        return result;
    }

    public ReportResult financialReport() {
        ReportResult result = new ReportResult("Financial Report",

            result.addRow("Fine - " + entry.getKey(), entry.getValue().toPlainString());

            result.addRow("Waiver - " + entry.getKey(), entry.getValue().toPlainString());
        }
        for (Map.Entry<String, BigDecimal> entry : financialService.paymentsByMethod().entrySet()) {
            result.addRow("Payment - " + entry.getKey(), entry.getValue().toPlainString());
        }
        return result;
    }

    public ReportResult behaviorAnalysis() {
        long checkouts = transactionService.count(TransactionType.CHECK_OUT);
        long renewals = transactionService.count(TransactionType.RENEWAL);
        double renewalPercent = checkouts == 0 ? 0.0 : 100.0 * renewals / checkouts;
        ReportResult result = new ReportResult("Patron Behavior and Use Analysis",
                List.of("Metric", "Value"), Map.of());
        result.addRow("Check-outs", String.valueOf(checkouts));
        result.addRow("Renewals", String.valueOf(renewals));
        result.addRow("Renewal % of check-outs", String.format(Locale.US, "%.2f", renewalPercent));
        long holdPlaced = transactionService.count(TransactionType.HOLD_PLACED);
        long holdFilled = transactionService.count(TransactionType.HOLD_FILLED);
        result.addRow("Holds placed", String.valueOf(holdPlaced));
        result.addRow("Holds filled", String.valueOf(holdFilled));
        return result;
    }

    public ReportResult demographicStatistics() {
        ReportResult result = new ReportResult("Demographic Statistics",

        addDemographicRows(result, "Zip Code", patronService.groupBy(Patron::getZipCode));

        addDemographicRows(result, "Home Branch", patronService.groupBy(

        addDemographicRows(result, "Patron Type", patronService.groupBy(
                patron -> patron.getPatronType() == null ? "Unknown" : patron.getPatronType().name()));
        addDemographicRows(result, "Preferred Language", patronService.groupBy(Patron::getPreferredLanguage));
        return result;

    private void addDemographicRows(ReportResult result, String dimension, Map<String, Long> counts) {
        for (Map.Entry<String, Long> entry : counts.entrySet()) {
            result.addRow(dimension, entry.getKey(), String.valueOf(entry.getValue()));
        }
    }

    public ReportResult deletionReport() {

        }
