public class FileAsset {
    private final String id;
    private String name;
    private String mimeType;
    private long sizeBytes;
    private byte[] content;
    private String doi;
    private String checksum;
    private List<StorageCopy> copies = new ArrayList<>();

    public FileAsset(String name, String mimeType, long sizeBytes, byte[] content) {
        this.id = UUID.randomUUID().toString();
        this.name = name;
        this.mimeType = mimeType;
        this.sizeBytes = sizeBytes;
        this.content = content == null ? new byte[0] : content.clone();
        this.checksum = computeChecksum(this.content);
        addCopy("primary");
        addCopy("redundant");
    }

    private String computeChecksum(byte[] data) {
        return Integer.toHexString(Arrays.hashCode(data));
    }

    public void addCopy(String location) {
        copies.add(new StorageCopy(location, checksum, false, true));
    }

    public boolean verifyCopies() {
        for (StorageCopy copy : copies) {
            if (!copy.verify(checksum)) {
                return false;
            }
        }
        return true;
    }

    public boolean isLargeFile() {
        return sizeBytes > 1024L * 1024L * 1024L;
    }

    public String getId() {

import java.time.LocalDate;

public class SearchQuery {
    private String keywords = "";
    private String title = "";
    private String author = "";
    private String subject = "";
    private LocalDate fromDate;
    private LocalDate toDate;
    private RepositoryCollection withinCollection;

    public boolean matches(Item item) {
        String keyword = keywords == null ? "" : keywords.trim().toLowerCase();
        if (!title.isEmpty() && !item.getTitle().toLowerCase().contains(title.toLowerCase())) {
            return false;
        }
        if (!author.isEmpty() && item.getAuthors().stream().noneMatch(a -> a.toLowerCase().contains(author.toLowerCase()))) {
            return false;
        }
        if (!subject.isEmpty() && item.getSubjects().stream().noneMatch(s ->
                s.getName().toLowerCase().contains(subject.toLowerCase()) ||
                s.getCode().equalsIgnoreCase(subject))) {
            return false;
        }
        if (!keyword.isEmpty()) {
            boolean found = item.getTitle().toLowerCase().contains(keyword)
                    || item.getDescription().toLowerCase().contains(keyword)
                    || item.getKeywords().stream().anyMatch(k -> k.toLowerCase().contains(keyword))
                    || item.getAuthors().stream().anyMatch(a -> a.toLowerCase().contains(keyword))
                    || item.getMetadata().values().stream().anyMatch(v -> v.toLowerCase().contains(keyword));
            if (!found) {
                return false;
            }
        }
        if (fromDate != null && (item.getPublicationDate() == null || item.getPublicationDate().isBefore(fromDate))) {
            return false;
        }
        if (toDate != null && (item.getPublicationDate() == null || item.getPublicationDate().isAfter(toDate))) {
            return false;
        }
        if (withinCollection != null && !item.getCollections().contains(withinCollection)) {
            return false;
        }
        return true;
    }

    public String getKeywords() {
        return keywords;
    }

    public void setKeywords(String keywords) {
        this.keywords = keywords == null ? "" : keywords;
    }

    public String getTitle() {
        return title;
    }

    public void setTitle(String title) {
        this.title = title == null ? "" : title;
    }

    public String getAuthor() {
        return author;
    }

    public void setAuthor(String author) {
        this.author = author == null ? "" : author;
    }

    public String getSubject() {
        return subject;
    }

    public void setSubject(String subject) {
        this.subject = subject == null ? "" : subject;
    }

    public LocalDate getFromDate() {
        return fromDate;
    }

    public void setFromDate(LocalDate fromDate) {
        this.fromDate = fromDate;

        return toDate;

public class RepositoryCollection implements Searchable {
    private final String id;
    private String name;
    private String description;
    private List<String> previousNames = new ArrayList<>();
    private AccessLevel accessLevel;
    private String branding;
    private List<Item> items = new ArrayList<>();
    private Map<String, String> customMetadataFields = new LinkedHashMap<>();
    private SortOrder sortOrder = SortOrder.TITLE;
    private String customSortKey;
    private RepositoryCollection parent;
    private String ownerId;

    public RepositoryCollection(String name, String description, AccessLevel accessLevel) {
        this.id = UUID.randomUUID().toString();
        this.name = name;
        this.description = description == null ? "" : description;
        this.accessLevel = accessLevel;
    }

    public void addItem(Item item) {
        if (!items.contains(item)) {
            items.add(item);
            if (!item.getCollections().contains(this)) {
                item.getCollections().add(this);
            }
        }
    }

    public void removeItem(Item item) {
        items.remove(item);
        item.getCollections().remove(this);
    }

    public void rename(String newName) {
        if (!name.equals(newName)) {
            previousNames.add(name);
            name = newName;
        }
    }

    public void moveTo(RepositoryCollection newParent) {
        this.parent = newParent;
    }

    public List<Item> getItemsSorted() {
        List<Item> sorted = new ArrayList<>(items);
        switch (sortOrder) {
            case TITLE:
                sorted.sort(Comparator.comparing(Item::getTitle, String.CASE_INSENSITIVE_ORDER));
                break;
            case DATE:
                sorted.sort((a, b) -> {
                    LocalDate da = a.getPublicationDate();
                    LocalDate db = b.getPublicationDate();
                    if (da == null && db == null) return 0;
                    if (da == null) return 1;
                    if (db == null) return -1;
                    return da.compareTo(db);
                });
                break;
            case AUTHOR:
                sorted.sort(Comparator.comparing(i -> i.getAuthors().isEmpty() ? "" : i.getAuthors().get(0),
                        String.CASE_INSENSITIVE_ORDER));
                break;
            case CUSTOM:
                sorted.sort(Comparator.comparing(i -> i.getCustomSortKey() == null ? "" : i.getCustomSortKey(),
                        String.CASE_INSENSITIVE_ORDER));
                break;
        }
        return sorted;
    }

    @Override
    public boolean matches(SearchQuery query) {
        String keyword = query.getKeywords() == null ? "" : query.getKeywords().toLowerCase();
        if (!keyword.isEmpty() && name.toLowerCase().contains(keyword)) {
            return true;
        }
        for (Item item : items) {
            if (query.matches(item)) {
                return true;
            }
        }
        return false;
    }

public class Subscription {
    private final String id;
    private String userId;
    private SubscriptionTargetType targetType;
    private String targetId;
    private boolean active = true;
    private boolean emailEnabled = true;

    public Subscription(String userId, SubscriptionTargetType targetType, String targetId) {
        this.id = java.util.UUID.randomUUID().toString();
        this.userId = userId;
        this.targetType = targetType;
        this.targetId = targetId;
    }

    public boolean matches(Item item) {
        if (!active) {
            return false;
        }
        switch (targetType) {
            case ITEM:
                return targetId.equals(item.getId());
            case COLLECTION:
                return item.getCollections().stream().anyMatch(c -> c.getId().equals(targetId));
            case AUTHOR:
                return item.getAuthors().stream().anyMatch(a -> a.equalsIgnoreCase(targetId));
            case TOPIC:
                return item.getKeywords().stream().anyMatch(k -> k.equalsIgnoreCase(targetId))
                        || item.getSubjects().stream().anyMatch(s -> s.getName().equalsIgnoreCase(targetId));
            default:
                return false;
        }
    }

    public String getId() {
        return id;
    }

        return userId;
    }

    public SubscriptionTargetType getTargetType() {
        return targetType;
    }

    public String getTargetId() {
        return targetId;
    }

        return active;
    }

        this.active = active;
    }

        return emailEnabled;
    }

import java.util.HashMap;

public class UsageStatistics {
    private Map<String, Integer> itemDownloads = new HashMap<>();
    private Map<String, Integer> fileDownloads = new HashMap<>();
    private Map<String, Integer> collectionViews = new HashMap<>();
    private Map<String, Integer> authorDownloads = new HashMap<>();

    public void recordDownload(Item item, FileAsset file) {
        itemDownloads.merge(item.getId(), 1, Integer::sum);
        if (file != null) {
            fileDownloads.merge(file.getId(), 1, Integer::sum);
        }
        for (String author : item.getAuthors()) {
            authorDownloads.merge(author, 1, Integer::sum);
        }
    }

    public void recordCollectionView(RepositoryCollection collection) {
        collectionViews.merge(collection.getId(), 1, Integer::sum);
    }

    public int getItemDownloads(String itemId) {
        return itemDownloads.getOrDefault(itemId, 0);
    }

    public int getFileDownloads(String fileId) {
        return fileDownloads.getOrDefault(fileId, 0);
    }

    public int getCollectionViews(String collectionId) {
        return collectionViews.getOrDefault(collectionId, 0);
    }

    public int getAuthorDownloads(String author) {
        return authorDownloads.getOrDefault(author, 0);
    }

    public Map<String, Integer> getAltmetrics(String itemId) {
        Map<String, Integer> metrics = new HashMap<>();
        metrics.put("downloads", getItemDownloads(itemId));
        metrics.put("shares", 0);
        metrics.put("views", 0);
        return metrics;
    }

    public String digest() {
        return "Item downloads=" + itemDownloads
                + ", File downloads=" + fileDownloads
                + ", Collection views=" + collectionViews
                + ", Author downloads=" + authorDownloads;
    }
}
