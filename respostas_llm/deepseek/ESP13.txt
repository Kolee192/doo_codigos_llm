import java.time.LocalDate;

public class UsePermission {
    private String act;
    private LocalDate startDate;
    private LocalDate endDate;

    public UsePermission(String act, LocalDate startDate, LocalDate endDate) {
        this.act = act;
        this.startDate = startDate;
        this.endDate = endDate;
    }

    public String getAct() { return act; }
    public LocalDate getStartDate() { return startDate; }
    public LocalDate getEndDate() { return endDate; }

    public boolean isActive(LocalDate date) {
        if (date == null) date = LocalDate.now();
        return (startDate == null || !date.isBefore(startDate)) &&
               (endDate == null || !date.isAfter(endDate));
    }

    @Override
    public String toString() {
        return act + " [" + startDate + " to " + endDate + "]";
    }
}

        List<Collection> result = new ArrayList<>();

            if (collection.getCurator() != null && collection.getCurator().equals(user)) {
                result.add(collection);
            }
        }
        return result;
    }

    @Override
    public String toString() {
        return name + " (" + id + ")";
    }
}

// File: AccessControl.java
public class AccessControl {
    public static boolean canCurate(User user, Repository repository) {
        if (user == null || repository == null) return false;
        if (user.getRole() == Role.DAMS_MANAGER) return true;
        return (user.getRole() == Role.STAFF || user.getRole() == Role.REPOSITORY_MANAGER)
                && repository.hasStaff(user);
    }

    public static boolean canView(User user, Repository repository) {
        if (user == null || repository == null) return false;
        if (user.getRole() == Role.DAMS_MANAGER) return true;
        if (user.getRole() == Role.STAFF || user.getRole() == Role.REPOSITORY_MANAGER) {
            return repository.hasStaff(user);
        }
        return true;
    }
}

import java.util.LinkedHashSet;
import java.util.Set;

public class NameRecord {
    private String id;
    private String uniqueIdentity;
    private String preferredName;
    private Set<String> variantNames = new LinkedHashSet<>();
    private String dates;
    private String description;

    public NameRecord(String id, String uniqueIdentity, String preferredName, String dates, String description) {
        this.id = id;
        this.uniqueIdentity = uniqueIdentity;
        this.preferredName = preferredName;
        this.dates = dates;
        this.description = description;
    }

    public String getId() { return id; }
    public String getUniqueIdentity() { return uniqueIdentity; }
    public String getPreferredName() { return preferredName; }
    public Set<String> getVariantNames() { return variantNames; }
    public String getDates() { return dates; }
    public String getDescription() { return description; }

    public void addVariantName(String name) { variantNames.add(name); }

    public boolean containsName(String text) {
        if (text == null) return false;
        String lower = text.toLowerCase();
        if (preferredName != null && preferredName.toLowerCase().contains(lower)) return true;
        for (String variant : variantNames) {
            if (variant != null && variant.toLowerCase().contains(lower)) return true;
        }
        return false;
    }

    @Override
    public String toString() {
        return preferredName + " (" + uniqueIdentity + ")";
    }
}

        this.lastRegisteredChecksum = checksum;
        this.checksumMethod = method;
    }

    public boolean updateChecksum(String newChecksum, ChecksumMethod method, Event event) {
        boolean changed = this.checksum != null && !this.checksum.equals(newChecksum);
        this.checksum = newChecksum;
        this.checksumMethod = method;
        if (changed) {
            events.add(event);
        }
        return changed;
    }

    public boolean hasChecksumChanged() {
        return checksum != null && lastRegisteredChecksum != null && !checksum.equals(lastRegisteredChecksum);
    }

        return "size=" + sizeBytes + "B, created=" + creationDate + ", format=" + format + "/" + formatVersion +
                ", checksum=" + checksum + " (" + checksumMethod + ")" +
                ", sourceCapture=" + sourceCapture + ", hardware=" + hardware + ", software=" + software;
    }

    @Override
    public String toString() {
        return filename + " [" + id + "] (" + format + ")";
    }
}

    public List<Component> getOrderedDescendants() {

        List<Component> sorted = new ArrayList<>(children);
        sorted.sort((a, b) -> Integer.compare(a.getOrder(), b.getOrder()));
        for (Component child : sorted) {
            result.add(child);
            result.addAll(child.getOrderedDescendants());
        }
        return result;
    }

    public String getHierarchyString() {
        StringBuilder sb = new StringBuilder();
        buildHierarchyString(sb, 0);
        return sb.toString();
    }

    private void buildHierarchyString(StringBuilder sb, int depth) {
        sb.append("  ".repeat(depth)).append(order).append(". ").append(title).append("\n");
        List<Component> sorted = new ArrayList<>(children);
        sorted.sort((a, b) -> Integer.compare(a.getOrder(), b.getOrder()));
        for (Component child : sorted) {
            child.buildHierarchyString(sb, depth + 1);
        }
    }

    public boolean hasName(String text) {
        for (Set<NameRecord> records : names.values()) {
            for (NameRecord record : records) {
                if (record.containsName(text)) return true;
            }
        }
        return false;
    }

    public boolean hasRole(NameRole role) {
        return names.containsKey(role) && !names.get(role).isEmpty();
    }

    @Override
    public String toString() {
        return title + " [" + id + "]";
    }
