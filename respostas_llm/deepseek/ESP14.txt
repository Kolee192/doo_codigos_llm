import java.util.LinkedHashMap;

public class UserManager {
    private final Map<String, User> users = new LinkedHashMap<>();
    private final Map<String, Role> roles = new LinkedHashMap<>();
    private final Map<String, UserGroup> groups = new LinkedHashMap<>();

    public User createUser(String username, String activeDirectoryId) {
        User user = new User(username, activeDirectoryId);
        users.put(username, user);
        return user;
    }

    public Role createRole(String name, Permission... permissions) {
        Role role = roles.get(name);
        if (role == null) {
            role = new Role(name);
            roles.put(name, role);
        }
        for (Permission permission : permissions) {
            role.addPermission(permission);
        }
        return role;
    }

    public UserGroup createUserGroup(String name, Permission... permissions) {
        UserGroup group = groups.get(name);
        if (group == null) {
            group = new UserGroup(name);
            groups.put(name, group);
        }
        for (Permission permission : permissions) {
            group.addPermission(permission);
        }
        return group;
    }

    public void assignRoleToUser(String username, String roleName) {
        User user = users.get(username);
        Role role = roles.get(roleName);
        if (user == null || role == null) {
            throw new IllegalArgumentException("Unknown user or role");
        }
        user.addRole(role);
    }

    public void assignUserToGroup(String username, String groupName) {
        User user = users.get(username);
        UserGroup group = groups.get(groupName);
        if (user == null || group == null) {
            throw new IllegalArgumentException("Unknown user or group");
        }
        user.addGroup(group);
        group.addMember(user);
    }

    public void assignPermissionToGroup(String groupName, Permission permission) {
        UserGroup group = groups.get(groupName);
        if (group == null) {
            throw new IllegalArgumentException("Unknown group");
        }
        group.addPermission(permission);
    }

    public void assignPermissionToRole(String roleName, Permission permission) {
        Role role = roles.get(roleName);
        if (role == null) {
            throw new IllegalArgumentException("Unknown role");
        }
        role.addPermission(permission);
    }

    public boolean authenticateActiveDirectory(String username, String adCredential) {
        User user = users.get(username);
        if (user == null || user.getActiveDirectoryId() == null) return false;
        return adCredential.equals("AD-" + user.getActiveDirectoryId());
    }

    public boolean resetPassword(String username, String newPassword) {

    }

    public User getUser(String username) {

// File: IpRange.java
public class IpRange {
    private final long start;
    private final long end;

    public IpRange(String startIp, String endIp) {
        this.start = parse(startIp);
        this.end = parse(endIp);
    }

    public boolean allows(String ip) {
        long value = parse(ip);
        return value >= start && value <= end;
    }

    private static long parse(String ip) {
        if (ip == null) {
            throw new IllegalArgumentException("IP address is null");
        }
        String[] parts = ip.split("\\.");
        if (parts.length != 4) {
            throw new IllegalArgumentException("Invalid IPv4 address: " + ip);
        }
        long result = 0;
        for (String part : parts) {
            int value = Integer.parseInt(part);
            if (value < 0 || value > 255) {
                throw new IllegalArgumentException("Invalid IPv4 address: " + ip);
            }
            result = (result << 8) | value;
        }
        return result;
    }
}

    public RightsStatement createStatement(RightsStatus status, String statement, String licenseName, String rightsUri, String copyrightHolder) {

        RightsStatement rightsStatement = new RightsStatement(id, status, statement, licenseName, rightsUri, copyrightHolder);
        statements.put(id, rightsStatement);

    public void applyRights(DigitalFile file, String statementId) {

            if (file.getRightsStatement() != null && file.getRightsStatement().getStatus() == status) {
                result.add(file);
            }
        }
        return result;
    }

            return AccessDecision.allow("Administrator rights");
        }
        Restriction restriction = file.getRestriction();
        if (restriction == null || !restriction.isViewRestricted()) {
            return AccessDecision.allow("Not view restricted");
        }
        LocalDate now = LocalDate.now();
        if (!restriction.isActiveOn(now)) {
            return AccessDecision.allow("Restriction expired");
        }
        if (user != null && (restriction.allowsUser(user.getUsername()) || user.hasPermission(Permission.VIEW_RESTRICTED))) {
            return AccessDecision.allow("User allowed");
        }
        if (ip != null && restriction.isIpAllowed(ip)) {
            return AccessDecision.allow("IP allowed");
        }
        return AccessDecision.deny("View restricted");
    }

    public AccessDecision canDownload(DigitalFile file, User user, String ip) {
        AccessDecision viewDecision = canView(file, user, ip);
        if (!viewDecision.isAllowed()) {
            return viewDecision;
        }
        Restriction restriction = file.getRestriction();
        if (restriction == null) {
            return AccessDecision.allow("No restriction");
        }
        LocalDate now = LocalDate.now();
        if (restriction.isAccessRestricted() && restriction.isActiveOn(now)) {
            if (user != null && (restriction.allowsUser(user.getUsername()) || user.hasPermission(Permission.VIEW_RESTRICTED))) {
                return AccessDecision.allow("Access allowed by user");
            }
            if (ip != null && restriction.isIpAllowed(ip)) {
                return AccessDecision.allow("Access allowed by IP");
            }
            return AccessDecision.deny("Access restricted");
        }
        if (restriction.isShareRestricted() && restriction.isActiveOn(now)) {
            if (user != null && (restriction.allowsUser(user.getUsername()) || user.hasPermission(Permission.SHARE) || user.hasPermission(Permission.VIEW_RESTRICTED))) {
                return AccessDecision.allow("Sharing allowed by user");
            }
            if (ip != null && restriction.isIpAllowed(ip)) {
                return AccessDecision.allow("Sharing allowed by IP");
            }
            return AccessDecision.deny("Sharing restricted");
        }
        return AccessDecision.allow("Download allowed");
    }

    public AccessDecision canShare(DigitalFile file, User user, String ip) {
        AccessDecision viewDecision = canView(file, user, ip);
        if (!viewDecision.isAllowed()) {
            return viewDecision;
        }
        Restriction restriction = file.getRestriction();
        if (restriction != null && restriction.isShareRestricted() && restriction.isActiveOn(LocalDate.now())) {
            if (user != null && (restriction.allowsUser(user.getUsername()) || user.hasPermission(Permission.SHARE) || user.hasPermission(Permission.VIEW_RESTRICTED))) {
                return AccessDecision.allow("Sharing allowed by user");
            }
            if (ip != null && restriction.isIpAllowed(ip)) {
                return AccessDecision.allow("Sharing allowed by IP");
            }
            return AccessDecision.deny("Sharing restricted");
        }
        return AccessDecision.allow("Sharing allowed");
    }

    public AccessDecision canRedact(User user) {
        if (user != null && (user.hasPermission(Permission.REDACT) || user.hasPermission(Permission.MANAGE_RIGHTS))) {
            return AccessDecision.allow("Redaction allowed");
        }
        return AccessDecision.deny("Redaction not allowed");
    }

    public AccessDecision canEditMetadata(User user) {
        if (user != null && (user.hasPermission(Permission.EDIT_METADATA) || user.hasPermission(Permission.WRITE) || user.hasPermission(Permission.MANAGE_RIGHTS))) {
            return AccessDecision.allow("Metadata edit allowed");
        }
        return AccessDecision.deny("Metadata edit not allowed");
    }

    public Map<String, String> filterMetadata(DigitalFile file, User user) {
        Map<String, String> visible = new LinkedHashMap<>();
        for (Map.Entry<String, String> entry : file.getMetadata().entrySet()) {
            if (isSensitiveField(entry.getKey()) && (user == null || !user.hasPermission(Permission.VIEW_RESTRICTED))) {
                continue;
            }
            visible.put(entry.getKey(), entry.getValue());
        }
        return visible;
    }

    private boolean isSensitiveField(String field) {
        String lower = field.toLowerCase(Locale.ROOT);
        return lower.contains("donor") || lower.contains("internal") || lower.contains("sensitive");
    }

            for (Map.Entry<String, String> criterion : criteria.entrySet()) {
                String key = criterion.getKey().toLowerCase(Locale.ROOT);
                String value = criterion.getValue();
                boolean fieldMatch;
                switch (key) {
                    case "title" -> fieldMatch = contains(file.getTitle(), value);
                    case "collection" -> fieldMatch = contains(file.getCollectionName(), value);
                    case "tag" -> fieldMatch = file.getTags().stream().anyMatch(t -> contains(t, value));
                    case "photographer" -> fieldMatch = contains(file.getPhotographer(), value);
                    case "person" -> fieldMatch = file.getPeople().stream().anyMatch(p -> contains(p, value));
                    case "place" -> fieldMatch = contains(file.getPlace(), value);
                    default -> fieldMatch = contains(file.getMetadataValue(criterion.getKey()), value);
                }
                if (!fieldMatch) {
                    matches = false;
                    break;
                }
            }
            if (matches) {
                result.add(file);
            }
        }
        return result;

    }

    }

        return new HashMap<>(downloads.getOrDefault(user.getUsername(), new HashMap<>()));
    }

        if (file.getType() != DigitalObjectType.AUDIO && file.getType() != DigitalObjectType.VIDEO) {
            throw new IllegalArgumentException("Not an audio or video file");
        }
        usageTracker.trackUse(file, user);
        return "STREAM[" + file.getId() + "] " + file.getContent();
    }

            return redacted.getContent();
        }
        LocalDate until = LocalDate.parse(untilValue);
        if (!LocalDate.now().isBefore(until)) {
            return redacted.getMetadataValue("originalContent");
        }
        return redacted.getContent();
    }
}
