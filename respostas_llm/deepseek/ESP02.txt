import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;

public class Address {
    private final String street;
    private final String city;
    private final String state;
    private final String zipCode;

    public Address(String street, String city, String state, String zipCode) {
        this.street = street;
        this.city = city;
        this.state = state;
        this.zipCode = zipCode;
    }

    public String getStreet() {
        return street;
    }

    public String getCity() {
        return city;
    }

    public String getState() {
        return state;
    }

    public String getZipCode() {
        return zipCode;
    }

    public boolean hasValidZipCode() {
        return zipCode != null && zipCode.matches("\\d{5}");
    }

    public String getDisplayText() {
        return street + ", " + city + ", " + state + " " + zipCode;
    }

    public String getGoogleMapsUrl() {
        String query = getDisplayText();
        return "https://www.google.com/maps/search/?api=1&query="
                + URLEncoder.encode(query, StandardCharsets.UTF_8);
    }

    @Override
    public String toString() {
        return getDisplayText();
    }
}

// File: GeoLocation.java
public record GeoLocation(double latitude, double longitude) {
    public double distanceTo(GeoLocation other) {
        double earthRadiusKm = 6371.0;
        double lat1 = Math.toRadians(this.latitude);
        double lat2 = Math.toRadians(other.latitude);
        double dLat = Math.toRadians(other.latitude - this.latitude);
        double dLon = Math.toRadians(other.longitude - this.longitude);
        double a = Math.sin(dLat / 2) * Math.sin(dLat / 2)
                + Math.cos(lat1) * Math.cos(lat2)
                * Math.sin(dLon / 2) * Math.sin(dLon / 2);
        double c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
        return earthRadiusKm * c;
    }
}

// File: TimeRange.java
import java.time.LocalTime;

public class TimeRange {
    private final LocalTime start;
    private final LocalTime end;

    public TimeRange(LocalTime start, LocalTime end) {
        this.start = start;
        this.end = end;
    }

    public LocalTime getStart() {
        return start;
    }

    public LocalTime getEnd() {
        return end;
    }

    public boolean contains(LocalTime time) {
        return !time.isBefore(start) && time.isBefore(end);
    }

    public boolean overlaps(TimeRange other) {
        return start.isBefore(other.end) && other.start.isBefore(end);
    }

    @Override
    public String toString() {
        return start + "-" + end;
    }
}

// File: OperatingHours.java
import java.time.DayOfWeek;
import java.time.LocalTime;
import java.util.ArrayList;
import java.util.EnumMap;
import java.util.List;
import java.util.Map;

public class OperatingHours {
    private final Map<DayOfWeek, List<TimeRange>> hours = new EnumMap<>(DayOfWeek.class);

    public void addHours(DayOfWeek day, TimeRange range) {
        hours.computeIfAbsent(day, d -> new ArrayList<>()).add(range);
    }

    public List<TimeRange> getHoursForDay(DayOfWeek day) {
        return List.copyOf(hours.getOrDefault(day, List.of()));
    }

    public boolean isOpen(DayOfWeek day, LocalTime time) {
        for (TimeRange range : hours.getOrDefault(day, List.of())) {
            if (range.contains(time)) {
                return true;
            }
        }
        return false;
    }

    public Map<DayOfWeek, List<TimeRange>> getWeeklyHours() {
        return hours;
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (DayOfWeek day : DayOfWeek.values()) {
            List<TimeRange> ranges = hours.get(day);
            if (ranges != null && !ranges.isEmpty()) {
                sb.append(day).append(": ").append(ranges).append("; ");
            }
        }
        return sb.toString().trim();
    }
}

    public void addFacility(Facility facility) {
        facilities.add(facility);
    }

    public void addEvent(SafeDisposalEvent event) {
        events.add(event);
    }

    public List<Facility> getPublicBins() {
        return facilities.stream()
                .filter(f -> f.getType() == FacilityType.PUBLIC_RECYCLING_BIN)
                .toList();
    }

    public List<Facility> getSpecialWasteDropOffs() {
        return facilities.stream()
                .filter(f -> f.getType() == FacilityType.SPECIAL_WASTE_DROP_OFF)
                .toList();
    }

    public String renderTextMap() {
        StringBuilder sb = new StringBuilder("Map Display:\n");
        for (Facility facility : facilities) {
            sb.append("- ").append(facility.getType()).append(": ")
              .append(facility.getName()).append(" | ")
              .append(facility.getAddress().getDisplayText()).append(" | ")
              .append(facility.getGoogleMapsUrl()).append("\n");
        }
        for (SafeDisposalEvent event : events) {
            sb.append("- Event: ").append(event.getTitle()).append(" | ")
              .append(event.getAddress().getDisplayText()).append(" | ")
              .append(event.getAddress().getGoogleMapsUrl()).append("\n");
        }
        return sb.toString();
    }
}

import java.time.DayOfWeek;
import java.time.LocalTime;
import java.util.ArrayList;
import java.util.EnumMap;
import java.util.List;
import java.util.Map;

public class WeeklySchedule {
    private final Map<DayOfWeek, List<TimeRange>> availability = new EnumMap<>(DayOfWeek.class);

    public void addAvailability(DayOfWeek day, TimeRange range) {
        availability.computeIfAbsent(day, d -> new ArrayList<>()).add(range);
    }

    public List<TimeRange> getAvailability(DayOfWeek day) {
        return List.copyOf(availability.getOrDefault(day, List.of()));
    }

    public Map<DayOfWeek, List<TimeRange>> getAvailabilityMap() {
        return availability;
    }

    public boolean isAvailable(DayOfWeek day, LocalTime time) {
        for (TimeRange range : availability.getOrDefault(day, List.of())) {
            if (range.contains(time)) {
                return true;
            }
        }
        return false;
    }

    public boolean hasOverlap(DayOfWeek day, TimeRange facilityRange) {
        for (TimeRange range : availability.getOrDefault(day, List.of())) {
            if (range.overlaps(facilityRange)) {
                return true;
            }
        }
        return false;
    }

    @Override
    public String toString() {
        return availability.toString();
    }
}

import java.util.ArrayList;
import java.util.List;

public class BillStatement {
    private final List<Transaction> transactions;

    public BillStatement(List<Transaction> transactions) {
        this.transactions = new ArrayList<>(transactions);
    }

    public List<Transaction> getTransactions() {
        return List.copyOf(transactions);
    }

    public double getTotalAmount() {
        return transactions.stream().mapToDouble(Transaction::getAmount).sum();
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder("Bill Statement\n");
        for (Transaction transaction : transactions) {
            sb.append("- ").append(transaction).append("\n");
        }
        sb.append("Total: $").append(getTotalAmount());
        return sb.toString();
    }
}

import java.nio.ByteBuffer;
import java.nio.charset.StandardCharsets;
import java.security.SecureRandom;
import java.util.Base64;
import javax.crypto.Cipher;
import javax.crypto.SecretKey;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;

public class EncryptionService {
    private static final String ALGORITHM = "AES/GCM/NoPadding";
    private static final int IV_LENGTH = 12;
    private static final int TAG_LENGTH = 128;
    private final SecretKey key;
    private final SecureRandom secureRandom;

    public EncryptionService() {
        byte[] keyBytes = "1234567890abcdef".getBytes(StandardCharsets.UTF_8);
        this.key = new SecretKeySpec(keyBytes, "AES");
        this.secureRandom = new SecureRandom();
    }

    public String encrypt(String plainText) {
        if (plainText == null) {
            return null;
        }
        try {
            byte[] iv = new byte[IV_LENGTH];
            secureRandom.nextBytes(iv);
            Cipher cipher = Cipher.getInstance(ALGORITHM);
            cipher.init(Cipher.ENCRYPT_MODE, key, new GCMParameterSpec(TAG_LENGTH, iv));
            byte[] cipherText = cipher.doFinal(plainText.getBytes(StandardCharsets.UTF_8));
            ByteBuffer buffer = ByteBuffer.allocate(iv.length + cipherText.length);
            buffer.put(iv);
            buffer.put(cipherText);
            return Base64.getEncoder().encodeToString(buffer.array());
        } catch (Exception e) {
            throw new IllegalStateException("Encryption failed", e);
        }
    }

    public String decrypt(String encryptedText) {
        if (encryptedText == null) {
            return null;
        }
        try {
            byte[] decoded = Base64.getDecoder().decode(encryptedText);
            ByteBuffer buffer = ByteBuffer.wrap(decoded);
            byte[] iv = new byte[IV_LENGTH];
            buffer.get(iv);
            byte[] cipherText = new byte[buffer.remaining()];
            buffer.get(cipherText);
            Cipher cipher = Cipher.getInstance(ALGORITHM);
            cipher.init(Cipher.DECRYPT_MODE, key, new GCMParameterSpec(TAG_LENGTH, iv));
            return new String(cipher.doFinal(cipherText), StandardCharsets.UTF_8);
        } catch (Exception e) {
            throw new IllegalStateException("Decryption failed", e);
        }
    }
}

public class AuthorizationService {
    public void requireRole(UserAccount account, Role role) {
        if (account == null || !account.hasRole(role)) {
            throw new SecurityException("Requires role: " + role);
        }
    }

    public void requireAnyRole(UserAccount account, Role... roles) {
        if (account == null) {
            throw new SecurityException("Authentication required");
        }
        for (Role role : roles) {
            if (account.hasRole(role)) {
                return;
            }
        }
        throw new SecurityException("Requires one of roles: " + java.util.Arrays.toString(roles));
    }

    public boolean isAnonymous(UserAccount account) {
        return account == null;
    }
}

    private String hashPassword(String password) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(password.getBytes(StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder();
            for (byte b : hash) {
                sb.append(String.format("%02x", b));
            }
            return sb.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException("SHA-256 not available", e);
        }
    }

    private boolean passwordMatches(String password, String passwordHash) {
        return hashPassword(password).equals(passwordHash);
    }

    public UserAccount createAccount(String username, String password, String email, String phone,
                                     Address address, String ipAddress) {
        if (username == null || username.isBlank()) {
            throw new IllegalArgumentException("Username is required");
        }
        if (password == null || password.length() < 4) {
            throw new IllegalArgumentException("Password must be at least 4 characters");
        }
        if (store.findUserByUsername(username).isPresent()) {
            throw new IllegalArgumentException("Username already exists");
        }
        String encryptedEmail = encryptionService.encrypt(email);
        String encryptedPhone = encryptionService.encrypt(phone);
        UserAccount user = new UserAccount(
                UUID.randomUUID().toString(),
                username,
                hashPassword(password),
                encryptedEmail,
                encryptedPhone,
                address,
                ipAddress,
                Set.of(Role.USER)
        );
        store.addUser(user);
        store.logActivity(user.getId(), "Created account", ipAddress);
        return user;
    }

    public UserAccount login(String username, String password) {
        UserAccount user = store.findUserByUsername(username)
                .orElseThrow(() -> new SecurityException("Invalid username or password"));
        if (user.isBlocked()) {
            throw new SecurityException("This account is blocked");
        }
        if (!passwordMatches(password, user.getPasswordHash())) {
            throw new SecurityException("Invalid username or password");
        }
        store.logActivity(user.getId(), "Logged in", user.getIpAddress());
        return user;
    }

    public String getEmail(UserAccount user) {
        return encryptionService.decrypt(user.getEncryptedEmail());
    }

    public String getPhone(UserAccount user) {
        return encryptionService.decrypt(user.getEncryptedPhone());
    }

    }

    public String requestTemporaryPassword(UserAccount user) {

        }

    public boolean redeemReward(UserAccount user, Reward reward) {
        if (user.getRewardPoints() < reward.getPointsRequired()) {
            return false;

        user.deductPoints(reward.getPointsRequired());
        user.addReward(reward);

import java.time.DayOfWeek;
import java.time.LocalTime;
import java.util.ArrayList;
import java.util.List;
import java.util.Set;

public class ScheduleService {
    public void uploadSchedule(UserAccount user, WeeklySchedule schedule) {
        user.setWeeklySchedule(schedule);
    }

    public List<Facility> recommendFacilities(UserAccount user, String zipCode,
                                              Set<WasteType> wasteTypes,
                                              FacilityService facilityService) {
        List<Facility> candidates = facilityService.searchByZipAndWaste(zipCode, wasteTypes);
        if (user.getWeeklySchedule() == null) {
            return candidates;
        }
        List<Facility> recommendations = new ArrayList<>();
        for (Facility facility : candidates) {
            if (hasOverlap(user.getWeeklySchedule(), facility.getOperatingHours())) {
                recommendations.add(facility);
            }
        }
        return recommendations;
    }

    public List<String> findFlexiblePickupWindows(UserAccount user, Facility facility) {
        if (user.getWeeklySchedule() == null) {
            return List.of();
        }
        WeeklySchedule schedule = user.getWeeklySchedule();
        OperatingHours hours = facility.getOperatingHours();
        List<String> windows = new ArrayList<>();
        for (DayOfWeek day : DayOfWeek.values()) {
            for (TimeRange availability : schedule.getAvailability(day)) {
                for (TimeRange openHours : hours.getHoursForDay(day)) {
                    if (availability.overlaps(openHours)) {
                        LocalTime start = availability.getStart().isAfter(openHours.getStart())
                                ? availability.getStart() : openHours.getStart();
                        LocalTime end = availability.getEnd().isBefore(openHours.getEnd())
                                ? availability.getEnd() : openHours.getEnd();
                        windows.add(day + " " + start + "-" + end);
                    }
                }
            }
        }
        return windows;
    }

    private boolean hasOverlap(WeeklySchedule schedule, OperatingHours hours) {
        for (DayOfWeek day : DayOfWeek.values()) {
            for (TimeRange availability : schedule.getAvailability(day)) {
                for (TimeRange openHours : hours.getHoursForDay(day)) {
                    if (availability.overlaps(openHours)) {
                        return true;
                    }
                }
            }
        }
        return false;
    }
}
