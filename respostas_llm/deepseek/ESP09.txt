public class Tag {
    private final String name;
    private final String vocabulary;

    public Tag(String name, String vocabulary) {
        this.name = name;
        this.vocabulary = vocabulary;
    }

    public String getName() {
        return name;
    }

    public String getVocabulary() {
        return vocabulary;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Tag)) return false;
        Tag tag = (Tag) o;
        return name.equals(tag.name) && vocabulary.equals(tag.vocabulary);
    }

    @Override
    public int hashCode() {
        return Objects.hash(name, vocabulary);
    }

    @Override
    public String toString() {
        return vocabulary + ":" + name;
    }
}

import java.text.ParseException;
import java.text.SimpleDateFormat;
import java.util.HashSet;
import java.util.Set;

public class MetadataField {
    private final String name;
    private final MetadataType type;
    private final boolean required;
    private final Set<String> allowedValues;
    private final String regex;
    private final String defaultValue;

    public MetadataField(String name, MetadataType type, boolean required) {
        this(name, type, required, new HashSet<>(), null, null);
    }

    public MetadataField(String name, MetadataType type, boolean required,
                         Set<String> allowedValues, String regex, String defaultValue) {
        this.name = name;
        this.type = type;
        this.required = required;
        this.allowedValues = allowedValues == null ? new HashSet<>() : new HashSet<>(allowedValues);
        this.regex = regex;
        this.defaultValue = defaultValue;
    }

    public String validate(String value) {
        if (value == null || value.isBlank()) {
            return required ? "Value is required" : null;
        }
        switch (type) {
            case NUMBER:
                try {
                    Double.parseDouble(value);
                } catch (NumberFormatException e) {
                    return "Value must be a number";
                }
                break;
            case DATE:
                SimpleDateFormat format = new SimpleDateFormat("yyyy-MM-dd");
                format.setLenient(false);
                try {
                    format.parse(value);
                } catch (ParseException e) {
                    return "Value must be a date in yyyy-MM-dd format";
                }
                break;
            case BOOLEAN:
                if (!value.equalsIgnoreCase("true") && !value.equalsIgnoreCase("false")) {
                    return "Value must be true or false";
                }
                break;
            case CONTROLLED_VOCABULARY:
                if (!allowedValues.contains(value)) {
                    return "Value is not in the controlled vocabulary";
                }
                break;
            case TEXT:
            case FILE_REFERENCE:
                if (regex != null && !value.matches(regex)) {
                    return "Value does not match the required format";
                }
                break;
        }
        return null;
    }

    public MetadataValue createValue(String value, User user) {
        String error = validate(value);
        if (error != null) {
            throw new IllegalArgumentException("Invalid metadata for " + name + ": " + error);
        }
        return new MetadataValue(this, value, user);
    }

    public String getName() {
        return name;
    }

    public MetadataType getType() {
        return type;
    }

    public boolean isRequired() {
        return required;
    }

    public Set<String> getAllowedValues() {
        return new HashSet<>(allowedValues);
    }

    public String getRegex() {
        return regex;
    }

    public String getDefaultValue() {
        return defaultValue;
    }
}

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public class PageTemplate {
    private final String name;
    private final List<MetadataField> fields;
    private final Map<String, String> defaultValues;

    public PageTemplate(String name, List<MetadataField> fields, Map<String, String> defaultValues) {
        this.name = name;
        this.fields = new ArrayList<>(fields);
        this.defaultValues = new LinkedHashMap<>(defaultValues);
    }

    public String getName() {
        return name;
    }

    public List<MetadataField> getFields() {
        return new ArrayList<>(fields);
    }

    public Map<String, String> prepopulate() {
        return new LinkedHashMap<>(defaultValues);
    }

    public void validateMetadata(Map<String, String> supplied) {
        for (MetadataField field : fields) {
            String value = supplied.containsKey(field.getName())
                    ? supplied.get(field.getName())
                    : defaultValues.get(field.getName());
            String error = field.validate(value);
            if (error != null) {
                throw new IllegalArgumentException("Invalid metadata for " + field.getName() + ": " + error);
            }
        }
    }
}

    private boolean deleted;

    public FileResource(String id, String fileName, String mimeType, User owner, byte[] content) {
        this.id = id;
        this.fileName = fileName;
        this.mimeType = mimeType;
        this.owner = owner;
        this.versions.add(new FileVersion(1, content, owner));
    }

    public void addVersion(User user, byte[] content) {
        if (immutable) {
            throw new IllegalStateException("File is immutable");
        }
        versions.add(new FileVersion(versions.size() + 1, content, user));
    }

    public void revertToVersion(int versionNumber, User user) {
        FileVersion version = getVersion(versionNumber);
        addVersion(user, version.getContent());
    }

    public FileVersion getVersion(int versionNumber) {
        for (FileVersion version : versions) {
            if (version.getVersionNumber() == versionNumber) {
                return version;
            }
        }
        throw new IllegalArgumentException("Unknown version: " + versionNumber);
    }

    public FileVersion getCurrentVersion() {
        return versions.get(versions.size() - 1);
    }

    }

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class BackupManager {
    private String schedule;
    private Path lastBackup;
    private boolean verified;

    public BackupManager() {
        this.schedule = "daily";
    }

    public void setSchedule(String schedule) {
        this.schedule = schedule;
    }

    public List<String> backup(Path source, Path destination) throws IOException {
        List<String> log = new ArrayList<>();
        if (!Files.exists(source)) {
            throw new IOException("Backup source does not exist: " + source);
        }
        Files.createDirectories(destination);
        Path target = destination.resolve("neurohub-backup-" + System.currentTimeMillis() + ".zip");
        Files.writeString(target, "Neurohub backup of " + source + " using schedule " + schedule);
        lastBackup = target;
        verified = Files.exists(target);
        log.add("Backed up " + source + " to " + target);
        log.add("Backup verified: " + verified);
        return log;
    }

    public boolean verify(Path backup) {
        return backup != null && Files.exists(backup);
    }

    public boolean restore(Path backup, Path destination) throws IOException {
        if (!verify(backup)) {
            return false;
        }
        Files.createDirectories(destination);
        Files.writeString(destination.resolve("restored-marker.txt"), "Restored from " + backup);
        return true;
    }

    public String getSchedule() {
        return schedule;
    }

    public Path getLastBackup() {
        return lastBackup;
    }

    public boolean isVerified() {
        return verified;
    }
}

import java.util.ArrayList;
import java.util.Collection;
import java.util.Date;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Set;

public class SearchService {
    public List<FileResource> searchFiles(Collection<FileResource> files, String query,
                                          String mimePrefix, Set<Tag> tags, Date from, Date to) {
        List<FileResource> results = new ArrayList<>();
        String q = query == null ? null : query.toLowerCase();
        for (FileResource file : files) {
            if (file.isDeleted()) {
                continue;
            }
            boolean matches = true;
            if (q != null && !q.isBlank()) {
                boolean textMatch = file.getFileName().toLowerCase().contains(q);
                if (!textMatch) {
                    for (MetadataValue value : file.getMetadata().values()) {
                        if (value.getValue().toLowerCase().contains(q)) {
                            textMatch = true;
                            break;
                        }
                    }
                }
                matches = textMatch;
            }
            if (matches && mimePrefix != null && !mimePrefix.isBlank()) {
                matches = file.getMimeType().startsWith(mimePrefix);
            }
            if (matches && tags != null && !tags.isEmpty()) {
                matches = file.getTags().containsAll(tags);
            }
            if (matches && from != null) {
                matches = !file.getUploadDate().before(from);
            }
            if (matches && to != null) {
                matches = !file.getUploadDate().after(to);
            }
            if (matches) {
                results.add(file);
            }
        }
        return results;
    }

    public List<LogBookPage> searchPages(Collection<LogBookPage> pages, String query) {
        List<LogBookPage> results = new ArrayList<>();
        String q = query == null ? "" : query.toLowerCase();
        for (LogBookPage page : pages) {
            if (page.getTitle().toLowerCase().contains(q) || page.getContent().toLowerCase().contains(q)) {
                results.add(page);
            }
        }
        return results;
    }

    public List<Protocol> searchProtocols(Collection<Protocol> protocols, String query) {
        List<Protocol> results = new ArrayList<>();
        String q = query == null ? "" : query.toLowerCase();
        for (Protocol protocol : protocols) {
            boolean match = protocol.getTitle().toLowerCase().contains(q);
            if (!match) {
                for (ProtocolVersion version : protocol.getVersions()) {
                    if (version.getContent().toLowerCase().contains(q)) {
                        match = true;
                        break;
                    }
                }
            }
            if (match) {
                results.add(protocol);
            }
        }
        return results;
    }

    public Map<String, byte[]> downloadMultiple(Collection<FileResource> files, User user) {
        Map<String, byte[]> result = new LinkedHashMap<>();
        if (!user.can(Permission.DOWNLOAD_MULTIPLE) && !user.can(Permission.DOWNLOAD_FILE)) {
            return result;
        }
        for (FileResource file : files) {
            boolean allowed = file.getOwner().equals(user)
                    || user.can(Permission.ADMIN_ACCESS)
                    || file.getShareLevel() == ShareLevel.PUBLIC
                    || file.getShareLevel() == ShareLevel.GROUP
                    || file.getSharedWith().contains(user);
            if (allowed && !file.isDeleted()) {
                result.put(file.getFileName(), file.getCurrentVersion().getContent());
            }
        }
        return result;
    }
}

                                           Map<String, byte[]> directoryFiles) {
        if (!uploader.can(Permission.BULK_UPLOAD) && !uploader.can(Permission.UPLOAD_FILE)) {
            throw new IllegalStateException("User cannot bulk upload files");
        }
        List<FileResource> uploaded = new ArrayList<>();
        for (Map.Entry<String, byte[]> entry : directoryFiles.entrySet()) {
            String fileId = id + "-" + entry.getKey();
            FileResource file = uploadFile(uploader, fileId, entry.getKey(),
                    "application/octet-stream", entry.getValue());
            uploaded.add(file);
        }
        return new DirectoryUpload(id, directoryName, uploader, uploaded);
    }

    public LogBookPage createPage(User author, String logBookId, String sectionName, String pageId,
                                  String title, PageTemplate template, Map<String, String> metadataValues) {
        if (!author.can(Permission.CREATE_PAGE)) {
            throw new IllegalStateException("User cannot create log book pages");
        }
        LogBook logBook = logBooks.get(logBookId);
        if (logBook == null) {
            throw new IllegalArgumentException("Unknown log book: " + logBookId);
        }
        LogBookSection section = logBook.getSection(sectionName).orElseGet(() -> logBook.addSection(sectionName));
        if (template != null) {
            template.validateMetadata(metadataValues == null ? Map.of() : metadataValues);
        }
        LogBookPage page = section.createPage(pageId, title, author, template);
        if (template != null && metadataValues != null) {
            for (Map.Entry<String, String> entry : metadataValues.entrySet()) {
                Optional<MetadataField> field = template.getFields().stream()
                        .filter(f -> f.getName().equals(entry.getKey()))
                        .findFirst();
                field.ifPresent(f -> page.setMetadata(f, entry.getValue(), author));
            }
        }
        return page;
    }

        if (!user.can(Permission.ATTACH_FILE)) {
