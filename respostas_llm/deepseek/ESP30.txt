import java.time.LocalDate;

public class SearchCriteria {
    private final String keyword;
    private final String author;
    private final LocalDate fromDate;
    private final LocalDate toDate;
    private final String advancedKeyword;
    private final String reference;

    public SearchCriteria(String keyword, String author, LocalDate fromDate, LocalDate toDate,
                          String advancedKeyword, String reference) {
        this.keyword = keyword;
        this.author = author;
        this.fromDate = fromDate;
        this.toDate = toDate;
        this.advancedKeyword = advancedKeyword;
        this.reference = reference;
    }

    public boolean matches(Publication publication) {
        if (publication == null) {
            return false;
        }

        if ((fromDate != null || toDate != null) && publication.getPublicationDate() == null) {
            return false;
        }
        if (fromDate != null && publication.getPublicationDate().isBefore(fromDate)) {
            return false;
        }
        if (toDate != null && publication.getPublicationDate().isAfter(toDate)) {
            return false;
        }

        if (!isBlank(keyword) && !publicationContains(publication, keyword)) {
            return false;
        }

        if (!isBlank(author)) {
            boolean authorFound = false;
            for (Author publicationAuthor : publication.getAuthors()) {
                if (containsIgnoreCase(publicationAuthor.getName(), author)) {
                    authorFound = true;
                    break;
                }
            }
            if (!authorFound) {
                return false;
            }
        }

        if (!isBlank(advancedKeyword) && !publicationContains(publication, advancedKeyword)) {
            return false;
        }

        if (!isBlank(reference)) {
            boolean referenceFound = false;
            for (Publication cited : publication.getCitedReferences()) {
                if (containsIgnoreCase(cited.getTitle(), reference)) {
                    referenceFound = true;
                    break;
                }
                for (Author citedAuthor : cited.getAuthors()) {
                    if (containsIgnoreCase(citedAuthor.getName(), reference)) {
                        referenceFound = true;
                        break;
                    }
                }
                if (referenceFound) {
                    break;
                }
            }
            if (!referenceFound) {
                return false;
            }
        }

        return true;
    }

    private boolean publicationContains(Publication publication, String query) {
        if (containsIgnoreCase(publication.getTitle(), query)) {
            return true;
        }
        if (containsIgnoreCase(publication.getAbstractText(), query)) {
            return true;
        }
        for (String keyword : publication.getKeywords()) {
            if (containsIgnoreCase(keyword, query)) {
                return true;
            }
        }
        return false;
    }

    private boolean containsIgnoreCase(String text, String query) {
        return text != null && query != null && text.toLowerCase().contains(query.toLowerCase());
    }

    private boolean isBlank(String value) {
        return value == null || value.isBlank();
    }
}

import java.util.ArrayList;
import java.util.List;

public class SearchResult {
    private final List<Publication> results;
    private final int pageSize;

    public SearchResult(List<Publication> results, int pageSize) {
        if (pageSize <= 0) {
            throw new IllegalArgumentException("Page size must be positive");
        }
        this.results = results == null ? new ArrayList<>() : new ArrayList<>(results);
        this.pageSize = pageSize;
    }

    public int getTotalCount() {
        return results.size();
    }

    public int getPageCount() {
        if (results.isEmpty()) {
            return 1;
        }
        return (int) Math.ceil((double) results.size() / pageSize);
    }

    public List<Publication> getPage(int pageNumber) {
        if (results.isEmpty()) {
            if (pageNumber == 1) {
                return new ArrayList<>();
            }
            throw new IllegalArgumentException("Page out of range");
        }
        if (pageNumber < 1 || pageNumber > getPageCount()) {
            throw new IllegalArgumentException("Page out of range");
        }
        int fromIndex = (pageNumber - 1) * pageSize;
        int toIndex = Math.min(fromIndex + pageSize, results.size());
        return new ArrayList<>(results.subList(fromIndex, toIndex));
    }

    public boolean hasNext(int currentPage) {
        return currentPage < getPageCount();
    }

    public boolean hasPrevious(int currentPage) {
        return currentPage > 1;
    }

    public List<Publication> getAllResults() {
        return new ArrayList<>(results);
    }
}

        if (publicationIds != null) {
            for (Integer id : publicationIds) {
                details.add(getPublication(id).toDetailedString());
            }
        }
        return details;
    }

    public String downloadPublication(int publicationId) {
        requireSearchAccess();
        Publication publication = getPublication(publicationId);
        if (!publication.isFullTextAvailable()) {
            throw new IllegalStateException("Full text is not available in the system");
        }
        return publication.getFullText();
    }

    public List<String> downloadPublications(List<Integer> publicationIds) {
        requireSearchAccess();
        if (publicationIds == null) {
            return new ArrayList<>();
        }
        if (publicationIds.size() > configuration.getMaxBatchDownload()) {
            throw new IllegalArgumentException("Too many publications requested for one download");
        }
        List<String> downloads = new ArrayList<>();
        for (Integer id : publicationIds) {
            Publication publication = getPublication(id);
            if (!publication.isFullTextAvailable()) {
                throw new IllegalStateException("Full text is not available for publication " + id);
            }
            downloads.add(publication.getFullText());
        }
        return downloads;
    }

    public List<Publication> consultReferences(int publicationId) {
        requireSearchAccess();
        return getPublication(publicationId).getCitedReferences();
    }

    public List<Publication> consultReferencesMultiple(List<Integer> publicationIds) {
        requireSearchAccess();
        LinkedHashSet<Publication> references = new LinkedHashSet<>();
        if (publicationIds != null) {
            for (Integer id : publicationIds) {
                references.addAll(getPublication(id).getCitedReferences());
            }
        }
        return new ArrayList<>(references);
    }

    public String displayFullText(int publicationId) {
        requireSearchAccess();
        Publication publication = getPublication(publicationId);
        if (!publication.isFullTextAvailable()) {
            throw new IllegalStateException("Full text is not available in the system");
        }
        return publication.getFullText();
    }

    public String viewReferenceDetails(int referenceId) {
        requireSearchAccess();
        return getPublication(referenceId).toDetailedString();
    }

    public String exportReferences(int publicationId, ExportFormat format) {
        requireSearchAccess();
        if (format == null) {
            throw new IllegalArgumentException("Export format is required");
        }
        Publication publication = getPublication(publicationId);
        StringBuilder sb = new StringBuilder();
        for (Publication reference : publication.getCitedReferences()) {
            if (format == ExportFormat.BIB) {
                sb.append(reference.toBibEntry());
            } else {
                sb.append(reference.toRisEntry());
            }
        }
        return sb.toString();
    }

    public String viewAuthorDetails(Author author) {
        requireSearchAccess();
        if (author == null) {
            throw new IllegalArgumentException("Author is required");
        }
        return author.getDetails();
    }

    public void editPersonalData(String password, String name, String email, String university, String department) {
        requireLoggedIn();
        User user = currentSession.getUser();
        if (password != null && !password.isBlank()) {
            user.setPassword(password);
        }
        user.updatePersonalData(name, email, university, department);
    }

    public List<Publication> getMyPublications() {
        requirePublisherRole();
        String username = currentSession.getUser().getUsername();
        List<Publication> ownPublications = new ArrayList<>();
        for (Publication publication : publications.values()) {
            if (username.equals(publication.getUploaderUsername())) {
                ownPublications.add(publication);
            }
        }
        return ownPublications;
    }

    public void editOwnPublication(int publicationId, String title, String abstractText,
                                   LocalDate publicationDate, List<Integer> citedReferenceIds) {
        requirePublisherRole();
        Publication publication = getPublication(publicationId);
        String username = currentSession.getUser().getUsername();
        if (!username.equals(publication.getUploaderUsername())) {
            throw new SecurityException("Only the owner can edit this publication");
        }
        if (title != null && !title.isBlank()) {
            publication.setTitle(title);
        }
        if (abstractText != null) {
            publication.setAbstractText(abstractText);
        }
        if (publicationDate != null) {
            publication.setPublicationDate(publicationDate);
        }
        if (citedReferenceIds != null) {
            List<Publication> references = new ArrayList<>();
            for (Integer referenceId : citedReferenceIds) {
                references.add(getPublication(referenceId));
            }
            publication.setCitedReferences(references);
        }
    }

    public Publication uploadPublication(PublicationFormat format, String fileName, String fullText,
                                         String title, List<Author> authors, LocalDate publicationDate,
                                         String abstractText, List<Integer> citedReferenceIds,
                                         List<String> keywords) {
        requirePublisherRole();
        if (format == null) {
            throw new IllegalArgumentException("Publication format is required");
        }
        if (title == null || title.isBlank()) {
            throw new IllegalArgumentException("Title is required");
        }
        User uploader = currentSession.getUser();
        int publicationId = nextPublicationId++;
        Publication publication = new Publication(publicationId, title, authors, publicationDate,
                abstractText, fullText, uploader.getUsername(), format, fileName);
        if (citedReferenceIds != null) {
            for (Integer referenceId : citedReferenceIds) {
                publication.addReference(getPublication(referenceId));
            }
        }
        if (keywords != null) {
            for (String keyword : keywords) {
                publication.addKeyword(keyword);
            }
        }
        publications.put(publicationId, publication);
        return publication;
    }

    public List<User> searchUsers(UserSearchCriteria criteria) {
        requireRole(Role.MODERATOR, Role.ADMINISTRATOR);
        List<User> found = new ArrayList<>();
        for (User user : users.values()) {
            if (criteria == null || criteria.matches(user)) {
                found.add(user);
            }
        }
        return found;
    }

    public void editUserSettings(String username, String newPassword, String email, String university,
                                 String department, Role role, Boolean active) {
        User actor = requireRole(Role.MODERATOR, Role.ADMINISTRATOR);
        User target = users.get(username);
        if (target == null) {
            throw new IllegalArgumentException("User does not exist");
        }
        if (actor.getRole() == Role.MODERATOR && target.getRole() == Role.ADMINISTRATOR) {
            throw new SecurityException("Moderators cannot edit administrators");
        }
        if (newPassword != null && !newPassword.isBlank()) {
            target.setPassword(newPassword);
        }
        if (email != null) {
            target.setEmail(email);
        }
        if (university != null) {
            target.setUniversity(university);
        }
        if (department != null) {
            target.setDepartment(department);
        }
        if (role != null) {
            target.setRole(role);
        }
        if (active != null) {
            target.setActive(active);
        }
    }

    public User createUser(String username, String password, String name, String email,
                           String university, String department, Role role) {
        requireRole(Role.ADMINISTRATOR);
        if (users.containsKey(username)) {
            throw new IllegalArgumentException("Username already exists");
        }
        User user = new User(username, password, name, email, university, department, role);
        users.put(username, user);
        return user;
    }

    public void deleteUser(String username) {
        requireRole(Role.ADMINISTRATOR);
        User user = users.get(username);
        if (user == null) {
            throw new IllegalArgumentException("Username does not exist");
        }
        users.remove(username);
        for (Group group : groups.values()) {
            group.removeUser(user);
        }
    }

    public Group createGroup(String groupName, List<String> usernames) {
        requireRole(Role.ADMINISTRATOR);
        if (groups.containsKey(groupName)) {
            throw new IllegalArgumentException("Group name already exists");
        }
        Group group = new Group(groupName);
        if (usernames != null) {
            for (String username : usernames) {
                User user = users.get(username);
                if (user == null) {
                    throw new IllegalArgumentException("User does not exist: " + username);
                }
                group.addUser(user);
            }
        }
        groups.put(groupName, group);
        return group;
    }

    public List<Group> searchGroups(GroupSearchCriteria criteria) {
        requireRole(Role.ADMINISTRATOR);
        List<Group> found = new ArrayList<>();
        for (Group group : groups.values()) {
            if (criteria == null || criteria.matches(group)) {
                found.add(group);
            }
        }
        return found;
    }

    public void deleteGroup(String groupName) {
        requireRole(Role.ADMINISTRATOR);
        Group group = groups.get(groupName);
        if (group == null) {
            throw new IllegalArgumentException("Group name does not exist");
        }
        if (!group.isEmpty()) {
            throw new IllegalStateException("Group is not empty");
        }
        groups.remove(groupName);
    }

    public void editGroupSettings(String oldName, String newName, List<String> usernames) {
        requireRole(Role.ADMINISTRATOR);
        Group group = groups.get(oldName);
        if (group == null) {
            throw new IllegalArgumentException("Group name does not exist");
        }
        if (newName != null && !newName.isBlank() && !newName.equals(oldName)) {
            if (groups.containsKey(newName)) {
                throw new IllegalArgumentException("New group name already exists");
            }
            groups.remove(oldName);
            group.setName(newName);
            groups.put(newName, group);
        }
        if (usernames != null) {
            for (String username : usernames) {
                if (!users.containsKey(username)) {
                    throw new IllegalArgumentException("User does not exist: " + username);
                }
            }
            group.clearUsers();
            for (String username : usernames) {
                group.addUser(users.get(username));
            }
        }
    }

    public void editSystemConfiguration(SystemConfiguration newConfiguration) {
        requireRole(Role.ADMINISTRATOR);
        if (newConfiguration == null) {
            throw new IllegalArgumentException("Configuration is required");
        }
        configuration = newConfiguration;
    }
}
