        boolean alreadyMember = members.keySet().stream()
                .anyMatch(u -> u.getEmail().equalsIgnoreCase(email));
        if (!alreadyMember) {
            pendingInvites.put(email.toLowerCase(), role);
        }
    }

        return pendingInvites.containsKey(email.toLowerCase());
    }

        Role role = pendingInvites.remove(user.getEmail().toLowerCase());
        if (role != null) {
            members.put(user, role);

        }
        return null;
    }

        if (user.equals(owner)) {

public class Resource {
    private String name;
    private String url;
    private String content;
    private String cachedContent;
    private String checksum;
    private boolean originalAvailable = true;

    public Resource(String name, String url) {
        this.name = name;
        this.url = url;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getUrl() {
        return url;
    }

    public void setUrl(String url) {
        this.url = url;
    }

    public String getContent() {
        if (originalAvailable && content != null) {
            return content;
        }
        return cachedContent != null ? cachedContent : content;
    }

    public void setContent(String content) {
        this.content = content;
    }

    public String getCachedContent() {
        return cachedContent;
    }

    public void cache() {
        if (content != null) {
            cachedContent = content;
        }
    }

    public boolean isCached() {
        return cachedContent != null;
    }

    public void markOriginalUnavailable() {
        originalAvailable = false;
    }

    public boolean isOriginalAvailable() {
        return originalAvailable;
    }

    public String getChecksum() {
        return checksum;
    }

    public void setChecksum(String checksum) {
        this.checksum = checksum;
    }

    public void computeChecksum() {
        checksum = Integer.toHexString(Objects.hashCode(content));
    }

    public boolean validate() {
        return name != null && !name.isEmpty()
                && url != null && !url.isEmpty()
                && getContent() != null && !getContent().isEmpty();
    }

    @Override
    public String toString() {
        return name + " -> " + url;
    }
}

import java.util.ArrayList;
import java.util.List;

public class ValidatorService {
    public ValidationStatus validate(PackageVersion version) {
        List<String> messages = new ArrayList<>();

        if (version.getDescriptorJson() == null || version.getDescriptorJson().trim().isEmpty()) {
            messages.add("Descriptor is empty");
        }

        for (Resource resource : version.getResources()) {
            if (!resource.validate()) {
                messages.add("Invalid resource: " + resource.getName());
            }
            if (resource.getChecksum() == null) {
                resource.computeChecksum();
            }
        }

        ValidationStatus status = messages.isEmpty() ? ValidationStatus.VALID : ValidationStatus.INVALID;
        version.setValidation(status, messages);
        return status;
    }

    public ValidationStatus validate(DataPackage dataPackage) {
        PackageVersion version = dataPackage.getCurrentVersion();
        if (version == null) {
            return ValidationStatus.INVALID;
        }
        return validate(version);
    }
}

public class CacheService {
    public void cache(PackageVersion version) {
        for (Resource resource : version.getResources()) {
            resource.cache();
        }
    }

    public String getResourceContent(PackageVersion version, String resourceName) {
        Resource resource = version.getResource(resourceName);
        if (resource == null) {
            return null;
        }
        return resource.getContent();
    }

    public boolean hasCache(PackageVersion version) {
        for (Resource resource : version.getResources()) {
            if (!resource.isCached()) {
                return false;
            }
        }
        return true;
    }
}

    }

        String q = query.toLowerCase();
        return packages.stream()
                .filter(p -> p.getState() == PackageState.PUBLISHED)
                .filter(p -> p.getDescription().toLowerCase().contains(q))
                .collect(Collectors.toList());
    }

    public List<DataPackage> searchByPublisher(Publisher publisher, String query) {
        return publisher.searchPackages(query);
    }

public class BillingSystem {
    private final List<PricingPlan> plans = new ArrayList<>();
    private final Map<Publisher, PricingPlan> subscriptions = new LinkedHashMap<>();

    public BillingSystem() {
        plans.add(new PricingPlan(PlanType.FREE, "Free", 0.0, "Public packages only", 0));
        plans.add(new PricingPlan(PlanType.BASIC, "Basic", 10.0, "5 private packages", 5));
        plans.add(new PricingPlan(PlanType.PRO, "Pro", 50.0, "50 private packages", 50));
        plans.add(new PricingPlan(PlanType.ENTERPRISE, "Enterprise", 200.0, "Unlimited private packages", Integer.MAX_VALUE));
    }

    public List<PricingPlan> listPlans() {
        return Collections.unmodifiableList(plans);
    }

    public PricingPlan subscribe(Publisher publisher, PlanType type) {
        PricingPlan plan = plans.stream()
                .filter(p -> p.getType() == type)
                .findFirst()
                .orElse(plans.get(0));
        subscriptions.put(publisher, plan);
        publisher.setPlan(type);
        return plan;
    }

    public PricingPlan getSubscription(Publisher publisher) {
        return subscriptions.getOrDefault(publisher, plans.get(0));
    }
}

import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Set;

public class NotificationService {
    private final Map<User, List<Notification>> notifications = new LinkedHashMap<>();
    private final Map<DataPackage, Set<User>> subscriptions = new HashMap<>();

    public void notify(User user, String message) {
        notifications.computeIfAbsent(user, k -> new ArrayList<>()).add(new Notification(user, message));
    }

    public void notifyPackageChanged(DataPackage dataPackage, String message) {
        Set<User> subscribers = subscriptions.getOrDefault(dataPackage, Collections.emptySet());
        for (User user : subscribers) {
            notify(user, message);
        }
    }

    public void subscribe(User user, DataPackage dataPackage) {
        subscriptions.computeIfAbsent(dataPackage, k -> new HashSet<>()).add(user);
    }

    public List<Notification> getNotifications(User user) {
        return notifications.getOrDefault(user, Collections.emptyList());
    }
}

    public String publish(DataPackage dataPackage, String version) {
        PackageVersion packageVersion = dataPackage.getVersion(version);
        if (packageVersion == null) {
            return "Version not found: " + version;
        }
        ValidationStatus status = registry.getValidatorService().validate(packageVersion);
        if (status == ValidationStatus.INVALID) {
            return "Validation failed: " + packageVersion.getValidationMessages();
        }
        registry.getCacheService().cache(packageVersion);
        dataPackage.publish();
        registry.getMetrics().incrementApiUsage();
        registry.getNotificationService().notifyPackageChanged(dataPackage,
                "Package " + dataPackage.getName() + " published version " + version);
        return "Published " + dataPackage.getName() + " version " + version;
    }

    public String validate(DataPackage dataPackage, String version) {
        PackageVersion packageVersion = dataPackage.getVersion(version);
        if (packageVersion == null) {
            return "Version not found: " + version;
        }
        ValidationStatus status = registry.getValidatorService().validate(packageVersion);
        return status + (packageVersion.getValidationMessages().isEmpty()
                ? ""
                : " " + packageVersion.getValidationMessages());
    }

    public String preview(DataPackage dataPackage) {
        PackageVersion packageVersion = dataPackage.getCurrentVersion();
        if (packageVersion == null) {
            return "No version available";
        }
        return "Preview of " + dataPackage.getName()
                + "\nView: " + dataPackage.getViewJson()
                + "\nDescriptor: " + packageVersion.getDescriptorJson();
    }

    public String restore(DataPackage dataPackage) {

        return "Restored " + dataPackage.getName();
    }

    public String undelete(DataPackage dataPackage) {
        dataPackage.restore();
        return "Undeleted " + dataPackage.getName();
    }

public class Cli {
    private final Registry registry;
    private final DataPackageClient client;
    private User currentUser;

    public Cli(Registry registry, DataPackageClient client) {
        this.registry = registry;
        this.client = client;
    }

    public void setCurrentUser(User currentUser) {
        this.currentUser = currentUser;
    }

    public String execute(String commandLine) {
        if (commandLine == null || commandLine.trim().isEmpty()) {
            return "";
        }
        String[] parts = commandLine.trim().split("\\s+");
        String command = parts[0].toLowerCase();

        switch (command) {
            case "config":
                if (parts.length >= 2) {
                    client.configure(new ClientConfig(parts[1]));
                    return "Configured client for " + parts[1];
                }
                return "Usage: config <registryUrl>";
            case "import":
                if (parts.length >= 2) {
                    DataPackage dataPackage = registry.getPackage(parts[1]);
                    return dataPackage != null
                            ? "Imported package " + parts[1]
                            : "Package not found: " + parts[1];
                }
                return "Usage: import <packageName>";
            case "publish":
                if (parts.length >= 3) {
                    DataPackage dataPackage = registry.getPackage(parts[1]);
                    if (dataPackage == null) {
                        return "Package not found: " + parts[1];
                    }
                    return client.publish(dataPackage, parts[2]);
                }
                return "Usage: publish <packageName> <version>";
            case "unpublish":
                if (parts.length >= 2) {
                    DataPackage dataPackage = registry.getPackage(parts[1]);
                    if (dataPackage != null) {
                        dataPackage.unpublish();
                        return "Unpublished " + dataPackage.getName();
                    }
                }
                return "Package not found";
            case "delete":
                if (parts.length >= 2) {
                    DataPackage dataPackage = registry.getPackage(parts[1]);
                    if (dataPackage != null) {
                        dataPackage.deletePermanently();
                        return "Deleted " + dataPackage.getName();
                    }
                }
                return "Package not found";
            case "restore":
                if (parts.length >= 2) {
                    DataPackage dataPackage = registry.getPackage(parts[1]);
                    if (dataPackage != null) {
                        dataPackage.restore();
                        return "Restored " + dataPackage.getName();
                    }
                }
                return "Package not found";
            case "undelete":
                if (parts.length >= 2) {
                    DataPackage dataPackage = registry.getPackage(parts[1]);
                    if (dataPackage != null) {
                        dataPackage.restore();
                        return "Undeleted " + dataPackage.getName();
                    }
                }
                return "Package not found";
            case "preview":
                if (parts.length >= 2) {
                    DataPackage dataPackage = registry.getPackage(parts[1]);
                    if (dataPackage != null) {
                        return client.preview(dataPackage);
                    }
                }
                return "Package not found";
            case "search":
                if (parts.length >= 2) {
                    StringBuilder query = new StringBuilder();
                    for (int i = 1; i < parts.length; i++) {
                        if (i > 1) {
                            query.append(" ");
                        }
                        query.append(parts[i]);
                    }
                    return "Found: " + registry.searchPackages(query.toString()).size() + " packages";
                }
                return "Usage: search <query>";
            case "download":
                if (parts.length >= 3) {
                    DataPackage dataPackage = registry.getPackage(parts[1]);
                    if (dataPackage != null) {
                        DownloadService downloadService = new DownloadService(registry.getMetrics());
                        return "Downloaded zip size: " + downloadService.downloadZip(dataPackage, parts[2]).length + " bytes";
                    }
                }
                return "Usage: download <packageName> <version>";
            case "tag":
                if (parts.length >= 4) {
                    DataPackage dataPackage = registry.getPackage(parts[1]);
                    if (dataPackage != null) {
                        return dataPackage.tag(parts[2], parts[3], false);
                    }
                }
                return "Usage: tag <packageName> <tag> <version>";
            default:
                return "Unknown command: " + command;
        }
    }
}

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.util.zip.ZipEntry;
import java.util.zip.ZipOutputStream;

public class DownloadService {
    private final Metrics metrics;

    public DownloadService(Metrics metrics) {
        this.metrics = metrics;
    }

    public byte[] downloadZip(DataPackage dataPackage, String version) {
        PackageVersion packageVersion = dataPackage.getVersion(version);
        if (packageVersion == null) {
            return new byte[0];
        }
        dataPackage.incrementDownloads();
        metrics.incrementDownloads();

        try (ByteArrayOutputStream baos = new ByteArrayOutputStream();
             ZipOutputStream zos = new ZipOutputStream(baos)) {
            zos.putNextEntry(new ZipEntry("descriptor.json"));
            String descriptor = packageVersion.getDescriptorJson();
            zos.write(descriptor == null ? new byte[0] : descriptor.getBytes());
            zos.closeEntry();

            for (Resource resource : packageVersion.getResources()) {
                zos.putNextEntry(new ZipEntry(resource.getName()));
                String content = resource.getContent();
                zos.write(content == null ? new byte[0] : content.getBytes());
                zos.closeEntry();
            }

            zos.finish();
            return baos.toByteArray();
        } catch (IOException e) {
            return new byte[0];
        }
    }

    public String downloadVersion(DataPackage dataPackage, String version) {
        PackageVersion packageVersion = dataPackage.getVersion(version);
        if (packageVersion == null) {
            return "Version not found: " + version;
        }
        dataPackage.incrementDownloads();
        metrics.incrementDownloads();
        return "Downloaded " + dataPackage.getName() + " version " + version;
    }

    public String updatePreviouslyDownloaded(DataPackage dataPackage, String version) {
        dataPackage.incrementDownloads();
        metrics.incrementDownloads();
        return "Updated local copy of " + dataPackage.getName() + " to version " + version;
    }

    public String downloadAsSQLite(DataPackage dataPackage, String version) {
        return new SQLiteExportService().export(dataPackage, version);
    }

public class SQLiteExportService {
    public String export(DataPackage dataPackage, String version) {
        PackageVersion packageVersion = dataPackage.getVersion(version);
        if (packageVersion == null) {
            return "Version not found: " + version;
        }
        StringBuilder builder = new StringBuilder();
        builder.append("-- SQLite export for ")
                .append(dataPackage.getName())
                .append(" ")
                .append(version)
                .append("\n");
        for (Resource resource : packageVersion.getResources()) {
            String table = resource.getName().replaceAll("\\W", "_");
            String content = resource.getContent() == null ? "" : resource.getContent().replace("'", "''");
            builder.append("CREATE TABLE ").append(table).append(" (data TEXT);\n");
            builder.append("INSERT INTO ").append(table).append(" VALUES ('").append(content).append("');\n");
        }
        return builder.toString();
    }

import java.util.HashSet;
import java.util.Objects;
import java.util.Set;

public class VersionDiffService {
    public SchemaDiff compare(PackageVersion v1, PackageVersion v2) {
        SchemaDiff diff = new SchemaDiff();
        Set<String> names1 = new HashSet<>();
        for (Resource resource : v1.getResources()) {
            names1.add(resource.getName());
        }
        Set<String> names2 = new HashSet<>();
        for (Resource resource : v2.getResources()) {
            names2.add(resource.getName());
        }
        for (String name : names2) {
            if (!names1.contains(name)) {
                diff.addAdded(name);
            }
        }
        for (String name : names1) {
            if (!names2.contains(name)) {
                diff.addRemoved(name);
            }
        }
        for (String name : names1) {
            if (names2.contains(name)) {
                Resource r1 = v1.getResource(name);
                Resource r2 = v2.getResource(name);
                if (!Objects.equals(r1.getChecksum(), r2.getChecksum())) {
                    diff.addChanged(name);
                }
            }
        }
        diff.setDescriptorDiff("v1: " + v1.getDescriptorJson() + " -> v2: " + v2.getDescriptorJson());
        return diff;
    }
}

import java.util.LinkedHashMap;
import java.util.Map;

public class RequirementsFile {
    private final Map<String, String> requirements = new LinkedHashMap<>();

    public void addRequirement(String packageName, String version) {
        requirements.put(packageName, version);
    }

    public String toFileContent() {
        StringBuilder builder = new StringBuilder();
        for (Map.Entry<String, String> entry : requirements.entrySet()) {
            builder.append(entry.getKey()).append("==").append(entry.getValue()).append("\n");
        }
        return builder.toString();
    }

    public void parse(String content) {
        for (String line : content.split("\n")) {
            line = line.trim();
            if (line.isEmpty() || line.startsWith("#")) {
                continue;
            }
            String[] parts = line.split("==");
            if (parts.length == 2) {
                requirements.put(parts[0].trim(), parts[1].trim());
            }
        }
    }

    public Map<String, String> getRequirements() {
        return new LinkedHashMap<>(requirements);
    }
}
