import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

public class User {
    private final long id;
    private final String username;
    private String passwordHash;
    private String ldapDn;
    private final Set<Role> globalRoles = new HashSet<>();
    private final Map<Long, Role> repositoryRoles = new HashMap<>();

    public User(long id, String username) {
        this.id = id;
        this.username = username;
    }

    public long getId() {
        return id;
    }

    public String getUsername() {
        return username;
    }

    public String getLdapDn() {
        return ldapDn;
    }

    public void setLdapDn(String ldapDn) {
        this.ldapDn = ldapDn;
    }

    public void setPassword(String password) {
        this.passwordHash = hash(password);
    }

    public boolean checkPassword(String password) {
        return passwordHash != null && passwordHash.equals(hash(password));
    }

    public Set<Role> getGlobalRoles() {
        return Set.copyOf(globalRoles);
    }

    public void addGlobalRole(Role role) {
        globalRoles.add(role);
    }

    public boolean hasGlobalRole(Role role) {
        return globalRoles.contains(role);
    }

    public void setRepositoryRole(long repositoryId, Role role) {
        repositoryRoles.put(repositoryId, role);
    }

    public Role getRepositoryRole(long repositoryId) {
        return repositoryRoles.get(repositoryId);
    }

    public boolean hasRepositoryRole(long repositoryId, Role role) {
        return repositoryRoles.get(repositoryId) == role;
    }

    public Map<Long, Role> getRepositoryRoles() {
        return Map.copyOf(repositoryRoles);
    }

    private String hash(String password) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] bytes = digest.digest(password.getBytes(StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder();
            for (byte b : bytes) {
                sb.append(String.format("%02x", b & 0xff));
            }
            return sb.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException(e);
        }
    }
}

// File: UserService.java
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Set;

public class UserService {
    private final Map<Long, User> usersById = new LinkedHashMap<>();
    private final Map<String, User> usersByUsername = new LinkedHashMap<>();
    private long nextId = 1;

    public User createInitialAdministrator(String username, String password) {
        return createUserInternal(username, password, Set.of(Role.ADMINISTRATOR));
    }

    public User createUser(User admin, String username, String password, Set<Role> globalRoles) {
        if (admin == null || !admin.hasGlobalRole(Role.ADMINISTRATOR)) {
            throw new SecurityException("Administrator required");
        }
        return createUserInternal(username, password, globalRoles);
    }

    private User createUserInternal(String username, String password, Set<Role> globalRoles) {
        if (usersByUsername.containsKey(username)) {
            throw new IllegalArgumentException("Username already exists: " + username);
        }
        User user = new User(nextId++, username);
        user.setPassword(password);
        for (Role role : globalRoles) {
            user.addGlobalRole(role);
        }
        usersById.put(user.getId(), user);
        usersByUsername.put(username, user);
        return user;
    }

    public User findById(long id) {
        return usersById.get(id);
    }

    public User findByUsername(String username) {
        return usersByUsername.get(username);
    }

    public void assignRepositoryRole(User admin, long userId, long repositoryId, Role role) {
        if (admin == null || !admin.hasGlobalRole(Role.ADMINISTRATOR)) {
            throw new SecurityException("Administrator required");
        }
        User user = usersById.get(userId);
        if (user == null) {
            throw new IllegalArgumentException("Unknown user: " + userId);
        }
        user.setRepositoryRole(repositoryId, role);
    }

    public void assignAdvancedDataEntry(User admin, long userId, AdvancedDataEntry ade) {
        if (admin == null || !admin.hasGlobalRole(Role.ADMINISTRATOR)) {
            throw new SecurityException("Administrator required");
        }
        User user = usersById.get(userId);
        if (user == null) {
            throw new IllegalArgumentException("Unknown user: " + userId);
        }
        ade.addUser(userId);
        user.setRepositoryRole(ade.getRepositoryId(), Role.ADVANCED_DATA_ENTRY);
    }
}

        this.ldapAuthenticator = ldapAuthenticator;
    }

    public User loginLocal(String username, String password) {
        User user = userService.findByUsername(username);
        if (user == null || !user.checkPassword(password)) {
            throw new SecurityException("Invalid local credentials");
        }
        return user;
    }

    public User loginLdap(String username, String password) {
        User user = userService.findByUsername(username);
        if (user == null || user.getLdapDn() == null || !ldapAuthenticator.authenticate(user.getLdapDn(), password)) {
            throw new SecurityException("Invalid LDAP credentials");
        }
        return user;
    }

// File: PermissionService.java
public class PermissionService {
    public void requireAuthenticated(User user) {
        if (user == null) {
            throw new SecurityException("Authentication required");
        }
    }

    public boolean isAdministrator(User user) {
        return user != null && user.hasGlobalRole(Role.ADMINISTRATOR);
    }

    public boolean canView(User user, long repositoryId) {
        if (user == null) {
            return false;
        }
        if (isAdministrator(user)) {
            return true;
        }
        return user.getRepositoryRole(repositoryId) != null;
    }

    public boolean canCreate(User user, long repositoryId) {
        if (user == null) {
            return false;
        }
        if (isAdministrator(user)) {
            return true;
        }
        Role role = user.getRepositoryRole(repositoryId);
        return role == Role.REPOSITORY_MANAGER || role == Role.BASIC_DATA_ENTRY || role == Role.ADVANCED_DATA_ENTRY;
    }

    public boolean canUpdate(User user, long repositoryId) {
        return canCreate(user, repositoryId);
    }

    public boolean canDelete(User user, long repositoryId) {
        if (user == null) {
            return false;
        }
        if (isAdministrator(user)) {
            return true;
        }
        return user.getRepositoryRole(repositoryId) == Role.REPOSITORY_MANAGER;
    }

    public boolean canManageRepository(User user, long repositoryId) {
        if (user == null) {
            return false;
        }
        if (isAdministrator(user)) {
            return true;
        }
        return user.getRepositoryRole(repositoryId) == Role.REPOSITORY_MANAGER;
    }

    public void requireView(User user, long repositoryId) {
        if (!canView(user, repositoryId)) {
            throw new SecurityException("View permission denied");
        }
    }

    public void requireCreate(User user, long repositoryId) {
        if (!canCreate(user, repositoryId)) {
            throw new SecurityException("Create permission denied");
        }
    }

    public void requireUpdate(User user, long repositoryId) {
        if (!canUpdate(user, repositoryId)) {
            throw new SecurityException("Update permission denied");
        }

public class ImportService {
    public ImportResult importEadArchon(String content) {
        return importEad(content, EadFormat.ARCHON);
    }

    public ImportResult importEadAt(String content) {
        return importEad(content, EadFormat.ARCHIVISTS_TOOLKIT);
    }

    public ImportResult importEad(String content, EadFormat format) {
        ImportResult result = new ImportResult();
        if (content == null || content.isBlank()) {
            result.addError(new ImportError("Empty EAD import content", 1, format.name()));
            return result;
        }
        String lower = content.toLowerCase();
        if (!lower.contains("<ead") && !lower.contains("<?xml")) {
            result.addError(new ImportError("EAD root element not found", 1, format.name()));
        }
        int count = countOccurrences(lower, "<archivalobject") + countOccurrences(lower, "<resource");
        if (count == 0) {
            result.addError(new ImportError("No importable EAD descriptions found", 1, format.name()));
        }
        for (int i = 0; i < count; i++) {
            result.addRecord(format + " EAD record " + (i + 1));
        }
        return result;
    }

    public ImportResult importMarcXml(String content, boolean onlyAgentsAndSubjects) {
        ImportResult result = new ImportResult();
        if (content == null || content.isBlank()) {
            result.addError(new ImportError("Empty MARCXML import content", 1, "MARCXML"));
            return result;
        }
        String lower = content.toLowerCase();
        int recordCount = countOccurrences(lower, "<record") + countOccurrences(lower, "<marc:record");
        if (recordCount == 0) {
            result.addError(new ImportError("No MARCXML records found", 1, "MARCXML"));
        }
        String type = onlyAgentsAndSubjects ? "Agent/Subject-only MARCXML" : "MARCXML";
        for (int i = 0; i < recordCount; i++) {
            result.addRecord(type + " record " + (i + 1));
        }
        return result;
    }

    public ImportResult importEacCpf(String content) {
        ImportResult result = new ImportResult();
        if (content == null || content.isBlank()) {
            result.addError(new ImportError("Empty EAC-CPF import content", 1, "EAC-CPF"));
            return result;
        }
        String lower = content.toLowerCase();
        int count = countOccurrences(lower, "<cpf");
        if (count == 0) {
            result.addError(new ImportError("EAC-CPF cpf element not found", 1, "EAC-CPF"));
        }
        for (int i = 0; i < count; i++) {
            result.addRecord("EAC-CPF record " + (i + 1));
        }
        return result;
    }

    public ImportResult importAccessionsCsv(String content) {
        ImportResult result = new ImportResult();
        if (content == null || content.isBlank()) {
            result.addError(new ImportError("Empty accession CSV content", 1, "CSV"));
            return result;
        }
        String[] lines = content.split("\\R");
        if (lines.length <= 1) {
            result.addError(new ImportError("No accession data rows found", 1, "CSV"));
        }
        for (int i = 1; i < lines.length; i++) {
            if (!lines[i].isBlank()) {
                result.addRecord("Accession CSV row " + i);
            }
        }
        return result;
    }

    public ImportResult importCustom(String content, CustomImporter importer) {
        if (importer == null) {
            throw new IllegalArgumentException("Custom importer required");
        }
        return importer.importData(content);
    }

    private int countOccurrences(String text, String needle) {
        int count = 0;
        int index = 0;
        while ((index = text.indexOf(needle, index)) >= 0) {
            count++;
            index += needle.length();
        }
        return count;
    }
}

public class ResourceService {

    private final Map<Long, Resource> resources = new LinkedHashMap<>();
    private final Map<Long, ResourceComponent> components = new LinkedHashMap<>();

    private long nextComponentId = 1;

    public ResourceService(PermissionService permissionService) {

    public Resource createResource(User user, long repositoryId, String title, LevelOfDescription level, String otherLevel) {
        permissionService.requireCreate(user, repositoryId);
        Resource resource = new Resource(nextResourceId++, repositoryId, title, level);
        resource.setOtherLevel(otherLevel);
        resources.put(resource.getId(), resource);
        return resource;
    }

    public Resource updateResource(User user, long resourceId, String title, LevelOfDescription level, String otherLevel, int expectedVersion) {
        Resource resource = requireResource(resourceId);
        permissionService.requireUpdate(user, resource.getRepositoryId());
        if (resource.getVersion() != expectedVersion) {
            throw new ConcurrentModificationException("Resource was modified by another user");
        }
        resource.setTitle(title);
        resource.setLevel(level);
        resource.setOtherLevel(otherLevel);
        resource.incrementVersion();
        return resource;
    }

        permissionService.requireDelete(user, resource.getRepositoryId());
        resources.remove(resourceId);
        components.values().removeIf(c -> c.getResourceId() == resourceId);
    }

        Resource resource = requireResource(resourceId);

            throw new IllegalArgumentException("Unknown parent component: " + parentId);

import java.util.ArrayList;
import java.util.List;

public class SearchService {
    public List<String> searchWithinRepository(long repositoryId, String query, ResourceService resourceService, AccessionService accessionService, DigitalObjectService digitalObjectService) {
        String q = query == null ? "" : query.toLowerCase();
        List<String> results = new ArrayList<>();

        for (Resource resource : resourceService.listResources()) {
            if (resource.getRepositoryId() == repositoryId && resource.getTitle().toLowerCase().contains(q)) {
                results.add("Resource: " + resource.getTitle());
            }
        }

        for (Accession accession : accessionService.listAccessions()) {
            if (accession.getRepositoryId() == repositoryId && accession.getTitle().toLowerCase().contains(q)) {
                results.add("Accession: " + accession.getTitle());
            }
        }

        for (DigitalObject digitalObject : digitalObjectService.listDigitalObjects()) {
            if (digitalObject.getRepositoryId() == repositoryId && digitalObject.getTitle().toLowerCase().contains(q)) {
                results.add("DigitalObject: " + digitalObject.getTitle());
            }
        }

        return results;
    }
}
