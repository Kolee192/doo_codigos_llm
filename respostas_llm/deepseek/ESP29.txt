import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Classroom {
    public static final int CAPACITY = 20;
    private final int classroomId;
    private String name;
    private String roomNo;
    private String phoneNumber;
    private String teacher;
    private String assistant;
    private final List<Child> children = new ArrayList<>();
    private final WaitingList waitingList;

    public Classroom(int classroomId, String name, String roomNo, String phoneNumber,
                     String teacher, String assistant) {
        this.classroomId = classroomId;
        this.name = name;
        this.roomNo = roomNo;
        this.phoneNumber = phoneNumber;
        this.teacher = teacher;
        this.assistant = assistant;
        this.waitingList = new WaitingList(this);
    }

    public boolean hasSpace() {
        return children.size() < CAPACITY;
    }

    public void addChild(Child child) {
        if (child == null) {
            throw new IllegalArgumentException("Child cannot be null");
        }
        if (children.size() >= CAPACITY) {
            throw new IllegalStateException("Classroom is full");
        }
        if (!children.contains(child)) {
            children.add(child);
            child.assignToClassroom(this, ChildStatus.ENROLLED);
        }
    }

    public void removeChild(Child child) {
        if (children.remove(child)) {
            child.clearClassroom();
        }
    }

    public void addToWaitingList(Child child) {
        waitingList.addChild(child);
    }

    public int getClassroomId() {
        return classroomId;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getRoomNo() {
        return roomNo;
    }

    public void setRoomNo(String roomNo) {
        this.roomNo = roomNo;
    }

    public String getPhoneNumber() {
        return phoneNumber;
    }

    public void setPhoneNumber(String phoneNumber) {
        this.phoneNumber = phoneNumber;
    }

    public String getTeacher() {
        return teacher;
    }

    public void setTeacher(String teacher) {
        this.teacher = teacher;
    }

    public String getAssistant() {
        return assistant;
    }

    public void setAssistant(String assistant) {
        this.assistant = assistant;
    }

    public List<Child> getChildren() {
        return Collections.unmodifiableList(children);
    }

    public WaitingList getWaitingList() {
        return waitingList;
    }
}

import java.math.BigDecimal;
import java.time.LocalDate;

public class Bill {
    private final int invoiceId;
    private final int parentId;
    private BigDecimal amountDue;
    private BigDecimal amountPaid;
    private LocalDate datePaid;

    public Bill(int invoiceId, Parent parent, BigDecimal amountDue) {
        if (parent == null) {
            throw new IllegalArgumentException("Parent cannot be null");
        }
        if (amountDue == null || amountDue.signum() < 0) {
            throw new IllegalArgumentException("Amount due must be non-negative");
        }
        this.invoiceId = invoiceId;
        this.parentId = parent.getParentId();
        this.amountDue = amountDue;
        this.amountPaid = BigDecimal.ZERO;
        this.datePaid = null;
        parent.addBill(this);
    }

    public void pay(BigDecimal amount, LocalDate date) {
        if (amount == null || amount.signum() <= 0) {
            throw new IllegalArgumentException("Payment amount must be positive");
        }
        if (date == null) {
            throw new IllegalArgumentException("Payment date cannot be null");
        }
        amountPaid = amountPaid.add(amount);
        if (amountPaid.compareTo(amountDue) >= 0) {
            datePaid = date;
        }
    }

    public BigDecimal getBalance() {
        BigDecimal balance = amountDue.subtract(amountPaid);
        return balance.signum() < 0 ? BigDecimal.ZERO : balance;
    }

    public int getInvoiceId() {
        return invoiceId;
    }

    public int getParentId() {
        return parentId;
    }

    public BigDecimal getAmountDue() {
        return amountDue;
    }

    public void setAmountDue(BigDecimal amountDue) {
        if (amountDue == null || amountDue.signum() < 0) {
            throw new IllegalArgumentException("Amount due must be non-negative");
        }
        this.amountDue = amountDue;
    }

    public BigDecimal getAmountPaid() {
        return amountPaid;
    }

    public LocalDate getDatePaid() {
        return datePaid;
    }
}

import java.time.LocalDate;

public class Registration {
    private final int childId;
    private final LocalDate registerDate;
    private LocalDate enrollDate;
    private LocalDate withdrawDate;

    public Registration(Child child, LocalDate registerDate) {
        if (child == null) {
            throw new IllegalArgumentException("Child cannot be null");
        }
        if (registerDate == null) {
            throw new IllegalArgumentException("Register date cannot be null");
        }
        this.childId = child.getChildId();
        this.registerDate = registerDate;
        child.addRegistration(this);
    }

    public void enroll(LocalDate enrollDate) {
        if (enrollDate == null) {
            throw new IllegalArgumentException("Enroll date cannot be null");
        }
        if (enrollDate.isBefore(registerDate)) {
            throw new IllegalArgumentException("Enroll date cannot be before register date");
        }
        this.enrollDate = enrollDate;
    }

    public void withdraw(LocalDate withdrawDate) {
        if (withdrawDate == null) {
            throw new IllegalArgumentException("Withdraw date cannot be null");
        }
        if (enrollDate == null) {
            throw new IllegalStateException("Cannot withdraw before enrollment");
        }
        if (withdrawDate.isBefore(enrollDate)) {
            throw new IllegalArgumentException("Withdraw date cannot be before enroll date");
        }
        this.withdrawDate = withdrawDate;
    }

    public int getChildId() {
        return childId;
    }

    public LocalDate getRegisterDate() {
        return registerDate;
    }

    public LocalDate getEnrollDate() {
        return enrollDate;
    }

    public LocalDate getWithdrawDate() {
        return withdrawDate;
    }
}

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Collection;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class DayCareCenter {
    private final Map<Integer, Parent> parents = new HashMap<>();
    private final Map<Integer, Child> children = new HashMap<>();
    private final Map<Integer, Employee> employees = new HashMap<>();
    private final Map<Integer, Classroom> classrooms = new HashMap<>();
    private final List<Bill> bills = new ArrayList<>();

    public void addParent(Parent parent) {
        if (parent == null) {
            throw new IllegalArgumentException("Parent cannot be null");
        }
        if (!parent.hasChildren()) {
            throw new IllegalArgumentException("Parent must have at least one child");
        }
        if (parents.putIfAbsent(parent.getParentId(), parent) != null) {
            throw new IllegalArgumentException("Parent ID already exists");
        }
    }

    public void addChild(Child child) {
        if (child == null) {
            throw new IllegalArgumentException("Child cannot be null");
        }
        if (children.putIfAbsent(child.getChildId(), child) != null) {
            throw new IllegalArgumentException("Child ID already exists");
        }
    }

    public void addEmployee(Employee employee) {
        if (employee == null) {
            throw new IllegalArgumentException("Employee cannot be null");
        }
        if (employees.putIfAbsent(employee.getUserId(), employee) != null) {
            throw new IllegalArgumentException("Employee ID already exists");
        }
    }

    public void addClassroom(Classroom classroom) {
        if (classroom == null) {
            throw new IllegalArgumentException("Classroom cannot be null");
        }
        if (classrooms.putIfAbsent(classroom.getClassroomId(), classroom) != null) {
            throw new IllegalArgumentException("Classroom ID already exists");
        }
    }

    public Registration registerChild(Child child, LocalDate registerDate) {
        if (child == null || !children.containsKey(child.getChildId())) {
            throw new IllegalArgumentException("Child is not registered with the center");
        }
        return new Registration(child, registerDate);
    }

    public Immunization addImmunization(Child child, int immuneId, String type, LocalDate date) {
        return new Immunization(immuneId, child, type, date);
    }

    public CallSheet addCallSheet(Child child, LocalDate date) {
        return new CallSheet(child, date);
    }

    public void assignChildToClassroom(Child child, Classroom classroom) {
        if (child == null || classroom == null) {
            throw new IllegalArgumentException("Child and classroom cannot be null");
        }
        if (!child.isAdmissible()) {
            throw new IllegalStateException("Child cannot be admitted without immunization");
        }
        if (child.getClassroom() != null) {
            if (child.getStatus() == ChildStatus.ENROLLED) {
                child.getClassroom().removeChild(child);
            } else {
                child.getClassroom().getWaitingList().removeChild(child);
            }
        }
        if (classroom.hasSpace()) {
            classroom.addChild(child);
        } else {
            classroom.addToWaitingList(child);
        }
    }

    public void addReminder(Employee employee, int reminderId, LocalDate date, String message) {
        if (employee == null || !employees.containsKey(employee.getUserId())) {
            throw new IllegalArgumentException("Employee is not registered with the center");
        }
        new Reminder(reminderId, employee, date, message);
    }

    public void recordChange(Employee employee, String description) {
        if (employee == null || !employees.containsKey(employee.getUserId())) {
            throw new IllegalArgumentException("Employee is not registered with the center");
        }
        new Change(employee, description);
    }

    public Bill generateMonthlyBill(Parent parent, int invoiceId, BigDecimal amountDue) {
        if (parent == null || !parents.containsKey(parent.getParentId())) {
            throw new IllegalArgumentException("Parent is not registered with the center");
        }
        Bill bill = new Bill(invoiceId, parent, amountDue);
        bills.add(bill);
        return bill;
    }

    public Collection<Parent> getParents() {
        return Collections.unmodifiableCollection(parents.values());
    }

    public Collection<Child> getChildren() {
        return Collections.unmodifiableCollection(children.values());
    }

    public Collection<Employee> getEmployees() {
        return Collections.unmodifiableCollection(employees.values());
    }

    public Collection<Classroom> getClassrooms() {
        return Collections.unmodifiableCollection(classrooms.values());
    }

    public List<Bill> getBills() {
        return Collections.unmodifiableList(bills);
    }
}
