import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.ArrayList;
import java.util.List;

public class Car {
    private CarModel model;
    private String color;
    private String licensePlate;
    private int yearOfProduction;
    private CarStatus status;
    private FleetLocation fleetLocation;
    private List<Rental> rentals = new ArrayList<>();

    public Car(CarModel model, String color, String licensePlate, int yearOfProduction,
               CarStatus status, FleetLocation fleetLocation) {
        this.model = model;
        this.color = color;
        this.licensePlate = licensePlate;
        this.yearOfProduction = yearOfProduction;
        this.status = status;
        this.fleetLocation = fleetLocation;
    }

    public CarModel getModel() {
        return model;
    }

    public void setModel(CarModel model) {
        this.model = model;
    }

    public String getColor() {
        return color;
    }

    public String getLicensePlate() {
        return licensePlate;
    }

    public int getYearOfProduction() {
        return yearOfProduction;
    }

    public CarStatus getStatus() {
        return status;
    }

    public void setStatus(CarStatus status) {
        this.status = status;
    }

    public FleetLocation getFleetLocation() {
        return fleetLocation;
    }

    public void setFleetLocation(FleetLocation fleetLocation) {
        this.fleetLocation = fleetLocation;
    }

    public List<Rental> getRentals() {
        return rentals;
    }

    public void addRental(Rental rental) {
        if (!rentals.contains(rental)) {
            rentals.add(rental);
        }
    }

    public BigDecimal averageRentalPrice() {
        if (rentals.isEmpty()) {
            return BigDecimal.ZERO;
        }
        BigDecimal total = BigDecimal.ZERO;
        for (Rental rental : rentals) {
            total = total.add(rental.getPrice());
        }
        return total.divide(BigDecimal.valueOf(rentals.size()), 2, RoundingMode.HALF_UP);
    }
}

import java.math.BigDecimal;
import java.time.LocalDate;

public class Rental {
    private Booking booking;
    private Car car;
    private BigDecimal price;
    private RentalType rentalType;
    private CarPickup pickup;
    private CarDropoff dropoff;

    public Rental(Car car, BigDecimal price, RentalType rentalType) {
        this.car = car;
        this.price = price;
        this.rentalType = rentalType;
    }

    public Booking getBooking() {
        return booking;
    }

    public void setBooking(Booking booking) {
        this.booking = booking;
    }

    public Car getCar() {
        return car;
    }

    public BigDecimal getPrice() {
        return price;
    }

    public RentalType getRentalType() {
        return rentalType;
    }

    public CarPickup getPickup() {
        return pickup;
    }

    public void setPickup(CarPickup pickup) {
        this.pickup = pickup;
        if (pickup != null) {
            pickup.setRental(this);
        }
    }

    public CarDropoff getDropoff() {
        return dropoff;
    }

    public void setDropoff(CarDropoff dropoff) {
        this.dropoff = dropoff;
        if (dropoff != null) {
            dropoff.setRental(this);
        }
    }

    public boolean isBookedForToday() {
        if (pickup == null) {
            return false;
        }
        LocalDate today = LocalDate.now();
        LocalDate start = pickup.getDate();
        LocalDate end = dropoff != null ? dropoff.getDate() : start;
        return !today.isBefore(start) && !today.isAfter(end);
    }
}

    public List<BookingInvoice> getInvoices() {
        return invoices;
    }

    public void addInvoice(BookingInvoice invoice) {
        if (!invoices.contains(invoice)) {
            invoices.add(invoice);
        }
        invoice.setBooking(this);
    }

    public List<Rental> getRentals() {

import java.util.ArrayList;
