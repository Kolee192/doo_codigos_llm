import java.util.*;

public class User {
    private final String id;
    private String name;
    private String email;
    private final Set<Role> roles = EnumSet.noneOf(Role.class);
    private boolean active = true;
    private boolean superuserScope = false;

    public User(String id, String name, String email, Role... roles) {
        this.id = id;
        this.name = name;
        this.email = email;
        this.roles.addAll(Arrays.asList(roles));
    }

    public String getId() { return id; }
    public String getName() { return name; }
    public String getEmail() { return email; }
    public Set<Role> getRoles() { return Collections.unmodifiableSet(roles); }
    public boolean isActive() { return active; }
    public void setActive(boolean active) { this.active = active; }

    public boolean hasRole(Role role) {
        return roles.contains(role);
    }

    public void addRole(Role role) {
        roles.add(role);
    }

    public boolean enterSuperuserScope() {
        if (hasRole(Role.SUPERUSER) || hasRole(Role.REPOSITORY_ADMIN)) {
            superuserScope = true;
            return true;
        }
        return false;
    }

    public void exitSuperuserScope() {
        superuserScope = false;
    }

    public boolean isInSuperuserScope() {
        return superuserScope;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof User)) return false;
        User user = (User) o;
        return id.equals(user.id);
    }

    @Override
    public int hashCode() {
        return Objects.hash(id);
    }
}

import java.time.LocalDate;
import java.util.*;

public class DepositGroup {
    private final String name;
    private final Set<User> members = new LinkedHashSet<>();
    private final Set<String> netIds = new LinkedHashSet<>();
    private final Set<String> submittedUserIds = new LinkedHashSet<>();
    private LocalDate windowStart;
    private LocalDate windowEnd;
    private AccessControl permissions = new AccessControl();

    public DepositGroup(String name) {
        this.name = name;
    }

    public String getName() { return name; }
    public Set<User> getMembers() { return Collections.unmodifiableSet(members); }
    public Set<String> getNetIds() { return Collections.unmodifiableSet(netIds); }
    public Set<String> getSubmittedUserIds() { return Collections.unmodifiableSet(submittedUserIds); }

    public void addMember(User user) { members.add(user); }
    public void removeMember(User user) { members.remove(user); }
    public void addNetId(String netId) { netIds.add(netId); }
    public void removeNetId(String netId) { netIds.remove(netId); }

    public void setWindow(LocalDate start, LocalDate end) {
        this.windowStart = start;
        this.windowEnd = end;
    }

    public boolean isOpen(LocalDate now) {
        boolean startOk = windowStart == null || !now.isBefore(windowStart);
        boolean endOk = windowEnd == null || !now.isAfter(windowEnd);
        return startOk && endOk;
    }

    public AccessControl getPermissions() { return permissions; }
    public void setPermissions(AccessControl permissions) { this.permissions = permissions; }

    public void markSubmitted(User user) {
        submittedUserIds.add(user.getId());
    }

    public Set<String> getNonSubmitters() {
        Set<String> nonSubmitters = new LinkedHashSet<>();
        for (User user : members) {
            if (!submittedUserIds.contains(user.getId())) {
                nonSubmitters.add(user.getName());
            }
        }
        for (String netId : netIds) {
            if (!submittedUserIds.contains(netId)) {
                nonSubmitters.add(netId);
            }
        }
        return nonSubmitters;
    }
}

// File: AccessControl.java
import java.util.*;

public class AccessControl {
    private final Set<String> allowedUserIds = new HashSet<>();
    private final Set<String> allowedRoles = new HashSet<>();
    private final Set<String> whitelistedIps = new HashSet<>();
    private boolean restricted = false;
    private boolean authenticatedOnly = false;
    private boolean dukeIpOnly = false;

    public void allowUser(User user) {
        allowedUserIds.add(user.getId());
        restricted = true;
    }

    public void allowRole(Role role) {
        allowedRoles.add(role.name());
        restricted = true;
    }

    public void addWhitelistIp(String ip) {
        whitelistedIps.add(ip);
    }

    public void setAuthenticatedOnly(boolean authenticatedOnly) {
        this.authenticatedOnly = authenticatedOnly;
        if (authenticatedOnly) restricted = true;
    }

    public void setDukeIpOnly(boolean dukeIpOnly) {
        this.dukeIpOnly = dukeIpOnly;
        if (dukeIpOnly) restricted = true;
    }

    public boolean isRestricted() { return restricted; }

    public boolean canAccess(User user, String ip) {
        if (ip != null) {
            if (whitelistedIps.contains(ip)) return true;
            if (dukeIpOnly && ip.startsWith("10.")) return true;
        }
        if (!restricted) return true;
        if (authenticatedOnly && user == null) return false;
        if (user != null) {
            if (allowedUserIds.contains(user.getId())) return true;
            for (Role role : user.getRoles()) {
                if (allowedRoles.contains(role.name())) return true;
            }
        }
        return false;
    }

    public boolean canDownloadEmbargoed(User user) {
        if (user == null) return false;
        return user.hasRole(Role.COLLECTION_CURATOR)
                || user.hasRole(Role.REPOSITORY_ADMIN)
                || user.hasRole(Role.DIGITAL_RECORDS_ARCHIVIST);
    }

    public String describe() {
        return "restricted=" + restricted
                + ", users=" + allowedUserIds
                + ", roles=" + allowedRoles
                + ", whitelistedIps=" + whitelistedIps
                + ", authenticatedOnly=" + authenticatedOnly
                + ", dukeIpOnly=" + dukeIpOnly;
    }
}

import java.time.LocalDate;

public class Embargo {
    private LocalDate releaseDate;
    private final boolean metadataDiscoverable;
    private final AccessControl accessAfterRelease;
    private boolean lifted;

    public Embargo(LocalDate releaseDate, boolean metadataDiscoverable, AccessControl accessAfterRelease) {
        this.releaseDate = releaseDate;
        this.metadataDiscoverable = metadataDiscoverable;
        this.accessAfterRelease = accessAfterRelease;
    }

    public LocalDate getReleaseDate() { return releaseDate; }
    public boolean isMetadataDiscoverable() { return metadataDiscoverable; }
    public AccessControl getAccessAfterRelease() { return accessAfterRelease; }
    public boolean isLifted() { return lifted; }

    public void setReleaseDate(LocalDate releaseDate) {
        this.releaseDate = releaseDate;
        this.lifted = false;
    }

    public boolean isActive(LocalDate now) {
        return !lifted && releaseDate != null && now.isBefore(releaseDate);
    }

    public void lift() {
        this.lifted = true;
        this.releaseDate = LocalDate.now();
    }
}

        if (embargo != null && embargo.getReleaseDate() != null) {
            updateMetadata("embargoRelease", embargo.getReleaseDate().toString());
        }
    }

    public void liftEmbargo() {
        if (embargo != null) {
            embargo.lift();
            updateMetadata("embargoRelease", "lifted");
        }
    }

    public boolean isEmbargoActive() {
        return embargo != null && embargo.isActive(LocalDate.now());
    }

        component.setItem(this);
        components.add(component);
    }

    public void addAuthor(AuthorIdentifier author) {
        authors.add(author);
    }

    public void mapTo(Collection otherCollection) {
        if (otherCollection != null && !otherCollection.equals(collection)) {
            mappedCollections.add(otherCollection);
        }
    }

    public void removeMapping(Collection otherCollection) {
        mappedCollections.remove(otherCollection);
    }

    public Set<Collection> getAllCollections() {
        Set<Collection> all = new LinkedHashSet<>();
        all.add(collection);
        all.addAll(mappedCollections);
        return all;
    }

    public long getTotalStorageBytes() {
        long total = storageBytes;
        for (Component component : components) {
            total += component.getStorageBytes();
        }
        return total;
    }

    public String getScheduledDeletionDisplay() {
        return scheduledDeletionDate == null ? "None" : scheduledDeletionDate.toString();
    }

    public String getEmbargoDisplay() {
        if (embargo == null) return "No embargo";
        return (embargo.isActive(LocalDate.now()) ? "EMBARGO ACTIVE" : "Embargo released")
                + " release=" + embargo.getReleaseDate();

    public String getPageDescription() {
        return "Item " + id + " [" + title + "]"
                + " | scheduledDeletion=" + getScheduledDeletionDisplay()
                + " | replication=" + replicationStatus
                + " | " + getEmbargoDisplay()
                + " | published=" + published;
    }

    public boolean canViewMetadata(User user, String ip) {
        if (embargo == null || !embargo.isActive(LocalDate.now()) || embargo.isMetadataDiscoverable()) {
            return true;
        }
        return accessControl.canAccess(user, ip);
    }

    public boolean canDownloadFiles(User user, String ip) {
        if (embargo == null || !embargo.isActive(LocalDate.now())) {
            return accessControl.canAccess(user, ip);
        }
        return embargo.getAccessAfterRelease().canAccess(user, ip)
                || embargo.getAccessAfterRelease().canDownloadEmbargoed(user);
    }

    public Version createVersion() {
        Version version = new Version("v-" + (versions.size() + 1), id, LocalDateTime.now(),
                title, getMetadataCopy());
        versions.add(version);
        return version;
    }

    public boolean restoreVersion(String versionId) {
        for (Version version : versions) {
            if (version.getId().equals(versionId)) {
                this.title = version.getTitle();
                metadata.clear();
                metadata.putAll(version.getMetadata());
                return true;
            }
        }
        return false;
    }

    public void setAllReplicationStatus(ReplicationStatus status) {
        this.replicationStatus = status;
        for (Component component : components) {
            component.setReplicationStatus(status);
        }
    }
}

import java.nio.charset.StandardCharsets;
import java.util.Base64;

public class EncryptionService {
    public String encrypt(String plainText, String key) {
        byte[] plain = plainText.getBytes(StandardCharsets.UTF_8);
        byte[] keyBytes = key.getBytes(StandardCharsets.UTF_8);
        byte[] encrypted = new byte[plain.length];
        for (int i = 0; i < plain.length; i++) {
            encrypted[i] = (byte) (plain[i] ^ keyBytes[i % keyBytes.length]);
        }
        return Base64.getEncoder().encodeToString(encrypted);
    }
}

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;

public class ChecksumService {
    public String sha256(byte[] data) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(data);
            StringBuilder sb = new StringBuilder();
            for (byte b : hash) {
                sb.append(String.format("%02x", b));
            }
            return sb.toString();
        } catch (NoSuchAlgorithmException e) {
            return "unavailable";
        }
    }

    public String sha256(String text) {
        return sha256(text.getBytes(StandardCharsets.UTF_8));
    }
}

        for (Item item : items) {
            item.setAllReplicationStatus(ReplicationStatus.QUEUED);
        }
    }

    public String replicateItem(Item item) {
        Collection collection = item.getCollection();
        ReplicationPolicy policy = collection.getReplicationPolicy();
        if (!policy.isEnabled()) {
            item.setAllReplicationStatus(ReplicationStatus.NOT_QUEUED);
            return "not-enabled";
        }
        String data = item.getTitle() + "|" + item.getMetadataCopy();
        if (policy.isEncrypt()) {
            data = encryptionService.encrypt(data, "server-key");
        }
        String checksum = policy.isChecksum()
                ? checksumService.sha256(data.getBytes(StandardCharsets.UTF_8))
                : "none";
        item.setAllReplicationStatus(ReplicationStatus.REPLICATED);
        return checksum;
    }
}

import java.time.LocalDate;

public class DateDelimiter {
    private final DateDelimiterMode mode;
    private final LocalDate start;
    private final LocalDate end;

    private DateDelimiter(DateDelimiterMode mode, LocalDate start, LocalDate end) {
        this.mode = mode;
        this.start = start;
        this.end = end;
    }

    public static DateDelimiter none() {
        return new DateDelimiter(DateDelimiterMode.NONE, null, null);
    }

    public static DateDelimiter before(LocalDate date) {
        return new DateDelimiter(DateDelimiterMode.BEFORE, date, null);
    }

    public static DateDelimiter after(LocalDate date) {
        return new DateDelimiter(DateDelimiterMode.AFTER, date, null);
    }

    public static DateDelimiter between(LocalDate start, LocalDate end) {
        return new DateDelimiter(DateDelimiterMode.BETWEEN, start, end);
    }

    public boolean matches(LocalDate date) {
        if (mode == DateDelimiterMode.NONE) return true;
        if (date == null) return false;
        switch (mode) {
            case BEFORE:
                return date.isBefore(start);
            case AFTER:
                return date.isAfter(start);
            case BETWEEN:
                return !date.isBefore(start) && !date.isAfter(end);
            default:
                return true;
        }
    }

    public DateDelimiterMode getMode() { return mode; }
    public LocalDate getStart() { return start; }
    public LocalDate getEnd() { return end; }
}

import java.nio.charset.StandardCharsets;

public class FormatValidator {
    public FileValidationResult validate(String format, byte[] data) {
        if (format == null || data == null) {
            return FileValidationResult.fail("Missing format or data");
        }
        String lower = format.toLowerCase();
        if (lower.contains("pdf")) {
            String header = new String(data, StandardCharsets.ISO_8859_1);
            return header.startsWith("%PDF")
                    ? FileValidationResult.ok("Valid PDF header")
                    : FileValidationResult.fail("Invalid PDF header");
        }
        if (lower.contains("xml")) {
            String text = new String(data, StandardCharsets.UTF_8).trim();
            return text.startsWith("<")
                    ? FileValidationResult.ok("Well-formed XML prefix")
                    : FileValidationResult.fail("Invalid XML prefix");
        }
        return FileValidationResult.ok("No specific validator; accepted");
    }
}

                sb.append(item.getId())
                        .append(",")
                        .append(item.getTitle())
                        .append(",")
                        .append(item.getEmbargo().getReleaseDate())
                        .append(",")
                        .append(item.isEmbargoActive())
                        .append("\n");
            }
        }
        return sb.toString();
    }

    public DashboardStats dashboardStats() {
        int componentCount = 0;
        long totalStorage = 0;
        Map<String, Long> storageByCollection = new LinkedHashMap<>();
        Map<String, Integer> itemCountByCollection = new LinkedHashMap<>();
        Map<String, Integer> depositActivityByDay = new LinkedHashMap<>();

        for (Item item : items.values()) {
            componentCount += item.getComponents().size();
            long bytes = item.getTotalStorageBytes();
            totalStorage += bytes;
            String collectionName = item.getCollection().getName();
            storageByCollection.merge(collectionName, bytes, Long::sum);
            itemCountByCollection.merge(collectionName, 1, Integer::sum);
            depositActivityByDay.merge(item.getDate().toString(), 1, Integer::sum);
        }

        return new DashboardStats(collections.size(), items.size(), componentCount,
                totalStorage, storageByCollection, itemCountByCollection, depositActivityByDay);
    }

    public Map<ReplicationStatus, Integer> collectionReplicationSummary(String collectionId) {
        Collection collection = getCollection(collectionId);
        Map<ReplicationStatus, Integer> summary = new EnumMap<>(ReplicationStatus.class);
        if (collection == null) return summary;
        for (Item item : itemsForCollection(collection)) {
            summary.merge(item.getReplicationStatus(), 1, Integer::sum);
        }
        return summary;
    }

    public List<Item> findByAuthorIdentifier(String identifier) {
        List<Item> result = new ArrayList<>();
        for (Item item : items.values()) {
            for (AuthorIdentifier author : item.getAuthors()) {
                if (author.getIdentifier().equals(identifier)) {
                    result.add(item);
                    break;
                }
            }
        }
        return result;
    }

    public void applyAuthorAuthority(String scheme, String identifier, String canonicalName) {
        for (Item item : items.values()) {
            for (AuthorIdentifier author : item.getAuthors()) {
                if (author.getScheme().equals(scheme) && author.getIdentifier().equals(identifier)) {
                    author.setName(canonicalName);
                }
            }
        }
    }
