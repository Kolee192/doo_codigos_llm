import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Order {
    private static int nextId = 1;
    private final int id;
    private final int tableNumber;
    private final List<OrderItem> items;
    private final List<DietaryRequirement> dietaryRequirements;
    private OrderStatus status;

    public Order(int tableNumber) {
        this.id = nextId++;
        this.tableNumber = tableNumber;
        this.items = new ArrayList<>();
        this.dietaryRequirements = new ArrayList<>();
        this.status = OrderStatus.PENDING;
    }

    public int getId() {
        return id;
    }

    public int getTableNumber() {
        return tableNumber;
    }

    public OrderStatus getStatus() {
        return status;
    }

    public boolean isPending() {
        return status == OrderStatus.PENDING;
    }

    public boolean isEmpty() {
        return items.isEmpty();
    }

    public List<OrderItem> getItems() {
        return Collections.unmodifiableList(items);
    }

    public List<DietaryRequirement> getDietaryRequirements() {
        return Collections.unmodifiableList(dietaryRequirements);
    }

    public OrderItem addItem(MenuItem menuItem) {
        if (status != OrderStatus.PENDING) {
            throw new IllegalStateException("Only pending orders can be modified");
        }
        OrderItem orderItem = new OrderItem(menuItem);
        items.add(orderItem);
        return orderItem;
    }

    public boolean removeItem(int orderItemId) {
        if (status != OrderStatus.PENDING) {
            return false;
        }
        return items.removeIf(item -> item.getId() == orderItemId);
    }

    public void addDietaryRequirement(DietaryRequirement requirement) {
        if (status != OrderStatus.PENDING) {
            throw new IllegalStateException("Dietary requirements can only be added to pending orders");
        }
        dietaryRequirements.add(requirement);
    }

    public boolean place() {
        if (status == OrderStatus.PENDING && !items.isEmpty()) {
            status = OrderStatus.PLACED;
            return true;
        }
        return false;
    }

    public boolean cancel() {
        if (status == OrderStatus.PENDING) {
            status = OrderStatus.CANCELLED;
            return true;
        }
        return false;
    }

    public void accept() {
        if (status == OrderStatus.PLACED) {
            status = OrderStatus.ACCEPTED;
        }
    }

    public void reject() {
        if (status == OrderStatus.PLACED || status == OrderStatus.PENDING) {
            status = OrderStatus.CANCELLED;
            for (OrderItem item : items) {
                item.setStatus(OrderItemStatus.REJECTED);
            }
        }
    }

    public void purge() {
        status = OrderStatus.PURGED;
        for (OrderItem item : items) {
            item.setStatus(OrderItemStatus.PURGED);
        }
    }

    public OrderItem findItem(int itemId) {
        for (OrderItem item : items) {
            if (item.getId() == itemId) {
                return item;
            }
        }
        return null;
    }

    public double subtotal() {
        double total = 0.0;
        for (OrderItem item : items) {
            total += item.getItem().getPrice();
        }
        return total;
    }
}

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public abstract class Payment {
    private final PaymentType type;
    private final List<OrderItem> meals;
    private final List<TipDenomination> tips;
    private boolean paid;

    protected Payment(PaymentType type) {
        this.type = type;
        this.meals = new ArrayList<>();
        this.tips = new ArrayList<>();
        this.paid = false;
    }

    public PaymentType getType() {
        return type;
    }

    public void addMeal(OrderItem item) {
        if (!meals.contains(item)) {
            meals.add(item);
        }
    }

    public boolean removeMeal(int itemId) {
        return meals.removeIf(meal -> meal.getId() == itemId);
    }

    public List<OrderItem> getMeals() {
        return Collections.unmodifiableList(meals);
    }

    public void addTip(TipDenomination tip) {
        tips.add(tip);
    }

    public boolean removeTip(double amount) {
        for (int i = 0; i < tips.size(); i++) {
            if (Double.compare(tips.get(i).getAmount(), amount) == 0) {
                tips.remove(i);
                return true;
            }
        }
        return false;
    }

    public List<TipDenomination> getTips() {
        return Collections.unmodifiableList(tips);
    }

    public double getAmount() {
        double total = 0.0;
        for (OrderItem meal : meals) {
            total += meal.getItem().getPrice();
        }
        for (TipDenomination tip : tips) {
            total += tip.getAmount();
        }
        return total;
    }

    public boolean isPaid() {
        return paid;
    }

    public void markPaid() {
        this.paid = true;
    }
}

import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

public class Account {
    private final int tableNumber;
    private boolean open;
    private boolean billMode;
    private final List<Order> orders;
    private CashPayment cashPayment;
    private final Map<String, BankCardPayment> bankCardPayments;

    public Account(int tableNumber) {
        this.tableNumber = tableNumber;
        this.open = false;
        this.billMode = false;
        this.orders = new ArrayList<>();
        this.bankCardPayments = new LinkedHashMap<>();
    }

    public int getTableNumber() {
        return tableNumber;
    }

    public boolean isOpen() {
        return open;
    }

    public void open() {
        this.open = true;
    }

    public void close() {
        this.open = false;
        this.billMode = false;
    }

    public boolean isBillMode() {
        return billMode;
    }

    public void engageBillMode() {
        if (!open) {
            throw new IllegalStateException("Cannot engage bill mode for a closed account");
        }
        billMode = true;
    }

    public void disengageBillMode() {
        billMode = false;
    }

    public void addOrder(Order order) {
        if (!open) {
            throw new IllegalStateException("Cannot add an order to a closed account");
        }
        orders.add(order);
    }

    public List<Order> getOrders() {
        return Collections.unmodifiableList(orders);
    }

    public Order getPendingOrder() {
        for (int i = orders.size() - 1; i >= 0; i--) {
            Order order = orders.get(i);
            if (order.getStatus() == OrderStatus.PENDING) {
                return order;
            }
        }
        return null;
    }

    public Order getLatestPlacedOrder() {
        for (int i = orders.size() - 1; i >= 0; i--) {
            Order order = orders.get(i);
            if (order.getStatus() == OrderStatus.PLACED) {
                return order;
            }
        }
        return null;
    }

    public OrderItem findOrderItem(int itemId) {
        for (Order order : orders) {
            OrderItem item = order.findItem(itemId);
            if (item != null) {
                return item;
            }
        }
        return null;
    }

    public CashPayment getCashPayment() {
        return cashPayment;
    }

    public CashPayment createCashPayment() {
        if (cashPayment == null) {
            cashPayment = new CashPayment();
        }
        return cashPayment;
    }

    public BankCardPayment getBankCardPayment(String customerId) {
        return bankCardPayments.get(customerId);
    }

    public BankCardPayment getOrCreateBankCardPayment(String customerId) {
        return bankCardPayments.computeIfAbsent(customerId, BankCardPayment::new);
    }

    public Map<String, BankCardPayment> getBankCardPayments() {
        return Collections.unmodifiableMap(bankCardPayments);
    }

    public Payment getOrCreatePayment(PaymentType type, String customerId) {
        if (type == PaymentType.CASH) {
            return createCashPayment();
        }
        return getOrCreateBankCardPayment(customerId);
    }

    public Payment getPayment(PaymentType type, String customerId) {
        if (type == PaymentType.CASH) {
            return cashPayment;
        }
        return bankCardPayments.get(customerId);
    }

    public void purgeAll() {
        for (Order order : orders) {
            order.purge();
        }
        orders.clear();
        cashPayment = null;
        bankCardPayments.clear();
        billMode = false;
    }
}

import java.util.ArrayList;

    private final Map<String, User> users;

    private final Map<String, List<Notification>> notifications;

    public RMOS() {
        this.menu = new Menu();
        this.users = new LinkedHashMap<>();
        this.tables = new LinkedHashMap<>();
        this.notifications = new LinkedHashMap<>();
        seedMenu();
    }

    private void seedMenu() {
        menu.addItem(new MenuItem(1, "Burger", 12.50));
        menu.addItem(new MenuItem(2, "Pasta", 10.00));
        menu.addItem(new MenuItem(3, "Salad", 8.00));
        menu.addItem(new MenuItem(4, "Steak", 20.00));
        menu.addDietaryRequirement(new DietaryRequirement("Gluten free"));
        menu.addDietaryRequirement(new DietaryRequirement("Vegetarian"));
        menu.addDietaryRequirement(new DietaryRequirement("Nut free"));
    }

    public Menu getMenu() {
        return menu;
    }

    public void addUser(User user) {
        users.put(user.getUsername(), user);
    }

    public User login(String username, String password) {
        User user = users.get(username);
        if (user == null || !user.checkPassword(password)) {
            throw new SecurityException("Invalid credentials");
        }
        return user;
    }

    public void addTable(Table table) {
        tables.put(table.getNumber(), table);
    }

    public Table getTable(int tableNumber) {
        Table table = tables.get(tableNumber);
        if (table == null) {
            throw new IllegalArgumentException("Unknown table: " + tableNumber);
        }
        return table;
    }

    private Table getActiveTable(int tableNumber) {
        Table table = getTable(tableNumber);
        if (!table.isActive()) {
            throw new IllegalStateException("Table " + tableNumber + " is not active");
        }
        return table;
    }

    private void requireRole(User user, Role... roles) {
        if (user == null || !user.hasRole(roles)) {
            throw new SecurityException("Access denied");
        }
    }

    private void requireWaiter(User user) {
        requireRole(user, Role.WAITER, Role.SUPERVISOR);
    }

    private void requireChef(User user) {
        requireRole(user, Role.CHEF, Role.SUPERVISOR);
    }

    private void requireSupervisor(User user) {
        requireRole(user, Role.SUPERVISOR);
    }

    public void activateTable(int tableNumber, User staff) {
        requireWaiter(staff);
        getTable(tableNumber).activate(staff);
    }

    public void deactivateTable(int tableNumber, User staff) {
        requireWaiter(staff);
        getTable(tableNumber).deactivate();
    }

    private void notifyAssignedWaiter(Table table, NotificationType type, String message) {
        User waiter = table.getAssignedWaiter();
        if (waiter != null) {
            notifications.computeIfAbsent(waiter.getUsername(), key -> new ArrayList<>())
                    .add(new Notification(type, table.getNumber(), message));
        }
    }

    public List<Notification> getNotifications(User user) {
        List<Notification> list = notifications.get(user.getUsername());
        if (list == null) {
            return List.of();
        }
        return Collections.unmodifiableList(list);
    }

    public List<Notification> drainNotifications(User user) {
        List<Notification> list = notifications.get(user.getUsername());
        if (list == null) {
            return List.of();
        }
        List<Notification> copy = new ArrayList<>(list);
        list.clear();
        return copy;
    }

    public Order getPendingOrder(int tableNumber) {

    public void processCashPayment(int tableNumber, User staff, Register register, double tendered) {
        requireWaiter(staff);
        Account account = getActiveTable(tableNumber).getAccount();
        CashPayment payment = account.getCashPayment();
        if (payment == null) {
            throw new IllegalStateException("No cash payment to process");
        }
        if (!register.processCashPayment(payment.getAmount(), tendered)) {

    public void processBankCardPayment(int tableNumber, String customerId, User staff, Register register, String cardNumber) {
        requireWaiter(staff);
        Account account = getActiveTable(tableNumber).getAccount();
        BankCardPayment payment = account.getBankCardPayment(customerId);
        if (payment == null) {
            throw new IllegalStateException("No bankcard payment to process");
        }
        if (!register.processCardPayment(payment.getAmount(), cardNumber)) {
            throw new IllegalStateException("Card payment declined");
        }
        payment.markPaid();
    }

    public void chefAcceptItem(int tableNumber, int itemId, User chef) {
        requireChef(chef);
        OrderItem item = findOrderItem(tableNumber, itemId);
        if (item == null) {
            throw new IllegalArgumentException("Unknown item");
        }
        item.setStatus(OrderItemStatus.ACCEPTED);
    }

    public void chefRejectItem(int tableNumber, int itemId, User chef) {
        requireChef(chef);
        OrderItem item = findOrderItem(tableNumber, itemId);
        if (item == null) {
            throw new IllegalArgumentException("Unknown item");
        }
        item.setStatus(OrderItemStatus.REJECTED);
        notifyAssignedWaiter(getTable(tableNumber), NotificationType.ITEM_REJECTED,
                "Item rejected: " + item.getItem().getName() + " at table " + tableNumber);
    }

    public void chefMarkItemReady(int tableNumber, int itemId, User chef) {
        requireChef(chef);
        OrderItem item = findOrderItem(tableNumber, itemId);
        if (item == null) {
            throw new IllegalArgumentException("Unknown item");
        }
        item.setStatus(OrderItemStatus.READY);
        notifyAssignedWaiter(getTable(tableNumber), NotificationType.ITEM_READY,
                "Item ready: " + item.getItem().getName() + " at table " + tableNumber);
    }

    public void purgeMeal(int tableNumber, int itemId, User staff) {
        requireSupervisor(staff);
        OrderItem item = findOrderItem(tableNumber, itemId);
        if (item == null) {
            throw new IllegalArgumentException("Unknown item");
        }
        item.setStatus(OrderItemStatus.PURGED);
    }

    public void purgeTableAccount(int tableNumber, User staff) {
        requireSupervisor(staff);
        getTable(tableNumber).getAccount().purgeAll();
    }

    public void refundItem(int tableNumber, int itemId, User staff) {
        requireSupervisor(staff);
        OrderItem item = findOrderItem(tableNumber, itemId);
        if (item == null) {
            throw new IllegalArgumentException("Unknown item");
        }
        item.setStatus(OrderItemStatus.REFUNDED);
    }
}

import java.util.List;

public class SurfaceComputer implements TouchScreen {
    private final RMOS server;
    private final int tableNumber;
    private String currentCustomerId;
    private final Menu menu;
    private boolean menuEngaged;
    private int menuIndex;

    public SurfaceComputer(RMOS server, int tableNumber, String currentCustomerId) {
        this.server = server;
        this.tableNumber = tableNumber;
        this.currentCustomerId = currentCustomerId;
        this.menu = server.getMenu();
        this.menuEngaged = false;
        this.menuIndex = 0;
    }

    public int getTableNumber() {
        return tableNumber;
    }

    public String getCurrentCustomerId() {
        return currentCustomerId;
    }

    public void setCurrentCustomerId(String currentCustomerId) {
        this.currentCustomerId = currentCustomerId;
    }

    public boolean isMenuEngaged() {
        return menuEngaged;
    }

    @Override
    public void performGesture(Gesture gesture, String target) {
        if (gesture == Gesture.DOUBLE_TAP) {
            String normalized = target.toLowerCase();
            if (normalized.equals("menu")) {
                engageMenu();
            } else if (normalized.equals("dismiss")) {
                dismissMenu();
            } else if (normalized.equals("bill")) {
                engageBillMode();
            } else if (normalized.equals("assistance")) {
                requestWaiterAssistance();
            }
        }
    }

    public void engageMenu() {
        if (!server.getTable(tableNumber).isActive()) {
            throw new IllegalStateException("Table is not active");
        }
        menuEngaged = true;
    }

    public void dismissMenu() {
        Order pending = server.getPendingOrder(tableNumber);
        if (pending != null && !pending.isEmpty()) {
            throw new IllegalStateException("Cannot dismiss menu while a non-empty pending order exists");
        }
        menuEngaged = false;
    }

    private void requireEngaged() {
        if (!menuEngaged) {
            throw new IllegalStateException("Menu is not engaged");
        }
    }

    public Order createPendingOrder() {
        requireEngaged();
        return server.createPendingOrder(tableNumber);
    }

    public MenuItem navigateMenu(int delta) {
        List<MenuItem> items = menu.getItems();
        if (items.isEmpty()) {
            throw new IllegalStateException("Menu has no items");
        }
        menuIndex = Math.floorMod(menuIndex + delta, items.size());
        return items.get(menuIndex);
    }

    public OrderItem addItemToOrder(MenuItem item) {
        requireEngaged();
        return server.addItemToPendingOrder(tableNumber, item);
    }

    public boolean removeItemFromOrder(int itemId) {
        requireEngaged();
        return server.removeItemFromPendingOrder(tableNumber, itemId);
    }

    public void addDietaryRequirement(DietaryRequirement requirement) {
        requireEngaged();
        server.addDietaryRequirementToPendingOrder(tableNumber, requirement);
    }

    public void addWildcardDietaryRequirement(String description) {
        requireEngaged();
        server.addDietaryRequirementToPendingOrder(tableNumber, new DietaryRequirement(description, true));
    }

    public void placeOrder() {
        requireEngaged();
        server.placeOrder(tableNumber);
    }

    public void cancelOrder() {
        requireEngaged();
        server.cancelOrder(tableNumber);
    }

    public void requestWaiterAssistance() {
        server.requestAssistance(tableNumber);
    }

    public void engageBillMode() {
        server.engageBillMode(tableNumber);
    }

    public void disengageBillMode() {
        server.disengageBillMode(tableNumber);
    }

    public Payment getCashPaymentRepresentation() {
        return server.getOrCreatePayment(tableNumber, PaymentType.CASH, currentCustomerId);
    }

    public Payment getBankCardPaymentRepresentation(String customerId) {
        return server.getOrCreatePayment(tableNumber, PaymentType.BANKCARD, customerId);
    }

    public List<OrderItem> getMealsForBilling() {
        return server.getOrderedMeals(tableNumber);
    }

    public List<TipDenomination> getTipDenominations() {
        return server.getTipDenominations();
    }

    public void dragMealIntoPayment(int itemId, PaymentType type, String customerId) {
        server.dragMealToPayment(tableNumber, itemId, type, customerId);
    }

    public void dragMealOutOfPayment(int itemId, PaymentType type, String customerId) {
        server.dragMealOutOfPayment(tableNumber, itemId, type, customerId);
    }

    public void dragTipIntoPayment(double amount, PaymentType type, String customerId) {
        server.dragTipToPayment(tableNumber, amount, type, customerId);
    }

    public void dragTipOutOfPayment(double amount, PaymentType type, String customerId) {
        server.dragTipOutOfPayment(tableNumber, amount, type, customerId);
    }
}

            System.out.println("Waiter alert: " + notification.getMessage());
        }

        tablet.acceptOrder(1);

        display.login("carol", "pass");
        display.acceptOrderItem(1, orderedBurger.getId());
