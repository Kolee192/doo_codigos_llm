public enum Certification {
    CSM, CSPO, CSP, CST;

    public boolean isPractitioner() {
        return this == CSP || this == CSM || this == CSPO;
    }

    public boolean isTrainer() {
        return this == CST;
    }

    public String badge() {
        switch (this) {
            case CSP: return "CSP badge";
            case CST: return "CST badge";
            case CSM: return "CSM badge";
            case CSPO: return "CSPO badge";
            default: return "";
        }
    }
}

    }

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Article {
    private final int id;
    private final String title;
    private final String teaser;
    private final String body;
    private final User author;
    private final LocalDate startPublish;
    private final LocalDate oldDate;
    private final LocalDate stopPublish;
    private final boolean publicAccess;
    private final boolean featuredHome;
    private final int priority;
    private final List<Comment> comments = new ArrayList<>();

    public Article(int id, String title, String teaser, String body, User author,
                   LocalDate startPublish, LocalDate oldDate, LocalDate stopPublish,
                   boolean publicAccess, boolean featuredHome, int priority) {
        this.id = id;
        this.title = title;
        this.teaser = teaser;
        this.body = body;
        this.author = author;
        this.startPublish = startPublish;
        this.oldDate = oldDate;
        this.stopPublish = stopPublish;
        this.publicAccess = publicAccess;
        this.featuredHome = featuredHome;
        this.priority = priority;
    }

    public boolean isVisible(LocalDate date) {
        return !date.isBefore(startPublish) && !date.isAfter(stopPublish);
    }

    public boolean isOld(LocalDate date) {
        return !date.isBefore(oldDate) && !date.isAfter(stopPublish);
    }

    public boolean canRead(User viewer) {
        if (!isVisible(LocalDate.now())) return false;
        if (publicAccess) return true;
        return viewer != null && viewer.hasRole(Role.MEMBER) && viewer.isMemberActive();
    }

    public String badgeGraphic() {
        if (author.hasCertification(Certification.CST)) return Certification.CST.badge();
        if (author.hasCertification(Certification.CSP)) return Certification.CSP.badge();
        return "";
    }

    public void addComment(Comment comment) { comments.add(comment); }

    public int getId() { return id; }
    public String getTitle() { return title; }
    public String getTeaser() { return teaser; }
    public String getBody() { return body; }
    public User getAuthor() { return author; }
    public LocalDate getStartPublish() { return startPublish; }
    public LocalDate getOldDate() { return oldDate; }
    public LocalDate getStopPublish() { return stopPublish; }
    public boolean isPublicAccess() { return publicAccess; }
    public boolean isFeaturedHome() { return featuredHome; }
    public int getPriority() { return priority; }
    public List<Comment> getComments() { return Collections.unmodifiableList(comments); }
}

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;

public class JobAd {
    private final int id;
    private String title;
    private String description;
    private final String recruiterName;
    private final String contactEmail;
    private final LocalDate postedDate;
    private JobStatus status = JobStatus.PENDING;
    private LocalDate expiryDate;
    private int extensions = 0;
    private LocalDate lastReminderSent;

    public JobAd(int id, String title, String description, String recruiterName, String contactEmail) {
        this.id = id;
        this.title = title;
        this.description = description;
        this.recruiterName = recruiterName;
        this.contactEmail = contactEmail;
        this.postedDate = LocalDate.now();
        this.expiryDate = postedDate.plusDays(30);
    }

    public void approve(LocalDate date) {
        status = JobStatus.ACTIVE;
        expiryDate = date.plusDays(30);
    }

    public void reject() { status = JobStatus.REJECTED; }

    public void extend(LocalDate date) {
        if (status == JobStatus.ACTIVE || status == JobStatus.EXPIRED) {
            expiryDate = date.plusDays(30);
            status = JobStatus.ACTIVE;
            extensions++;
        }
    }

    public void expireIfNeeded(LocalDate date) {
        if (status == JobStatus.ACTIVE && date.isAfter(expiryDate)) {
            status = JobStatus.EXPIRED;
        }
    }

    public long daysUntilExpiry(LocalDate date) {
        return ChronoUnit.DAYS.between(date, expiryDate);
    }

    public boolean reminderDue(LocalDate date) {
        long days = daysUntilExpiry(date);
        return status == JobStatus.ACTIVE && days >= 0 && days <= 7 &&
               (lastReminderSent == null || lastReminderSent.isBefore(date.minusDays(6)));
    }

    public void markReminderSent(LocalDate date) { lastReminderSent = date; }

    public int getId() { return id; }
    public String getTitle() { return title; }
    public void setTitle(String title) { this.title = title; }
    public String getDescription() { return description; }
    public void setDescription(String description) { this.description = description; }
    public String getRecruiterName() { return recruiterName; }
    public String getContactEmail() { return contactEmail; }
    public LocalDate getPostedDate() { return postedDate; }
    public JobStatus getStatus() { return status; }
    public LocalDate getExpiryDate() { return expiryDate; }
    public int getExtensions() { return extensions; }
}

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class RssFeed {
    private final String title;
    private final List<String> items = new ArrayList<>();

    public RssFeed(String title) {
        this.title = title;
    }

    public void addItem(String item) { items.add(item); }

    public String toXml() {
        StringBuilder sb = new StringBuilder();
        sb.append("<rss><channel><title>").append(escape(title)).append("</title>");
        for (String item : items) {
            sb.append("<item><title>").append(escape(item)).append("</title></item>");
        }
        sb.append("</channel></rss>");
        return sb.toString();
    }

    private String escape(String text) {
        return text.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;");
    }

    public List<String> getItems() { return Collections.unmodifiableList(items); }
}

public class SearchCriteria {
    private String name;
    private Certification certification;
    private String location;
    private Role role;

    public SearchCriteria() {}

    public SearchCriteria(String name, Certification certification, String location, Role role) {
        this.name = name;
        this.certification = certification;
        this.location = location;
        this.role = role;
    }

    public boolean matches(User user) {
        if (name != null && !user.getName().toLowerCase().contains(name.toLowerCase())) return false;
        if (certification != null && !user.hasCertification(certification)) return false;
        if (location != null && (user.getProfile() == null || user.getProfile().getLocation() == null ||
                !user.getProfile().getLocation().toLowerCase().contains(location.toLowerCase()))) return false;
        if (role != null && !user.hasRole(role)) return false;
        return true;
    }

    public String getName() { return name; }
    public void setName(String name) { this.name = name; }
    public Certification getCertification() { return certification; }
    public void setCertification(Certification certification) { this.certification = certification; }
    public String getLocation() { return location; }
    public void setLocation(String location) { this.location = location; }
    public Role getRole() { return role; }
    public void setRole(Role role) { this.role = role; }
}

            if (app.getType() == ApplicationType.PRACTITIONER) {
                applicant.addRole(Role.PRACTITIONER);
                applicant.addCertification(new CertificationRecord(Certification.CSP, LocalDate.now()));

    public List<CourseEvent> upcomingCourses(CourseKind kind, LocalDate date, int page, int pageSize) {
        List<CourseEvent> filtered = new ArrayList<>();
        for (CourseEvent c : courses) {
            if (c.isUpcoming(date) && (kind == null || c.getKind() == kind)) filtered.add(c);
        }
        filtered.sort(Comparator.comparing(CourseEvent::getStartDate));
        int from = Math.max(0, page * pageSize);
        int to = Math.min(filtered.size(), from + pageSize);
        if (from >= filtered.size()) return Collections.emptyList();
        return new ArrayList<>(filtered.subList(from, to));
    }

    public double submitRating(CourseEvent course, String participantEmail, int score, String comment) {
        for (Rating r : ratings) {
            if (r.getCourseId() == course.getId() && r.getParticipantEmail().equalsIgnoreCase(participantEmail)) {
                throw new IllegalStateException("Duplicate survey submission");
            }
        }
        ratings.add(new Rating(course.getTrainer().getId(), course.getId(), participantEmail, score, comment));
        return getTrainerRating(course.getTrainer());
    }

    public double getTrainerRating(User trainer) {
        int sum = 0;
        int count = 0;
        for (Rating r : ratings) {
            if (r.getTrainerId() == trainer.getId()) {
                sum += r.getScore();
                count++;
            }
        }
        return count == 0 ? 0.0 : (double) sum / count;
    }
