// File: CountyPermitSystem.java
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class CountyPermitSystem {
    private final Map<String, Customer> customersById = new HashMap<>();
    private final Map<String, Customer> customersByUsername = new HashMap<>();
    private final Map<String, StaffMember> staffById = new HashMap<>();
    private final Map<String, Application> applications = new HashMap<>();
    private final Map<String, Property> properties = new HashMap<>();
    private final Map<String, Parcel> parcels = new HashMap<>();
    private final List<PreApplicationRequest> preAppRequests = new ArrayList<>();
    private final List<Proffer> proffers = new ArrayList<>();
    private final Map<String, Ordinance> ordinances = new HashMap<>();
    private final List<Meeting> meetings = new ArrayList<>();
    private final List<PlanningStudy> planningStudies = new ArrayList<>();
    private final List<PlanAmendment> planAmendments = new ArrayList<>();
    private final List<Report> reports = new ArrayList<>();
    private final List<ReviewAssignment> reviewAssignments = new ArrayList<>();
    private final List<StaffReport> staffReports = new ArrayList<>();
    private final List<Hearing> hearings = new ArrayList<>();
    private final List<Affidavit> affidavits = new ArrayList<>();
    private final List<Appeal> appeals = new ArrayList<>();
    private final List<PlanReview> planReviews = new ArrayList<>();
    private final List<CodeModification> codeModifications = new ArrayList<>();
    private final List<Permit> permits = new ArrayList<>();
    private final List<Bond> bonds = new ArrayList<>();
    private final List<Inspection> inspections = new ArrayList<>();
    private final List<Violation> violations = new ArrayList<>();
    private final List<CondoUnit> condoUnits = new ArrayList<>();
    private final List<Citation> citations = new ArrayList<>();
    private final List<ZoningInfoRequest> zoningInfoRequests = new ArrayList<>();
    private final List<Notification> notifications = new ArrayList<>();
    private final List<FormalResponse> formalResponses = new ArrayList<>();
    private final List<InternalNote> internalNotes = new ArrayList<>();
    private final List<Hold> holds = new ArrayList<>();
    private int idCounter = 0;

    private String nextId(String prefix) {
        return prefix + "-" + (++idCounter);
    }

    private Customer requireCustomer(String id) {
        Customer c = customersById.get(id);
        if (c == null) throw new IllegalArgumentException("Unknown customer " + id);
        return c;
    }

    private StaffMember requireStaff(String id) {
        StaffMember s = staffById.get(id);
        if (s == null) throw new IllegalArgumentException("Unknown staff " + id);
        return s;
    }

    private Application requireApplication(String id) {
        Application a = applications.get(id);
        if (a == null) throw new IllegalArgumentException("Unknown application " + id);
        return a;
    }

    public StaffMember addStaff(String name, UserRole role) {
        StaffMember s = new StaffMember(nextId("STAFF"), name, role, 0);
        staffById.put(s.id, s);
        return s;
    }

    public Customer createCustomerPortalAccount(String username, String password, String name, String email) {
        if (customersByUsername.containsKey(username)) throw new IllegalArgumentException("Username exists");
        Customer c = new Customer(nextId("CUST"), username, password, name, email);
        customersById.put(c.id, c);
        customersByUsername.put(c.username, c);
        return c;
    }

    public Customer login(String username, String password) {
        Customer c = customersByUsername.get(username);
        if (c != null && c.password.equals(password)) return c;
        return null;
    }

    public Property createProperty(String parcelId, String address, List<String> owners) {
        Property p = new Property(parcelId, address, new ArrayList<>(owners), new ArrayList<>(), new ArrayList<>(), new ArrayList<>(), null);
        properties.put(parcelId, p);
        return p;
    }

    public Parcel createParcel(String parcelId, String parentParcelId, String address, GISLayer gisLayer) {
        Parcel p = new Parcel(parcelId, parentParcelId, address, gisLayer);
        parcels.put(parcelId, p);
        return p;
    }

    public List<SearchResult> searchInformation(String query) {
        String q = query.toLowerCase();
        List<SearchResult> results = new ArrayList<>();
        for (Property p : properties.values()) {
            if ((p.parcelId + " " + p.address + " " + String.join(" ", p.owners)).toLowerCase().contains(q)) {
                results.add(new SearchResult("PROPERTY", p.parcelId, p.address));
            }
        }
        for (Ordinance o : ordinances.values()) {
            if ((o.title + " " + o.description).toLowerCase().contains(q)) {
                results.add(new SearchResult("ORDINANCE", o.id, o.title));
            }
        }
        for (ServiceType s : ServiceType.values()) {
            if (s.name().toLowerCase().contains(q) || s.toString().toLowerCase().contains(q)) {
                results.add(new SearchResult("SERVICE", s.name(), s.name()));
            }
        }
        return results;
    }

    public ServiceType selectService(ServiceType type) {
        return type;
    }

    public PreApplicationRequest requestPreApplicationAssistance(String customerId, ServiceType serviceType, String description) {
        requireCustomer(customerId);
        PreApplicationRequest r = new PreApplicationRequest(nextId("PREAPP"), customerId, serviceType, description, "SUBMITTED", LocalDateTime.now());
        preAppRequests.add(r);
        return r;
    }

    public Application submitApplication(String customerId, ServiceType serviceType, String description, String parcelId) {
        Customer c = requireCustomer(customerId);
        if (!properties.containsKey(parcelId)) {
            createProperty(parcelId, "Unknown address for " + parcelId, List.of(c.name));
        }
        Property p = properties.get(parcelId);
        double base = switch (serviceType) {
            case SITE_PLAN -> 500.0;
            case REZONING -> 750.0;
            case BUILDING_PERMIT -> 300.0;
            case DEVELOPMENT_PERMIT -> 600.0;
            case PLAN_REVIEW -> 200.0;
            default -> 100.0;
        };
        Application app = new Application(nextId("APP"), customerId, serviceType, description, parcelId, ApplicationStatus.SUBMITTED, new Fee(nextId("FEE"), base));
        applications.put(app.id, app);
        c.applicationIds.add(app.id);
        p.applicationIds.add(app.id);
        return app;
    }

    public SupportingDocument submitSupportingDocument(String applicationId, String name, String content) {
        Application app = requireApplication(applicationId);
        SupportingDocument doc = new SupportingDocument(nextId("DOC"), name, content, LocalDateTime.now());
        app.supportingDocuments.add(doc);
        if (app.status == ApplicationStatus.SUBMITTED) app.status = ApplicationStatus.DOCUMENTS_PENDING;
        return doc;
    }

    public Payment payFee(String applicationId, double amount) {
        Application app = requireApplication(applicationId);
        Payment p = new Payment(nextId("PAY"), amount, LocalDateTime.now(), "ONLINE");
        app.payments.add(p);
        app.fee.pay(amount);
        if (app.fee.isPaid()) app.status = ApplicationStatus.FEE_PAID;
        return p;
    }

    public ApplicationStatus checkStatus(String applicationId) {
        return requireApplication(applicationId).status;
    }

    public void modifyApplication(String applicationId, String newDescription) {
        requireApplication(applicationId).description = newDescription;
    }

    public void performCompletenessReview(String applicationId, String staffId, boolean complete, String comments) {
        Application app = requireApplication(applicationId);
        requireStaff(staffId);
        app.status = ApplicationStatus.COMPLETENESS_REVIEW;
        recordInternalNote(applicationId, staffId, "Completeness review: " + complete + ". " + comments);
        if (complete) app.status = ApplicationStatus.IN_REVIEW;
        else app.status = ApplicationStatus.INCOMPLETE;
    }

    public InternalNote recordInternalNote(String applicationId, String staffId, String note) {
        Application app = requireApplication(applicationId);
        requireStaff(staffId);
        InternalNote n = new InternalNote(nextId("NOTE"), staffId, note, LocalDateTime.now());
        app.notes.add(n);
        internalNotes.add(n);
        return n;
    }

    public ReviewAssignment assignApplicationForDetailedReview(String applicationId, String staffId) {
        Application app = requireApplication(applicationId);
        StaffMember s = requireStaff(staffId);
        ReviewAssignment r = new ReviewAssignment(nextId("REVIEW"), applicationId, staffId, LocalDateTime.now());
        reviewAssignments.add(r);
        app.status = ApplicationStatus.DETAILED_REVIEW;
        s.workload++;
        return r;
    }

    public Hold applyHold(String applicationId, String staffId, String reason) {
        Application app = requireApplication(applicationId);
        requireStaff(staffId);
        Hold h = new Hold(nextId("HOLD"), applicationId, staffId, reason, true, LocalDateTime.now());
        app.holds.add(h);
        holds.add(h);
        app.onHold = true;
        app.status = ApplicationStatus.ON_HOLD;
        return h;
    }

    public void removeHold(String applicationId, String staffId, String holdId) {
        requireStaff(staffId);
        Application app = requireApplication(applicationId);
        Hold target = null;
        for (Hold h : holds) {
            if (h.id.equals(holdId) && h.applicationId.equals(applicationId)) target = h;
        }
        if (target == null) throw new IllegalArgumentException("Unknown hold");
        target.active = false;
        boolean anyActive = false;
        for (Hold h : app.holds) if (h.active) anyActive = true;
        if (!anyActive) {
            app.onHold = false;
            if (app.status == ApplicationStatus.ON_HOLD) app.status = ApplicationStatus.IN_REVIEW;
        }
    }

    public FormalResponse sendFormalResponse(String applicationId, String staffId, String message) {
        Application app = requireApplication(applicationId);
        requireStaff(staffId);
        FormalResponse r = new FormalResponse(nextId("RESP"), applicationId, staffId, message, LocalDateTime.now());
        app.responses.add(r);
        formalResponses.add(r);
        return r;
    }

    public Proffer trackDraftProffer(String applicationId, String customerId, String statement) {
        Application app = requireApplication(applicationId);
        requireCustomer(customerId);
        Proffer p = new Proffer(nextId("PROFFER"), applicationId, customerId, statement, "DRAFT");
        app.proffers.add(p);
        proffers.add(p);
        return p;
    }

    public Proffer manageApprovedProffer(String profferId, boolean approved) {
        Proffer p = null;
        for (Proffer x : proffers) if (x.id.equals(profferId)) p = x;
        if (p == null) throw new IllegalArgumentException("Unknown proffer");
        p.status = approved ? "APPROVED" : "REJECTED";
        return p;
    }

    public Ordinance manageOrdinance(String ordinanceId, String title, String description) {
        Ordinance o = new Ordinance(ordinanceId, title, description);
        ordinances.put(o.id, o);
        return o;
    }

    public Meeting scheduleMeeting(LocalDateTime dateTime, List<String> attendees, String topic) {
        Meeting m = new Meeting(nextId("MEET"), topic, dateTime, attendees, "");
        meetings.add(m);
        return m;
    }

    public void denoteFinalAction(String applicationId, String staffId, String decision) {
        Application app = requireApplication(applicationId);
        requireStaff(staffId);
        app.status = ApplicationStatus.FINAL_ACTION;
        app.finalDecision = decision;
        recordInternalNote(applicationId, staffId, "Final action: " + decision);
    }

    public PlanningStudy conductPlanningStudy(String title, String location, String outcome) {
        PlanningStudy s = new PlanningStudy(nextId("STUDY"), title, location, outcome);
        planningStudies.add(s);
        return s;
    }

    public PlanAmendment trackPlanAmendment(String title, String location, String status) {
        PlanAmendment a = new PlanAmendment(nextId("AMEND"), title, location, status);
        planAmendments.add(a);
        return a;
    }

    public Report generatePlanningReport(String format, String filter) {
        StringBuilder sb = new StringBuilder("Planning Report\nFilter: " + filter + "\n");
        for (PlanningStudy s : planningStudies) sb.append("Study: ").append(s.title).append(" - ").append(s.outcome).append("\n");
        for (PlanAmendment a : planAmendments) sb.append("Amendment: ").append(a.title).append(" - ").append(a.status).append("\n");
        Report r = new Report(nextId("REPORT"), "PLANNING", format, sb.toString(), LocalDateTime.now());
        reports.add(r);
        return r;
    }

    public void evaluateComprehensivePlanConsistency(String applicationId, String staffId, boolean consistent, String comments) {
        Application app = requireApplication(applicationId);
        requireStaff(staffId);
        app.consistentWithComprehensivePlan = consistent;
        recordInternalNote(applicationId, staffId, "Comprehensive plan consistency: " + consistent + ". " + comments);
    }

    public boolean performStaffReview(String applicationId, String staffId, boolean compliant, String comments) {
        Application app = requireApplication(applicationId);
        requireStaff(staffId);
        app.staffReviewCompliant = compliant;
        recordInternalNote(applicationId, staffId, "Staff review compliant=" + compliant + ". " + comments);
        if (!compliant) app.status = ApplicationStatus.DENIED;
        else app.status = ApplicationStatus.IN_REVIEW;
        return compliant;
    }

    public ZoningInfoRequest processZoningInfoRequest(String customerId, String parcelId, String question) {
        requireCustomer(customerId);
        ZoningInfoRequest r = new ZoningInfoRequest(nextId("ZINFO"), customerId, parcelId, question, "", "OPEN");
        zoningInfoRequests.add(r);
        return r;
    }

    public FormalResponse respondToZoningInfoRequest(String requestId, String staffId, String response) {
        ZoningInfoRequest req = null;
        for (ZoningInfoRequest r : zoningInfoRequests) if (r.id.equals(requestId)) req = r;
        if (req == null) throw new IllegalArgumentException("Unknown zoning info request");
        req.response = response;
        req.status = "RESPONDED";
        FormalResponse fr = new FormalResponse(nextId("RESP"), "ZINFO-" + requestId, staffId, response, LocalDateTime.now());
        formalResponses.add(fr);
        return fr;
    }

    public Permit issueZoningPermit(String applicationId, String staffId) {
        Application app = requireApplication(applicationId);
        requireStaff(staffId);
        Permit p = new Permit(nextId("PERMIT"), applicationId, PermitType.ZONING, "ISSUED", staffId);
        permits.add(p);
        app.permitIds.add(p.id);
        return p;
    }

    public StaffReport developStaffReport(String applicationId, String staffId, String content) {
        requireApplication(applicationId);
        requireStaff(staffId);
        StaffReport r = new StaffReport(nextId("SREPORT"), applicationId, staffId, content, false);
        staffReports.add(r);
        return r;
    }

    public void publishStaffReport(String reportId) {
        for (StaffReport r : staffReports) {
            if (r.id.equals(reportId)) {
                r.published = true;
                return;
            }
        }
        throw new IllegalArgumentException("Unknown staff report");
    }

    public Hearing assignHearingDate(String applicationId, LocalDateTime dateTime, String body) {
        requireApplication(applicationId);
        Hearing h = new Hearing(nextId("HEAR"), applicationId, dateTime, body, HearingDecision.NO_DECISION, "", false);
        hearings.add(h);
        return h;
    }

    public Affidavit createAffidavit(String applicationId, String staffId, boolean signed, String details) {
        requireApplication(applicationId);
        requireStaff(staffId);
        Affidavit a = new Affidavit(nextId("AFF"), applicationId, staffId, signed, details);
        affidavits.add(a);
        return a;
    }

    public boolean satisfyHearingRequirements(String applicationId, String customerId) {
        Application app = requireApplication(applicationId);
        requireCustomer(customerId);
        app.hearingRequirementsSatisfied = true;
        return true;
    }

    public boolean prepareAndSupportHearing(String hearingId) {
        Hearing h = null;
        for (Hearing x : hearings) if (x.id.equals(hearingId)) h = x;
        if (h == null) throw new IllegalArgumentException("Unknown hearing");
        Application app = requireApplication(h.applicationId);
        boolean affidavitSigned = false;
        for (Affidavit a : affidavits) {
            if (a.applicationId.equals(app.id) && a.signed) affidavitSigned = true;
        }
        boolean ready = app.hearingRequirementsSatisfied && affidavitSigned;
        h.prepared = ready;
        return ready;
    }

    public Hearing captureHearingDecision(String hearingId, HearingDecision decision, String notes) {
        Hearing h = null;
        for (Hearing x : hearings) if (x.id.equals(hearingId)) h = x;
        if (h == null) throw new IllegalArgumentException("Unknown hearing");
        h.decision = decision;
        h.notes = notes;
        Application app = requireApplication(h.applicationId);
        if (decision == HearingDecision.APPROVED) app.status = ApplicationStatus.APPROVED;
        else if (decision == HearingDecision.DENIED) app.status = ApplicationStatus.DENIED;
        return h;
    }

    public Appeal processAppeal(String applicationId, String appellant, String reason) {
        requireApplication(applicationId);
        Appeal a = new Appeal(nextId("APPEAL"), applicationId, appellant, reason, "OPEN", "");
        appeals.add(a);
        return a;
    }

    public Appeal decideAppeal(String appealId, String outcome, String notes) {
        for (Appeal a : appeals) {
            if (a.id.equals(appealId)) {
                a.outcome = outcome;
                a.notes = notes;
                return a;
            }
        }
        throw new IllegalArgumentException("Unknown appeal");
    }

    public PlanReview assignPlanForReview(String applicationId, String reviewerId) {
        Application app = requireApplication(applicationId);
        StaffMember s = requireStaff(reviewerId);
        PlanReview pr = new PlanReview(nextId("PLANREV"), applicationId, reviewerId, PlanReviewStatus.ASSIGNED, "", "", "");
        planReviews.add(pr);
        app.planReviewIds.add(pr.id);
        s.workload++;
        return pr;
    }

    public int managePlanReviewerWorkload(String reviewerId) {
        return requireStaff(reviewerId).workload;
    }

    public Meeting requestPlanReviewMeeting(String applicationId, String customerId, LocalDateTime dateTime) {
        Application app = requireApplication(applicationId);
        Customer c = requireCustomer(customerId);
        List<String> attendees = new ArrayList<>();
        attendees.add(c.name);
        attendees.add("Plan Review Staff");
        Meeting m = new Meeting(nextId("MEET"), "Plan review meeting for " + applicationId, dateTime, attendees, "");
        meetings.add(m);
        return m;
    }

    public Meeting conductPlanReviewMeeting(String meetingId, String outcome) {
        for (Meeting m : meetings) {
            if (m.id.equals(meetingId)) {
                m.outcome = outcome;
                return m;
            }
        }
        throw new IllegalArgumentException("Unknown meeting");
    }

    public PlanReview reviewPlans(String planReviewId, String staffId, boolean approved, String comments, String conditions) {
        PlanReview pr = null;
        for (PlanReview x : planReviews) if (x.id.equals(planReviewId)) pr = x;
        if (pr == null) throw new IllegalArgumentException("Unknown plan review");
        requireStaff(staffId);
        pr.status = approved ? PlanReviewStatus.APPROVED : PlanReviewStatus.DENIED;
        pr.comments = comments;
        pr.conditions = conditions;
        Application app = requireApplication(pr.applicationId);
        recordInternalNote(app.id, staffId, "Plan review " + (approved ? "approved" : "denied") + ". " + comments + " " + conditions);
        return pr;
    }

    public PlanReview resubmitPlans(String planReviewId, String customerId, String revisedPlan) {
        requireCustomer(customerId);
        PlanReview pr = null;
        for (PlanReview x : planReviews) if (x.id.equals(planReviewId)) pr = x;
        if (pr == null) throw new IllegalArgumentException("Unknown plan review");
        pr.status = PlanReviewStatus.RESUBMITTED;
        pr.revisedPlan = revisedPlan;
        return pr;
    }

    public CodeModification reviewCodeModification(String applicationId, String staffId, String code, String description, boolean approved) {
        requireApplication(applicationId);
        requireStaff(staffId);
        CodeModification cm = new CodeModification(nextId("CODEMOD"), applicationId, staffId, code, description, approved);
        codeModifications.add(cm);
        return cm;
    }

    public String trackCompletionOfRequiredPlanReviews(String applicationId) {
        requireApplication(applicationId);
        int total = 0;
        int completed = 0;
        for (PlanReview pr : planReviews) {
            if (pr.applicationId.equals(applicationId)) {
                total++;
                if (pr.status == PlanReviewStatus.APPROVED || pr.status == PlanReviewStatus.COMPLETED) completed++;
            }
        }
        return "Completed " + completed + " of " + total + " required plan reviews for " + applicationId;
    }

    public Permit issueDevelopmentPermit(String applicationId, String staffId) {
        Application app = requireApplication(applicationId);
        requireStaff(staffId);
        Permit p = new Permit(nextId("PERMIT"), applicationId, PermitType.DEVELOPMENT, "ISSUED", staffId);
        permits.add(p);
        app.permitIds.add(p.id);
        return p;
    }

    public Bond manageBond(String applicationId, double amount, String type) {
        Application app = requireApplication(applicationId);
        Bond b = new Bond(nextId("BOND"), applicationId, amount, type, BondStatus.POSTED);
        bonds.add(b);
        app.bondIds.add(b.id);
        return b;
    }

    public Bond releaseBond(String bondId) {
        for (Bond b : bonds) {
            if (b.id.equals(bondId)) {
                b.status = BondStatus.RELEASED;
                return b;
            }
        }
        throw new IllegalArgumentException("Unknown bond");
    }

    public void completeBuildingDevelopmentProject(String applicationId, String customerId) {
        Application app = requireApplication(applicationId);
        requireCustomer(customerId);
        app.status = ApplicationStatus.COMPLETED;
        for (Bond b : bonds) {
            if (b.applicationId.equals(applicationId) && b.status == BondStatus.POSTED) b.status = BondStatus.RELEASED;
        }
    }

    public Inspection requestInspection(String customerId, String applicationId, LocalDate date, String description) {
        requireCustomer(customerId);

        return i;
    }

    public Inspection createInspection(String staffId, String applicationId, LocalDate date, String inspectorId) {
        requireStaff(staffId);
        Application app = requireApplication(applicationId);
        requireStaff(inspectorId);
        Inspection i = new Inspection(nextId("INSP"), applicationId, null, inspectorId, date, "Scheduled by staff", InspectionStatus.SCHEDULED, null, "");
        inspections.add(i);
        app.inspectionIds.add(i.id);
        return i;
    }

    public Inspection assignInspection(String inspectionId, String inspectorId) {
        StaffMember s = requireStaff(inspectorId);
        for (Inspection i : inspections) {
            if (i.id.equals(inspectionId)) {
                i.assignedInspectorId = inspectorId;
                i.status = InspectionStatus.SCHEDULED;
                s.workload++;
                return i;
            }
        }
        throw new IllegalArgumentException("Unknown inspection");
    }

    public int manageInspectorWorkload(String inspectorId) {
        return requireStaff(inspectorId).workload;
    }

    public Inspection conductInspection(String inspectionId, String inspectorId, boolean passed, String notes) {
        requireStaff(inspectorId);
        for (Inspection i : inspections) {
            if (i.id.equals(inspectionId)) {
                i.status = InspectionStatus.COMPLETED;
                i.passed = passed;
                i.notes = notes;
                return i;
            }
        }
        throw new IllegalArgumentException("Unknown inspection");
    }

        notifications.add(n);
        return n;
    }

    public Report createCustomQueryReport(String query) {
        Report r = new Report(nextId("REPORT"), "CUSTOM_QUERY", "TEXT", "Query: " + query + "\nGenerated: " + LocalDateTime.now(), LocalDateTime.now());
        reports.add(r);
        return r;
    }

    public CondoUnit createCondoUnit(String childParcelId, String address, String unitNumber) {
        CondoUnit u = new CondoUnit(nextId("CONDO"), childParcelId, address, unitNumber);
        condoUnits.add(u);
        Parcel p = parcels.get(childParcelId);
        if (p != null) p.units.add(u);
        return u;
    }

    public Parcel updateGISPropertyManagement(String parcelId, GISLayer gisLayer) {
        Parcel p = parcels.get(parcelId);
        if (p == null) {
            p = new Parcel(parcelId, null, "Unknown address", gisLayer);
            parcels.put(parcelId, p);
        } else {
            p.gisLayer = gisLayer;
        }
        return p;
    }

    public Citation sendOrPostCitation(String violationId, String staffId) {
        requireStaff(staffId);
        Violation v = null;
        for (Violation x : violations) if (x.id.equals(violationId)) v = x;
        if (v == null) throw new IllegalArgumentException("Unknown violation");
        Citation c = new Citation(nextId("CITATION"), violationId, staffId, true, LocalDateTime.now());
        citations.add(c);
        v.status = "CITED";
        return c;
    }
