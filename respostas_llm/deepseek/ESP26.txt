// File: DatabaseCredentials.java
public final class DatabaseCredentials {
    private final MasterPassword masterPassword;
    private final KeyFile keyFile;

    public DatabaseCredentials(MasterPassword masterPassword, KeyFile keyFile) {
        if (masterPassword == null && keyFile == null) {
            throw new IllegalArgumentException("At least one unlock credential is required.");
        }
        this.masterPassword = masterPassword;
        this.keyFile = keyFile;
    }

    public boolean requiresMasterPassword() {
        return masterPassword != null;
    }

    public boolean requiresKeyFile() {
        return keyFile != null;
    }

    public boolean isComposite() {
        return requiresMasterPassword() && requiresKeyFile();
    }

    public boolean unlock(String providedPassword, KeyFile providedKeyFile) {
        if (requiresMasterPassword()) {
            if (providedPassword == null || !masterPassword.matches(providedPassword)) {
                return false;
            }
        }

        if (requiresKeyFile()) {
            if (providedKeyFile == null || !providedKeyFile.isPresent()) {
                return false;
            }
            if (!keyFile.getPath().toAbsolutePath().normalize()
                    .equals(providedKeyFile.getPath().toAbsolutePath().normalize())) {
                return false;
            }
        }

        return true;
    }
}

// File: DatabaseFileValidator.java
import java.nio.file.Files;
import java.nio.file.Path;

public final class DatabaseFileValidator {
    private DatabaseFileValidator() {
    }

    public static void validate(Path file) {
        if (file == null) {
            throw new IllegalArgumentException("Database file cannot be null.");
        }

        Path fileNamePath = file.getFileName();
        if (fileNamePath == null) {
            throw new IllegalArgumentException("Database file must have a name.");
        }

        String fileName = fileNamePath.toString();
        String lowerName = fileName.toLowerCase();
        if (!lowerName.endsWith(".kdb")) {
            throw new IllegalArgumentException("Database file must be of type name.kdb.");
        }
        if (fileName.length() <= ".kdb".length()) {
            throw new IllegalArgumentException("Database file must have a non-empty name before .kdb.");
        }
        if (Files.exists(file) && !Files.isReadable(file)) {
            throw new IllegalArgumentException("Database file must be readable.");
        }
    }
}

import java.util.ArrayList;
import java.util.List;

public final class DatabaseManager {
    private final List<PasswordDatabase> databases = new ArrayList<>();
    private PasswordDatabase selectedDatabase;

    public void addDatabase(PasswordDatabase database) {
        if (database == null) {
            throw new IllegalArgumentException("Database cannot be null.");
        }

        int existingIndex = -1;
        for (int i = 0; i < databases.size(); i++) {
            if (databases.get(i).getName().equals(database.getName())) {
                existingIndex = i;
                break;
            }
        }

        if (existingIndex >= 0) {
            PasswordDatabase previous = databases.get(existingIndex);
            databases.set(existingIndex, database);
            if (selectedDatabase == previous) {
                selectedDatabase = database;
            }
        } else {
            databases.add(database);
        }

        if (selectedDatabase == null) {
            selectedDatabase = database;
        }
    }

    public void selectDatabase(String name) {
        for (PasswordDatabase database : databases) {
            if (database.getName().equals(name)) {
                selectedDatabase = database;
                return;
            }
        }
        throw new IllegalArgumentException("Database not found: " + name);
    }

    public PasswordDatabase getSelectedDatabase() {
        return selectedDatabase;
    }

    public List<PasswordDatabase> getDatabases() {
        return new ArrayList<>(databases);
    }
}

import java.util.ArrayList;
import java.util.List;

public class Group {
    private String name;
    private final Group parent;
    private final List<Group> subgroups = new ArrayList<>();
    private final List<Entry> entries = new ArrayList<>();

    public Group(String name, Group parent) {
        validateName(name);
        this.name = name;
        this.parent = parent;
    }

    private static void validateName(String name) {
        if (name == null || name.trim().isEmpty()) {
            throw new IllegalArgumentException("A name is required.");
        }
    }

    public Group addSubgroup(String name) {
        validateName(name);
        Group subgroup = new Group(name, this);
        subgroups.add(subgroup);
        return subgroup;
    }

    public void rename(String newName) {
        validateName(newName);
        this.name = newName;
    }

    public void addEntry(Entry entry) {
        if (entry == null) {
            throw new IllegalArgumentException("Entry cannot be null.");
        }
        if (entry.getGroup() != this) {
            throw new IllegalArgumentException("Entry must belong to this group.");
        }
        entries.add(entry);
    }

    public void removeEntry(Entry entry) {
        entries.remove(entry);
    }

    public String getName() {
        return name;
    }

    public Group getParent() {
        return parent;
    }

    public List<Entry> getEntries() {
        return new ArrayList<>(entries);
    }

    public List<Group> getSubgroups() {
        return new ArrayList<>(subgroups);
    }

    public List<Entry> search(String word) {
        if (word == null || word.trim().isEmpty()) {
            throw new IllegalArgumentException("A search word must be placed in the find field.");
        }
        List<Entry> result = new ArrayList<>();
        collectMatches(word, result);
        return result;
    }

    private void collectMatches(String word, List<Entry> result) {
        for (Entry entry : entries) {
            if (entry.matches(word)) {
                result.add(entry);
            }
        }
        for (Group subgroup : subgroups) {
            subgroup.collectMatches(word, result);
        }
    }

    public List<Entry> getAllEntriesRecursive() {
        List<Entry> result = new ArrayList<>(entries);
        for (Group subgroup : subgroups) {
            result.addAll(subgroup.getAllEntriesRecursive());
        }
        return result;
    }

    public boolean containsEntryRecursive(Entry entry) {
        if (entries.contains(entry)) {
            return true;
        }
        for (Group subgroup : subgroups) {
            if (subgroup.containsEntryRecursive(entry)) {
                return true;
            }
        }
        return false;
    }

    public boolean containsGroupRecursive(Group target) {
        if (this == target) {
            return true;
        }
        for (Group subgroup : subgroups) {
            if (subgroup.containsGroupRecursive(target)) {
                return true;
            }
        }
        return false;
    }
}

public class Entry {
    private final Group group;
    private String title;
    private String username;
    private String url;
    private String password;
    private String notes;

    public Entry(
            Group group,
            String title,
            String username,
            String url,
            String password,
            String repeatPassword,
            String notes
    ) {
        if (group == null) {
            throw new IllegalArgumentException("An entry must belong to a group to be created.");
        }
        validatePasswords(password, repeatPassword);
        this.group = group;
        this.title = safe(title);
        this.username = safe(username);
        this.url = safe(url);
        this.password = password == null ? "" : password;
        this.notes = safe(notes);
    }

    private static String safe(String value) {
        return value == null ? "" : value;
    }

    private static void validatePasswords(String password, String repeatPassword) {
        String first = password == null ? "" : password;
        String second = repeatPassword == null ? "" : repeatPassword;

        if (!first.isEmpty() && second.isEmpty()) {
            throw new IllegalArgumentException("Repeat password is required when password field is completed.");
        }
        if (!first.equals(second)) {
            throw new IllegalArgumentException("Password field and repeat password field must be identical.");
        }
    }

    public Group getGroup() {
        return group;
    }

    public String getTitle() {
        return title;
    }

    public void setTitle(String title) {
        this.title = safe(title);
    }

    public String getUsername() {
        return username;
    }

    public void setUsername(String username) {
        this.username = safe(username);
    }

    public String getUrl() {
        return url;
    }

    public void setUrl(String url) {
        this.url = safe(url);
    }

    public String getPassword() {
        return password;
    }

    public void updatePassword(String newPassword, String repeatPassword) {
        validatePasswords(newPassword, repeatPassword);
        this.password = newPassword == null ? "" : newPassword;
    }

    public String getNotes() {
        return notes;
    }

    public void setNotes(String notes) {
        this.notes = safe(notes);
    }

    public boolean matches(String word) {
        if (word == null || word.trim().isEmpty()) {
            throw new IllegalArgumentException("A search word is required.");
        }
        String lowerWord = word.toLowerCase();
        return contains(title, lowerWord)
                || contains(username, lowerWord)
                || contains(url, lowerWord)
                || contains(password, lowerWord)
                || contains(notes, lowerWord);
    }

    private static boolean contains(String value, String lowerWord) {
        return value != null && value.toLowerCase().contains(lowerWord);
    }

    public Entry duplicate() {
        return new Entry(group, title, username, url, password, password, notes);
    }

    @Override
    public String toString() {
        return "Entry{title='" + title + '\''
                + ", username='" + username + '\''
                + ", url='" + url + '\''
                + ", password='" + password + '\''
                + ", notes='" + notes + '\''
                + '}';
    }
}

// File: AutoType.java
public final class AutoType {
    public static final String PREFIX = "Auto-Type:";
    private final String sequence;

    public AutoType(String sequence) {
        if (sequence == null || !sequence.startsWith(PREFIX)) {
            throw new IllegalArgumentException("The prefix \"Auto-Type:\" is required in front of each sequence.");
        }

        String keystrokes = sequence.substring(PREFIX.length());
        if (keystrokes.length() > 59) {
            throw new IllegalArgumentException("Sequence length must not be more than 59 characters.");
        }
        this.sequence = sequence;
    }

    public static AutoType firstFromNote(String note) {
        if (note == null) {
            return null;
        }
        String[] lines = note.split("\\R");
        for (String line : lines) {
            if (line.startsWith(PREFIX)) {
                return new AutoType(line);
            }
        }
        return null;
    }

    public String getSequence() {
        return sequence;
    }

    public String getKeystrokes() {
        return sequence.substring(PREFIX.length());
    }
}

        List<CommandLineToken> tokens = tokenize(line);
        List<CommandLineToken> databaseTokens = new ArrayList<>();

        for (CommandLineToken token : tokens) {
            if (token.getText().toLowerCase().endsWith(".kdb")) {
                if (token.getText().contains(" ") && !token.isQuoted()) {
                    throw new IllegalArgumentException("A path containing spaces must be enclosed in quotes.");
                }
                databaseTokens.add(token);
            }
        }

        if (databaseTokens.size() > 1) {
            throw new IllegalArgumentException("Only one database file is allowed in command line options.");
        }

        if (databaseTokens.size() == 1 && tokens.size() > 1 && !databaseTokens.get(0).isQuoted()) {
            for (CommandLineToken token : tokens) {
                if (token != databaseTokens.get(0) && !token.getText().startsWith("-")) {
                    throw new IllegalArgumentException("A path containing spaces must be enclosed in quotes.");
                }
            }
        }

        String databaseFile = databaseTokens.isEmpty() ? null : databaseTokens.get(0).getText();
        return new CommandLineOptions(databaseFile);
    }

    public String getDatabaseFile() {
        return databaseFile;
    }

    private static List<CommandLineToken> tokenize(String commandLine) {
        List<CommandLineToken> tokens = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        boolean inQuotes = false;
        boolean tokenWasQuoted = false;

        for (int i = 0; i < commandLine.length(); i++) {
            char c = commandLine.charAt(i);
            if (c == '"') {
                inQuotes = !inQuotes;
                tokenWasQuoted = true;
            } else if (Character.isWhitespace(c) && !inQuotes) {
                if (current.length() > 0 || tokenWasQuoted) {
                    tokens.add(new CommandLineToken(current.toString(), tokenWasQuoted));
                    current.setLength(0);
                    tokenWasQuoted = false;
                }
            } else {
                current.append(c);
            }
        }

        if (inQuotes) {
            throw new IllegalArgumentException("Unclosed quote in command line options.");
        }

        if (current.length() > 0 || tokenWasQuoted) {
            tokens.add(new CommandLineToken(current.toString(), tokenWasQuoted));
        }

        return tokens;
    }
}
