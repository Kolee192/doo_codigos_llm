    public synchronized User signUpWithOAuth(String email, String name, String provider) {

    private User signUp(String email, String name, String provider) {
        if (email == null || email.isBlank()) throw new IllegalArgumentException("Email is required");
        String normalizedEmail = email.trim().toLowerCase(Locale.ROOT);
        String existingId = userIdsByEmail.get(normalizedEmail);
        if (existingId != null) return users.get(existingId);

        User user = new User(id(), normalizedEmail, name == null ? normalizedEmail : name, provider);
        users.put(user.getId(), user);
        userIdsByEmail.put(normalizedEmail, user.getId());
        activity.add("User joined: " + user.getId());
        if (adminUserId == null) adminUserId = user.getId();

        String invitedPublisherId = invitations.remove(normalizedEmail);
        if (invitedPublisherId != null) {
            Publisher publisher = publishers.get(invitedPublisherId);
            if (publisher != null) publisher.members.put(user.getId(), Role.MEMBER);
        }
        return user;
    }

    public synchronized DataPackage publish(String userId, String packageId) {
        DataPackage dataPackage = requirePackage(packageId);
        requirePublisherMember(userId, dataPackage.getPublisherId());
        ensureNotPermanentlyDeleted(dataPackage);
        ValidationResult validation = validate(dataPackage);
        if (!validation.isValid()) {
            throw new IllegalStateException("Package validation failed: " + validation.getErrors());
        }
        dataPackage.setDeleted(false);
        String version = nextVersion(dataPackage);
        dataPackage.addVersion(version);
        activity.add("Package published: " + packageId + "@" + version);
        notifySubscribers(dataPackage, "Package updated: " + packageId + "@" + version);
        return dataPackage;
    }

    public synchronized ValidationResult validate(String userId, String packageId) {
        DataPackage dataPackage = requirePackage(packageId);
        requirePublisherMember(userId, dataPackage.getPublisherId());
        return validate(dataPackage);
    }

    private ValidationResult validate(DataPackage dataPackage) {
        ValidationResult result = new ValidationResult();
        if (dataPackage.getTitle() == null || dataPackage.getTitle().isBlank()) {
            result.addError("Package title is missing");
        }
        if (dataPackage.mutableResources().isEmpty()) {
            result.addError("At least one resource is required");
        }
        for (PackageResource resource : dataPackage.mutableResources()) {
            if (resource.getData().length == 0) result.addError("Resource is empty: " + resource.getName());
            if (resource.getName().contains("..")) result.addError("Invalid resource path: " + resource.getName());
        }
        Object declaredName = dataPackage.getDescriptor().get("name");
        if (declaredName != null && !(declaredName instanceof String)) {
            result.addError("Descriptor name must be text");
        }
        if (dataPackage.getDescription().isBlank()) result.addWarning("Package description is empty");
        return result;
    }

    public synchronized PackageSnapshot tag(String userId, String packageId, String tag,
                                            boolean overwrite) {
        DataPackage dataPackage = requirePackage(packageId);
        requirePublisherMember(userId, dataPackage.getPublisherId());
        ensureNotPermanentlyDeleted(dataPackage);
        if (tag == null || tag.isBlank()) throw new IllegalArgumentException("Tag is required");
        PackageSnapshot previous = dataPackage.getTags().get(tag);
        if (previous != null && !overwrite) {
            throw new IllegalStateException("Tag already exists; pass overwrite=true to replace it");
        }
        String version = dataPackage.getCurrentVersion();
        if (version == null) throw new IllegalStateException("Publish the package before tagging it");
        PackageSnapshot snapshot = dataPackage.getVersions().get(version);
        PackageSnapshot tagged = new PackageSnapshot(snapshot.getVersion(), snapshot.getDescriptor(),
                snapshot.getResources());
        dataPackage.mutableTags().put(tag, tagged);
        return tagged;
    }

    public synchronized List<DataPackage> search(String query) {
        countApi("search");
        String term = query == null ? "" : query.toLowerCase(Locale.ROOT);
        List<DataPackage> result = new ArrayList<>();
        for (DataPackage dataPackage : packages.values()) {
            if (!isDiscoverable(dataPackage)) continue;
            String text = dataPackage.getTitle() + " " + dataPackage.getDescription() + " "
                    + dataPackage.getDescriptor();
            if (text.toLowerCase(Locale.ROOT).contains(term)) result.add(dataPackage);
        }
        result.sort(Comparator.comparing(DataPackage::getTitle, String.CASE_INSENSITIVE_ORDER));
        return Collections.unmodifiableList(result);
    }

    public synchronized byte[] downloadPackageZip(String userId, String packageId) {
        DataPackage dataPackage = requireVisiblePackage(userId, packageId);
        try {
            ByteArrayOutputStream bytes = new ByteArrayOutputStream();
            try (ZipOutputStream zip = new ZipOutputStream(bytes)) {
                String descriptor = descriptorJson(dataPackage);
                zip.putNextEntry(new ZipEntry("datapackage.json"));
                zip.write(descriptor.getBytes(StandardCharsets.UTF_8));
                zip.closeEntry();
                for (PackageResource resource : dataPackage.mutableResources()) {
                    zip.putNextEntry(new ZipEntry(resource.getName()));
                    zip.write(resource.getData());
                    zip.closeEntry();
                }
            }
            dataPackage.incrementDownloads();
            countApi("download");
            return bytes.toByteArray();
        } catch (IOException exception) {
            throw new IllegalStateException("Unable to create package archive", exception);
        }
    }

    public synchronized byte[] downloadAsSql(String userId, String packageId) {
        DataPackage dataPackage = requireVisiblePackage(userId, packageId);
        StringBuilder sql = new StringBuilder("-- SQLite-compatible import script\n");
        for (PackageResource resource : dataPackage.mutableResources()) {
            String table = sqlIdentifier(resource.getName());
            sql.append("CREATE TABLE IF NOT EXISTS \"").append(table)
                    .append("\" (\"value\" TEXT);\n");
            String content = new String(resource.getData(), StandardCharsets.UTF_8);
            String[] lines = content.split("\\R");
            for (String line : lines) {
                sql.append("INSERT INTO \"").append(table).append("\" (\"value\") VALUES ('")
                        .append(line.replace("'", "''")).append("');\n");
            }
        }
        dataPackage.incrementDownloads();
        countApi("download");
        return sql.toString().getBytes(StandardCharsets.UTF_8);
    }

    public synchronized Map<String, String> preview(String userId, String packageId) {

            StringBuilder preview = new StringBuilder();

    public synchronized String compareVersions(String userId, String packageId,
                                               String firstVersion, String secondVersion) {
        PackageSnapshot first = getVersion(userId, packageId, firstVersion);
        PackageSnapshot second = getVersion(userId, packageId, secondVersion);
        Map<String, Object> a = first.getDescriptor();
        Map<String, Object> b = second.getDescriptor();
        List<String> changes = new ArrayList<>();
        for (String key : a.keySet()) {
            if (!b.containsKey(key)) changes.add("Removed descriptor field: " + key);
            else if (!Objects.equals(a.get(key), b.get(key))) changes.add("Changed descriptor field: " + key);
        }
        for (String key : b.keySet()) {
            if (!a.containsKey(key)) changes.add("Added descriptor field: " + key);
        }
        List<String> oldResources = resourceNames(first);
        List<String> newResources = resourceNames(second);
        for (String name : oldResources) if (!newResources.contains(name)) changes.add("Removed resource: " + name);
        for (String name : newResources) if (!oldResources.contains(name)) changes.add("Added resource: " + name);
        return changes.isEmpty() ? "No descriptor or resource changes" : String.join("\n", changes);
    }

    public synchronized void makeMember(String ownerUserId, String publisherId, String memberUserId) {
        requireOwner(ownerUserId, publisherId);
        if (!publishers.get(publisherId).members.containsKey(memberUserId)) {
            throw new IllegalArgumentException("User is not a publisher member");
        }
        long owners = publishers.get(publisherId).members.values().stream()
                .filter(role -> role == Role.OWNER).count();
        if (publishers.get(publisherId).roleOf(memberUserId) == Role.OWNER && owners <= 1) {
            throw new IllegalStateException("Publisher must have at least one owner");
        }
        publishers.get(publisherId).members.put(memberUserId, Role.MEMBER);
    }

    public synchronized Publisher getPublisher(String publisherId) {
        Publisher publisher = publishers.get(publisherId);
        if (publisher == null) throw new IllegalArgumentException("Unknown publisher: " + publisherId);
        return publisher;
    }

    public synchronized List<Publisher> searchPublishers(String query) {
        String term = query == null ? "" : query.toLowerCase(Locale.ROOT);
        List<Publisher> result = new ArrayList<>();
        for (Publisher publisher : publishers.values()) {
            if ((publisher.getName() + " " + publisher.getDescription()).toLowerCase(Locale.ROOT).contains(term)) {
                result.add(publisher);
            }
        }
        return Collections.unmodifiableList(result);
    }

    public synchronized List<DataPackage> packagesForPublisher(String publisherId, String query) {
        String term = query == null ? "" : query.toLowerCase(Locale.ROOT);
        List<DataPackage> result = new ArrayList<>();
        for (DataPackage dataPackage : packages.values()) {
            if (dataPackage.getPublisherId().equals(publisherId) && isDiscoverable(dataPackage)
                    && (dataPackage.getTitle() + " " + dataPackage.getDescription())
                    .toLowerCase(Locale.ROOT).contains(term)) result.add(dataPackage);
        }
        return Collections.unmodifiableList(result);
    }

    public synchronized void editProfile(String userId, String name, Map<String, String> profile) {
        User user = requireUser(userId);
        if (name != null && !name.isBlank()) user.setName(name);
        if (profile != null) profile.forEach(user::setProfileValue);
    }

    public synchronized void editPublisherProfile(String userId, String publisherId,
                                                  String name, String description) {
        requireOwner(userId, publisherId);
        Publisher publisher = publishers.get(publisherId);
        if (name != null && !name.isBlank()) publisher.setName(name);
        if (description != null) publisher.setDescription(description);
    }

    public synchronized void setPricingPlan(String userId, String publisherId, String planId) {
        requirePublisherMember(userId, publisherId);
        if (!plans.containsKey(planId)) throw new IllegalArgumentException("Unknown pricing plan");
        publishers.get(publisherId).setPricingPlanId(planId);
    }

    public synchronized void addPricingPlan(PricingPlan plan) {
        plans.put(plan.getId(), plan);
    }

    public synchronized List<PricingPlan> getPricingPlans() {
        return Collections.unmodifiableList(new ArrayList<>(plans.values()));
    }

    public synchronized void configureSite(String userId, String key, String value) {
        requireAdmin(userId);
        siteConfig.put(key, value);
    }

    public synchronized Map<String, String> getSiteConfig() {
        return Collections.unmodifiableMap(new LinkedHashMap<>(siteConfig));
    }

    public synchronized Map<String, Long> getAdminMetrics(String userId) {
        requireAdmin(userId);
        Map<String, Long> result = new LinkedHashMap<>();
        result.put("users", (long) users.size());
        result.put("publishers", (long) publishers.size());
        result.put("packages", (long) packages.size());
        result.put("downloads", packages.values().stream().mapToLong(DataPackage::getDownloads).sum());
        apiUsage.forEach((key, value) -> result.put("api." + key, value));
        return Collections.unmodifiableMap(result);
    }

    public synchronized List<String> getActivity() {
        return Collections.unmodifiableList(new ArrayList<>(activity));
    }

    public synchronized List<DataPackage> mostDownloaded(int limit) {
        return packages.values().stream().filter(this::isDiscoverable)
                .sorted(Comparator.comparingLong(DataPackage::getDownloads).reversed())
                .limit(Math.max(0, limit)).toList();
    }

    public synchronized Map<String, Object> nextSteps(String userId) {
        requireUser(userId);
        Map<String, Object> result = new LinkedHashMap<>();
        result.put("suggestions", List.of("Create a publisher profile", "Configure your client",
                "Create a data package", "Validate and publish your package"));
        result.put("publishers", publishers.values().stream()
                .filter(publisher -> publisher.roleOf(userId) != null).map(Publisher::getId).toList());
        return Collections.unmodifiableMap(result);
    }

    private void notifySubscribers(DataPackage dataPackage, String message) {
        // Subscription membership is recorded; the activity log provides the notification feed.
        activity.add(message);
    }

    private boolean isDiscoverable(DataPackage dataPackage) {
        return !dataPackage.isDeleted() && !dataPackage.isPermanentlyDeleted()
                && dataPackage.getVisibility() == Visibility.PUBLIC;
    }

    private boolean canAccess(DataPackage dataPackage, String userId) {
        if (dataPackage.isDeleted() || dataPackage.isPermanentlyDeleted()) return false;
        if (dataPackage.getVisibility() == Visibility.PUBLIC) return true;
        if (userId == null) return false;
        if (dataPackage.getVisibility() == Visibility.SELECTED_USERS
                && dataPackage.hasPermittedUser(userId)) return true;
        Publisher publisher = publishers.get(dataPackage.getPublisherId());
        return publisher != null && publisher.roleOf(userId) != null;
    }

    private DataPackage requireVisiblePackage(String userId, String packageId) {
        DataPackage dataPackage = requirePackage(packageId);
        if (!canAccess(dataPackage, userId)) throw new SecurityException("Package is not available to this user");
        return dataPackage;
    }

    private DataPackage requirePackage(String packageId) {
        DataPackage dataPackage = packages.get(packageId);
        if (dataPackage == null) throw new IllegalArgumentException("Unknown package: " + packageId);
        return dataPackage;
    }

    private User requireUser(String userId) {
        User user = users.get(userId);
        if (user == null) throw new IllegalArgumentException("Unknown user: " + userId);
        return user;
    }

    private void requirePublisherMember(String userId, String publisherId) {
        requireUser(userId);
        Publisher publisher = publishers.get(publisherId);
        if (publisher == null || publisher.roleOf(userId) == null) {
            throw new SecurityException("User is not a member of this publisher");
        }
    }

    private void requireOwner(String userId, String publisherId) {
        requirePublisherMember(userId, publisherId);
        if (publishers.get(publisherId).roleOf(userId) != Role.OWNER) {
            throw new SecurityException("Publisher owner role required");
        }
    }

    private void requireAdmin(String userId) {
        requireUser(userId);
        if (!userId.equals(adminUserId)) throw new SecurityException("Administrator role required");
    }

    private void ensureNotPermanentlyDeleted(DataPackage dataPackage) {
        if (dataPackage.isPermanentlyDeleted()) {
            throw new IllegalStateException("Package has been permanently deleted");
        }
    }

    private String nextVersion(DataPackage dataPackage) {
        int maximum = 0;
        for (String version : dataPackage.getVersions().keySet()) {
            try {
                maximum = Math.max(maximum, Integer.parseInt(version.replaceFirst("^v", "")));
            } catch (NumberFormatException ignored) {
                // Existing custom versions do not affect the sequential version counter.
            }
        }
        return "v" + (maximum + 1);
    }

    private void countApi(String operation) {
        apiUsage.merge(operation, 1L, Long::sum);
    }

    private static String id() { return UUID.randomUUID().toString(); }

    private static String slug(String value) {
        return value.toLowerCase(Locale.ROOT).replaceAll("[^a-z0-9]+", "-")
                .replaceAll("(^-|-$)", "");
    }

    private static String sqlIdentifier(String value) {
        return value.replaceAll("[^A-Za-z0-9_]", "_");
    }

    private static List<String> resourceNames(PackageSnapshot snapshot) {
        return snapshot.getResources().stream().map(PackageResource::getName).toList();
    }

    private static String descriptorJson(DataPackage dataPackage) {
        StringBuilder json = new StringBuilder("{\n");
        json.append("  \"name\": \"").append(escape(dataPackage.getId())).append("\",\n");
        json.append("  \"title\": \"").append(escape(dataPackage.getTitle())).append("\",\n");
        json.append("  \"description\": \"").append(escape(dataPackage.getDescription())).append("\",\n");
        json.append("  \"resources\": [");
        List<PackageResource> resources = dataPackage.getResources();
        for (int i = 0; i < resources.size(); i++) {
            if (i > 0) json.append(", ");
            json.append("{\"path\": \"").append(escape(resources.get(i).getName())).append("\"}");
        }
        return json.append("]\n}\n").toString();
    }

    private static String escape(String value) {
        return value.replace("\\", "\\\\").replace("\"", "\\\"")
                .replace("\n", "\\n").replace("\r", "\\r");
    }

    public synchronized User getUser(String userId) { return requireUser(userId); }
}

        System.out.println("Published " + dataPackage.getId() + " at " + dataPackage.getCurrentVersion());

    }
