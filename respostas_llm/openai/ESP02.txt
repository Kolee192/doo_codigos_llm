public record GeoPoint(double latitude, double longitude) {
    public GeoPoint {
        if (latitude < -90 || latitude > 90 || longitude < -180 || longitude > 180) {
            throw new IllegalArgumentException("Invalid geographic coordinates.");
        }
    }

    public double distanceKilometersTo(GeoPoint other) {
        final double earthRadiusKm = 6371.0;
        double lat1 = Math.toRadians(latitude);
        double lat2 = Math.toRadians(other.latitude);
        double deltaLat = Math.toRadians(other.latitude - latitude);
        double deltaLon = Math.toRadians(other.longitude - longitude);

        double a = Math.sin(deltaLat / 2) * Math.sin(deltaLat / 2)
                + Math.cos(lat1) * Math.cos(lat2)
                * Math.sin(deltaLon / 2) * Math.sin(deltaLon / 2);
        return earthRadiusKm * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    }
}

// File: TimeWindow.java
import java.time.LocalTime;

public record TimeWindow(LocalTime start, LocalTime end) {
    public TimeWindow {
        if (start == null || end == null || !start.isBefore(end)) {
            throw new IllegalArgumentException("A time window must have a start before its end.");
        }
    }

    public boolean contains(TimeWindow other) {
        return !other.start().isBefore(start) && !other.end().isAfter(end);
    }

    public boolean overlaps(TimeWindow other) {
        return start.isBefore(other.end()) && other.start().isBefore(end);
    }

    public TimeWindow intersection(TimeWindow other) {
        LocalTime intersectionStart = start.isAfter(other.start()) ? start : other.start();
        LocalTime intersectionEnd = end.isBefore(other.end()) ? end : other.end();
        return intersectionStart.isBefore(intersectionEnd)
                ? new TimeWindow(intersectionStart, intersectionEnd)
                : null;
    }
}

import javax.crypto.Cipher;
import javax.crypto.SecretKey;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;
import java.nio.ByteBuffer;
import java.nio.charset.StandardCharsets;
import java.security.GeneralSecurityException;
import java.security.SecureRandom;
import java.util.Base64;

public class DataProtector {
    private static final int NONCE_LENGTH = 12;
    private static final int TAG_LENGTH_BITS = 128;
    private final SecretKey key;
    private final SecureRandom random = new SecureRandom();

    public DataProtector(byte[] keyBytes) {
        if (keyBytes == null || (keyBytes.length != 16 && keyBytes.length != 24 && keyBytes.length != 32)) {
            throw new IllegalArgumentException("AES key must be 16, 24, or 32 bytes.");
        }
        key = new SecretKeySpec(keyBytes.clone(), "AES");
    }

    public static DataProtector create() {
        byte[] key = new byte[32];
        new SecureRandom().nextBytes(key);
        return new DataProtector(key);
    }

    public String encrypt(String plaintext) {
        try {
            byte[] nonce = new byte[NONCE_LENGTH];
            random.nextBytes(nonce);
            Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
            cipher.init(Cipher.ENCRYPT_MODE, key, new GCMParameterSpec(TAG_LENGTH_BITS, nonce));
            byte[] encrypted = cipher.doFinal(plaintext.getBytes(StandardCharsets.UTF_8));
            ByteBuffer output = ByteBuffer.allocate(nonce.length + encrypted.length);
            output.put(nonce).put(encrypted);
            return Base64.getEncoder().encodeToString(output.array());
        } catch (GeneralSecurityException e) {
            throw new IllegalStateException("Unable to encrypt data.", e);
        }
    }

    public String decrypt(String ciphertext) {
        try {
            byte[] input = Base64.getDecoder().decode(ciphertext);
            if (input.length <= NONCE_LENGTH) {
                throw new IllegalArgumentException("Invalid encrypted value.");
            }
            ByteBuffer buffer = ByteBuffer.wrap(input);
            byte[] nonce = new byte[NONCE_LENGTH];
            buffer.get(nonce);
            byte[] encrypted = new byte[buffer.remaining()];
            buffer.get(encrypted);
            Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
            cipher.init(Cipher.DECRYPT_MODE, key, new GCMParameterSpec(TAG_LENGTH_BITS, nonce));
            return new String(cipher.doFinal(encrypted), StandardCharsets.UTF_8);
        } catch (GeneralSecurityException | IllegalArgumentException e) {
            throw new IllegalArgumentException("Unable to decrypt data.", e);
        }
    }
}

    private void requireRole(UserAccount user, UserRole... roles) {
        if (user == null) {
            throw new SecurityException("Authentication required.");
        }
        for (UserRole role : roles) {
            if (user.getRole() == role) {
                return;
            }
        }
        throw new SecurityException("You do not have permission to perform this action.");

    private String hashPassword(String password) {
        if (password == null || password.length() < 8) {
            throw new IllegalArgumentException("Password must be at least eight characters.");
        }
        try {
            byte[] salt = new byte[16];
            new SecureRandom().nextBytes(salt);
            PBEKeySpec spec = new PBEKeySpec(password.toCharArray(), salt, 120_000, 256);
            byte[] hash = SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256")
                    .generateSecret(spec).getEncoded();
            spec.clearPassword();
            return Base64.getEncoder().encodeToString(salt) + ":"
                    + Base64.getEncoder().encodeToString(hash);
        } catch (GeneralSecurityException e) {
            throw new IllegalStateException("Unable to hash password.", e);
        }
    }

    public UserAccount createAccount(String username, String email, String password, String ipAddress) {
        if (username == null || username.isBlank() || email == null || !email.matches("^.+@.+\\..+$")) {
            throw new IllegalArgumentException("A username and valid email address are required.");
        }
        if (blockedIpAddresses.contains(ipAddress)) {
            throw new SecurityException("Requests from this IP address are blocked.");
        }
        boolean duplicate = users.values().stream()
                .anyMatch(user -> user.getUsername().equalsIgnoreCase(username));
        if (duplicate) {
            throw new IllegalArgumentException("Username is already in use.");
        }
        UserAccount account = new UserAccount(
                newId(), username, hashPassword(password), UserRole.USER,
                dataProtector.encrypt(email), ipAddress);
        users.put(account.getId(), account);
        return account;
    }

    public List<RecyclingFacility> findFacilities(String zipCode, double radiusKm, Set<WasteType> wasteTypes) {
        validateZipCode(zipCode);
        if (radiusKm < 0) {
            throw new IllegalArgumentException("Radius cannot be negative.");
        }
        GeoPoint origin = approximateZipLocation(zipCode);
        Set<WasteType> requested = wasteTypes == null ? Set.of() : Set.copyOf(wasteTypes);

        List<RecyclingFacility> results = facilities.values().stream()
                .filter(RecyclingFacility::isActive)
                .filter(facility -> requested.isEmpty() || facility.getAcceptedWaste().containsAll(requested))
                .filter(facility -> facility.getLocation() != null
                        && facility.getLocation().distanceKilometersTo(origin) <= radiusKm)
                .sorted(Comparator.comparingDouble(facility ->
                        facility.getLocation().distanceKilometersTo(origin)))
                .toList();
        results.forEach(RecyclingFacility::recordView);
        siteVisits++;
        return results;
    }

    private GeoPoint approximateZipLocation(String zipCode) {
        int value = Integer.parseInt(zipCode);
        double latitude = 24.0 + (value % 5000) / 100.0;
        double longitude = -125.0 + (value % 6000) / 100.0;
        return new GeoPoint(Math.min(latitude, 49.0), Math.min(longitude, -67.0));
    }

                .filter(user -> user.getRole() != UserRole.USER)
                .toList();
    }

    public Map<String, Object> getAdminDashboard(UserAccount administrator) {
        requireRole(administrator, UserRole.ADMIN, UserRole.SUPERUSER);
        Map<String, Object> result = new HashMap<>();
        result.put("activeFacilities", facilities.values().stream().filter(RecyclingFacility::isActive).count());
        result.put("users", users.size());
        result.put("siteVisits", siteVisits);
        result.put("feedbackCount", feedback.size());
        result.put("pickupRequests", pickupRequests.size());
        Map<String, Long> visitsByZip = new HashMap<>();
        facilities.values().stream().filter(RecyclingFacility::isActive)
                .forEach(facility -> visitsByZip.merge(facility.getZipCode(), facility.getViews(), Long::sum));
        result.put("facilityViewsByZip", Map.copyOf(visitsByZip));
        result.put("serviceStatus", "OPERATIONAL");
        return Map.copyOf(result);
    }

    public Map<String, Long> getFacilityUsageStats(UserAccount requester, String facilityId) {
        requireRole(requester, UserRole.SUPERUSER, UserRole.FACILITY_REPRESENTATIVE, UserRole.ADMIN);
        RecyclingFacility facility = getFacility(facilityId);
        return Map.of("views", facility.getViews(),
                "relatedPickupRequests", pickupRequests.stream()
                        .filter(request -> request.facilityId().equals(facilityId)).count());
    }

    public Collection<RecyclingFacility> getPublicFacilities() {
        siteVisits++;
        return facilities.values().stream()
                .filter(RecyclingFacility::isActive)
                .toList();
    }

    public void setActivityFee(UserAccount accountant, String activity, BigDecimal fee) {
        requireRole(accountant, UserRole.ACCOUNTANT, UserRole.ADMIN, UserRole.SUPERUSER);
        if (fee == null || fee.signum() < 0) {
            throw new IllegalArgumentException("Fee must be non-negative.");
        }
        activityFees.put(activity, fee);
    }

    public Map<String, BigDecimal> getActivityFees(UserAccount requester) {
        requireRole(requester, UserRole.ACCOUNTANT, UserRole.ADMIN, UserRole.SUPERUSER);
        return Map.copyOf(activityFees);
    }

    public List<RecyclingFacility> getApiFacilities(String zipCode) {
        return findFacilitiesByZip(zipCode, Set.of());
    }

    public String getTemporaryPasswordResetToken(String email) {
        if (email == null || email.isBlank()) {
            throw new IllegalArgumentException("Email address is required.");
        }
        String token = UUID.randomUUID().toString();
        users.values().stream()
                .filter(user -> {
                    try {
                        return getDecryptedEmail(user).equalsIgnoreCase(email);
                    } catch (RuntimeException ignored) {
                        return false;
                    }
                })
                .forEach(user -> user.addNotification("Password reset requested. Reset token: " + token));
        return token;
    }

    public List<String> getRewards(UserAccount user) {
        long completedTransactions = user.getTransactionIds().size();
        List<String> rewards = new ArrayList<>();
        if (completedTransactions >= 1) {
            rewards.add("Eco Starter reward unlocked");
        }
        if (completedTransactions >= 5) {
            rewards.add("Green Champion reward unlocked");
        }
        if (completedTransactions >= 10) {
            rewards.add("Planet Protector reward unlocked");
        }
        return List.copyOf(rewards);
    }
