import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collection;
import java.util.Comparator;
import java.util.HashMap;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Set;
import java.util.UUID;
import java.util.stream.Collectors;

public class CampSystem {
    private final Map<String, Camper> campers = new LinkedHashMap<>();
    private final Map<String, Parent> parents = new LinkedHashMap<>();
    private final Map<String, StaffMember> staff = new LinkedHashMap<>();
    private final Map<String, CampGroup> groups = new LinkedHashMap<>();
    private final Map<String, Activity> activities = new LinkedHashMap<>();
    private final Map<String, ConsentForm> forms = new LinkedHashMap<>();
    private final Map<String, FormSubmission> submissions = new LinkedHashMap<>();
    private final Map<String, AttendanceRecord> attendance = new LinkedHashMap<>();
    private final Map<String, Reminder> reminders = new LinkedHashMap<>();
    private final Map<String, CampRule> rules = new LinkedHashMap<>();
    private final Map<String, Feedback> feedback = new LinkedHashMap<>();
    private final Map<String, Message> messages = new LinkedHashMap<>();
    private final Map<String, Payment> payments = new LinkedHashMap<>();
    private final Map<String, CampReport> reports = new LinkedHashMap<>();
    private final Map<String, CampPhoto> photos = new LinkedHashMap<>();
    private final Map<String, RegistrationForm> registrationForms = new LinkedHashMap<>();
    private final Map<String, List<String>> biWeeklyFeedback = new HashMap<>();
    private final List<String> notifications = new ArrayList<>();

    private String id() { return UUID.randomUUID().toString(); }
    private static void requireText(String value, String name) {
        if (value == null || value.isBlank()) throw new IllegalArgumentException(name + " is required");
    }
    private static <T> T required(Map<String, T> map, String id, String kind) {
        T value = map.get(id);
        if (value == null) throw new IllegalArgumentException(kind + " not found: " + id);
        return value;
    }

    public Parent createParentAccount(String name, String email, String phone, String passwordHash) {
        requireText(name, "name");
        requireText(email, "email");
        for (Parent parent : parents.values()) {
            if (parent.getEmail().equalsIgnoreCase(email)) throw new IllegalArgumentException("Email already registered");
        }
        Parent parent = new Parent(id(), name, email, phone, passwordHash);
        parents.put(parent.getId(), parent);
        return parent;
    }

    public Parent addParent(String name, String email, String phone) {
        return createParentAccount(name, email, phone, null);
    }

    public Parent getParent(String parentId) { return required(parents, parentId, "Parent"); }
    public Collection<Parent> getParents() { return List.copyOf(parents.values()); }

    public void updateParent(String parentId, String name, String email, String phone) {
        getParent(parentId).update(name, email, phone);
    }

    public void deleteParent(String parentId) {
        Parent parent = parents.remove(parentId);
        if (parent == null) throw new IllegalArgumentException("Parent not found: " + parentId);
    }

    public Camper addCamper(String parentId, String firstName, String lastName, int birthYear, int campYear) {
        Parent parent = getParent(parentId);
        Camper camper = new Camper(id(), firstName, lastName, birthYear, parentId, campYear);
        campers.put(camper.getId(), camper);
        parent.addCamper(camper.getId());
        return camper;
    }

    public Camper registerCamper(String parentId, String firstName, String lastName, int birthYear, int campYear) {
        return addCamper(parentId, firstName, lastName, birthYear, campYear);
    }

    public Camper getCamper(String camperId) { return required(campers, camperId, "Camper"); }
    public Collection<Camper> getCampers() { return List.copyOf(campers.values()); }

    public List<Camper> getActiveCampers() {
        return campers.values().stream().filter(c -> c.getStatus() != CamperStatus.ARCHIVED).toList();
    }

    public List<Camper> getCampersForYear(int year) {
        return campers.values().stream().filter(c -> c.getCampYears().contains(year)).toList();
    }

    public List<Camper> getChildren(String parentId) {
        Parent parent = getParent(parentId);
        return parent.getCamperIds().stream().map(this::getCamper).toList();
    }

    public void updateCamper(String camperId, String firstName, String lastName, int birthYear, String parentId) {
        Camper camper = getCamper(camperId);
        Parent newParent = getParent(parentId);
        if (!Objects.equals(camper.getParentId(), parentId)) {
            Parent oldParent = parents.get(camper.getParentId());
            if (oldParent != null) oldParent.removeCamper(camperId);
            newParent.addCamper(camperId);
        }
        camper.update(firstName, lastName, birthYear, parentId);
    }

    public void deleteCamper(String camperId) {
        Camper camper = getCamper(camperId);
        camper.archive();
        Parent parent = parents.get(camper.getParentId());
        if (parent != null) parent.removeCamper(camperId);
        for (CampGroup group : groups.values()) group.removeCamper(camperId);
    }

    public void setCamperEmergencyInformation(String camperId, String information) {
        getCamper(camperId).setEmergencyInformation(information);
    }

    public void addEmergencyContact(String camperId, String name, String contactInformation) {
        getCamper(camperId).addEmergencyContact(name, contactInformation);
    }

    public void setCamperMedicalInformation(String camperId, String information) {
        getCamper(camperId).setMedicalInformation(information);
    }

    public void enrollCamperForYear(String camperId, int year) {
        getCamper(camperId).enrollForYear(year);
    }

    public void setCamperAdmissionStatus(String camperId, CamperStatus status) {
        if (status == CamperStatus.ARCHIVED) throw new IllegalArgumentException("Use deleteCamper to archive a camper");
        getCamper(camperId).setStatus(status);
    }

    public void suspendCamper(String camperId, String reason) {
        Camper camper = getCamper(camperId);
        camper.setStatus(CamperStatus.SUSPENDED);
        createReport("ADMIN", "BEHAVIOR", camperId, reason);
        Parent parent = parents.get(camper.getParentId());
        if (parent != null) notifications.add("To " + parent.getEmail() + ": camper " + camper.getFullName() + " has been suspended. " + reason);
    }

    public StaffMember addStaff(String name, String email, String phone, String position) {
        StaffMember member = new StaffMember(id(), name, email, phone, position);
        staff.put(member.getId(), member);
        return member;
    }

    public StaffMember getStaff(String staffId) { return required(staff, staffId, "Staff member"); }
    public Collection<StaffMember> getStaffMembers() { return List.copyOf(staff.values()); }

    public void updateStaff(String staffId, String name, String email, String phone, String position) {
        getStaff(staffId).update(name, email, phone, position);
    }

    public void assignPosition(String staffId, String position) {
        StaffMember member = getStaff(staffId);
        member.update(member.getName(), member.getEmail(), member.getPhone(), position);
    }

    public void warnStaff(String staffId, String warning) {
        requireText(warning, "warning");
        getStaff(staffId).addWarning(warning);
    }

    public CampGroup createGroup(String name, String description, int campYear) {
        CampGroup group = new CampGroup(id(), name, description, campYear);
        groups.put(group.getId(), group);
        return group;
    }

    public CampGroup getGroup(String groupId) { return required(groups, groupId, "Group"); }
    public Collection<CampGroup> getGroups() { return List.copyOf(groups.values()); }

    public void updateGroup(String groupId, String name, String description) {
        getGroup(groupId).update(name, description);
    }

    public void addCamperToGroup(String camperId, String groupId) {
        Camper camper = getCamper(camperId);
        CampGroup target = getGroup(groupId);
        if (camper.getStatus() == CamperStatus.ARCHIVED) throw new IllegalStateException("Archived camper cannot be assigned");
        if (camper.getGroupId() != null) groups.get(camper.getGroupId()).removeCamper(camperId);
        target.addCamper(camperId);
        camper.setGroupId(groupId);
    }

    public void removeCamperFromGroup(String camperId, String groupId) {
        getGroup(groupId).removeCamper(camperId);
        Camper camper = getCamper(camperId);
        if (groupId.equals(camper.getGroupId())) camper.setGroupId(null);
    }

    public void assignStaffToGroup(String staffId, String groupId) {
        StaffMember member = getStaff(staffId);
        CampGroup group = getGroup(groupId);
        member.addGroup(groupId);
        group.addStaff(staffId);
    }

    public List<StaffMember> getCounselorsForCamper(String camperId) {
        Camper camper = getCamper(camperId);
        if (camper.getGroupId() == null) return List.of();
        return getGroup(camper.getGroupId()).getStaffIds().stream().map(this::getStaff).toList();
    }

    public Activity scheduleActivity(String title, String description, String groupId, String facility,
                                     LocalDateTime startsAt, LocalDateTime endsAt, String recurrence) {
        getGroup(groupId);
        validateTimes(startsAt, endsAt);
        Activity activity = new Activity(id(), title, description, groupId, facility, startsAt, endsAt, recurrence);
        ensureFacilityAvailable(activity, null);
        activities.put(activity.getId(), activity);
        return activity;
    }

    public Activity getActivity(String activityId) { return required(activities, activityId, "Activity"); }
    public Collection<Activity> getActivities() { return List.copyOf(activities.values()); }

    public void updateActivity(String activityId, String title, String description, String groupId,
                               String facility, LocalDateTime startsAt, LocalDateTime endsAt, String recurrence) {
        Activity activity = getActivity(activityId);
        getGroup(groupId);
        validateTimes(startsAt, endsAt);
        Activity proposed = new Activity(activityId, title, description, groupId, facility, startsAt, endsAt, recurrence);
        ensureFacilityAvailable(proposed, activityId);
        activity.update(title, description, groupId, facility, startsAt, endsAt, recurrence);
    }

    public void deleteActivity(String activityId) {
        if (activities.remove(activityId) == null) throw new IllegalArgumentException("Activity not found: " + activityId);
    }

    public void assignStaffToActivity(String staffId, String activityId) {
        getStaff(staffId);
        getActivity(activityId).assignStaff(staffId);
    }

    private void validateTimes(LocalDateTime start, LocalDateTime end) {
        if (start == null || end == null || !end.isAfter(start)) {
            throw new IllegalArgumentException("Activity end must be after its start");
        }
    }

    private void ensureFacilityAvailable(Activity candidate, String ignoredId) {
        if (candidate.getFacility() == null || candidate.getFacility().isBlank()) return;
        for (Activity other : activities.values()) {
            if (other.getId().equals(ignoredId)) continue;
            if (candidate.getFacility().equalsIgnoreCase(other.getFacility())
                    && candidate.getStartsAt().isBefore(other.getEndsAt())
                    && other.getStartsAt().isBefore(candidate.getEndsAt())) {
                throw new IllegalStateException("Facility is already scheduled during that time");
            }
        }
    }

    public List<Activity> getActivitiesForGroup(String groupId) {
        getGroup(groupId);
        return activities.values().stream().filter(a -> a.getGroupId().equals(groupId))
                .sorted(Comparator.comparing(Activity::getStartsAt)).toList();
    }

    public List<Activity> getCalendarEvents(Set<String> groupIds) {
        return activities.values().stream()
                .filter(a -> groupIds == null || groupIds.isEmpty() || groupIds.contains(a.getGroupId()))
                .sorted(Comparator.comparing(Activity::getStartsAt)).toList();
    }

    public List<Activity> getChildSchedule(String parentId, String camperId) {
        Parent parent = getParent(parentId);
        Camper camper = getCamper(camperId);
        if (!parent.getCamperIds().contains(camperId)) throw new IllegalArgumentException("Camper is not linked to parent");
        if (camper.getGroupId() == null) return List.of();
        return getActivitiesForGroup(camper.getGroupId());
    }

    public List<Camper> getStaffCampers(String staffId) {
        StaffMember member = getStaff(staffId);
        return member.getGroupIds().stream().map(this::getGroup).flatMap(g -> g.getCamperIds().stream())
                .distinct().map(this::getCamper).toList();
    }

    public List<Activity> getStaffSchedule(String staffId) {
        StaffMember member = getStaff(staffId);
        return activities.values().stream().filter(a -> a.getAssignedStaffIds().contains(staffId)
                || member.getGroupIds().contains(a.getGroupId()))
                .sorted(Comparator.comparing(Activity::getStartsAt)).toList();
    }

    public List<String> generateNametags(String groupId) {
        CampGroup group = getGroup(groupId);
        return group.getCamperIds().stream().map(this::getCamper)
                .filter(c -> c.getStatus() != CamperStatus.ARCHIVED)
                .map(c -> c.getFullName() + (c.getGroupId() == null ? "" : " — " + group.getName()))
                .sorted().toList();
    }

    public ConsentForm uploadConsentForm(String name, String description, String content, boolean medical) {
        ConsentForm form = new ConsentForm(id(), name, description, content, medical);
        forms.put(form.getId(), form);
        return form;
    }

    public ConsentForm getConsentForm(String formId) { return required(forms, formId, "Consent form"); }
    public Collection<ConsentForm> getConsentForms() { return List.copyOf(forms.values()); }

    public FormSubmission submitForm(String formId, String camperId, String parentId,
                                     String signedBy, String submittedContent) {
        ConsentForm form = getConsentForm(formId);
        Camper camper = getCamper(camperId);
        Parent parent = getParent(parentId);
        if (!parent.getCamperIds().contains(camperId) || !parentId.equals(camper.getParentId())) {
            throw new IllegalArgumentException("Parent is not authorized for this camper");
        }
        FormSubmission submission = new FormSubmission(id(), form.getId(), camperId, parentId, signedBy, submittedContent);
        submissions.put(submission.getId(), submission);
        return submission;
    }

    public List<FormSubmission> getSubmissionsForCamper(String camperId) {
        getCamper(camperId);
        return submissions.values().stream().filter(s -> s.getCamperId().equals(camperId)).toList();
    }

    public List<ConsentForm> getMissingForms(String camperId) {
        getCamper(camperId);
        Set<String> submittedFormIds = submissions.values().stream()
                .filter(s -> s.getCamperId().equals(camperId))
                .map(FormSubmission::getFormId).collect(Collectors.toSet());
        return forms.values().stream().filter(f -> !submittedFormIds.contains(f.getId())).toList();
    }

    public List<ConsentForm> getMedicalForms() {
        return forms.values().stream().filter(ConsentForm::isMedical).toList();
    }

    public AttendanceRecord logAttendance(String camperId, String groupId, String staffId,
                                          LocalDate date, boolean present, String note) {
        Camper camper = getCamper(camperId);
        CampGroup group = getGroup(groupId);
        getStaff(staffId);
        if (!group.getCamperIds().contains(camperId)) throw new IllegalArgumentException("Camper is not in this group");
        if (date == null) date = LocalDate.now();
        String key = date + ":" + camperId;
        AttendanceRecord record = new AttendanceRecord(camperId, groupId, staffId, date, present, note);
        attendance.put(key, record);
        return record;
    }

    public List<AttendanceRecord> getAttendanceForDate(LocalDate date) {
        return attendance.values().stream().filter(a -> a.getDate().equals(date)).toList();
    }

    public List<AttendanceRecord> getAttendanceForCamper(String camperId) {
        getCamper(camperId);
        return attendance.values().stream().filter(a -> a.getCamperId().equals(camperId)).toList();
    }

    public void sendBiWeeklyFeedback(String camperId, String message) {
        Camper camper = getCamper(camperId);
        Parent parent = getParent(camper.getParentId());
        biWeeklyFeedback.computeIfAbsent(camperId, ignored -> new ArrayList<>()).add(message);
        notifications.add("Bi-weekly update to " + parent.getEmail() + " about " + camper.getFullName() + ": " + message);
    }

    public List<String> notifyParents(String message) {
        notifications.clear();
        for (Parent parent : parents.values()) notifications.add("To " + parent.getEmail() + ": " + message);
        return List.copyOf(notifications);
    }

    public Message sendMessage(String senderId, String recipientId, String camperId, String text) {
        if (!parents.containsKey(senderId) && !staff.containsKey(senderId)) throw new IllegalArgumentException("Unknown sender");
        if (!parents.containsKey(recipientId) && !staff.containsKey(recipientId)) throw new IllegalArgumentException("Unknown recipient");
        if (camperId != null) getCamper(camperId);
        Message message = new Message(id(), senderId, recipientId, camperId, text);
        messages.put(message.getId(), message);
        return message;
    }

    public List<Message> getMessagesFor(String participantId) {
        return messages.values().stream().filter(m -> m.getSenderId().equals(participantId)
                || m.getRecipientId().equals(participantId)).toList();
    }

    public Payment recordPayment(String parentId, String camperId, BigDecimal amount, String description) {
        Parent parent = getParent(parentId);
        if (!parent.getCamperIds().contains(camperId)) throw new IllegalArgumentException("Camper is not linked to parent");
        if (amount == null || amount.signum() <= 0) throw new IllegalArgumentException("Payment must be positive");
        Payment payment = new Payment(id(), parentId, camperId, amount, description);
        payments.put(payment.getId(), payment);
        return payment;
    }

    public List<Payment> getPaymentsForCamper(String parentId, String camperId) {
        Parent parent = getParent(parentId);
        if (!parent.getCamperIds().contains(camperId)) throw new IllegalArgumentException("Camper is not linked to parent");
        return payments.values().stream().filter(p -> p.getParentId().equals(parentId)
                && p.getCamperId().equals(camperId)).toList();
    }

    public BigDecimal getTotalPayments(String parentId, String camperId) {
        return getPaymentsForCamper(parentId, camperId).stream().map(Payment::getAmount)
                .reduce(BigDecimal.ZERO, BigDecimal::add);
    }
