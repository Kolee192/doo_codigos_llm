                || requireRepository(repositoryId).hasStaff(user.getId()));
    }

    public synchronized void addCollectionMember(User actor, String collectionId, String resourceId) {
        CollectionRecord collection = requireCollection(collectionId);
        Resource member = getResource(resourceId);
        requireCuratorialAccess(actor, collection.getRepositoryId());
        if (!collection.getRepositoryId().equals(member.getRepositoryId())) {
            throw new IllegalArgumentException("Resources must belong to the same repository");
        }
        collection.addMember(member);
        collection.recordEvent("composition-change", actor.getName(), "Member added", resourceId);
    }

    public synchronized void relateCollections(User actor, String firstId, String secondId) {
        CollectionRecord first = requireCollection(firstId);
        CollectionRecord second = requireCollection(secondId);
        requireCuratorialAccess(actor, first.getRepositoryId());
        requireCuratorialAccess(actor, second.getRepositoryId());
        first.relateTo(second);
        second.relateTo(first);
        first.recordEvent("relationship-change", actor.getName(), "Collections related", secondId);
    }

    public synchronized void addComponent(User actor, String objectId, String componentId) {
        DigitalObject object = requireObject(objectId);
        Component component = requireComponent(componentId);
        requireCuratorialAccess(actor, object.getRepositoryId());
        object.addComponent(component);
        object.recordEvent("structure-change", actor.getName(), "Component added", componentId);
    }

    public synchronized void addChildComponent(User actor, String parentId, String childId) {
        Component parent = requireComponent(parentId);
        Component child = requireComponent(childId);
        requireCuratorialAccess(actor, parent.getRepositoryId());
        parent.addChild(child);
        parent.recordEvent("structure-change", actor.getName(), "Child component added", childId);
    }

    public synchronized void associateFile(User actor, String ownerResourceId, String fileId) {
        Resource owner = getResource(ownerResourceId);
        ContentFile file = getContentFile(fileId);
        requireCuratorialAccess(actor, owner.getRepositoryId());
        if (!owner.getRepositoryId().equals(file.getRepositoryId())) {
            throw new IllegalArgumentException("File and resource must belong to the same repository");
        }
        if (owner instanceof DigitalObject object) {
            object.addFile(file);
        } else if (owner instanceof Component component) {
            component.addFile(file);
        } else {
            throw new IllegalArgumentException("Files can only be associated with objects or components");
        }
    }

    public synchronized void addTitle(User actor, String resourceId, String title, String type) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.addTitle(new Title(title, type), actor.getName());
    }

    public synchronized NameRecord registerName(User actor, String repositoryId,
                                                 String name, String description) {
        requireCuratorialAccess(actor, repositoryId);
        String key = name.trim().toLowerCase();
        NameRecord record = nameRegistry.get(key);
        if (record == null) {
            record = new NameRecord(name, description);
            nameRegistry.put(key, record);
        }
        return record;
    }

    public synchronized void addName(User actor, String resourceId, NameRecord name, String role) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.addName(name, role);
    }

    public synchronized void addSubject(User actor, String resourceId, Subject subject) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.addSubject(subject);
    }

    public synchronized void addLanguage(User actor, String resourceId, String language) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.addLanguage(language, actor.getName());
    }

    public synchronized void addNote(User actor, String resourceId, String note) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.addNote(note, actor.getName());
    }

    public synchronized void addExternalReference(User actor, String resourceId,
                                                   URI uri, String description) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.addExternalReference(new ExternalReference(uri, description));
    }

    public synchronized void setGeoLocation(User actor, String resourceId, GeoLocation location) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.setGeoLocation(location);
    }

    public synchronized void setOriginInfo(User actor, String resourceId, OriginInfo originInfo) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.setOriginInfo(originInfo, actor.getName());
    }

    public synchronized void updateRights(User actor, String resourceId, RightsInfo rights) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        RightsInfo destination = resource.getRights();
        destination.setCopyrightStatus(rights.getCopyrightStatus());
        destination.setRightsHolder(rights.getRightsHolder());
        destination.setStatutes(rights.getStatutes());
        destination.setLicense(rights.getLicense());
        destination.setOtherRights(rights.getOtherRights());
        destination.setLibraryPolicy(rights.getLibraryPolicy());
        destination.setCopyrightExpiry(rights.getCopyrightExpiry());
        destination.setStatuteExpiry(rights.getStatuteExpiry());
        destination.setPolicyExpiry(rights.getPolicyExpiry());
        rights.getPermittedUses().forEach(destination::permitUse);
        rights.getRestrictedUses().forEach(destination::restrictUse);
        resource.recordEvent("rights-change", actor.getName(), "Rights updated", "");
    }

    public synchronized void recordEvent(User actor, String resourceId, String type,
                                         String outcome, String details) {
        Resource resource = getResource(resourceId);
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.recordEvent(type, actor.getName(), outcome, details);
    }

    public synchronized List<Resource> browseFormalCollections() {
        return resources.values().stream()
                .filter(CollectionRecord.class::isInstance)
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Resource> resourcesForRepository(String repositoryId) {
        requireRepository(repositoryId);
        return resources.values().stream()
                .filter(r -> r.getRepositoryId().equals(repositoryId))
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Resource> curatorialResources(User user) {
        if (user == null) {
            return List.of();
        }
        return resources.values().stream()
                .filter(r -> canCurate(user, r.getRepositoryId()))
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Resource> searchByTitle(String query) {
        if (query == null || query.isBlank()) {
            return List.of();
        }
        return resources.values().stream().filter(r -> r.matchesTitle(query))
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Resource> searchByTitle(String query, Set<String> collectionIds,
                                                      String language) {
        Set<String> allowed = collectionIds == null ? Set.of() : collectionIds;
        return resources.values().stream()
                .filter(r -> query == null || query.isBlank() || r.matchesTitle(query))
                .filter(r -> language == null || language.isBlank()
                        || r.getLanguages().contains(language))
                .filter(r -> allowed.isEmpty() || belongsToAnyCollection(r, allowed))
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Resource> searchByName(String query, String role) {
        if (query == null || query.isBlank()) {
            return List.of();
        }
        String normalized = query.toLowerCase();
        return resources.values().stream().filter(resource ->
                resource.getNames().entrySet().stream().anyMatch(entry ->
                        entry.getKey().getName().toLowerCase().contains(normalized)
                                && (role == null || role.isBlank()
                                || entry.getValue().contains(role))))
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Resource> searchBySubject(String heading) {
        if (heading == null || heading.isBlank()) {
            return List.of();
        }
        String normalized = heading.toLowerCase();
        return resources.values().stream().filter(resource ->
                resource.getSubjects().stream()
                        .anyMatch(s -> s.getHeading().toLowerCase().contains(normalized)
                                || s.getType().toLowerCase().contains(normalized)))
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<ContentFile> filesForRepository(String repositoryId) {
        requireRepository(repositoryId);
        return files.values().stream()
                .filter(file -> file.getRepositoryId().equals(repositoryId))
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<ContentFile> filesWithChangedChecksums() {
        return files.values().stream().filter(ContentFile::hasChecksumChanged)
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Resource> resourcesUnderCopyright() {
        return resources.values().stream().filter(r -> r.getRights().isUnderCopyright())
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Resource> resourcesWithExpiringCopyright() {
        return resources.values().stream()
                .filter(r -> r.getRights().getCopyrightExpiry() != null)
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Resource> resourcesWithApplicableStatutes() {
        return resources.values().stream()
                .filter(r -> !r.getRights().getStatutes().isBlank())
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized String repositoryForObject(String objectId) {
        return requireObject(objectId).getRepositoryId();
    }

    public synchronized List<CollectionRecord> collectionsForResource(String resourceId) {
        Resource target = getResource(resourceId);
        return resources.values().stream()
                .filter(CollectionRecord.class::isInstance)
                .map(CollectionRecord.class::cast)
                .filter(collection -> collection.getMemberResourceIds().contains(target.getId()))
                .collect(Collectors.toUnmodifiableList());
    }

    public synchronized List<Component> componentsInOrder(String objectId) {
        return requireObject(objectId).getComponents();
    }

    public synchronized List<ContentFile> contentFiles(String resourceId) {
        Resource resource = getResource(resourceId);
        if (resource instanceof DigitalObject object) {
            return object.getFiles();
        }
        if (resource instanceof Component component) {
            return component.getFiles();
        }
        return List.of();
    }

    public synchronized boolean userMayAccessFile(User user, String fileId) {
        ContentFile file = getContentFile(fileId);
        Resource owner = resources.get(file.getOwnerId());
        if (owner == null) {
            return false;
        }
        RightsInfo rights = owner.getRights();
        return !rights.isUnderCopyright() || rights.hasApplicableLicense()
                || (user != null && canCurate(user, file.getRepositoryId()));
    }

    public synchronized List<Resource> allResources() {
        return Collections.unmodifiableList(new ArrayList<>(resources.values()));
    }

    private boolean belongsToAnyCollection(Resource resource, Set<String> collectionIds) {
        if (resource instanceof CollectionRecord collection
                && collectionIds.contains(collection.getId())) {
            return true;
        }
        return collectionsForResource(resource.getId()).stream()
                .anyMatch(collection -> collectionIds.contains(collection.getId()));
    }

    private void recordCreation(Resource resource, User actor) {
        requireCuratorialAccess(actor, resource.getRepositoryId());
        resource.recordEvent("creation", actor.getName(), "Resource created", "");
    }

    private Repository requireRepository(String id) {
        Repository repository = repositories.get(id);
        if (repository == null) {
            throw new IllegalArgumentException("Unknown repository: " + id);
        }
        return repository;
    }

    private CollectionRecord requireCollection(String id) {
        Resource resource = getResource(id);
        if (!(resource instanceof CollectionRecord collection)) {
            throw new IllegalArgumentException("Not a collection: " + id);
        }
        return collection;
    }

    private DigitalObject requireObject(String id) {
        Resource resource = getResource(id);
        if (!(resource instanceof DigitalObject object)) {
            throw new IllegalArgumentException("Not an object: " + id);
        }
        return object;
    }

    private Component requireComponent(String id) {
        Resource resource = getResource(id);
        if (!(resource instanceof Component component)) {
            throw new IllegalArgumentException("Not a component: " + id);
        }
        return component;
    }

    private void requireCuratorialAccess(User user, String repositoryId) {
        if (!canCurate(user, repositoryId)) {
            throw new SecurityException("User does not have curatorial access to this repository");
        }
    }
}

import java.time.Instant;
import java.util.Set;

public class Main {
    public static void main(String[] args) {
        Dams dams = new Dams();
        Repository repository = dams.createRepository(
                "City Archives", "archives@example.org", "100 Main Street");
        User manager = new User("DAMS Manager", true);
        User curator = new User("Repository Curator", false);
        dams.addRepositoryStaff(manager, repository.getId(), curator);

        CollectionRecord collection = dams.createCollection(curator, repository.getId(),
                "Photographs of the City");
        collection.setProvenance("City Historical Office");

        DigitalObject object = dams.createObject(curator, repository.getId(),
                "Main Street, 1924");
        object.setProvenance("City Historical Office");
        dams.addCollectionMember(curator, collection.getId(), object.getId());

        Component component = dams.createComponent(curator, repository.getId(), "Street view");
        dams.addComponent(curator, object.getId(), component.getId());

        ContentFile file = dams.createContentFile(curator, repository.getId(),
                "main-street.tif", 2_400_000, Instant.now(),
                "image/tiff", "6.0", "abc123", "SHA-256");
        dams.associateFile(curator, object.getId(), file.getId());

        System.out.println("Collections: " + dams.browseFormalCollections().size());
        System.out.println("Title matches: " + dams.searchByTitle("Main Street").size());
        System.out.println("Curator resources: " + dams.curatorialResources(curator).size());
        System.out.println("Object repository: " + dams.getRepository(
                dams.repositoryForObject(object.getId())).getName());
        System.out.println("Object files: " + dams.contentFiles(object.getId()).size());
        System.out.println("Collection filter results: "
                + dams.searchByTitle("", Set.of(collection.getId()), null).size());
    }
}
