// File: SharingPolicy.java
import java.util.Set;
import java.util.concurrent.ConcurrentHashMap;

public final class SharingPolicy {
    private volatile boolean publicRead;
    private final Set<String> userIds = ConcurrentHashMap.newKeySet();
    private final Set<String> groups = ConcurrentHashMap.newKeySet();

    public boolean isPublicRead() { return publicRead; }
    public void setPublicRead(boolean publicRead) { this.publicRead = publicRead; }
    public Set<String> getUserIds() { return Set.copyOf(userIds); }
    public Set<String> getGroups() { return Set.copyOf(groups); }
    public void shareWithUser(String userId) { userIds.add(userId); }
    public void revokeFromUser(String userId) { userIds.remove(userId); }
    public void shareWithGroup(String group) { groups.add(group); }
    public void revokeFromGroup(String group) { groups.remove(group); }

    public boolean permits(User user, String ownerId) {
        if (user == null) return false;
        if (user.isAdministrator() || user.getId().equals(ownerId) || publicRead) return true;
        if (userIds.contains(user.getId())) return true;
        return user.getGroups().stream().anyMatch(groups::contains);
    }
}

    public LogBookPage(String authorId, String title, String content, String section) {
        this.authorId = authorId;
        this.title = title == null ? "" : title;
        this.content = content == null ? "" : content;
        this.section = section == null ? "General" : section;
        this.contentHash = hash(authorId + "\n" + this.title + "\n" + this.content + "\n" + createdAt);

        try {

        } catch (java.security.NoSuchAlgorithmException e) {
            throw new IllegalStateException(e);
        }
    }

import java.util.List;

public final class LogBookNavigator {
    private final List<String> pageIds;
    private int position = -1;

    public LogBookNavigator(List<LogBookPage> pages) {
        this.pageIds = pages.stream().map(LogBookPage::getId).toList();
    }

    public String current() { return position < 0 || position >= pageIds.size() ? null : pageIds.get(position); }
    public String next() {
        if (position + 1 < pageIds.size()) position++;
        return current();
    }
    public String previous() {
        if (position > 0) position--;
        return current();
    }
}

import java.util.Set;

public record MetadataRule(boolean required, Set<String> allowedValues) {
    public MetadataRule {
        allowedValues = allowedValues == null ? Set.of() : Set.copyOf(allowedValues);
    }

    public void validate(String value) {
        if (required && (value == null || value.isBlank())) {
            throw new IllegalArgumentException("Required metadata is missing");
        }
        if (value != null && !allowedValues.isEmpty() && !allowedValues.contains(value)) {
            throw new IllegalArgumentException("Metadata value is not allowed: " + value);
        }
    }
}

import java.io.IOException;
import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;

import java.security.NoSuchAlgorithmException;

public final class NeurohubNode {

    public NeurohubNode(Path storageRoot) throws IOException {
        this.storageRoot = storageRoot.toAbsolutePath().normalize();
        Files.createDirectories(this.storageRoot);
    }

    public Path getStorageRoot() { return storageRoot; }
    public int getUserCount() { return users.size(); }
    public boolean isStorageAvailable() { return Files.isDirectory(storageRoot) && Files.isWritable(storageRoot); }
    public void registerUser(User user) { users.put(user.getId(), user); }
    public User getUser(String id) { return require(users.get(id), "Unknown user"); }
    public Collection<FileAsset> getFiles() { return List.copyOf(files.values()); }

    public LogBook createLogBook(User owner, String name) {
        LogBook book = new LogBook(owner.getId(), name);
        logBooks.put(book.getId(), book);

    }

    public LogBookPage createPage(User author, String logBookId, String title, String content, String section) {
        LogBook book = require(logBooks.get(logBookId), "Unknown log book");
        requireAccess(book.getSharing(), author, book.getOwnerId());
        LogBookPage page = new LogBookPage(author.getId(), title, content, section);
        book.addPage(page);
        return page;

    public FileAsset upload(User owner, String filename, InputStream input, String mediaType) throws IOException {
        if (filename == null || filename.isBlank()) throw new IllegalArgumentException("Filename is required");
        String safeName = Path.of(filename).getFileName().toString();
        Path directory = storageRoot.resolve(owner.getId());
        Files.createDirectories(directory);
        Path target = Files.createTempFile(directory, "upload-", ".data");
        try (InputStream in = input) {
            Files.copy(in, target, StandardCopyOption.REPLACE_EXISTING);
        } catch (IOException e) {
            Files.deleteIfExists(target);
            throw e;
        }
        FileVersion initial = createVersion(1, target);
        FileAsset asset = new FileAsset(owner.getId(), safeName, mediaType, target, initial);
        files.put(asset.getId(), asset);
        return asset;
    }

    public List<FileAsset> uploadDirectory(User owner, Path directory) throws IOException {
        if (!Files.isDirectory(directory)) throw new IllegalArgumentException("Not a directory");
        List<FileAsset> uploaded = new ArrayList<>();
        try (var paths = Files.walk(directory)) {
            for (Path path : paths.filter(Files::isRegularFile).toList()) {
                try (InputStream in = Files.newInputStream(path)) {
                    uploaded.add(upload(owner, path.getFileName().toString(), in, Files.probeContentType(path)));
                }
            }
        }
        return List.copyOf(uploaded);
    }

    public void attachFile(User user, String logBookId, String pageId, String fileId) {
        LogBook book = require(logBooks.get(logBookId), "Unknown log book");
        LogBookPage page = findPage(book, pageId);
        FileAsset file = require(files.get(fileId), "Unknown file");
        requireAccess(book.getSharing(), user, book.getOwnerId());
        requireAccess(page.getSharing(), user, page.getAuthorId());
        requireAccess(file.getSharing(), user, file.getOwnerId());
        page.attachFile(fileId);
    }

    public void attachFiles(User user, String logBookId, String pageId, List<String> fileIds) {
        for (String fileId : fileIds) attachFile(user, logBookId, pageId, fileId);
    }

    public List<FileAsset> unattachedFiles(User user) {
        var attached = logBooks.values().stream()
                .flatMap(book -> book.getPages().stream())
                .flatMap(page -> page.getFileIds().stream())
                .collect(java.util.stream.Collectors.toSet());
        return files.values().stream()
                .filter(file -> file.getOwnerId().equals(user.getId()) && !file.isDeleted())
                .filter(file -> !attached.contains(file.getId()))
                .sorted(Comparator.comparing(FileAsset::getUploadedAt))
                .toList();
    }

    public List<FileAsset> searchFiles(User user, SearchQuery query) {
        String text = query.getText() == null ? "" : query.getText().toLowerCase();
        return files.values().stream()
                .filter(file -> !file.isDeleted())
                .filter(file -> !file.isHiddenAsBadData())
                .filter(file -> file.getSharing().permits(user, file.getOwnerId()))
                .filter(file -> query.getMediaType() == null || file.getMediaType().equalsIgnoreCase(query.getMediaType()))
                .filter(file -> query.getFrom() == null || !file.getUploadedAt().isBefore(query.getFrom()))
                .filter(file -> query.getTo() == null || !file.getUploadedAt().isAfter(query.getTo()))
                .filter(file -> file.getTags().containsAll(query.getTags()))
                .filter(file -> text.isEmpty() || file.getFilename().toLowerCase().contains(text)
                        || file.getMetadata().values().stream().anyMatch(v -> v.toLowerCase().contains(text)))
                .filter(file -> matchesBookAndSection(user, file, query))
                .sorted(Comparator.comparing(FileAsset::getUploadedAt).reversed())
                .toList();
    }

    private boolean matchesBookAndSection(User user, FileAsset file, SearchQuery query) {
        if (query.getLogBookId() == null && query.getSection() == null) return true;
        return logBooks.values().stream()
                .filter(book -> query.getLogBookId() == null || book.getId().equals(query.getLogBookId()))
                .filter(book -> book.getSharing().permits(user, book.getOwnerId()))
                .flatMap(book -> book.getPages().stream())
                .filter(page -> query.getSection() == null || page.getSection().equalsIgnoreCase(query.getSection()))
                .anyMatch(page -> page.getFileIds().contains(file.getId())
                        && page.getSharing().permits(user, page.getAuthorId()));
    }

    public void addControlledTag(String tag) {
        if (tag == null || tag.isBlank()) throw new IllegalArgumentException("Tag is required");
        controlledTags.put(tag.toLowerCase(), tag);
    }

    public void tagFile(User user, String fileId, String tag) {
        FileAsset file = accessibleFile(user, fileId);
        String canonical = controlledTags.get(tag.toLowerCase());
        if (canonical == null) throw new IllegalArgumentException("Tag is not in the controlled vocabulary");
        file.addTag(canonical);
    }

    public void setFileMetadata(User user, String fileId, String key, String value, MetadataTemplate template) {
        FileAsset file = accessibleFile(user, fileId);
        if (template != null) template.validate(Map.of(key, value));
        file.setMetadata(key, value);
    }

    public void updateFile(User user, String fileId, InputStream input) throws IOException {
        FileAsset file = accessibleFile(user, fileId);
        if (file.isImmutable()) throw new IllegalStateException("This file is immutable");
        Path directory = storageRoot.resolve(file.getOwnerId());
        Path target = Files.createTempFile(directory, "version-", ".data");
        try (InputStream in = input) {
            Files.copy(in, target, StandardCopyOption.REPLACE_EXISTING);
        }
        file.addVersion(createVersion(file.getVersions().size() + 1, target));
    }

    public void revertFile(User user, String fileId, int version) {
        accessibleFile(user, fileId).revertToVersion(version);
    }

    public void deleteFile(User user, String fileId) {
        FileAsset file = accessibleFile(user, fileId);
        if (file.isImmutable()) throw new IllegalStateException("This file is immutable");
        file.markDeleted();
    }

    public void restoreFile(User user, String fileId) {
        FileAsset file = require(files.get(fileId), "Unknown file");
        requireAccess(file.getSharing(), user, file.getOwnerId());
        file.restoreFromTrash();
    }

    public void moveFile(User user, String fileId, Path destination) throws IOException {
        FileAsset file = accessibleFile(user, fileId);
        Path normalized = destination.toAbsolutePath().normalize();
        if (!normalized.startsWith(storageRoot)) throw new IllegalArgumentException("Destination must be inside storage");
        Files.createDirectories(normalized.getParent());
        Files.move(file.getCurrentPath(), normalized, StandardCopyOption.REPLACE_EXISTING);
        FileVersion previous = file.getVersions().get(file.getVersions().size() - 1);
        file.addVersion(new FileVersion(previous.number() + 1, normalized, Files.size(normalized),
                checksum(normalized), Instant.now()));
    }

    public Path download(User user, String fileId) throws IOException {
        FileAsset file = accessibleFile(user, fileId);
        if (file.isDeleted()) throw new IllegalStateException("File is in trash");
        return file.getCurrentPath();
    }

    public Protocol createProtocol(User owner, String title, String content) {
        Protocol protocol = new Protocol(owner.getId(), title, content, null);
        protocols.put(protocol.getId(), protocol);
        return protocol;
    }

    public Protocol branchProtocol(User owner, String protocolId, String title, String content) {
        Protocol original = require(protocols.get(protocolId), "Unknown protocol");
        Protocol branch = new Protocol(owner.getId(), title, content, original.getId());
        protocols.put(branch.getId(), branch);
        original.addRevision(branch);
        return branch;
    }

    public WorkflowRun startWorkflow(User owner, String plan) {
        WorkflowRun run = new WorkflowRun(owner.getId(), plan);
        workflows.put(run.getId(), run);
        return run;
    }

    public LogBookPage recordWorkflow(User owner, String logBookId, WorkflowRun run, String title) {
        String details = "Workflow plan:\n" + run.getPlan() + "\nStatus: " + run.getStatus()
                + "\nResults: " + run.getResults();
        LogBookPage page = createPage(owner, logBookId, title, details, "Workflows");
        page.setMetadata("workflowId", run.getId());
        return page;
    }

    public Path createBackup(Path destination) throws IOException {
        Path target = destination.toAbsolutePath().normalize();
        if (target.getParent() != null) Files.createDirectories(target.getParent());
        try (var out = new java.util.zip.ZipOutputStream(Files.newOutputStream(target));
             var paths = Files.walk(storageRoot)) {
            for (Path path : paths.filter(Files::isRegularFile).toList()) {
                String entryName = storageRoot.relativize(path).toString().replace('\\', '/');
                out.putNextEntry(new java.util.zip.ZipEntry(entryName));
                Files.copy(path, out);
                out.closeEntry();
            }
        }
        return target;
    }

    public List<LogBookPage> pagesForPeriod(User viewer, String logBookId, Instant from, Instant to) {
        LogBook book = require(logBooks.get(logBookId), "Unknown log book");
        requireAccess(book.getSharing(), viewer, book.getOwnerId());
        return book.getPages().stream()
                .filter(page -> page.getSharing().permits(viewer, page.getAuthorId()))
                .filter(page -> from == null || !page.getCreatedAt().isBefore(from))
                .filter(page -> to == null || !page.getCreatedAt().isAfter(to))
                .toList();
    }

    public LogBook getLogBook(String id) { return require(logBooks.get(id), "Unknown log book"); }
    public FileAsset getFile(String id) { return require(files.get(id), "Unknown file"); }
    public void registerTemplate(MetadataTemplate template) { templates.put(template.getName(), template); }
    public MetadataTemplate getTemplate(String name) { return templates.get(name); }

    private FileAsset accessibleFile(User user, String fileId) {
        FileAsset file = require(files.get(fileId), "Unknown file");
        requireAccess(file.getSharing(), user, file.getOwnerId());
        return file;
    }

    private static LogBookPage findPage(LogBook book, String pageId) {
        return book.getPages().stream().filter(page -> page.getId().equals(pageId))
                .findFirst().orElseThrow(() -> new IllegalArgumentException("Unknown page"));
    }

    private static void requireAccess(SharingPolicy sharing, User user, String ownerId) {
        if (!sharing.permits(user, ownerId)) throw new SecurityException("Access denied");
    }

    private static FileVersion createVersion(int number, Path path) throws IOException {
        return new FileVersion(number, path, Files.size(path), checksum(path), Instant.now());
    }

    private static String checksum(Path path) throws IOException {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            try (InputStream input = Files.newInputStream(path)) {
                byte[] buffer = new byte[1024 * 1024];
                int read;
                while ((read = input.read(buffer)) != -1) digest.update(buffer, 0, read);
            }
            return HexFormat.of().formatHex(digest.digest());
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException(e);
        }
    }

    private static <T> T require(T value, String message) {
        if (value == null) throw new IllegalArgumentException(message);
        return value;
    }
