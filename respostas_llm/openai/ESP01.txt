import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.Collections;
import java.util.EnumSet;
import java.util.Set;
import java.util.UUID;

public final class User {
    private final String id;
    private final String name;
    private final String email;
    private final String passwordHash;
    private final Set<UserRole> roles;

    public User(String name, String email, String password, Set<UserRole> roles) {
        if (name == null || name.isBlank() || email == null || email.isBlank()) {
            throw new IllegalArgumentException("Name and email are required.");
        }
        this.id = UUID.randomUUID().toString();
        this.name = name;
        this.email = email.trim().toLowerCase();
        this.passwordHash = hash(password == null ? "" : password);
        this.roles = roles == null || roles.isEmpty()
                ? EnumSet.of(UserRole.PUBLIC_USER)
                : EnumSet.copyOf(roles);
    }

    public String getId() {
        return id;
    }

    public String getName() {
        return name;
    }

    public String getEmail() {
        return email;
    }

    public Set<UserRole> getRoles() {
        return Collections.unmodifiableSet(roles);
    }

    public boolean hasRole(UserRole role) {
        return roles.contains(role);
    }

    public boolean verifyPassword(String password) {
        return passwordHash.equals(hash(password == null ? "" : password));
    }

    private static String hash(String value) {
        try {
            byte[] digest = MessageDigest.getInstance("SHA-256")
                    .digest(value.getBytes(StandardCharsets.UTF_8));
            StringBuilder result = new StringBuilder();
            for (byte b : digest) {
                result.append(String.format("%02x", b));
            }
            return result.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException("SHA-256 is unavailable.", e);
        }
    }
}

    public synchronized User createAccount(String name, String email, String password) {
        String key = email == null ? "" : email.trim().toLowerCase(Locale.ROOT);
        if (key.isBlank() || usersByEmail.containsKey(key)) {
            throw new IllegalArgumentException("A unique email address is required.");
        }
        User user = new User(name, key, password,
                java.util.Set.of(UserRole.CUSTOMER, UserRole.APPLICANT));
        usersById.put(user.getId(), user);
        usersByEmail.put(key, user);

    public User registerStaff(String name, String email, String password, UserRole role) {
        if (role == null || role == UserRole.PUBLIC_USER || role == UserRole.APPLICANT
                || role == UserRole.CUSTOMER || role == UserRole.PROSPECTIVE_APPLICANT) {
            throw new IllegalArgumentException("A staff role is required.");

        String key = email.trim().toLowerCase(Locale.ROOT);

        }

        usersById.put(user.getId(), user);
        usersByEmail.put(key, user);
        return user;

    public User authenticate(String email, String password) {

    public List<ServiceDefinition> searchServices(String query, User user) {

        return services.values().stream()
                .filter(s -> user == null || s.isAvailableTo(user))
                .filter(s -> q.isBlank() || (s.getName() + " " + s.getDescription() + " "

                .sorted((a, b) -> a.getName().compareToIgnoreCase(b.getName()))
                .collect(Collectors.toUnmodifiableList());
    }

    public List<String> searchPublicInformation(String query) {

        List<String> results = new ArrayList<>();
        for (ServiceDefinition service : services.values()) {
            String content = service.getName() + " — " + service.getDescription()
                    + " Requirements: " + service.getRequirements();
            if (q.isBlank() || content.toLowerCase(Locale.ROOT).contains(q)) {
                results.add(content);
            }

        for (PropertyRecord property : properties.values()) {
            String content = "Property " + property.getId() + ": " + property.getAddress()

        int version = (int) application.getDocuments().stream()
                .filter(d -> d.getFileName().equalsIgnoreCase(fileName)).count() + 1;

                fileName, documentType, applicant.getId(), description, version);
        application.addDocument(document);
        record(application, applicant, "DOCUMENT_SUBMITTED", fileName + " (version " + version + ")");
        return document;
    }

    private void refreshPlanReviewCompletion(ApplicationRecord app) {
        List<PlanReviewTask> tasks = planReviews.values().stream()
                .filter(t -> t.getApplicationId().equals(app.getId())).toList();
        app.setPlanReviewsComplete(!tasks.isEmpty() && tasks.stream()
                .allMatch(t -> !"PENDING".equals(t.getResult())));
    }

    private void record(ApplicationRecord application, User actor, String category, String detail) {
        application.addActivity(new Activity(actor == null ? "System" : actor.getName(), category, detail));
    }

    private void addSystemActivity(User actor, String category, String detail) {
        systemActivities.computeIfAbsent(UUID.randomUUID().toString(), ignored -> new ArrayList<>())
                .add(new Activity(actor == null ? "System" : actor.getName(), category, detail));
    }

    private ApplicationRecord requireApplication(String id) {
        return require(applications, id, "Application");
    }

    private void requireApplicant(User user) {
        if (user == null || !(user.hasRole(UserRole.APPLICANT) || user.hasRole(UserRole.CUSTOMER)
                || user.hasRole(UserRole.PROSPECTIVE_APPLICANT))) {
            throw new SecurityException("Applicant or customer account required.");
        }
    }

    private void requireStaff(User user) {
        if (user == null || user.getRoles().stream().noneMatch(this::isStaffRole)) {
            throw new SecurityException("Staff account required.");
        }
    }

    private void requireRole(User user, UserRole role) {
        if (user == null || !user.hasRole(role)) {
            throw new SecurityException("Required role: " + role);
        }
    }

    private boolean isStaffRole(UserRole role) {
        return switch (role) {
            case STAFF, PLANNING_STAFF, ZONING_STAFF, PLAN_REVIEW_STAFF, PLAN_REVIEW_SUPERVISOR,
                    DEVELOPMENT_STAFF, INSPECTION_STAFF, INSPECTION_SUPERVISOR, INSPECTOR,
                    ENFORCEMENT_STAFF, TECHNICAL_STAFF, COUNTY_STAFF -> true;
            default -> false;
        };
    }

    private void requireOwnerOrStaff(User user, ApplicationRecord app) {
        if (user == null) {
            throw new SecurityException("Authentication required.");
        }
        if (!app.getApplicantId().equals(user.getId())) {
            requireStaff(user);
        }
    }

    private void ensureNotOnHold(ApplicationRecord application) {
        if (application.isOnHold()) {
            throw new IllegalStateException("Application is on hold: " + application.getHoldReason());
        }
    }

    private void ensureCanProgress(ApplicationRecord application) {
        ensureNotOnHold(application);
        ensureNotFinal(application);
    }

    private void ensureNotFinal(ApplicationRecord application) {
        if (application.getStatus() == ApplicationStatus.CLOSED
                || application.getStatus() == ApplicationStatus.COMPLETED
                || application.getStatus() == ApplicationStatus.DENIED) {
            throw new IllegalStateException("Application is in a final state.");
        }
    }

    private static String requireText(String value, String name) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(name + " is required.");
        }
        return value.trim();
    }

    private static String safe(String value) {
        return value == null ? "" : value;
    }

    private static double roundCurrency(double amount) {
        return Math.round(amount * 100.0) / 100.0;
    }

    private static <T> T require(Map<String, T> map, String id, String type) {
        T value = map.get(id);
        if (value == null) {
            throw new IllegalArgumentException(type + " not found: " + id);
        }
        return value;
    }
}
