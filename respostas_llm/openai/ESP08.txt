public record SearchQuery(String text, LocalDate fromDate, LocalDate toDate) {
    public SearchQuery {

    public synchronized Item ingest(
            String collectionId,
            String itemId,
            User depositor,
            Map<String, String> metadata,
            List<Component> components,
            String license,
            LocalDate embargoReleaseDate) {

        RepositoryCollection collection = getCollection(collectionId);
        LocalDate today = LocalDate.now();

        if (depositor != null && !collection.getDepositorIds().isEmpty()
                && !collection.canDeposit(depositor.getId(), today)) {
            throw new SecurityException("Depositor is not authorized or the deposit window has closed");
        }
        if (collection.getDepositForm().isAllowEmbargo() == false && embargoReleaseDate != null) {
            throw new IllegalArgumentException("This collection does not allow embargoes");
        }
        List<String> acceptedLicenses = collection.getDepositForm().getAcceptedLicenses();
        if (!acceptedLicenses.isEmpty() && (license == null || !acceptedLicenses.contains(license))) {
            throw new SecurityException("A license accepted by this collection is required");
        }

        for (String required : collection.getDepositForm().getRequiredFields()) {
            if (metadata == null || metadata.get(required) == null || metadata.get(required).isBlank()) {
                throw new IllegalArgumentException("Required metadata is missing: " + required);
            }
        }

        Item item = new Item(itemId, collectionId);
        item.setMetadata(metadata == null ? Map.of() : metadata);
        item.setAccessLevel(collection.getDefaultAccess());
        item.setLicense(license);
        if (components != null) {
            components.forEach(item::addComponent);
        }
        if (embargoReleaseDate != null) {
            item.setEmbargo(embargoReleaseDate, collection.getDefaultAccess());
        }

        Item previous = items.putIfAbsent(itemId, item);
        if (previous != null) {
            throw new IllegalArgumentException("Item id already exists: " + itemId);
        }
        collection.addItem(itemId);

        if (collection.getReplicationPolicy() != ReplicationPolicy.DISABLED) {
            item.setReplicationStatus(ReplicationStatus.QUEUED);
        }
        if (collection.isEncrypted()) {
            encryptItemComponents(collection, item);
        }

        auditEvents.add(new AuditEvent(itemId, "INGESTED", depositor == null ? "system" : depositor.getId(), null));
        return item;
    }

    public List<String> processExpirations(LocalDate today, boolean hideBeforeDelete) {
        List<String> deleted = new ArrayList<>();
        for (Item item : items.values()) {
            LocalDate expiry = item.getExpirationDate();
            if (expiry == null || today.isBefore(expiry)) {
                continue;
            }
            if (hideBeforeDelete && item.isPublished()) {
                item.setPublished(false, today);
                auditEvents.add(new AuditEvent(item.getId(), "HIDDEN_FOR_EXPIRATION", "system", null));
            } else {
                items.remove(item.getId());
                for (RepositoryCollection collection : collections.values()) {
                    collection.removeItem(item.getId());
                }
                deleted.add(item.getId());
                auditEvents.add(new AuditEvent(item.getId(), "EXPIRED_AND_DELETED", "system", null));
            }
        }
        return List.copyOf(deleted);
    }

    public List<Item> search(SearchQuery query) {
        String text = query.text().toLowerCase();
        return items.values().stream()
                .filter(item -> query.matchesDate(item))
                .filter(Item::isPublished)
                .filter(item -> {
                    if (text.isBlank()) {
                        return true;
                    }
                    return item.getMetadata().entrySet().stream()
                            .anyMatch(entry -> entry.getKey().toLowerCase().contains(text)
                                    || entry.getValue().toLowerCase().contains(text));
                })
                .sorted(Comparator.comparing(Item::getId))
                .toList();
    }

    public Map<String, Map<String, Long>> collectionStatistics() {

    public void encryptCollection(String collectionId) {
        RepositoryCollection collection = getCollection(collectionId);
        collection.setEncrypted(true);
        ensureCollectionKey(collectionId);
        for (String itemId : collection.getItemIds()) {
            Item item = items.get(itemId);
            if (item != null) {
                encryptItemComponents(collection, item);
            }
        }
    }

    private void encryptItemComponents(RepositoryCollection collection, Item item) {
        SecretKey key = ensureCollectionKey(collection.getId());
        for (Component component : item.getComponents()) {
            if (component.isEncrypted()) {
                continue;
            }
            try {
                byte[] plain = component.getData();
                byte[] checksum = MessageDigest.getInstance("SHA-256").digest(plain);
                byte[] nonce = new byte[12];
                new SecureRandom().nextBytes(nonce);
                Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding");
                cipher.init(Cipher.ENCRYPT_MODE, key, new GCMParameterSpec(128, nonce));
                byte[] encrypted = cipher.doFinal(plain);
                byte[] packed = new byte[nonce.length + encrypted.length];
                System.arraycopy(nonce, 0, packed, 0, nonce.length);
                System.arraycopy(encrypted, 0, packed, nonce.length, encrypted.length);
                component.replaceData(packed);
                component.setChecksum(HexFormat.of().formatHex(checksum));
                component.setEncrypted(true);
                if (collection.getReplicationPolicy() != ReplicationPolicy.DISABLED) {
                    component.setReplicationStatus(ReplicationStatus.QUEUED);
                }
            } catch (Exception exception) {
                throw new IllegalStateException("Unable to encrypt component", exception);
            }
        }
    }

    private SecretKey ensureCollectionKey(String collectionId) {
        return collectionKeys.computeIfAbsent(collectionId, ignored -> {
            try {
                KeyGenerator generator = KeyGenerator.getInstance("AES");
                generator.init(256);
                return generator.generateKey();
            } catch (Exception exception) {
                throw new IllegalStateException("Unable to create collection encryption key", exception);
            }
        });
    }

    public String exposeOaiPmh() {
        StringBuilder xml = new StringBuilder("<?xml version=\"1.0\"?><records>");
        for (Item item : items.values()) {
            if (item.isPublished() && item.isMetadataDiscoverable(LocalDate.now())) {
                xml.append("<record id=\"").append(escape(item.getId())).append("\">");
                item.getMetadata().forEach((key, value) -> xml.append("<metadata key=\"")
                        .append(escape(key)).append("\">").append(escape(value)).append("</metadata>"));
                xml.append("</record>");
            }
        }
        return xml.append("</records>").toString();
    }

    public String linkedData(String itemId) {
        Item item = getItem(itemId);
        StringBuilder json = new StringBuilder("{\"@id\":\"").append(escape(item.getId())).append("\",\"metadata\":{");
        boolean first = true;
        for (Map.Entry<String, String> entry : item.getMetadata().entrySet()) {
            if (!first) {
                json.append(',');
            }
            json.append('"').append(escape(entry.getKey())).append("\":\"").append(escape(entry.getValue())).append('"');
            first = false;
        }
        return json.append("}}").toString();
    }

    private static String escape(String value) {
        return Objects.toString(value, "")
                .replace("&", "&amp;")
                .replace("<", "&lt;")
                .replace(">", "&gt;")
                .replace("\"", "&quot;");
    }

    public List<String> getActiveMessages(Instant now) {
        List<String> result = new ArrayList<>();
        for (String value : scheduledMessages.values()) {
            String[] parts = value.split("\\|", 3);
            Instant start = Instant.parse(parts[0]);
            Instant end = Instant.parse(parts[1]);
            if (!now.isBefore(start) && now.isBefore(end)) {
                result.add(parts[2]);
            }
        }
        return List.copyOf(result);
    }
