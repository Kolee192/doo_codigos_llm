        fields.put("collection", asset.getCollection());
        fields.put("identifier", asset.getId());
        fields.put("repository", "RAC");
        fields.put("date", asset.getMetadataValue("date") == null ? "" : asset.getMetadataValue("date"));

    }

        return Map.copyOf(fields);
    }

        return "\"" + fields.get("title") + "\", " + fields.get("collection")
                + ", RAC, digital object " + assetId + ".";
    }

        return "@misc{" + assetId + ",\n"
                + "  title = {" + fields.get("title") + "},\n"
                + "  howpublished = {" + fields.get("collection") + ", RAC},\n"
                + "  note = {Digital object " + assetId + "}\n}";
    }
}

    public synchronized boolean canAccess(String assetId, User user, String ipAddress) {

            return false;

        boolean userAllowed = asset.isUserExplicitlyAllowed(user == null ? null : user.getId());
        boolean ipAllowed = asset.isIpExplicitlyAllowed(ipAddress);
        if (asset.hasExplicitUserOrIpAllowList() && !userAllowed && !ipAllowed) {
            return false;
        }
        if (asset.isRestricted() || asset.isRestrictedFromView()) {
            return userAllowed || ipAllowed
                    || (user != null && userHasPermission(user.getId(), "view-restricted"));
        }
        return true;
    }

    public synchronized DigitalAsset rotateImage(String assetId, double degrees) throws IOException {
        DigitalAsset asset = requireAsset(assetId);
        BufferedImage source = readImage(asset);
        if (source == null) {
            throw new IOException("Asset is not a supported image");
        }
        double radians = Math.toRadians(degrees);
        double sin = Math.abs(Math.sin(radians));
        double cos = Math.abs(Math.cos(radians));
        int width = (int) Math.floor(source.getWidth() * cos + source.getHeight() * sin);
        int height = (int) Math.floor(source.getHeight() * cos + source.getWidth() * sin);
        BufferedImage rotated = new BufferedImage(width, height,
                source.getColorModel().hasAlpha() ? BufferedImage.TYPE_INT_ARGB : BufferedImage.TYPE_INT_RGB);
        Graphics2D graphics = rotated.createGraphics();
        graphics.translate((width - source.getWidth()) / 2.0, (height - source.getHeight()) / 2.0);
        graphics.rotate(radians, source.getWidth() / 2.0, source.getHeight() / 2.0);
        graphics.drawRenderedImage(source, null);
        graphics.dispose();
        writeImage(asset, rotated);
        return asset;
    }

    public synchronized DigitalAsset adjustContrast(String assetId, float scale, float offset) throws IOException {
        DigitalAsset asset = requireAsset(assetId);
        BufferedImage source = readImage(asset);
        if (source == null) {
            throw new IOException("Asset is not a supported image");
        }
        BufferedImage adjusted = new BufferedImage(source.getWidth(), source.getHeight(),
                source.getColorModel().hasAlpha() ? BufferedImage.TYPE_INT_ARGB : BufferedImage.TYPE_INT_RGB);
        for (int y = 0; y < source.getHeight(); y++) {
            for (int x = 0; x < source.getWidth(); x++) {
                int argb = source.getRGB(x, y);
                int alpha = (argb >>> 24) & 0xff;
                int red = adjustChannel((argb >>> 16) & 0xff, scale, offset);
                int green = adjustChannel((argb >>> 8) & 0xff, scale, offset);
                int blue = adjustChannel(argb & 0xff, scale, offset);
                adjusted.setRGB(x, y, (alpha << 24) | (red << 16) | (green << 8) | blue);
            }
        }
        writeImage(asset, adjusted);
        return asset;
    }

    public synchronized List<DigitalAsset> findByUploadedImage(byte[] image, User user) {
        String fingerprint = sha256(image == null ? new byte[0] : image);
        return assets.values().stream()
                .filter(asset -> canAccess(asset.getId(), user, null))
                .filter(asset -> fingerprint.equals(asset.getFingerprint()))
                .toList();
    }

    }

    public synchronized String exportDescriptiveDataCsv(User user) {
        StringBuilder csv = new StringBuilder("id,title,collection,materialType,rights\n");
        for (DigitalAsset asset : assets.values()) {
            if (canAccess(asset.getId(), user, null)) {
                csv.append(csv(asset.getId())).append(',')
                        .append(csv(asset.getTitle())).append(',')
                        .append(csv(asset.getCollection())).append(',')
                        .append(csv(asset.getMaterialType())).append(',')
                        .append(csv(asset.getRightsStatement() == null
                                ? "" : asset.getRightsStatement().getIdentifier()))
                        .append('\n');
            }
        }
        return csv.toString();
    }

    private DigitalAsset requireAsset(String id) {
        DigitalAsset asset = assets.get(id);
        if (asset == null) {
            throw new IllegalArgumentException("Unknown asset: " + id);
        }
        return asset;
    }

    private String searchableText(DigitalAsset asset) {
        StringBuilder text = new StringBuilder(asset.getTitle()).append(' ')
                .append(asset.getCollection()).append(' ')
                .append(asset.getMaterialType()).append(' ')
                .append(asset.getOcrText());
        asset.getMetadata().forEach((key, value) -> text.append(' ').append(key).append(' ').append(value));
        asset.getTags().forEach(tag -> text.append(' ').append(tag));
        return text.toString().toLowerCase();
    }

    private BufferedImage readImage(DigitalAsset asset) throws IOException {
        return ImageIO.read(new ByteArrayInputStream(asset.getContent()));
    }

    private void writeImage(DigitalAsset asset, BufferedImage image) throws IOException {
        String format = asset.getMimeType().toLowerCase().contains("png") ? "png" : "jpg";
        ByteArrayOutputStream output = new ByteArrayOutputStream();
        if (!ImageIO.write(image, format, output)) {
            throw new IOException("No image writer for " + format);
        }
        asset.replaceContent(output.toByteArray());
        asset.setMetadata("fileSize", Long.toString(asset.getFileSize()));
        asset.setMetadata("modifiedAt", Instant.now().toString());
        asset.setFingerprint(sha256(asset.getContent()));
    }

    private static int adjustChannel(int value, float scale, float offset) {
        return Math.max(0, Math.min(255, Math.round(value * scale + offset)));
    }

    private static String sha256(byte[] data) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            return HexFormat.of().formatHex(digest.digest(data));
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException(e);
        }
    }

    private static String csv(String value) {
        String safe = value == null ? "" : value.replace("\"", "\"\"");
        return "\"" + safe + "\"";
    }

    public record UploadItem(String title, String collection, String materialType,
                             String mimeType, byte[] content) {
        public UploadItem {
            content = content == null ? new byte[0] : content.clone();
        }

        @Override
        public byte[] content() {
            return content.clone();
        }
    }
