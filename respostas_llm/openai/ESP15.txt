    public RepositoryItem(String ownerId, ItemMetadata metadata, List<DigitalFile> files,

        this.id = UUID.randomUUID().toString();
        this.persistentUrl = "https://repository.example.edu/items/" + id;
        this.ownerId = ownerId;
        this.accessLevel = accessLevel == null ? AccessLevel.PUBLIC : accessLevel;
        this.license = license == null ? License.ALL_RIGHTS_RESERVED : license;
        this.published = true;
        this.emailEnabled = emailEnabled;
        this.lastUpdated = Instant.now();
        versions.add(new ItemVersion(1, metadata, files, uploadedBy, "Initial deposit"));
        revisionHistory.add("Initial deposit by " + uploadedBy + " at " + lastUpdated);
    }

    public synchronized String getId() {
        return id;
    }

    public String getPersistentUrl() {
        return persistentUrl;
    }

        return ownerId;

        return currentVersion().getMetadata();
    }

        return currentVersion().getFiles();
    }

    public synchronized List<ItemVersion> getVersions() {
        return List.copyOf(versions);
    }

    public synchronized ItemVersion getVersion(int number) {
        return versions.stream().filter(v -> v.getVersionNumber() == number).findFirst().orElse(null);
    }

    public synchronized int addVersion(ItemMetadata metadata, List<DigitalFile> files,
                                       String uploadedBy, String revisionNote) {
        String oldTitle = currentVersion().getMetadata().getTitle();
        String newTitle = metadata.getTitle();
        if (!oldTitle.equals(newTitle)) {
            previousNames.add(oldTitle);
        }
        int number = versions.size() + 1;
        versions.add(new ItemVersion(number, metadata, files, uploadedBy, revisionNote));
        lastUpdated = Instant.now();
        revisionHistory.add("Version " + number + " uploaded by " + uploadedBy + ": "
                + (revisionNote == null ? "" : revisionNote) + " at " + lastUpdated);
        return number;
    }

    public synchronized boolean isAccessibleTo(User user) {
        if (!published || (embargoActive && embargoUntil != null && LocalDate.now().isBefore(embargoUntil))) {
            return false;
        }
        if (user != null && (user.getId().equals(ownerId)
                || user.getRole() == Role.ADMINISTRATOR
                || user.getRole() == Role.DB_IR_ADMIN
                || user.getRole() == Role.LIBRARY_STAFF
                || user.getRole() == Role.REPOSITORY_SUPPORT)) {
            return true;
        }
        return switch (accessLevel) {
            case PUBLIC -> true;
            case UNIVERSITY -> user != null && user.isInstitutionalUser();
            case PRIVATE -> false;
        };
    }

    private ItemVersion currentVersion() {
        return versions.get(versions.size() - 1);
    }

    }

        views.incrementAndGet();

    }

        return views.get();
    }

    public RepositoryItem getItem(String itemId, User viewer) {
        RepositoryItem item = requireItem(itemId);
        if (!item.isAccessibleTo(viewer)) {
            throw new SecurityException("Item is not accessible to this user.");
        }
        itemStatistics.computeIfAbsent(itemId, ignored -> new UsageStatistics()).recordView();
        return item;
    }

    public String getPersistentUrl(String itemId) {
        return requireItem(itemId).getPersistentUrl();
    }

    public List<RepositoryItem> getCollectionItems(String collectionId, User viewer) {
        RepositoryCollection collection = requireCollection(collectionId);
        if (!collectionAccessible(collection, viewer)) {
            throw new SecurityException("Collection is not accessible to this user.");
        }
        return collection.getItemIds().stream()
                .map(items::get)
                .filter(item -> item != null && item.isAccessibleTo(viewer))
                .toList();
    }

    public List<RepositoryItem> search(String query, String collectionId, User viewer) {
        Set<String> candidateIds = null;
        if (collectionId != null) {
            RepositoryCollection collection = requireCollection(collectionId);
            if (!collectionAccessible(collection, viewer)) {
                throw new SecurityException("Collection is not accessible to this user.");
            }
            candidateIds = new LinkedHashSet<>(collection.getItemIds());
        }
        Set<String> ids = candidateIds;
        return items.values().stream()
                .filter(item -> ids == null || ids.contains(item.getId()))
                .filter(item -> item.isAccessibleTo(viewer))
                .filter(item -> item.getMetadata().containsText(query))
                .sorted(Comparator.comparing(i -> i.getMetadata().getTitle(),
                        String.CASE_INSENSITIVE_ORDER))
                .toList();
    }

    public List<RepositoryItem> searchByAuthor(String author, User viewer) {
        return search(author, null, viewer).stream()
                .filter(item -> item.getMetadata().getAuthors().stream()
                        .anyMatch(a -> a.toLowerCase(Locale.ROOT)
                                .contains(author == null ? "" : author.toLowerCase(Locale.ROOT))))
                .toList();
    }

    public synchronized void setPublicationState(String adminId, String itemId, boolean published) {

    public synchronized void setEmailEnabled(String adminId, String itemId, boolean enabled) {

        requireItem(itemId).setEmailEnabled(enabled);
        audit(adminId, "SET_EMAIL_ENABLED", itemId, String.valueOf(enabled));

    public synchronized void subscribe(String userId, String itemId) {
        User user = requireUser(userId);
        RepositoryItem item = requireItem(itemId);
        if (!item.isEmailEnabled()) {
            throw new IllegalStateException("Email notifications are disabled for this item.");
        }
        subscriptions.computeIfAbsent(userId, ignored -> ConcurrentHashMap.newKeySet()).add(itemId);
        audit(userId, "SUBSCRIBE", itemId, user.getEmail());
    }

    public synchronized void unsubscribe(String userId, String itemId) {
        requireUser(userId);
        subscriptions.computeIfAbsent(userId, ignored -> ConcurrentHashMap.newKeySet()).remove(itemId);
        audit(userId, "UNSUBSCRIBE", itemId, "");
    }

    public synchronized void setSubscriptions(String userId, Collection<String> itemIds) {
        requireUser(userId);
        Set<String> replacement = ConcurrentHashMap.newKeySet();
        for (String itemId : itemIds) {
            RepositoryItem item = requireItem(itemId);
            if (item.isEmailEnabled()) {
                replacement.add(itemId);
            }
        }
        subscriptions.put(userId, replacement);
        audit(userId, "SET_SUBSCRIPTIONS", userId, replacement.toString());
    }

    public List<User> getSubscribers(String adminId, String itemId) {
        requireAdmin(adminId);
        requireItem(itemId);
        return subscriptions.entrySet().stream()
                .filter(entry -> entry.getValue().contains(itemId))
                .map(entry -> users.get(entry.getKey()))
                .filter(user -> user != null)
                .toList();
    }

    public synchronized void announceNewVersion(String actorId, String itemId,
            ItemMetadata metadata, List<DigitalFile> files, String note) {
        requireStaffOrAdmin(actorId);
        RepositoryItem item = requireItem(itemId);
        item.addVersion(metadata, files, actorId, note);
        String message = "New version available: " + metadata.getTitle() + " " + item.getPersistentUrl();
        for (Map.Entry<String, Set<String>> entry : subscriptions.entrySet()) {
            if (entry.getValue().contains(itemId)) {
                User subscriber = users.get(entry.getKey());
                if (subscriber != null) {
                    subscriber.notify(message);
                }
            }
        }
        User owner = users.get(item.getOwnerId());
        if (owner != null) {
            owner.notify("A new version of your item was deposited: " + item.getPersistentUrl());
        }
        audit(actorId, "ANNOUNCE_VERSION", itemId, note);
    }

    public synchronized void recordDownload(String itemId, String fileId, User downloader) {
        RepositoryItem item = requireItem(itemId);
        if (!item.isAccessibleTo(downloader)) {
            throw new SecurityException("Item is not accessible to this user.");
        }
        boolean exists = item.getFiles().stream().anyMatch(file -> file.getId().equals(fileId));
        if (!exists) {
            throw new IllegalArgumentException("File does not belong to the current item version.");
        }
        itemStatistics.computeIfAbsent(itemId, ignored -> new UsageStatistics()).recordDownload();
        User owner = users.get(item.getOwnerId());
        if (owner != null && (downloader == null || !owner.getId().equals(downloader.getId()))) {
            owner.notify("A file was downloaded from '" + item.getMetadata().getTitle() + "'.");
        }
        audit(downloader == null ? "anonymous" : downloader.getId(), "DOWNLOAD", itemId, fileId);
    }

        requireAdmin(adminId);

    public synchronized String exportReleaseCalendar() {
        StringBuilder csv = new StringBuilder("itemId,title,releaseDate\n");
        releaseCalendar.forEach((id, date) -> {
            RepositoryItem item = items.get(id);
            if (item != null) {
                csv.append(id).append(",\"")
                        .append(item.getMetadata().getTitle().replace("\"", "\"\""))
                        .append("\",").append(date).append('\n');
            }
        });
        return csv.toString();
    }

    public UsageStatistics getItemStatistics(String itemId) {
        requireItem(itemId);
        return itemStatistics.computeIfAbsent(itemId, ignored -> new UsageStatistics());
    }

    public UsageStatistics getCollectionStatistics(String collectionId) {
        RepositoryCollection collection = requireCollection(collectionId);
        UsageStatistics aggregate = new UsageStatistics();
        for (String itemId : collection.getItemIds()) {
            UsageStatistics stats = getItemStatistics(itemId);
            for (long i = 0; i < stats.getViews(); i++) {
                aggregate.recordView();
            }
            for (long i = 0; i < stats.getDownloads(); i++) {
                aggregate.recordDownload();
            }
        }
        return aggregate;
    }

    public synchronized List<Map<String, String>> harvestOai(String sinceItemId) {
        return items.values().stream()
                .filter(item -> sinceItemId == null || item.getId().compareTo(sinceItemId) > 0)
                .sorted(Comparator.comparing(RepositoryItem::getId))
                .map(item -> {
                    Map<String, String> record = new LinkedHashMap<>();
                    record.put("id", item.getId());
                    record.put("url", item.getPersistentUrl());
                    record.put("title", item.getMetadata().getTitle());
                    record.put("description", item.getMetadata().getDescription());
                    record.put("authors", String.join("; ", item.getMetadata().getAuthors()));
                    record.put("subject", item.getMetadata().getSubjectArea());
                    record.put("updated", item.getLastUpdated().toString());
                    return Map.copyOf(record);
                })
                .toList();
    }

    public synchronized List<AuditEntry> getAuditLog(String actorId) {
        requireStaffOrAdmin(actorId);

    }

    private boolean collectionAccessible(RepositoryCollection collection, User viewer) {
        if (viewer != null && (viewer.getId().equals(collection.getOwnerId())
                || viewer.getRole() == Role.ADMINISTRATOR
                || viewer.getRole() == Role.DB_IR_ADMIN
                || viewer.getRole() == Role.LIBRARY_STAFF
                || viewer.getRole() == Role.REPOSITORY_SUPPORT)) {
            return true;
        }
        return switch (collection.getAccessLevel()) {
            case PUBLIC -> true;
            case UNIVERSITY -> viewer != null && viewer.isInstitutionalUser();
            case PRIVATE -> false;
        };
    }

    private User requireUser(String userId) {
        User user = users.get(userId);
        if (user == null) {
            throw new IllegalArgumentException("Unknown user: " + userId);
        }
        return user;
    }

    private RepositoryItem requireItem(String itemId) {
        RepositoryItem item = items.get(itemId);
        if (item == null) {
            throw new IllegalArgumentException("Unknown item: " + itemId);
        }
        return item;
    }

    private RepositoryCollection requireCollection(String collectionId) {
        RepositoryCollection collection = collections.get(collectionId);
        if (collection == null) {
            throw new IllegalArgumentException("Unknown collection: " + collectionId);
        }
        return collection;
    }

    private void requireAdmin(String userId) {
        User user = requireUser(userId);
        if (user.getRole() != Role.ADMINISTRATOR && user.getRole() != Role.DB_IR_ADMIN) {
            throw new SecurityException("Administrator permission required.");
        }
    }

    private void requireStaffOrAdmin(String userId) {
        User user = requireUser(userId);
        if (user.getRole() != Role.ADMINISTRATOR && user.getRole() != Role.DB_IR_ADMIN
                && user.getRole() != Role.LIBRARY_STAFF
                && user.getRole() != Role.REPOSITORY_SUPPORT) {
            throw new SecurityException("Staff permission required.");
        }
    }

    private void requireStaffOrOwner(String userId, String itemId) {
        User user = requireUser(userId);
        RepositoryItem item = requireItem(itemId);
        if (!item.getOwnerId().equals(userId)
                && user.getRole() != Role.ADMINISTRATOR
                && user.getRole() != Role.DB_IR_ADMIN
                && user.getRole() != Role.LIBRARY_STAFF
                && user.getRole() != Role.REPOSITORY_SUPPORT) {
            throw new SecurityException("Item owner or staff permission required.");
        }
    }

    private synchronized void audit(String actorId, String action, String targetId, String details) {
        auditLog.add(new AuditEntry(actorId == null ? "system" : actorId, action, targetId, details));
    }
}

import java.time.LocalDate;
import java.util.List;

public class Main {
    public static void main(String[] args) {
        Repository repository = new Repository("Cornell Research Repository");

        User faculty = repository.createUser(
                "Dr. Ada Researcher", "ada@example.edu", Role.FACULTY, true);
        User patron = repository.createUser(
                "Sam Patron", "sam@example.org", Role.PATRON, false);
        User admin = repository.createUser(
                "Repository Admin", "admin@example.edu", Role.DB_IR_ADMIN, true);

        RepositoryCollection collection = repository.createCollection(
                faculty.getId(), "Climate Research", "Research outputs about climate.",
                AccessLevel.PUBLIC, "Cornell University");

        ItemMetadata metadata = new ItemMetadata(
                "Climate Data 2025",
                "A sample dataset for repository demonstration.",
                List.of(faculty.getName()),
                "Dataset",
                "Climate science",
                List.of("climate", "data", "research"));
        metadata.setCustomField("Project", "North Campus Study");

        DigitalFile dataFile = new DigitalFile(
                "climate-data.csv", "text/csv", 1024, "10.1234/example", "sha256:demo", false);

        RepositoryItem item = repository.deposit(
                faculty.getId(), metadata, List.of(dataFile), AccessLevel.PUBLIC, License.CC_BY, true);
        repository.addItemToCollection(faculty.getId(), item.getId(), collection.getId());
        repository.subscribe(patron.getId(), item.getId());

        ItemMetadata revisedMetadata = metadata.copy();
        revisedMetadata.setDescription("Updated and quality-checked climate dataset.");
        repository.announceNewVersion(admin.getId(), item.getId(),
                revisedMetadata, List.of(dataFile), "Corrected data values");

        System.out.println("Collection: " + collection.getName());
        System.out.println("Persistent URL: " + item.getPersistentUrl());
        System.out.println("Search results: " + repository.search("climate", null, patron).size());
        System.out.println("Patron notifications: " + patron.getNotifications().size());
        System.out.println("Release calendar CSV:\n" + repository.exportReleaseCalendar());

        repository.addReleaseDate(admin.getId(), item.getId(), LocalDate.now().plusMonths(1));
        System.out.println("License: " + item.getLicense());
    }
