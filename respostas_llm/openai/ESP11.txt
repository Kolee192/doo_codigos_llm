import java.util.HashMap;
import java.util.Map;

public class HelpConfiguration {
    private String baseUrl;
    private final Map<String, String> topicMappings = new HashMap<>();

    public String getBaseUrl() { return baseUrl; }
    public void setBaseUrl(String value) { baseUrl = value; }
    public Map<String, String> getTopicMappings() { return topicMappings; }

    public String linkFor(String topic) {
        if (baseUrl == null || baseUrl.isBlank()) return null;
        String path = topicMappings.getOrDefault(topic, topic);
        return baseUrl.endsWith("/") ? baseUrl + path : baseUrl + "/" + path;
    }
}

    public User createUser(String username, String password, String email, User administrator) {
        requireRole(administrator, Role.APPLICATION_ADMINISTRATOR, Role.SYSTEM_ADMINISTRATOR);
        if (users.containsKey(username)) throw new IllegalArgumentException("Username already exists");
        User user = new User(username, hash(password), email);
        users.put(username, user);
        return user;
    }

    public User createInitialUser(String username, String password, String email, Role role) {

        user.getRoles().add(role);
        users.put(username, user);
        return user;
    }

    public void resetPassword(User administrator, String username, String newPassword) {
        requireRole(administrator, Role.APPLICATION_ADMINISTRATOR, Role.SYSTEM_ADMINISTRATOR);
        User user = Optional.ofNullable(users.get(username))
                .orElseThrow(() -> new IllegalArgumentException("Unknown user: " + username));
        user.setPasswordHash(hash(newPassword));
    }

    public boolean authenticateLocally(String username, String password) {
        User user = users.get(username);
        return user != null && !user.isLdapAccount()
                && constantTimeEquals(user.getPasswordHash(), hash(password));
    }

        User user = users.get(username);
        if (user == null || !user.isLdapAccount() || authenticator == null) return false;
        return authenticator.authenticate(username, password);
    }

    public ResourceComponent addComponent(User user, String resourceId, String parentId, String title) {
        Resource resource = getResource(user, resourceId);
        requireWriteAccess(user, resource.getRepositoryId(), false);
        ResourceComponent component = new ResourceComponent(title);
        if (parentId == null) {
            resource.getComponents().add(component);
        } else {
            findComponent(resource.getComponents(), parentId)
                    .orElseThrow(() -> new IllegalArgumentException("Component not found"))
                    .getChildren().add(component);

    public void setComponentIdentifier(User user, String resourceId, String componentId, String value) {
        Resource resource = getResource(user, resourceId);
        requireWriteAccess(user, resource.getRepositoryId(), false);
        if (value != null && !value.isBlank()
                && findComponentByIdentifier(resource.getComponents(), value).isPresent()) {
            throw new IllegalArgumentException("Component identifier must be unique within the resource");
        }
        ResourceComponent component = findComponent(resource.getComponents(), componentId)
                .orElseThrow(() -> new IllegalArgumentException("Component not found"));
        component.setComponentIdentifier(value);
        resource.incrementVersion();
    }

    public void setComponentLevel(User user, String resourceId, String componentId, String level) {
        Resource resource = getResource(user, resourceId);
        requireWriteAccess(user, resource.getRepositoryId(), false);
        ResourceComponent component = findComponent(resource.getComponents(), componentId)
                .orElseThrow(() -> new IllegalArgumentException("Component not found"));
        if (supportedLevels.contains(level.toLowerCase(Locale.ROOT))) {
            component.setLevelOfDescription(level);
            component.setCustomLevelOfDescription(null);
        } else {
            component.setLevelOfDescription("otherlevel");
            component.setCustomLevelOfDescription(level);

        resource.incrementVersion();

            resource.setLevelOfDescription(level);
            resource.setCustomLevelOfDescription(null);

            resource.setCustomLevelOfDescription(level);

    public void moveComponent(User user, String resourceId, String componentId,
                              String newParentId, int position) {
        Resource resource = getResource(user, resourceId);
        requireWriteAccess(user, resource.getRepositoryId(), false);
        List<ResourceComponent> oldList = findContainingList(resource.getComponents(), componentId);
        if (oldList == null) throw new IllegalArgumentException("Component not found");
        ResourceComponent component = oldList.stream()
                .filter(c -> c.getId().equals(componentId)).findFirst().orElseThrow();
        List<ResourceComponent> destination;
        if (newParentId == null) {
            destination = resource.getComponents();
        } else {
            ResourceComponent parent = findComponent(resource.getComponents(), newParentId)
                    .orElseThrow(() -> new IllegalArgumentException("Parent component not found"));
            if (contains(component, parent.getId())) {
                throw new IllegalArgumentException("Cannot move a component into its own descendant");
            }
            destination = parent.getChildren();
        }
        oldList.remove(component);
        destination.add(Math.max(0, Math.min(position, destination.size())), component);
        resource.incrementVersion();
    }

    public void associateSubject(User user, String resourceId, String subjectId) {
        Resource resource = getResource(user, resourceId);
        requireWriteAccess(user, resource.getRepositoryId(), false);
        requireRecord(subjects, subjectId);
        if (!resource.getSubjectIds().contains(subjectId)) resource.getSubjectIds().add(subjectId);
        resource.incrementVersion();
    }

    public void associateAgent(User user, ArchiveRecord record, String agentId, String relationship) {
        requireWriteAccess(user, record.getRepositoryId(), false);
        Agent agent = requireRecord(agents, agentId);
        if (!Objects.equals(agent.getRepositoryId(), record.getRepositoryId())) {
            throw new IllegalArgumentException("Agent belongs to another repository");
        }
        if (record instanceof Resource r && !r.getAgentIds().contains(agentId)) r.getAgentIds().add(agentId);
        if (record instanceof Accession a && !a.getAgentIds().contains(agentId)) a.getAgentIds().add(agentId);
        if (record instanceof ResourceComponent c && !c.getAgentIds().contains(agentId)) c.getAgentIds().add(agentId);
        record.addNote("Agent relationship (" + relationship + "): " + agentId);
        record.incrementVersion();
    }

    public void addInstance(User user, Resource resource, ResourceComponent component, Instance instance) {
        requireWriteAccess(user, resource.getRepositoryId(), false);
        if (component == null) resource.addNote("Instance associated: " + instance.getId());
        else component.getInstances().add(instance);
        resource.incrementVersion();
    }

    public void addInstance(User user, Accession accession, Instance instance) {

    public String findingAid(User user, String resourceId) {
        Resource resource = getResource(user, resourceId);
        StringBuilder out = new StringBuilder();
        out.append("Finding Aid: ").append(resource.getFindingAidTitle() == null
                ? resource.getTitle() : resource.getFindingAidTitle()).append('\n');
        if (resource.getFindingAidDate() != null) out.append("Date: ").append(resource.getFindingAidDate()).append('\n');
        out.append("Collection: ").append(resource.getTitle()).append('\n');
        out.append("Dates: ").append(String.join(", ", resource.getDates())).append('\n');
        out.append("Extents: ").append(String.join(", ", resource.getExtents())).append('\n');
        for (String subjectId : resource.getSubjectIds()) {
            Subject subject = subjects.get(subjectId);
            if (subject != null) out.append("Subject: ").append(subject.getTerm()).append('\n');
        }
        if (resource.getFindingAidNote() != null) out.append(resource.getFindingAidNote()).append('\n');
        appendComponentText(out, resource.getComponents(), 0);
        return out.toString();
    }

    public String exportEad(User user, String resourceId) {
        Resource resource = getResource(user, resourceId);
        StringBuilder xml = new StringBuilder();
        xml.append("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n")
                .append("<ead><archdesc level=\"").append(xmlEscape(resource.getLevelOfDescription()))
                .append("\"><did><unittitle>").append(xmlEscape(resource.getTitle()))
                .append("</unittitle>");
        for (String date : resource.getDates()) xml.append("<unitdate>").append(xmlEscape(date)).append("</unitdate>");
        for (String extent : resource.getExtents()) xml.append("<extent>").append(xmlEscape(extent)).append("</extent>");
        xml.append("</did>");
        appendEadComponents(xml, resource.getComponents());
        xml.append("</archdesc></ead>");
        return xml.toString();
    }

    public ImportResult<Resource> importEad(User user, String repositoryId, String xml) {
        requireWriteAccess(user, repositoryId, false);
        ImportResult<Resource> result = new ImportResult<>();
        try {
            Document doc = parseXml(xml);
            Element archdesc = firstElement(doc, "archdesc");
            String title = textOfFirst(archdesc, "unittitle");
            if (title.isBlank()) title = "Untitled imported resource";
            Resource resource = new Resource(repositoryId, title);
            String level = archdesc.getAttribute("level");
            if (!level.isBlank()) resource.setLevelOfDescription(level);
            for (Element date : descendants(archdesc, "unitdate")) resource.getDates().add(date.getTextContent().trim());
            for (Element extent : descendants(archdesc, "extent")) resource.getExtents().add(extent.getTextContent().trim());
            importComponents(archdesc, resource.getComponents());
            resource.setSourceRecordId(findAttribute(doc.getDocumentElement(), "id"));
            resources.put(resource.getId(), resource);
            result.getImported().add(resource);
        } catch (Exception e) {
            result.getErrors().add("Unable to import EAD: " + usefulMessage(e));
        }
        return result;
    }

    public ImportResult<Accession> importAccessionsCsv(User user, String repositoryId, String csv) {
        requireWriteAccess(user, repositoryId, false);
        ImportResult<Accession> result = new ImportResult<>();
        String[] lines = csv.split("\\R");
        if (lines.length == 0 || lines[0].isBlank()) {
            result.getErrors().add("CSV is empty");
            return result;
        }
        List<String> headers = parseCsvLine(lines[0]);
        int titleColumn = headers.indexOf("title");
        if (titleColumn < 0) {
            result.getErrors().add("CSV must contain a 'title' column");
            return result;
        }
        for (int line = 1; line < lines.length; line++) {
            if (lines[line].isBlank()) continue;
            try {
                List<String> row = parseCsvLine(lines[line]);
                if (titleColumn >= row.size() || row.get(titleColumn).isBlank()) {
                    throw new IllegalArgumentException("title is missing");
                }
                Accession accession = new Accession(repositoryId, row.get(titleColumn));
                int dateColumn = headers.indexOf("date");
                int extentColumn = headers.indexOf("extent");
                if (dateColumn >= 0 && dateColumn < row.size()) accession.getDates().add(row.get(dateColumn));
                if (extentColumn >= 0 && extentColumn < row.size()) accession.getExtents().add(row.get(extentColumn));
                int sourceColumn = headers.indexOf("source_id");
                if (sourceColumn >= 0 && sourceColumn < row.size()) accession.setSourceRecordId(row.get(sourceColumn));
                accessions.put(accession.getId(), accession);
                result.getImported().add(accession);
            } catch (RuntimeException e) {
                result.getErrors().add("CSV line " + (line + 1) + ": " + usefulMessage(e));
            }
        }
        return result;
    }

    public ImportResult<Resource> importMarcXml(User user, String repositoryId, String xml) {
        requireWriteAccess(user, repositoryId, false);
        ImportResult<Resource> result = new ImportResult<>();
        try {
            Document doc = parseXml(xml);
            NodeList records = doc.getElementsByTagNameNS("*", "record");
            if (records.getLength() == 0) records = doc.getElementsByTagName("record");
            if (records.getLength() == 0) throw new IllegalArgumentException("No MARC record elements found");
            for (int i = 0; i < records.getLength(); i++) {
                Element record = (Element) records.item(i);
                String title = "";
                NodeList datafields = record.getElementsByTagNameNS("*", "datafield");
                if (datafields.getLength() == 0) datafields = record.getElementsByTagName("datafield");
                for (int j = 0; j < datafields.getLength() && title.isBlank(); j++) {
                    Element field = (Element) datafields.item(j);
                    if (!"245".equals(field.getAttribute("tag"))) continue;
                    NodeList subs = field.getElementsByTagNameNS("*", "subfield");
                    if (subs.getLength() == 0) subs = field.getElementsByTagName("subfield");
                    for (int k = 0; k < subs.getLength(); k++) {
                        Element sub = (Element) subs.item(k);
                        if ("a".equals(sub.getAttribute("code"))) title += sub.getTextContent().trim();
                    }
                }
                if (title.isBlank()) title = "Untitled MARC resource";
                Resource resource = new Resource(repositoryId, title);
                resource.setSourceRecordId(findMarcControlNumber(record));
                resources.put(resource.getId(), resource);
                result.getImported().add(resource);
            }
        } catch (Exception e) {
            result.getErrors().add("Unable to import MARCXML: " + usefulMessage(e));
        }
        return result;
    }

    public ImportResult<ArchiveRecord> importMarcAgentsAndSubjects(User user, String repositoryId, String xml) {
        requireWriteAccess(user, repositoryId, true);
        ImportResult<ArchiveRecord> result = new ImportResult<>();
        try {
            Document doc = parseXml(xml);
            NodeList records = doc.getElementsByTagNameNS("*", "record");
            if (records.getLength() == 0) records = doc.getElementsByTagName("record");
            if (records.getLength() == 0) throw new IllegalArgumentException("No MARC record elements found");
            for (int i = 0; i < records.getLength(); i++) {
                Element record = (Element) records.item(i);
                NodeList fields = record.getElementsByTagNameNS("*", "datafield");
                if (fields.getLength() == 0) fields = record.getElementsByTagName("datafield");
                for (int j = 0; j < fields.getLength(); j++) {
                    Element field = (Element) fields.item(j);
                    String tag = field.getAttribute("tag");
                    String value = concatenateSubfields(field);
                    if (value.isBlank()) continue;
                    if (tag.startsWith("1")) {
                        Agent agent = new Agent(repositoryId, value, "person");
                        agent.setSourceRecordId(findMarcControlNumber(record));
                        agents.put(agent.getId(), agent);
                        result.getImported().add(agent);
                    } else if (tag.startsWith("6") || tag.startsWith("65")) {
                        Subject subject = new Subject(repositoryId, value);
                        subject.setSourceRecordId(findMarcControlNumber(record));
                        subjects.put(subject.getId(), subject);
                        result.getImported().add(subject);
                    }
                }
            }
        } catch (Exception e) {
            result.getErrors().add("Unable to import MARC agent/subject data: " + usefulMessage(e));
        }
        return result;
    }

    public ImportResult<Agent> importEacCpf(User user, String repositoryId, String xml) {
        requireWriteAccess(user, repositoryId, true);
        ImportResult<Agent> result = new ImportResult<>();
        try {
            Document doc = parseXml(xml);
            NodeList names = doc.getElementsByTagNameNS("*", "part");
            StringBuilder name = new StringBuilder();
            for (int i = 0; i < names.getLength(); i++) {
                String part = names.item(i).getTextContent().trim();
                if (!part.isBlank()) {
                    if (name.length() > 0) name.append(' ');
                    name.append(part);
                }
            }
            if (name.length() == 0) name.append(textOfFirst(doc.getDocumentElement(), "nameEntry"));
            if (name.length() == 0) throw new IllegalArgumentException("No agent name found");
            Agent agent = new Agent(repositoryId, name.toString(), "person");
            agent.setSourceRecordId(findAttribute(doc.getDocumentElement(), "id"));
            agents.put(agent.getId(), agent);
            result.getImported().add(agent);
        } catch (Exception e) {
            result.getErrors().add("Unable to import EAC-CPF: " + usefulMessage(e));
        }
        return result;
    }

    private void requireReadAccess(User user, String repositoryId) {
        if (user == null) throw new SecurityException("Authentication required");
        if (hasGlobal(user, Role.SYSTEM_ADMINISTRATOR, Role.APPLICATION_ADMINISTRATOR)) return;
        Role role = getRepository(repositoryId).getUserRoles().get(user.getUsername());
        if (role == null) role = highestApplicableGlobalRole(user);
        if (role == null) throw new SecurityException("No access to repository");
    }

    private void requireWriteAccess(User user, String repositoryId, boolean advanced) {
        requireReadAccess(user, repositoryId);
        if (hasGlobal(user, Role.SYSTEM_ADMINISTRATOR, Role.APPLICATION_ADMINISTRATOR)) return;
        Role role = getRepository(repositoryId).getUserRoles().get(user.getUsername());
        if (role == null) role = highestApplicableGlobalRole(user);
        if (role == Role.REPOSITORY_MANAGER) return;
        if (advanced && role == Role.ADVANCED_DATA_ENTRY) return;
        if (!advanced && (role == Role.BASIC_DATA_ENTRY || role == Role.ADVANCED_DATA_ENTRY)) return;
        throw new SecurityException("Insufficient permission to modify this record");
    }

    private void requireManager(User user, String repositoryId) {
        requireReadAccess(user, repositoryId);
        if (hasGlobal(user, Role.SYSTEM_ADMINISTRATOR, Role.APPLICATION_ADMINISTRATOR)) return;
        Role role = getRepository(repositoryId).getUserRoles().get(user.getUsername());
        if (role == null) role = highestApplicableGlobalRole(user);
        if (role != Role.REPOSITORY_MANAGER) throw new SecurityException("Repository Manager permission required");
    }

    private void requireRole(User user, Role... allowed) {
        if (user == null) throw new SecurityException("Authentication required");
        if (!hasGlobal(user, allowed)) throw new SecurityException("Administrator permission required");
    }

    private boolean hasGlobal(User user, Role... roles) {
        if (user == null) return false;
        for (Role role : roles) if (user.getRoles().contains(role)) return true;
        return false;
    }

    private Role highestApplicableGlobalRole(User user) {
        if (user.getRoles().contains(Role.REPOSITORY_MANAGER)) return Role.REPOSITORY_MANAGER;
        if (user.getRoles().contains(Role.ADVANCED_DATA_ENTRY)) return Role.ADVANCED_DATA_ENTRY;
        if (user.getRoles().contains(Role.BASIC_DATA_ENTRY)) return Role.BASIC_DATA_ENTRY;
        if (user.getRoles().contains(Role.READ_ONLY)) return Role.READ_ONLY;
        return null;
    }

    private static <T> T requireRecord(Map<String, T> records, String id) {
        return Optional.ofNullable(records.get(id))
                .orElseThrow(() -> new IllegalArgumentException("Record not found: " + id));
    }

    private static void verifyVersion(ArchiveRecord record, long expectedVersion) {
        if (record.getVersion() != expectedVersion) {
            throw new IllegalStateException("Record was changed by another user; reload before saving");
        }
    }

    private static Optional<ResourceComponent> findComponent(List<ResourceComponent> components, String id) {
        for (ResourceComponent component : components) {
            if (component.getId().equals(id)) return Optional.of(component);
            Optional<ResourceComponent> nested = findComponent(component.getChildren(), id);
            if (nested.isPresent()) return nested;
        }
        return Optional.empty();
    }

    private static Optional<ResourceComponent> findComponentByIdentifier(
            List<ResourceComponent> components, String identifier) {
        for (ResourceComponent component : components) {
            if (identifier.equals(component.getComponentIdentifier())) return Optional.of(component);
            Optional<ResourceComponent> nested = findComponentByIdentifier(component.getChildren(), identifier);
            if (nested.isPresent()) return nested;
        }
        return Optional.empty();
    }

    private static List<ResourceComponent> findContainingList(List<ResourceComponent> list, String id) {
        for (ResourceComponent component : list) {
            if (component.getId().equals(id)) return list;
            List<ResourceComponent> nested = findContainingList(component.getChildren(), id);
            if (nested != null) return nested;
        }
        return null;
    }

    private static boolean contains(ResourceComponent root, String id) {
        if (root.getId().equals(id)) return true;
        for (ResourceComponent child : root.getChildren()) if (contains(child, id)) return true;
        return false;
    }

    private static void appendComponentText(StringBuilder out, List<ResourceComponent> components, int depth) {
        for (ResourceComponent component : components) {
            out.append("  ".repeat(depth)).append(component.getTitle()).append('\n');
            if (!component.getDates().isEmpty()) out.append("  ".repeat(depth + 1))
                    .append("Dates: ").append(String.join(", ", component.getDates())).append('\n');
            if (!component.getExtents().isEmpty()) out.append("  ".repeat(depth + 1))
                    .append("Extents: ").append(String.join(", ", component.getExtents())).append('\n');
            appendComponentText(out, component.getChildren(), depth + 1);
        }
    }

    private static void appendEadComponents(StringBuilder xml, List<ResourceComponent> components) {
        for (ResourceComponent component : components) {
            xml.append("<c level=\"").append(xmlEscape(component.getLevelOfDescription() == null
                    ? "otherlevel" : component.getLevelOfDescription())).append("\"><did><unittitle>")
                    .append(xmlEscape(component.getTitle())).append("</unittitle></did>");
            appendEadComponents(xml, component.getChildren());
            xml.append("</c>");
        }
    }

    private static void importComponents(Element parent, List<ResourceComponent> destination) {
        NodeList children = parent.getChildNodes();
        for (int i = 0; i < children.getLength(); i++) {
            Node node = children.item(i);
            if (!(node instanceof Element element) || !element.getTagName().toLowerCase(Locale.ROOT).startsWith("c")) continue;
            String title = textOfFirst(element, "unittitle");
            if (title.isBlank()) title = "Untitled component";
            ResourceComponent component = new ResourceComponent(title);
            String level = element.getAttribute("level");
            if (!level.isBlank()) component.setLevelOfDescription(level);
            for (Element date : descendants(element, "unitdate")) component.getDates().add(date.getTextContent().trim());
            for (Element extent : descendants(element, "extent")) component.getExtents().add(extent.getTextContent().trim());
            destination.add(component);
            importComponents(element, component.getChildren());
        }
    }

    private static Document parseXml(String xml) throws Exception {
        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
        factory.setNamespaceAware(true);
        factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
        factory.setFeature("http://xml.org/sax/features/external-general-entities", false);
        factory.setFeature("http://xml.org/sax/features/external-parameter-entities", false);
        factory.setAttribute(XMLConstants.ACCESS_EXTERNAL_DTD, "");
        factory.setAttribute(XMLConstants.ACCESS_EXTERNAL_SCHEMA, "");
        return factory.newDocumentBuilder().parse(new InputSource(new StringReader(xml)));
    }

    private static Element firstElement(Document doc, String name) {
        NodeList list = doc.getElementsByTagNameNS("*", name);
        if (list.getLength() == 0) list = doc.getElementsByTagName(name);
        if (list.getLength() == 0) throw new IllegalArgumentException("Missing <" + name + "> element");
        return (Element) list.item(0);
    }

    private static List<Element> descendants(Element parent, String name) {
        NodeList list = parent.getElementsByTagNameNS("*", name);
        if (list.getLength() == 0) list = parent.getElementsByTagName(name);
        List<Element> result = new ArrayList<>();
        for (int i = 0; i < list.getLength(); i++) result.add((Element) list.item(i));
        return result;
    }

    private static String textOfFirst(Element parent, String name) {
        NodeList list = parent.getElementsByTagNameNS("*", name);
        if (list.getLength() == 0) list = parent.getElementsByTagName(name);
        return list.getLength() == 0 ? "" : list.item(0).getTextContent().trim();
    }

    private static String findAttribute(Element element, String attribute) {
        if (element.hasAttribute(attribute)) return element.getAttribute(attribute);
        return "";
    }

    private static String findMarcControlNumber(Element record) {
        NodeList controls = record.getElementsByTagNameNS("*", "controlfield");
        if (controls.getLength() == 0) controls = record.getElementsByTagName("controlfield");
        for (int i = 0; i < controls.getLength(); i++) {
            Element control = (Element) controls.item(i);
            if ("001".equals(control.getAttribute("tag"))) return control.getTextContent().trim();
        }
        return "";
    }

    private static String concatenateSubfields(Element field) {
        NodeList subs = field.getElementsByTagNameNS("*", "subfield");
        if (subs.getLength() == 0) subs = field.getElementsByTagName("subfield");
        StringBuilder out = new StringBuilder();
        for (int i = 0; i < subs.getLength(); i++) {
            String value = subs.item(i).getTextContent().trim();
            if (!value.isBlank()) {
                if (out.length() > 0) out.append(' ');
                out.append(value);
            }
        }
        return out.toString();
    }

    private static List<String> parseCsvLine(String line) {
        List<String> values = new ArrayList<>();
        StringBuilder value = new StringBuilder();
        boolean quoted = false;
        for (int i = 0; i < line.length(); i++) {
            char c = line.charAt(i);
            if (c == '"') {
                if (quoted && i + 1 < line.length() && line.charAt(i + 1) == '"') {
                    value.append('"');
                    i++;
                } else quoted = !quoted;
            } else if (c == ',' && !quoted) {
                values.add(value.toString().trim());
                value.setLength(0);
            } else value.append(c);
        }
        values.add(value.toString().trim());
        return values;
    }

    private static String xmlEscape(String value) {
        if (value == null) return "";
        return value.replace("&", "&amp;").replace("<", "&lt;")
                .replace(">", "&gt;").replace("\"", "&quot;").replace("'", "&apos;");
    }

    private static String safe(String value) {
        return value == null ? "" : value.toLowerCase(Locale.ROOT);
    }

    private static String usefulMessage(Exception e) {
        return e.getMessage() == null ? e.getClass().getSimpleName() : e.getMessage();
    }

    private static String hash(String value) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] bytes = digest.digest(value.getBytes(StandardCharsets.UTF_8));
            StringBuilder out = new StringBuilder();
            for (byte b : bytes) out.append(String.format("%02x", b));
            return out.toString();
        } catch (Exception e) {
            throw new IllegalStateException("Unable to hash password", e);
        }
    }

    private static boolean constantTimeEquals(String left, String right) {
        return MessageDigest.isEqual(left.getBytes(StandardCharsets.UTF_8),
                right.getBytes(StandardCharsets.UTF_8));
    }
