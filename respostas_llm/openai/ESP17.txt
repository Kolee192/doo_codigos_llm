import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public abstract class Person {
    private final String name;
    private final String title;
    private final String notes;
    private final List<Booking> bookings = new ArrayList<>();
    private final List<BankAccount> bankAccounts = new ArrayList<>();
    private final List<Fax> faxes = new ArrayList<>();
    private final List<Phone> phones = new ArrayList<>();
    private final List<Email> emails = new ArrayList<>();
    private final List<SocialMediaAccount> socialMediaAccounts = new ArrayList<>();
    private final List<PersonLocation> locations = new ArrayList<>();
    private final List<Connection> connections = new ArrayList<>();

    protected Person(String name, String title, String notes) {
        this.name = name;
        this.title = title;
        this.notes = notes;
    }

    public String getName() { return name; }
    public String getTitle() { return title; }
    public String getNotes() { return notes; }

    public List<Booking> getBookings() { return Collections.unmodifiableList(bookings); }
    public List<BankAccount> getBankAccounts() { return Collections.unmodifiableList(bankAccounts); }
    public List<Fax> getFaxes() { return Collections.unmodifiableList(faxes); }
    public List<Phone> getPhones() { return Collections.unmodifiableList(phones); }
    public List<Email> getEmails() { return Collections.unmodifiableList(emails); }
    public List<SocialMediaAccount> getSocialMediaAccounts() { return Collections.unmodifiableList(socialMediaAccounts); }
    public List<PersonLocation> getLocations() { return Collections.unmodifiableList(locations); }
    public List<Connection> getConnections() { return Collections.unmodifiableList(connections); }

    void addBooking(Booking booking) { bookings.add(booking); }
    void addBankAccount(BankAccount account) { bankAccounts.add(account); }
    void addFax(Fax fax) { faxes.add(fax); }
    void addPhone(Phone phone) { phones.add(phone); }
    void addEmail(Email email) { emails.add(email); }
    void addSocialMediaAccount(SocialMediaAccount account) { socialMediaAccounts.add(account); }
    void addLocation(PersonLocation location) { locations.add(location); }
    void addConnection(Connection connection) { connections.add(connection); }
}

import java.math.BigDecimal;
import java.math.MathContext;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Car {
    private final CarModel model;
    private final String color;
    private final String licensePlate;
    private final int productionYear;
    private String status;
    private final FleetLocation fleetLocation;
    private final List<Rental> rentals = new ArrayList<>();

    public Car(CarModel model, String color, String licensePlate, int productionYear,
               String status, FleetLocation fleetLocation) {
        if (model == null || fleetLocation == null) throw new IllegalArgumentException("Model and fleet location are required.");
        this.model = model;
        this.color = color;
        this.licensePlate = licensePlate;
        this.productionYear = productionYear;
        this.status = status;
        this.fleetLocation = fleetLocation;
    }

    public CarModel getModel() { return model; }
    public String getColor() { return color; }
    public String getLicensePlate() { return licensePlate; }
    public int getProductionYear() { return productionYear; }
    public String getStatus() { return status; }
    public void setStatus(String status) { this.status = status; }
    public FleetLocation getFleetLocation() { return fleetLocation; }
    public List<Rental> getRentals() { return Collections.unmodifiableList(rentals); }

    void addRental(Rental rental) { rentals.add(rental); }

    public BigDecimal getAverageRentalPrice() {
        if (rentals.isEmpty()) return BigDecimal.ZERO;
        BigDecimal total = rentals.stream().map(Rental::getPrice).reduce(BigDecimal.ZERO, BigDecimal::add);
        return total.divide(BigDecimal.valueOf(rentals.size()), MathContext.DECIMAL64);
    }
}

// File: Rental.java
import java.math.BigDecimal;

public class Rental {
    private final Booking booking;
    private final Car car;
    private final BigDecimal price;
    private final RentalType type;
    private CarPickup pickup;
    private CarDropoff dropoff;

    public Rental(Booking booking, Car car, BigDecimal price, RentalType type) {
        if (booking == null || car == null || price == null || type == null) {
            throw new IllegalArgumentException("Booking, car, price, and rental type are required.");
        }
        this.booking = booking;
        this.car = car;
        this.price = price;
        this.type = type;
        booking.addRental(this);
        car.addRental(this);
    }

    public Booking getBooking() { return booking; }
    public Car getCar() { return car; }
    public BigDecimal getPrice() { return price; }
    public RentalType getType() { return type; }
    public CarPickup getPickup() { return pickup; }
    public CarDropoff getDropoff() { return dropoff; }

    public void setPickup(CarPickup pickup) {
        if (pickup != null && pickup.getRental() != this) throw new IllegalArgumentException("Pickup belongs to another rental.");
        this.pickup = pickup;
    }

    public void setDropoff(CarDropoff dropoff) {
        if (dropoff != null && dropoff.getRental() != this) throw new IllegalArgumentException("Dropoff belongs to another rental.");
        this.dropoff = dropoff;
    }

    public boolean isBookedToday() {
        if (pickup == null) return false;
        return !java.time.LocalDate.now().isBefore(pickup.getDate())
                && (dropoff == null || !java.time.LocalDate.now().isAfter(dropoff.getDate()));
    }
}
