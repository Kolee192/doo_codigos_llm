// File: LocalFileStorage.java
import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.UUID;

public final class LocalFileStorage implements StorageBackend {
    private final Path root;

    public LocalFileStorage(Path root) throws IOException {
        this.root = root.toAbsolutePath().normalize();
        Files.createDirectories(this.root);
    }

    @Override
    public StoredObject store(InputStream input, String originalFileName) throws IOException {
        String key = UUID.randomUUID().toString();
        Path target = resolve(key);
        long size;
        try (OutputStream output = Files.newOutputStream(target)) {
            size = input.transferTo(output);
        }
        return new StoredObject(key, size);
    }

    @Override
    public InputStream open(String storageKey) throws IOException {
        return Files.newInputStream(resolve(storageKey));
    }

    @Override
    public void copyTo(String storageKey, OutputStream output) throws IOException {
        try (InputStream input = open(storageKey)) {
            input.transferTo(output);
        }
    }

    @Override
    public void delete(String storageKey) throws IOException {
        Files.deleteIfExists(resolve(storageKey));
    }

    private Path resolve(String key) {
        Path path = root.resolve(key).normalize();
        if (!path.startsWith(root)) {
            throw new IllegalArgumentException("Invalid storage key");
        }
        return path;
    }
}

    private final String prefix;

    public SimpleIdentifierProvider(String prefix) {

    @Override
    public String mint(String datasetId) {
        return prefix + "/" + sequence.incrementAndGet();

public final class ArchiveService {
    private final Map<String, Dataset> datasets = new ConcurrentHashMap<>();
    private final Map<String, Map<String, String>> rememberedMetadata = new ConcurrentHashMap<>();
    private final Map<String, java.util.Set<String>> delegates = new ConcurrentHashMap<>();

    private final List<DownloadEvent> downloads = new CopyOnWriteArrayList<>();
    private final List<String> requiredMetadata = new CopyOnWriteArrayList<>(

    public ArchiveService(StorageBackend storage, IdentifierProvider identifiers, IdentityProvider identities) {
        this.storage = storage;

        this.identities = identities;

    public Dataset createDataset(
            String depositorId,
            String ownerId,
            String title,
            String description,
            DepositChannel channel) {
        requireKnownUser(depositorId);
        requireKnownUser(ownerId);
        if (!depositorId.equals(ownerId)
                && !delegates.getOrDefault(ownerId, java.util.Set.of()).contains(depositorId)) {
            throw new SecurityException("Depositor is not authorized to act for this owner");
        }

        Dataset dataset = new Dataset(UUID.randomUUID().toString(), ownerId, title, description, channel);
        Map<String, String> remembered = rememberedMetadata.get(ownerId);
        if (remembered != null) {
            remembered.forEach(dataset::setMetadata);
        }
        datasets.put(dataset.getId(), dataset);
        return dataset;
    }

    public void authorizeDelegate(String ownerId, String delegateId) {
        requireKnownUser(ownerId);
        requireKnownUser(delegateId);
        delegates.computeIfAbsent(ownerId, ignored -> ConcurrentHashMap.newKeySet()).add(delegateId);
    }

    public void updateMetadata(String actorId, String datasetId, String key, String value) {
        Dataset dataset = getDataset(datasetId);
        requireCanDeposit(actorId, dataset);
        synchronized (dataset) {
            if ("title".equalsIgnoreCase(key)) {
                dataset.setTitle(value);
            } else if ("description".equalsIgnoreCase(key)) {
                dataset.setDescription(value);
            } else if ("license".equalsIgnoreCase(key)) {
                dataset.setLicense(value);
                dataset.setMetadata("license", value);
            } else if ("language".equalsIgnoreCase(key)) {
                dataset.setLanguage(value);
            } else {
                dataset.setMetadata(key, value);
            }
        }
    }

    public void rememberMetadata(String userId, Map<String, String> metadata) {
        requireKnownUser(userId);
        rememberedMetadata.computeIfAbsent(userId, ignored -> new ConcurrentHashMap<>()).putAll(metadata);
    }

    public Map<String, String> getRememberedMetadata(String userId) {
        return Map.copyOf(rememberedMetadata.getOrDefault(userId, Map.of()));
    }

    public DatasetFile uploadFile(
            String actorId,
            String datasetId,
            String fileName,
            String mediaType,
            InputStream input) throws IOException {
        Dataset dataset = getDataset(datasetId);
        requireCanDeposit(actorId, dataset);

        MessageDigest digest = sha256();
        StoredObject stored;
        try (DigestInputStream digestInput = new DigestInputStream(input, digest)) {
            stored = storage.store(digestInput, fileName);
        }
        DatasetFile file = new DatasetFile(
                fileName,
                mediaType == null ? "application/octet-stream" : mediaType,
                stored.getKey(),
                stored.getSize(),
                HexFormat.of().formatHex(digest.digest()));

        synchronized (dataset) {
            dataset.addFile(file);
        }
        return file;
    }

    public void addExternalRepositoryLink(String actorId, String datasetId, String url) {

        if (url == null || !(url.startsWith("https://") || url.startsWith("http://"))) {
            throw new IllegalArgumentException("A valid HTTP(S) repository URL is required");

    public void submitForReview(String actorId, String datasetId) {
        Dataset dataset = getDataset(datasetId);
        requireCanDeposit(actorId, dataset);
        synchronized (dataset) {
            if (dataset.getStatus() == DatasetStatus.DISPOSED) {
                throw new IllegalStateException("Disposed datasets cannot be submitted");
            }
            List<String> missing = new ArrayList<>();
            for (String required : requiredMetadata) {
                String value = "license".equals(required)
                        ? dataset.getLicense()
                        : dataset.getMetadata().get(required);
                if (value == null || value.isBlank()) {
                    missing.add(required);
                }
            }
            if ((dataset.getTitle() == null || dataset.getTitle().isBlank()) && !missing.contains("title")) {
                missing.add("title");
            }
            if (!missing.isEmpty()) {
                throw new IllegalStateException("Missing required metadata: " + String.join(", ", missing));
            }
            dataset.setStatus(DatasetStatus.IN_REVIEW);
        }
    }

    public void setRequiredMetadata(List<String> fields) {
        requiredMetadata.clear();
        requiredMetadata.addAll(fields);
    }

    public void approvePublication(String administratorId, String datasetId) {
        requireKnownUser(administratorId);
        Dataset dataset = getDataset(datasetId);
        synchronized (dataset) {
            if (dataset.getStatus() != DatasetStatus.IN_REVIEW) {
                throw new IllegalStateException("Dataset is not awaiting review");
            }
            if (dataset.getDoi() == null) {
                dataset.setDoi(identifiers.mint(dataset.getId()));
            }
            dataset.addVersion();
            dataset.setStatus(DatasetStatus.PUBLISHED);
            rememberMetadata(dataset.getOwnerId(), dataset.getMetadata());
        }
    }

    public void rejectPublication(String administratorId, String datasetId, String reason) {
        requireKnownUser(administratorId);
        Dataset dataset = getDataset(datasetId);
        if (dataset.getStatus() != DatasetStatus.IN_REVIEW) {
            throw new IllegalStateException("Dataset is not awaiting review");
        }
        dataset.setMetadata("reviewRejectionReason", reason);
        dataset.setStatus(DatasetStatus.DRAFT);
    }

    public void scheduleDisposal(String actorId, String datasetId, Instant disposalDate) {
        Dataset dataset = getDataset(datasetId);
        requireCanDeposit(actorId, dataset);
        if (disposalDate == null || !disposalDate.isAfter(Instant.now())) {
            throw new IllegalArgumentException("Disposal date must be in the future");
        }
        dataset.setDisposalDate(disposalDate);
    }

    public void approveScheduledDisposal(String administratorId, String datasetId) {
        requireKnownUser(administratorId);
        Dataset dataset = getDataset(datasetId);
        if (dataset.getDisposalDate() == null) {
            throw new IllegalStateException("No disposal is scheduled");
        }
        dataset.setMetadata("disposalApproved", "true");
    }

    public void disposeDueDatasets(Instant now) throws IOException {
        for (Dataset dataset : datasets.values()) {
            if (dataset.getDisposalDate() != null
                    && !dataset.getDisposalDate().isAfter(now)
                    && "true".equals(dataset.getMetadata().get("disposalApproved"))) {
                synchronized (dataset) {
                    for (DatasetFile file : dataset.getFiles()) {
                        storage.delete(file.getStorageKey());
                    }
                    dataset.setStatus(DatasetStatus.DISPOSED);
                }
            }
        }
    }

    public Dataset registerExternalDataset(
            String actorId,
            String title,
            String description,

        Dataset dataset = createDataset(actorId, actorId, title, description, DepositChannel.ADMIN_IMPORT);

        dataset.setExternalUrl(repositoryUrl);
        dataset.setLicense(metadata.get("license"));
        dataset.setStatus(DatasetStatus.PUBLISHED);
        dataset.setDoi(identifiers.mint(dataset.getId()));

    public Dataset getDataset(String datasetId) {

            throw new IllegalArgumentException("Unknown dataset: " + datasetId);

    public List<Dataset> queryArchive(String administratorId) {

        return datasets.values().stream()
                .sorted(Comparator.comparing(Dataset::getCreatedAt))

    public List<Dataset> search(String query, String language) {
        String term = query == null ? "" : query.toLowerCase();
        return datasets.values().stream()
                .filter(this::isPubliclyAvailable)
                .filter(dataset -> language == null || language.isBlank()
                        || language.equalsIgnoreCase(dataset.getLanguage()))
                .filter(dataset -> searchableText(dataset).contains(term))
                .sorted(Comparator.comparing(Dataset::getTitle, String.CASE_INSENSITIVE_ORDER))
                .toList();
    }

    public List<Dataset> searchViaPrimo(String query, String language) {
        return search(query, language);
    }

    public List<DatasetFile> examineFiles(String userId, String datasetId) {
        Dataset dataset = getDataset(datasetId);
        requireCanRead(userId, dataset);
        return dataset.getFiles();
    }

    public InputStream downloadFile(
            String userId,
            String datasetId,
            String storageKey,
            String country,
            String sector) throws IOException {
        Dataset dataset = getDataset(datasetId);
        requireCanRead(userId, dataset);
        DatasetFile file = dataset.getFiles().stream()
                .filter(candidate -> candidate.getStorageKey().equals(storageKey))
                .findFirst()
                .orElseThrow(() -> new IllegalArgumentException("File is not part of this dataset"));
        downloads.add(new DownloadEvent(datasetId, Instant.now(), country, sector));
        return storage.open(file.getStorageKey());
    }

    public boolean verifyIntegrity(String datasetId) throws IOException {
        Dataset dataset = getDataset(datasetId);
        for (DatasetFile file : dataset.getFiles()) {
            MessageDigest digest = sha256();
            try (InputStream input = storage.open(file.getStorageKey())) {
                byte[] buffer = new byte[8192];
                int read;
                while ((read = input.read(buffer)) != -1) {
                    digest.update(buffer, 0, read);
                }
            }
            if (!HexFormat.of().formatHex(digest.digest()).equalsIgnoreCase(file.getSha256())) {
                return false;
            }
        }
        return true;
    }

    public void recordCitation(String datasetId) {

    public String getExampleCitation(String datasetId) {

        String year = String.valueOf(dataset.getCreatedAt().atZone(java.time.ZoneOffset.UTC).getYear());
        return dataset.getOwnerId() + " (" + year + "). " + dataset.getTitle() + ". "

    public void exportAll(OutputStream output) throws IOException {
        try (ZipOutputStream zip = new ZipOutputStream(output)) {
            for (Dataset dataset : datasets.values()) {
                String manifestName = "datasets/" + dataset.getId() + "/metadata.tsv";
                zip.putNextEntry(new ZipEntry(manifestName));
                String metadata = "id\ttitle\towner\tstatus\tdoi\texternalUrl\n"
                        + tsv(dataset.getId()) + "\t" + tsv(dataset.getTitle()) + "\t"
                        + tsv(dataset.getOwnerId()) + "\t" + dataset.getStatus() + "\t"
                        + tsv(dataset.getDoi()) + "\t" + tsv(dataset.getExternalUrl()) + "\n";
                zip.write(metadata.getBytes(java.nio.charset.StandardCharsets.UTF_8));
                zip.closeEntry();

                for (DatasetFile file : dataset.getFiles()) {
                    zip.putNextEntry(new ZipEntry(
                            "datasets/" + dataset.getId() + "/files/" + safeFileName(file.getFileName())));
                    storage.copyTo(file.getStorageKey(), zip);
                    zip.closeEntry();
                }
            }
            zip.finish();
        }
    }

    private boolean isPubliclyAvailable(Dataset dataset) {
        return dataset.getStatus() == DatasetStatus.PUBLISHED
                && (dataset.getEmbargoUntil() == null || !dataset.getEmbargoUntil().isAfter(Instant.now()));
    }

    private String searchableText(Dataset dataset) {
        return (String.valueOf(dataset.getTitle()) + " " + String.valueOf(dataset.getDescription()) + " "
                + dataset.getMetadata().toString()).toLowerCase();
    }

    private void requireCanRead(String userId, Dataset dataset) {
        requireKnownUser(userId);
        if (isPubliclyAvailable(dataset)
                || dataset.getOwnerId().equals(userId)
                || dataset.getCollaborators().containsKey(userId)) {
            return;
        }
        throw new SecurityException("User is not authorized to access this dataset");
    }

    private void requireCanDeposit(String userId, Dataset dataset) {
        requireKnownUser(userId);
        if (dataset.getOwnerId().equals(userId)
                || dataset.getCollaborators().get(userId) == AccessRole.DEPOSIT) {
            return;
        }
        throw new SecurityException("User is not authorized to modify this dataset");
    }

    private void requireOwner(String userId, Dataset dataset) {
        requireKnownUser(userId);
        if (!dataset.getOwnerId().equals(userId)) {
            throw new SecurityException("Only the dataset owner may perform this action");
        }
    }

    private void requireKnownUser(String userId) {
        if (!identities.isKnownUser(userId)) {
            throw new SecurityException("Unknown user");
        }
    }

    private static MessageDigest sha256() {
        try {
            return MessageDigest.getInstance("SHA-256");
        } catch (NoSuchAlgorithmException exception) {
            throw new IllegalStateException("SHA-256 is unavailable", exception);
        }
    }

    private static String normalize(String value) {
        return value == null || value.isBlank() ? "unknown" : value;
    }

    private static String tsv(String value) {
        return value == null ? "" : value.replace('\t', ' ').replace('\n', ' ');
    }

    private static String safeFileName(String fileName) {
        if (fileName == null || fileName.isBlank()) {
            return "unnamed";
        }
        return fileName.replace("/", "_").replace("\\", "_").replace("..", "_");
    }
