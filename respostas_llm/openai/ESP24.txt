import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Set;

public class GatewayDevice implements Device {
    private final String id;
    private final String name;
    private final Set<String> connectedDeviceIds = new LinkedHashSet<>();
    private final List<String> communicationLog = new ArrayList<>();

    public GatewayDevice(String id, String name) {
        this.id = id;
        this.name = name;
    }

    @Override
    public String getId() {
        return id;
    }

    @Override
    public String getName() {
        return name;
    }

    public void register(Device device) {
        connectedDeviceIds.add(device.getId());
    }

    public boolean isConnected(String deviceId) {
        return connectedDeviceIds.contains(deviceId);
    }

    public void transmit(String deviceId, String command) {
        if (!isConnected(deviceId)) {
            throw new IllegalArgumentException("Device is not registered with the gateway: " + deviceId);
        }
        communicationLog.add(LocalDateTime.now() + " -> " + deviceId + ": " + command);
    }

    public List<String> getCommunicationLog() {
        return Collections.unmodifiableList(communicationLog);
    }
}

import java.time.DayOfWeek;
import java.time.LocalDateTime;
import java.util.EnumMap;
import java.util.HashMap;
import java.util.Map;

public class Thermostat implements Device {
    private final String id;
    private final String name;
    private final String room;
    private final Map<DayOfWeek, Map<Integer, Integer>> hourlyTargets = new EnumMap<>(DayOfWeek.class);
    private double currentTemperatureFahrenheit;
    private int targetTemperatureFahrenheit;
    private int defaultTemperatureFahrenheit = 70;
    private LocalDateTime manualOverrideUntil;

    public Thermostat(String id, String name, String room, double currentTemperatureFahrenheit) {
        this.id = id;
        this.name = name;
        this.room = room;
        this.currentTemperatureFahrenheit = currentTemperatureFahrenheit;
        this.targetTemperatureFahrenheit = defaultTemperatureFahrenheit;
    }

    @Override
    public String getId() {
        return id;
    }

    @Override
    public String getName() {
        return name;
    }

    public String getRoom() {
        return room;
    }

    public double readTemperature(TemperatureUnit unit) {
        if (unit == TemperatureUnit.CELSIUS) {
            return (currentTemperatureFahrenheit - 32.0) * 5.0 / 9.0;
        }
        return currentTemperatureFahrenheit;
    }

    public double getCurrentTemperatureFahrenheit() {
        return currentTemperatureFahrenheit;
    }

    public void setCurrentTemperatureFahrenheit(double temperature) {
        if (!Double.isFinite(temperature)) {
            throw new IllegalArgumentException("Temperature must be finite");
        }
        currentTemperatureFahrenheit = temperature;
    }

    public int getTargetTemperatureFahrenheit() {
        return targetTemperatureFahrenheit;
    }

    public void setTargetTemperature(double value, TemperatureUnit unit) {
        setTargetTemperature(value, unit, LocalDateTime.now());
    }

    public void setTargetTemperature(double value, TemperatureUnit unit, LocalDateTime now) {
        int fahrenheit = toFahrenheit(value, unit);
        validateTarget(fahrenheit);
        targetTemperatureFahrenheit = fahrenheit;
        manualOverrideUntil = nextHour(now);
    }

    public void setPlannedTarget(int fahrenheit) {
        validateTarget(fahrenheit);
        defaultTemperatureFahrenheit = fahrenheit;
        targetTemperatureFahrenheit = fahrenheit;
        manualOverrideUntil = null;
    }

    public void setDefaultTarget(int fahrenheit) {
        validateTarget(fahrenheit);
        defaultTemperatureFahrenheit = fahrenheit;
    }

    public void scheduleTarget(DayOfWeek day, int hour, int fahrenheit) {
        validateHour(hour);
        validateTarget(fahrenheit);
        hourlyTargets.computeIfAbsent(day, ignored -> new HashMap<>()).put(hour, fahrenheit);
    }

    public void applySchedule(LocalDateTime time) {
        if (manualOverrideUntil != null && time.isBefore(manualOverrideUntil)) {
            return;
        }
        manualOverrideUntil = null;
        targetTemperatureFahrenheit = hourlyTargets
                .getOrDefault(time.getDayOfWeek(), Map.of())
                .getOrDefault(time.getHour(), defaultTemperatureFahrenheit);
    }

    private static int toFahrenheit(double value, TemperatureUnit unit) {
        if (!Double.isFinite(value)) {
            throw new IllegalArgumentException("Temperature must be finite");
        }
        double fahrenheit = unit == TemperatureUnit.CELSIUS ? value * 9.0 / 5.0 + 32.0 : value;
        return (int) Math.round(fahrenheit);
    }

    private static void validateTarget(int fahrenheit) {
        if (fahrenheit < 60 || fahrenheit > 80) {
            throw new IllegalArgumentException("Thermostat target must be between 60 and 80 F");
        }
    }

    private static void validateHour(int hour) {
        if (hour < 0 || hour > 23) {
            throw new IllegalArgumentException("Hour must be between 0 and 23");
        }
    }

    private static LocalDateTime nextHour(LocalDateTime now) {
        return now.withMinute(0).withSecond(0).withNano(0).plusHours(1);
    }
}

            this.sensorId = sensorId;

        @Override

            return timestamp + " breach at " + location + " (" + sensorId + ")";
        }
    }

    private boolean armed = true;
    private boolean soundAlarmActive;
    private boolean lightAlarmActive;
    private final List<BreachEvent> events = new ArrayList<>();

    public boolean isArmed() {
        return armed;
    }

    public void setArmed(boolean armed) {
        this.armed = armed;
    }

    public boolean isSoundAlarmActive() {
        return soundAlarmActive;
    }

    public boolean isLightAlarmActive() {
        return lightAlarmActive;
    }

    public void activateFor(ContactSensor sensor, LocalDateTime timestamp) {
        if (armed && sensor.isOpen()) {
            soundAlarmActive = true;
            lightAlarmActive = true;
            events.add(new BreachEvent(sensor.getId(), sensor.getLocation(), timestamp));
        }
    }

    public void silence() {
        soundAlarmActive = false;
        lightAlarmActive = false;
    }

    public List<BreachEvent> getEvents() {
        return Collections.unmodifiableList(events);
    }

import java.time.LocalDate;
import java.time.YearMonth;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class MonthlyPlan {
    private final YearMonth month;
    private final Map<Integer, List<PlannedPeriod>> dailyPeriods = new HashMap<>();

    public MonthlyPlan(YearMonth month) {
        this.month = month;
    }

    public YearMonth getMonth() {
        return month;
    }

    public void addOrReplacePeriod(int dayOfMonth, PlannedPeriod period) {
        if (dayOfMonth < 1 || dayOfMonth > month.lengthOfMonth()) {
            throw new IllegalArgumentException("Day is not in the plan month");
        }
        List<PlannedPeriod> periods = dailyPeriods.computeIfAbsent(dayOfMonth, ignored -> new ArrayList<>());
        periods.removeIf(existing -> existing.getStartTime().equals(period.getStartTime()));
        if (periods.size() >= 4) {
            throw new IllegalStateException("A day can contain at most four planned periods");
        }
        periods.add(period);
        periods.sort(Comparator.comparing(PlannedPeriod::getStartTime));
    }

    public List<PlannedPeriod> getPeriods(LocalDate date) {
        if (!YearMonth.from(date).equals(month)) {
            return List.of();
        }
        return Collections.unmodifiableList(dailyPeriods.getOrDefault(date.getDayOfMonth(), List.of()));
    }
}

        gateway.transmit(humidistatId, "set humidity target");
        humidistat.setTargetHumidity(value, time);
    }

    public void setPowerSwitch(String switchId, boolean on, LocalDateTime time) {
        PowerSwitch powerSwitch = requireDevice(powerSwitches, switchId, "power switch");
        gateway.transmit(switchId, on ? "turn on" : "turn off");
        powerSwitch.setOnManually(on, time);
    }

    public void updateContactSensor(String sensorId, ContactState state, LocalDateTime time) {
        ContactSensor sensor = requireDevice(contactSensors, sensorId, "contact sensor");
        gateway.transmit(sensorId, "contact state " + state);
        sensor.setState(state);
        if (state == ContactState.OPEN) {
            securitySystem.activateFor(sensor, time);
        }
    }

    public void applySchedules(LocalDateTime time) {
        thermostats.values().forEach(device -> device.applySchedule(time));
        humidistats.values().forEach(device -> device.applySchedule(time));
        powerSwitches.values().forEach(device -> device.applySchedule(time));
        applyMonthlyPlan(time);
    }

    public void applyMonthlyPlan(LocalDateTime time) {
        MonthlyPlan plan = planner.getPlan(YearMonth.from(time));
        if (plan == null) {
            return;
        }
        List<PlannedPeriod> periods = new ArrayList<>(plan.getPeriods(time.toLocalDate()));
        periods.sort(Comparator.comparing(PlannedPeriod::getStartTime));
        PlannedPeriod selected = null;
        for (PlannedPeriod period : periods) {
            if (!period.getStartTime().isAfter(time.toLocalTime())) {
                selected = period;
            }
        }
        if (selected == null) {
            return;
        }
        selected.getTemperaturesFahrenheit().forEach((id, value) -> {
            Thermostat device = thermostats.get(id);
            if (device != null) {
                gateway.transmit(id, "apply planned temperature");
                device.setPlannedTarget(value);
            }
        });
        selected.getHumidities().forEach((id, value) -> {
            Humidistat device = humidistats.get(id);
            if (device != null) {
                gateway.transmit(id, "apply planned humidity");
                device.setPlannedTarget(value);
            }
        });
        selected.getSwitchStates().forEach((id, value) -> {
            PowerSwitch device = powerSwitches.get(id);
            if (device != null) {
                gateway.transmit(id, "apply planned switch state");
                device.setPlannedState(value);
            }
        });
        selected.getContactStates().forEach((id, value) -> {
            ContactSensor device = contactSensors.get(id);
            if (device != null) {
                gateway.transmit(id, "apply planned contact state");
                device.setState(value);
                if (value == ContactState.OPEN) {
                    securitySystem.activateFor(device, time);
                }
            }
        });
        if (selected.getSecurityArmed() != null) {
            securitySystem.setArmed(selected.getSecurityArmed());
        }
    }

    public void recordSystemOutage(LocalDateTime startedAt) {
        if (outageStartedAt != null) {
            throw new IllegalStateException("An outage is already in progress");
        }
        outageStartedAt = startedAt;
    }

    public void recordSystemRestored(LocalDateTime restoredAt) {
        if (outageStartedAt == null) {
            throw new IllegalStateException("No outage is in progress");
        }
        downtimePeriods.add(new MonthlyReport.DowntimePeriod(outageStartedAt, restoredAt));
        outageStartedAt = null;
    }

    public MonthlyReport generateMonthlyReport(YearMonth month, LocalDate today) {
        if (month.isAfter(YearMonth.from(today)) || month.isBefore(YearMonth.from(today).minusYears(2))) {
            throw new IllegalArgumentException("Reports are available only for the current and previous two years");
        }
        MonthlyReport report = new MonthlyReport(month);
        for (String id : thermostats.keySet()) {
            report.addTemperatureStatistics(id, calculateDailyStats(id, ReadingType.TEMPERATURE, month));
        }
        for (String id : humidistats.keySet()) {
            report.addHumidityStatistics(id, calculateDailyStats(id, ReadingType.HUMIDITY, month));
        }
        securitySystem.getEvents().stream()
                .filter(event -> YearMonth.from(event.getTimestamp()).equals(month))
                .forEach(report::addSecurityBreach);
        for (MonthlyReport.DowntimePeriod period : downtimePeriods) {
            if (YearMonth.from(period.getStartedAt()).equals(month)
                    || YearMonth.from(period.getEndedAt()).equals(month)) {
                report.addDowntime(period);
            }
        }
        if (outageStartedAt != null && YearMonth.from(outageStartedAt).equals(month)) {
            report.addDowntime(new MonthlyReport.DowntimePeriod(outageStartedAt, null));
        }
        return report;
    }

    private List<MonthlyReport.DailyStats> calculateDailyStats(String deviceId, ReadingType type, YearMonth month) {
        Map<LocalDate, List<SensorReading>> byDay = new LinkedHashMap<>();
        for (SensorReading reading : readings) {
            if (reading.getDeviceId().equals(deviceId)
                    && reading.getType() == type
                    && YearMonth.from(reading.getTimestamp()).equals(month)) {
                byDay.computeIfAbsent(reading.getTimestamp().toLocalDate(), ignored -> new ArrayList<>()).add(reading);
            }
        }
        List<MonthlyReport.DailyStats> result = new ArrayList<>();
        byDay.forEach((date, samples) -> {
            double sum = 0;
            SensorReading minimum = samples.get(0);
            SensorReading maximum = samples.get(0);
            for (SensorReading sample : samples) {
                sum += sample.getValue();
                if (sample.getValue() < minimum.getValue()) {
                    minimum = sample;
                }
                if (sample.getValue() > maximum.getValue()) {
                    maximum = sample;
                }
            }
            result.add(new MonthlyReport.DailyStats(date, sum / samples.size(),
                    minimum.getValue(), minimum.getTimestamp(),
                    maximum.getValue(), maximum.getTimestamp(), samples.size()));
        });
        result.sort(Comparator.comparing(MonthlyReport.DailyStats::getDate));
        return result;
    }

    private void ensureUniqueId(String id) {
        if (thermostats.containsKey(id) || humidistats.containsKey(id)
                || contactSensors.containsKey(id) || powerSwitches.containsKey(id)
                || gateway.getId().equals(id)) {
            throw new IllegalArgumentException("Device ID already exists: " + id);
        }
    }

    private static void requireCapacity(Map<?, ?> devices, int maximum, String type) {
        if (devices.size() >= maximum) {
            throw new IllegalStateException("Maximum number of " + type + " reached");
        }
    }

    private static <T> T requireDevice(Map<String, T> devices, String id, String type) {
        T device = devices.get(id);
        if (device == null) {
            throw new IllegalArgumentException("Unknown " + type + ": " + id);
        }
        return device;
    }
}
