import java.util.Set;

public enum Role {
    ADMINISTRATOR,
    SECURITY_OFFICER,
    DATA_MANAGER,
    PORTAL_MANAGER,
    REGISTERED_VISITOR;

    public static boolean hasAny(UserAccount account, Role... required) {
        if (account == null) {
            return false;
        }
        Set<Role> roles = account.getRoles();
        for (Role role : required) {
            if (roles.contains(role) || roles.contains(ADMINISTRATOR)) {
                return true;
            }
        }
        return false;
    }
}

    private final ResourceType type;
    private final Map<String, String> metadata = new LinkedHashMap<>();
    private byte[] content = new byte[0];

    public Resource(ResourceType type, Map<String, String> metadata) {
        this(UUID.randomUUID().toString(), type, metadata);
    }

    public Resource(String id, ResourceType type, Map<String, String> metadata) {
        if (id == null || id.isBlank()) {
            throw new IllegalArgumentException("id must not be blank");
        }
        this.id = id;
        this.type = Objects.requireNonNull(type);
        if (metadata != null) {
            this.metadata.putAll(metadata);
        }
    }

    public String getId() {
        return id;
    }

    public ResourceType getType() {
        return type;
    }

    public synchronized Map<String, String> getMetadata() {
        return Collections.unmodifiableMap(new LinkedHashMap<>(metadata));
    }

    public synchronized void updateMetadata(Map<String, String> changes) {
        if (changes == null) {
            return;
        }
        changes.forEach((key, value) -> {
            if (key == null || key.isBlank()) {
                throw new IllegalArgumentException("Metadata keys must not be blank");
            }
            metadata.put(key, value == null ? "" : value);
        });
    }

    public synchronized String getMetadataValue(String key) {
        return metadata.get(key);
    }

    public synchronized void setContent(byte[] content) {
        this.content = content == null ? new byte[0] : content.clone();
    }

    public synchronized byte[] getContent() {
        return content.clone();
    }

    public synchronized String searchableText() {
        return String.join(" ", metadata.values());
    }
}

import java.util.ArrayList;
import java.util.Collection;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

public class ResourceCatalog {
    private final Map<String, Resource> resources = new ConcurrentHashMap<>();

    public Resource importResource(Resource resource) {
        if (resource == null) throw new IllegalArgumentException("Resource is required");
        if (resources.putIfAbsent(resource.getId(), resource) != null) {
            throw new IllegalArgumentException("A resource with this id already exists");
        }
        return resource;
    }

    public Resource create(ResourceType type, Map<String, String> metadata) {
        return importResource(new Resource(type, metadata));
    }

    public void delete(String id) {
        if (resources.remove(id) == null) throw new IllegalArgumentException("Unknown resource: " + id);
    }

    public Resource require(String id) {
        Resource resource = resources.get(id);
        if (resource == null) throw new IllegalArgumentException("Unknown resource: " + id);
        return resource;
    }

    public void editMetadata(String id, Map<String, String> changes) {
        require(id).updateMetadata(changes);
    }

    public List<Resource> search(ResourceType type, String keywords) {
        String[] terms = keywords == null ? new String[0] : keywords.toLowerCase().trim().split("\\s+");
        List<Resource> result = new ArrayList<>();
        for (Resource resource : resources.values()) {
            if (resource.getType() != type) continue;
            String text = resource.searchableText().toLowerCase();
            boolean matches = true;
            for (String term : terms) {
                if (!term.isBlank() && !text.contains(term)) {
                    matches = false;
                    break;
                }
            }
            if (matches) result.add(resource);
        }
        return List.copyOf(result);
    }

    public List<Resource> searchAll(String keywords) {
        List<Resource> result = new ArrayList<>();
        for (ResourceType type : ResourceType.values()) result.addAll(search(type, keywords));
        return List.copyOf(result);
    }

    public Collection<Resource> all() {
        return List.copyOf(resources.values());
    }

    public List<Resource> all(ResourceType type) {
        return resources.values().stream().filter(r -> r.getType() == type).toList();
    }
}

import java.time.Instant;
import java.util.UUID;

public class HarvestJob {
    public enum Status { SCHEDULED, RUNNING, COMPLETED, CANCELLED, FAILED }

    private final String id = UUID.randomUUID().toString();
    private final String dataSourceId;
    private final Instant createdAt = Instant.now();
    private volatile Status status = Status.SCHEDULED;
    private volatile String report = "Not yet run";

    public HarvestJob(String dataSourceId) {
        if (dataSourceId == null || dataSourceId.isBlank()) {
            throw new IllegalArgumentException("Data source id is required");
        }
        this.dataSourceId = dataSourceId;
    }

    public String getId() { return id; }
    public String getDataSourceId() { return dataSourceId; }
    public Instant getCreatedAt() { return createdAt; }
    public Status getStatus() { return status; }
    public String getReport() { return report; }

    public synchronized void cancel() {
        if (status == Status.RUNNING || status == Status.COMPLETED) {
            throw new IllegalStateException("A running or completed job cannot be cancelled");
        }
        status = Status.CANCELLED;
        report = "Job cancelled";
    }

    public synchronized void run() {
        if (status != Status.SCHEDULED) throw new IllegalStateException("Job is not scheduled");
        status = Status.RUNNING;
        report = "Harvest completed successfully";
        status = Status.COMPLETED;
    }

    public synchronized void fail(String reason) {
        status = Status.FAILED;
        report = reason == null ? "Harvest failed" : reason;
    }
}

public class FilterCriterion {
    public enum Type {
        TAXONOMIC, DATASET, DEPTH, ALTITUDE, BASIS_OF_RECORD, GEOGRAPHIC, DATE
    }

    public FilterCriterion(Type type, String value) {

    }

    public FilterCriterion(Type type, Double minimum, Double maximum) {
        this(type, null, minimum, maximum);

            throw new IllegalArgumentException("Minimum must not exceed maximum");
        }
    }

    private FilterCriterion(Type type, String value, Double minimum, Double maximum) {
        this.type = Objects.requireNonNull(type);
        this.value = value;
        this.minimum = minimum;
        this.maximum = maximum;
    }

    public Type getType() { return type; }
    public String getValue() { return value; }
    public Double getMinimum() { return minimum; }
    public Double getMaximum() { return maximum; }

import java.nio.charset.StandardCharsets;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Collection;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

public class OccurrenceService {
    private final Map<String, Occurrence> occurrences = new ConcurrentHashMap<>();

    public void addOccurrence(Occurrence occurrence) {
        if (occurrences.putIfAbsent(occurrence.getId(), occurrence) != null) {
            throw new IllegalArgumentException("Duplicate occurrence id");
        }
    }

    public List<Occurrence> select(SelectionFilter filter) {
        if (filter == null) throw new IllegalArgumentException("Filter is required");
        return occurrences.values().stream().filter(o -> matches(o, filter.getCriteria())).toList();
    }

    private boolean matches(Occurrence occurrence, List<FilterCriterion> criteria) {
        for (FilterCriterion criterion : criteria) {
            String value = criterion.getValue();
            switch (criterion.getType()) {
                case TAXONOMIC -> {
                    if (!contains(occurrence.getTaxon(), value)) return false;
                }
                case DATASET -> {
                    if (!contains(occurrence.getDataset(), value)) return false;
                }
                case BASIS_OF_RECORD -> {
                    if (!contains(occurrence.getBasisOfRecord(), value)) return false;
                }
                case DEPTH -> {
                    if (!within(occurrence.getDepth(), criterion)) return false;
                }
                case ALTITUDE -> {
                    if (!within(occurrence.getAltitude(), criterion)) return false;
                }
                case GEOGRAPHIC -> {
                    if (!matchesGeography(occurrence, value)) return false;
                }
                case DATE -> {
                    if (!matchesDate(occurrence.getEventDate(), value)) return false;
                }
            }
        }
        return true;
    }

    private boolean contains(String actual, String expected) {
        return expected == null || actual.toLowerCase().contains(expected.toLowerCase());
    }

    private boolean within(Double value, FilterCriterion criterion) {
        if (value == null) return false;
        return (criterion.getMinimum() == null || value >= criterion.getMinimum())
                && (criterion.getMaximum() == null || value <= criterion.getMaximum());
    }

    // Geographic value is "south,west,north,east".
    private boolean matchesGeography(Occurrence o, String value) {
        if (value == null || value.isBlank()) return true;
        String[] parts = value.split(",");
        if (parts.length != 4) throw new IllegalArgumentException("Geographic criterion must be south,west,north,east");
        double south = Double.parseDouble(parts[0].trim());
        double west = Double.parseDouble(parts[1].trim());
        double north = Double.parseDouble(parts[2].trim());
        double east = Double.parseDouble(parts[3].trim());
        return o.getLatitude() >= south && o.getLatitude() <= north
                && o.getLongitude() >= west && o.getLongitude() <= east;
    }

    // Date value accepts an ISO date or an inclusive "start/end" range.
    private boolean matchesDate(LocalDate date, String value) {
        if (value == null || value.isBlank()) return true;
        if (date == null) return false;
        String[] range = value.split("/", -1);
        if (range.length == 1) return date.equals(LocalDate.parse(range[0]));
        if (range.length != 2) throw new IllegalArgumentException("Date criterion must be a date or start/end");
        LocalDate start = range[0].isBlank() ? LocalDate.MIN : LocalDate.parse(range[0]);
        LocalDate end = range[1].isBlank() ? LocalDate.MAX : LocalDate.parse(range[1]);
        return !date.isBefore(start) && !date.isAfter(end);
    }

    public String exportCsv(Collection<Occurrence> records) {
        StringBuilder output = new StringBuilder("id,taxon,dataset,depth,altitude,basisOfRecord,latitude,longitude,eventDate\n");
        for (Occurrence o : records) {
            output.append(csv(o.getId())).append(',').append(csv(o.getTaxon())).append(',')
                    .append(csv(o.getDataset())).append(',').append(o.getDepth()).append(',')
                    .append(o.getAltitude()).append(',').append(csv(o.getBasisOfRecord())).append(',')
                    .append(o.getLatitude()).append(',').append(o.getLongitude()).append(',')
                    .append(o.getEventDate()).append('\n');
        }
        return output.toString();
    }

    private String csv(String text) {
        if (text == null) return "";
        return "\"" + text.replace("\"", "\"\"") + "\"";
    }

    public byte[] exportMapImage(Collection<Occurrence> records) {
        StringBuilder svg = new StringBuilder(
                "<svg xmlns=\"http://www.w3.org/2000/svg\" width=\"800\" height=\"400\" viewBox=\"-180 -90 360 180\">"
                + "<rect x=\"-180\" y=\"-90\" width=\"360\" height=\"180\" fill=\"#eef5ff\"/>");
        for (Occurrence o : records) {
            svg.append("<circle cx=\"").append(o.getLongitude()).append("\" cy=\"")
                    .append(-o.getLatitude()).append("\" r=\"1.5\" fill=\"#c33\"/>");
        }
        svg.append("</svg>");
        return svg.toString().getBytes(StandardCharsets.UTF_8);
    }

    public int size() { return occurrences.size(); }
}

import java.time.LocalDate;
import java.util.Map;

public class Main {
    public static void main(String[] args) {
        PortalSystem portal = new PortalSystem();

        UserAccount securityOfficer = new UserAccount("security", "security@example.org", "demo");
        securityOfficer.addRole(Role.SECURITY_OFFICER);
        UserAccount dataManager = new UserAccount("data-manager", "data@example.org", "demo");
        dataManager.addRole(Role.DATA_MANAGER);
        UserAccount portalManager = new UserAccount("portal-manager", "portal@example.org", "demo");
        portalManager.addRole(Role.PORTAL_MANAGER);
        UserAccount administrator = new UserAccount("admin", "admin@example.org", "demo");
        administrator.addRole(Role.ADMINISTRATOR);

        AdminService admin = portal.getAdminService();
        UserAccount visitor = admin.createUser(securityOfficer, "visitor", "visitor@example.org", "secret");
        admin.assignRole(securityOfficer, visitor.getId(), Role.REGISTERED_VISITOR);

        admin.installModule(administrator, "distribution-map");
        SpatialLayer layer = admin.addLayer(administrator, "Protected areas", "https://example.org/layers/protected");

        portal.getPortalContentService().configureContact(portalManager, "Portal Team", "contact@example.org");
        portal.getPortalContentService().setDefaultLanguage(portalManager, "en");
        portal.getPortalContentService().contributeNews(portalManager, "Portal launched", "Welcome to the biodiversity portal.");

        portal.getDataManagerService().importResource(dataManager, ResourceType.CHECKLIST,
                Map.of("title", "Demo checklist", "description", "Regional species checklist"));
        portal.getOccurrenceService().addOccurrence(new Occurrence(
                "occ-1", "Quercus robur", "dataset-1", 2.0, 15.0, "HumanObservation",
                51.5, -0.1, LocalDate.of(2024, 5, 12)));

        SelectionFilter filter = portal.getVisitorService().buildOccurrenceFilter();
        portal.getVisitorService().addCriterion(filter,
                new FilterCriterion(FilterCriterion.Type.TAXONOMIC, "Quercus"));
        var selection = portal.getVisitorService().selectOccurrences(filter);
        portal.getVisitorService().saveCustomFilter(visitor, filter);

        System.out.println("Selected occurrences: " + selection.size());
        System.out.println(portal.getVisitorService().viewOccurrencesAsTable(selection));
        System.out.println("Map layer: " + layer.getName());
    }
}
