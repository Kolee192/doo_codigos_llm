import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

public class SearchResultState {
    private final List<RestaurantResult> matchingResults;
    private final boolean priceSearch;
    private final boolean destinationSearch;
    private List<RestaurantResult> visibleResults;
    private ViewMode viewMode;
    private SortField sortField;
    private boolean descending;

    public SearchResultState(List<RestaurantResult> matchingResults, boolean priceSearch,
                             boolean destinationSearch, ViewMode defaultView) {
        this.matchingResults = new ArrayList<>(matchingResults);
        this.priceSearch = priceSearch;
        this.destinationSearch = destinationSearch;
        this.viewMode = defaultView;
        this.visibleResults = new ArrayList<>(matchingResults);
        sortByDefault();
        limitVisibleResults();
    }

    public List<RestaurantResult> getResults() {
        return List.copyOf(visibleResults);
    }

    public ViewMode getViewMode() {
        return viewMode;
    }

    public void setViewMode(ViewMode viewMode) {
        this.viewMode = viewMode;
    }

    public int getResultCount() {
        return visibleResults.size();
    }

    public void switchView() {
        viewMode = viewMode == ViewMode.MAP ? ViewMode.LIST : ViewMode.MAP;
    }

    public void sort(SortField field) {
        if (field == sortField) {
            descending = !descending;
        } else {
            sortField = field;
            descending = false;
        }
        sortVisible();
    }

    public void filter(Integer maximumDistanceKm, Integer maximumPrice,
                       String restaurantType, String specificDish) {
        visibleResults = new ArrayList<>(matchingResults);
        visibleResults.removeIf(result -> {
            Restaurant restaurant = result.getRestaurant();
            return (maximumDistanceKm != null && result.getDistanceKm() > maximumDistanceKm)
                    || (maximumPrice != null && restaurant.getAveragePrice() > maximumPrice)
                    || (restaurantType != null && !restaurant.getType().equalsIgnoreCase(restaurantType))
                    || (specificDish != null && restaurant.getMenu().stream()
                    .noneMatch(dish -> dish.getName().equalsIgnoreCase(specificDish)));
        });
        sortVisible();
        limitVisibleResults();
    }

    private void sortByDefault() {
        Comparator<RestaurantResult> comparator;
        if (priceSearch) {
            comparator = Comparator.comparingInt((RestaurantResult r) ->
                            r.getRestaurant().getAveragePrice())
                    .thenComparingDouble(RestaurantResult::getDistanceKm)
                    .thenComparing(r -> safe(r.getRestaurant().getType()))
                    .thenComparing(r -> firstDish(r.getRestaurant()));
        } else {
            comparator = Comparator.comparingDouble(RestaurantResult::getDistanceKm)
                    .thenComparingInt(r -> r.getRestaurant().getAveragePrice())
                    .thenComparing(r -> safe(r.getRestaurant().getType()))
                    .thenComparing(r -> firstDish(r.getRestaurant()));
        }
        visibleResults.sort(comparator);
    }

    private void sortVisible() {
        if (sortField == null) {
            sortByDefault();
            return;
        }

        Comparator<RestaurantResult> comparator = switch (sortField) {
            case PRICE -> Comparator.comparingInt(r -> r.getRestaurant().getAveragePrice());
            case DISTANCE -> Comparator.comparingDouble(RestaurantResult::getDistanceKm);
            case RESTAURANT_TYPE -> Comparator.comparing(r -> safe(r.getRestaurant().getType()));
            case SPECIFIC_DISH -> Comparator.comparing(SearchResultState::firstDish);
            case RESTAURANT_NAME -> Comparator.comparing(r -> safe(r.getRestaurant().getName()));
        };
        if (descending) {
            comparator = comparator.reversed();
        }
        visibleResults.sort(comparator);
    }

    private void limitVisibleResults() {
        if (visibleResults.size() > 100) {
            visibleResults = new ArrayList<>(visibleResults.subList(0, 100));
        }
    }

    private static String firstDish(RestaurantResult result) {
        return firstDish(result.getRestaurant());
    }

    private static String firstDish(Restaurant restaurant) {
        return restaurant.getMenu().stream()
                .map(Dish::getName)
                .min(String.CASE_INSENSITIVE_ORDER)
                .orElse("");
    }

    private static String safe(String text) {
        return text == null ? "" : text.toLowerCase();
    }
}

    public SearchResultState search(String username, SearchCriteria criteria,
                                    double userLatitude, double userLongitude) {
        requireMobileLogin(username);
        Objects.requireNonNull(criteria, "Search criteria are required.");

        if (criteria.getMinimumPrice() != null && criteria.getMaximumPrice() != null
                && criteria.getMinimumPrice() > criteria.getMaximumPrice()) {
            throw new IllegalArgumentException("Minimum price cannot exceed maximum price.");
        }

        int minimumDistance = criteria.getMinimumDistanceKm() == null
                ? 0 : criteria.getMinimumDistanceKm();
        int maximumDistance = criteria.getMaximumDistanceKm() == null
                ? (criteria.searchesByDestination() ? 10 : Integer.MAX_VALUE)
                : criteria.getMaximumDistanceKm();

        if (minimumDistance > maximumDistance) {
            throw new IllegalArgumentException("Minimum distance cannot exceed maximum distance.");
        }

        List<RestaurantResult> results = new ArrayList<>();
        for (Restaurant restaurant : restaurants.values()) {
            double distance = distanceKm(userLatitude, userLongitude,
                    restaurant.getLatitude(), restaurant.getLongitude());
            if (!matches(restaurant, criteria, distance, minimumDistance, maximumDistance)) {
                continue;
            }
            results.add(new RestaurantResult(restaurant, distance));
        }

        ViewMode defaultView = criteria.searchesByPrice() ? ViewMode.LIST : ViewMode.MAP;
        return new SearchResultState(results, criteria.searchesByPrice(),
                criteria.searchesByDestination(), defaultView);
    }

    private boolean matches(Restaurant restaurant, SearchCriteria criteria, double distance,
                            int minimumDistance, int maximumDistance) {
        if (criteria.getMinimumPrice() != null
                && restaurant.getAveragePrice() < criteria.getMinimumPrice()) {
            return false;
        }
        if (criteria.getMaximumPrice() != null
                && restaurant.getAveragePrice() > criteria.getMaximumPrice()) {
            return false;
        }
        if ((criteria.searchesByDestination() || minimumDistance > 0)
                && (distance < minimumDistance || distance > maximumDistance)) {
            return false;
        }
        if (criteria.getRestaurantType() != null
                && !restaurant.getType().equalsIgnoreCase(criteria.getRestaurantType())) {
            return false;
        }
        if (criteria.getSpecificDish() != null
                && restaurant.getMenu().stream().noneMatch(dish ->
                dish.getName().equalsIgnoreCase(criteria.getSpecificDish()))) {
            return false;
        }
        if (criteria.getFreeText() != null && !matchesFreeText(restaurant, criteria.getFreeText())) {
            return false;
        }
        return true;
    }

    private boolean matchesFreeText(Restaurant restaurant, String text) {
        String query = text.toLowerCase(Locale.ROOT);
        return contains(restaurant.getName(), query)
                || contains(restaurant.getDescription(), query)
                || contains(restaurant.getAddress(), query)
                || contains(restaurant.getType(), query)
                || restaurant.getMenu().stream().anyMatch(dish ->
                contains(dish.getName(), query) || contains(dish.getDescription(), query));
    }

        validateMandatoryRestaurantFields(restaurant);
        String restaurantId = ownerRestaurantIds.get(key(ownerUsername));
        if (restaurantId != null && !restaurantId.equals(restaurant.getId())) {
            restaurants.remove(restaurantId);
        }
        restaurants.put(restaurant.getId(), restaurant);
        ownerRestaurantIds.put(key(ownerUsername), restaurant.getId());
        if (!restaurant.getType().isBlank()) {
            restaurantTypes.add(restaurant.getType());
        }
        restaurant.getMenu().forEach(dish -> dishes.add(dish.getName()));
    }

    public Restaurant getOwnerRestaurant(String ownerUsername) {
        requireOwnerLogin(ownerUsername);
        String id = ownerRestaurantIds.get(key(ownerUsername));
        return id == null ? null : restaurants.get(id);
    }

    public void deleteOwnerRestaurant(String ownerUsername) {
        requireOwnerLogin(ownerUsername);
        String id = ownerRestaurantIds.remove(key(ownerUsername));
        if (id != null) {
            restaurants.remove(id);
        }
    }

    public void editDish(String oldName, String newName) {
        requireAdministrator();
        String replacement = requireText(newName, "Dish name");
        if (!dishes.remove(oldName)) {
            throw new IllegalArgumentException("Dish not found.");
        }
        dishes.add(replacement);
        for (Restaurant restaurant : restaurants.values()) {
            List<Dish> updated = new ArrayList<>();
            for (Dish dish : restaurant.getMenu()) {
                if (dish.getName().equalsIgnoreCase(oldName)) {
                    updated.add(new Dish(replacement, dish.getDescription(), dish.getPrice()));
                } else {
                    updated.add(dish);
                }
            }
            restaurant.setMenu(updated);
        }
    }

        dishes.remove(dishName);

    private void requireMobileLogin(String username) {
        if (!loggedInMobileUsers.contains(key(username))) {
            throw new IllegalStateException("Mobile user must be logged in.");
        }
    }

    private void requireOwnerLogin(String username) {
        if (!loggedInOwners.contains(key(username))) {
            throw new IllegalStateException("Restaurant owner must be logged in.");
        }
    }

    private void requireAdministrator() {
        if (!administratorLoggedIn) {
            throw new IllegalStateException("Administrator must be logged in.");
        }
    }

    private RestaurantOwner requireOwner(String username) {
        RestaurantOwner owner = owners.get(key(username));
        if (owner == null) {
            throw new IllegalArgumentException("Restaurant owner not found.");
        }
        return owner;
    }

    private void ensureEmailAvailable(String email) {
        String normalized = requireText(email, "Email").toLowerCase(Locale.ROOT);
        boolean usedByMobile = mobileUsers.values().stream()
                .anyMatch(user -> user.getEmail().equalsIgnoreCase(normalized));
        boolean usedByOwner = owners.values().stream()
                .anyMatch(owner -> owner.getEmail().equalsIgnoreCase(normalized));
        if (usedByMobile || usedByOwner) {
            throw new IllegalArgumentException("Email address is already registered.");
        }
    }

    private void sendEmail(String email, String subject, String body) {
        sentEmails.add("To: " + email + " | Subject: " + subject + " | " + body);
    }

    private static void validateMandatoryRestaurantFields(Restaurant restaurant) {
        Objects.requireNonNull(restaurant, "Restaurant is required.");
        requireText(restaurant.getName(), "Restaurant name");
        requireText(restaurant.getAddress(), "Restaurant address");
        requireText(restaurant.getPhoneNumber(), "Restaurant phone number");
        requireText(restaurant.getEmail(), "Restaurant email");
        if (restaurant.getAveragePrice() < 0) {
            throw new IllegalArgumentException("Average price cannot be negative.");
        }
    }

    private static String requireText(String value, String label) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(label + " is required.");
        }
        return value.trim();
    }

    private static String key(String value) {
        return value == null ? "" : value.trim().toLowerCase(Locale.ROOT);
    }

    private static boolean contains(String source, String query) {
        return source != null && source.toLowerCase(Locale.ROOT).contains(query);
    }

    private static double distanceKm(double lat1, double lon1, double lat2, double lon2) {
        double earthRadiusKm = 6371.0;
        double dLat = Math.toRadians(lat2 - lat1);
        double dLon = Math.toRadians(lon2 - lon1);
        double a = Math.sin(dLat / 2) * Math.sin(dLat / 2)
                + Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2))
                * Math.sin(dLon / 2) * Math.sin(dLon / 2);
        return earthRadiusKm * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    }

    private static List<Integer> versionParts(String version) {
        List<Integer> parts = new ArrayList<>();
        for (String part : version.split("\\.")) {
            try {
                parts.add(Integer.parseInt(part.replaceAll("[^0-9].*$", "")));
            } catch (NumberFormatException exception) {
                parts.add(0);
            }
        }
        return parts;
    }
