import java.util.EnumSet;
import java.util.Set;
import java.util.UUID;

public class Member {
    private final String id;
    private String name;
    private String email;
    private String password;
    private final Profile profile = new Profile();
    private final Set<MemberRole> roles = EnumSet.of(MemberRole.MEMBER);
    private final Set<Certification> certifications = EnumSet.noneOf(Certification.class);
    private double accountBalance;
    private double trainerRating;
    private int trainerRatingCount;

    public Member(String name, String email, String password) {
        this.id = UUID.randomUUID().toString();
        this.name = require(name, "name");
        this.email = require(email, "email");
        this.password = require(password, "password");
    }

    public String getId() {
        return id;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = require(name, "name");
    }

    public String getEmail() {
        return email;
    }

    public void setEmail(String email) {
        this.email = require(email, "email");
    }

    public Profile getProfile() {
        return profile;
    }

    public Set<MemberRole> getRoles() {
        return Set.copyOf(roles);
    }

    public boolean hasRole(MemberRole role) {
        return roles.contains(role);
    }

    public void addRole(MemberRole role) {
        roles.add(role);
    }

    public Set<Certification> getCertifications() {
        return Set.copyOf(certifications);
    }

    public boolean hasCertification(Certification certification) {
        return certifications.contains(certification);
    }

    public void addCertification(Certification certification) {
        certifications.add(certification);
        if (certification == Certification.CST) {
            roles.add(MemberRole.TRAINER);
        }
    }

    public double getAccountBalance() {
        return accountBalance;
    }

    public void charge(double amount) {
        if (amount < 0) {
            throw new IllegalArgumentException("Amount must not be negative");
        }
        accountBalance += amount;
    }

    public void pay(double amount) {
        if (amount < 0 || amount > accountBalance) {
            throw new IllegalArgumentException("Invalid payment amount");
        }
        accountBalance -= amount;
    }

    public double getTrainerRating() {
        return trainerRatingCount == 0 ? 0 : trainerRating;
    }

    public int getTrainerRatingCount() {
        return trainerRatingCount;
    }

    public void addTrainerRating(int rating) {
        if (rating < 1 || rating > 5) {
            throw new IllegalArgumentException("Rating must be between 1 and 5");
        }
        trainerRating = ((trainerRating * trainerRatingCount) + rating) / ++trainerRatingCount;
    }

    public boolean passwordMatches(String candidate) {
        return password.equals(candidate);
    }

    public void resetPassword(String newPassword) {
        password = require(newPassword, "password");
    }

    private static String require(String value, String field) {
        if (value == null || value.isBlank()) {
            throw new IllegalArgumentException(field + " is required");
        }
        return value.trim();
    }
}

    public void payForApplication(String memberId, String applicationId) {
        Application application = require(applications, applicationId, "application");
        if (!application.getMemberId().equals(memberId)) {
            throw new SecurityException("This application belongs to another member");
        }
        if (application.getStatus() != ApplicationStatus.APPROVED_AWAITING_PAYMENT) {
            throw new IllegalStateException("Application is not awaiting payment");
        }
        Member member = requireMember(memberId);
        double fee = annualFees.get(application.getType() == ApplicationType.PRACTITIONER
                ? Certification.CSP : Certification.CST);
        member.pay(fee);

    public Activity createActivity(String creatorId, String trainerId, String title,
                                   String description, String url, String location,
                                   LocalDate start, LocalDate end, ActivityType type) {
        Member creator = requireMember(creatorId);
        requireMember(trainerId);
        if (!creatorId.equals(trainerId) && !creator.hasRole(MemberRole.ADMIN)) {
            throw new SecurityException("Only a trainer or administrator may create an activity");
        }
        boolean adminCreated = creator.hasRole(MemberRole.ADMIN);
        boolean feeRequired = !adminCreated
                && (type == ActivityType.OTHER_COURSE || type == ActivityType.EVENT);
        if (feeRequired) {
            creator.charge(otherActivityListingFee);
        }
        Activity activity = new Activity(title, description, url, location, start, end,
                type, creatorId, trainerId, feeRequired);
        activities.put(activity.getId(), activity);
        return activity;
    }

    public List<Activity> upcomingActivities(ActivityType type, int page, int pageSize) {
        if (page < 0 || pageSize < 1) {
            throw new IllegalArgumentException("Invalid page settings");
        }
        List<Activity> result = activities.values().stream()
                .filter(a -> a.isApproved() && !a.getStartDate().isBefore(LocalDate.now()))
                .filter(a -> type == null || a.getType() == type)
                .sorted(Comparator.comparing(Activity::getStartDate))
                .toList();
        int from = Math.min(page * pageSize, result.size());
        return result.subList(from, Math.min(from + pageSize, result.size()));
    }

                .toList();

    public ClassRoster importRoster(String trainerId, String courseName, String csv) {
        Member trainer = requireMember(trainerId);
        if (!trainer.hasRole(MemberRole.TRAINER) && !trainer.hasCertification(Certification.CST)) {
            throw new SecurityException("Only trainers can import class rosters");
        }
        ClassRoster roster = new ClassRoster(trainerId, courseName);
        if (csv != null) {
            for (String line : csv.split("\\R")) {
                String[] cells = line.split(",", -1);
                if (cells.length >= 2 && !cells[0].isBlank() && !cells[1].isBlank()) {
                    roster.addParticipant(cells[0].trim(), cells[1].trim(), true);
                }
            }
        }
        rosters.put(roster.getId(), roster);
        return roster;
    }

    public void approveRoster(String administratorId, String rosterId, boolean proofOfPaymentReceived) {
        requireRole(administratorId, MemberRole.ADMIN);
        if (!proofOfPaymentReceived) {
            throw new IllegalStateException("Proof of payment is required");
        }
        ClassRoster roster = require(rosters, rosterId, "roster");
        roster.approve();
        roster.registerPendingParticipants();
        roster.getParticipants().forEach(p -> {
            RegistryEntry entry = addRegistryEntry(p.getName(), p.getEmail(), Certification.CSM);
            entry.activate();
        });
    }

    private String emailTemplate(Certification certification, String name) {
        String key = certification == Certification.CSPO ? "CSPO" : "CSM";
        return editorEmailTemplates.getOrDefault(key, "Welcome, {name}!")
                .replace("{name}", name);

    private String memberName(String memberId) {
        Member member = members.get(memberId);
        return member == null ? "" : member.getName();
    }

    private void requireActivityEditPermission(String actorId, Activity activity) {
        Member actor = requireMember(actorId);
        if (!activity.getOwnerId().equals(actorId)
                && !activity.getTrainerId().equals(actorId)
                && !actor.hasRole(MemberRole.ADMIN)
                && !actor.hasRole(MemberRole.EDITOR)) {
            throw new SecurityException("Not permitted to edit this activity");
        }
    }

    private void requireRole(String memberId, MemberRole role) {
        if (!requireMember(memberId).hasRole(role)) {
            throw new SecurityException("Required role: " + role);
        }
    }

    private Member requireMember(String id) {
        return require(members, id, "member");
    }

    private static <T> T require(Map<String, T> map, String id, String type) {
        T value = map.get(id);
        if (value == null) {
            throw new IllegalArgumentException("Unknown " + type + ": " + id);
        }
        return value;
    }

    private void ensureUniqueEmail(String email, String exceptMemberId) {
        if (email == null || email.isBlank()) {
            throw new IllegalArgumentException("Email is required");
        }
        boolean exists = members.values().stream()
                .anyMatch(m -> !m.getId().equals(exceptMemberId)
                        && m.getEmail().equalsIgnoreCase(email.trim()));
        if (exists) {
            throw new IllegalArgumentException("Email is already registered");
        }
    }

    private static boolean contains(String text, String query) {
        return query == null || query.isBlank()
                || text.toLowerCase(Locale.ROOT).contains(query.toLowerCase(Locale.ROOT));
    }

    private static void validateAmount(double amount) {
        if (amount < 0 || Double.isNaN(amount) || Double.isInfinite(amount)) {
            throw new IllegalArgumentException("Amount must be a valid non-negative number");
        }
    }

    private static String rss(String title, List<String> items) {
        StringBuilder xml = new StringBuilder("<?xml version=\"1.0\" encoding=\"UTF-8\"?>")
                .append("<rss version=\"2.0\"><channel><title>")
                .append(escape(title)).append("</title>");
        for (String item : items) {
            xml.append("<item><title>").append(escape(item)).append("</title></item>");
        }
        return xml.append("</channel></rss>").toString();
    }

    private static String escape(String text) {
        return text.replace("&", "&amp;").replace("<", "&lt;")
                .replace(">", "&gt;").replace("\"", "&quot;")
                .replace("'", "&apos;");
    }

    private static void copyProfile(Profile from, Profile to) {
        if (from == null) {
            return;
        }
        to.setLocation(from.getLocation());
        to.setBiography(from.getBiography());
        to.setWebsite(from.getWebsite());
        to.setExperience(from.getExperience());
        to.setTrainerDetails(from.getTrainerDetails());
        to.setPrivateProfile(from.isPrivateProfile());
        to.setEmailPrivate(from.isEmailPrivate());
        from.getFields().forEach(to::setField);
    }
