public final class Fee {
    private final BigDecimal amount;
    private final String reason;

    public Fee(BigDecimal amount, String reason) {
        this.amount = Objects.requireNonNull(amount, "amount");
        if (amount.signum() < 0) {
            throw new IllegalArgumentException("Fee amount cannot be negative");
        }
        this.reason = Objects.requireNonNull(reason, "reason");
    }

    public BigDecimal getAmount() {
        return amount;
    }

    public String getReason() {
        return reason;
    }

import java.util.ArrayList;

public abstract class Medium {
    private final String partialShelfmark;
    private final MediumType type;
    private final List<String> authors;
    private String title;
    private String additionalTitle;
    private int publicationYear;
    private final List<String> keywords;

    protected Medium(String partialShelfmark, MediumType type, List<String> authors,
                     String title, String additionalTitle, int publicationYear,
                     List<String> keywords) {
        if (partialShelfmark == null || partialShelfmark.isBlank()) {
            throw new IllegalArgumentException("Partial shelfmark is required");
        }
        this.partialShelfmark = partialShelfmark;
        this.type = type;
        this.authors = new ArrayList<>(authors == null ? List.of() : authors);
        this.title = title;
        this.additionalTitle = additionalTitle;
        this.publicationYear = publicationYear;
        this.keywords = new ArrayList<>(keywords == null ? List.of() : keywords);
    }

    public String getPartialShelfmark() {
        return partialShelfmark;
    }

    public MediumType getType() {
        return type;
    }

    public List<String> getAuthors() {
        return Collections.unmodifiableList(authors);
    }

    public String getTitle() {
        return title;
    }

    public void setTitle(String title) {
        this.title = title;
    }

    public String getAdditionalTitle() {
        return additionalTitle;
    }

    public void setAdditionalTitle(String additionalTitle) {
        this.additionalTitle = additionalTitle;
    }

    public int getPublicationYear() {
        return publicationYear;
    }

    public void setPublicationYear(int publicationYear) {
        this.publicationYear = publicationYear;
    }

    public List<String> getKeywords() {
        return Collections.unmodifiableList(keywords);
    }

    public void addKeyword(String keyword) {
        keywords.add(keyword);
    }

    public boolean matchesText(String query) {
        if (query == null || query.isBlank()) {
            return true;
        }
        String normalized = query.toLowerCase();
        return contains(title, normalized)
                || contains(additionalTitle, normalized)
                || authors.stream().anyMatch(a -> contains(a, normalized))
                || keywords.stream().anyMatch(k -> contains(k, normalized));
    }

    private boolean contains(String value, String query) {
        return value != null && value.toLowerCase().contains(query);
    }

    @Override
    public String toString() {
        return type + ": " + title + " [" + partialShelfmark + "]";
    }
}

import java.time.LocalDate;
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Deque;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Set;

public class LibraryInstance {
    private final String shelfmark;
    private final Medium medium;
    private InstanceStatus status;
    private LocalDate dueDate;
    private LocalDate availableForReservationSince;
    private String currentBorrower;
    private int extensionDaysThisRental;
    private final Deque<String> reservationQueue = new ArrayDeque<>();
    private final List<BorrowRecord> borrowingHistory = new ArrayList<>();
    private final List<String> components = new ArrayList<>();
    private String comments = "";

    public LibraryInstance(String shelfmark, Medium medium, InstanceStatus initialStatus) {
        if (shelfmark == null || shelfmark.isBlank()) {
            throw new IllegalArgumentException("Shelfmark is required");
        }
        this.shelfmark = shelfmark;
        this.medium = medium;
        this.status = initialStatus == null ? InstanceStatus.AVAILABLE : initialStatus;
    }

    public String getShelfmark() {
        return shelfmark;
    }

    public Medium getMedium() {
        return medium;
    }

    public InstanceStatus getStatus() {
        return status;
    }

    public LocalDate getDueDate() {
        return dueDate;
    }

    public String getCurrentBorrower() {
        return currentBorrower;
    }

    public int getExtensionDaysThisRental() {
        return extensionDaysThisRental;
    }

    public List<String> getReservationQueue() {
        return Collections.unmodifiableList(new ArrayList<>(reservationQueue));
    }

    public List<BorrowRecord> getBorrowingHistory() {
        return Collections.unmodifiableList(borrowingHistory);
    }

    public List<String> getComponents() {
        return Collections.unmodifiableList(components);
    }

    public void addComponent(String component) {
        components.add(component);
    }

    public String getComments() {
        return comments;
    }

    public void setComments(String comments) {
        this.comments = comments == null ? "" : comments;
    }

    boolean isReservedBy(String userNumber) {
        return reservationQueue.contains(userNumber);
    }

    boolean isFirstInReservationQueue(String userNumber) {
        return !reservationQueue.isEmpty() && reservationQueue.peekFirst().equals(userNumber);
    }

    boolean addReservation(String userNumber) {
        if (reservationQueue.contains(userNumber)) {
            return false;
        }
        reservationQueue.addLast(userNumber);
        updateDerivedStatus();
        return true;
    }

    boolean removeReservation(String userNumber) {
        boolean removed = reservationQueue.remove(userNumber);
        if (removed) {
            updateDerivedStatus();
        }
        return removed;
    }

    String releaseExpiredFirstReservation() {
        String removed = reservationQueue.pollFirst();
        if (removed != null) {
            updateDerivedStatus();
        }
        return removed;
    }

    LocalDate getAvailableForReservationSince() {
        return availableForReservationSince;
    }

    void borrow(String userNumber, LocalDate borrowedOn, int rentalDays) {
        currentBorrower = userNumber;
        dueDate = borrowedOn.plusDays(rentalDays);
        extensionDaysThisRental = 0;
        borrowingHistory.add(new BorrowRecord(userNumber, borrowedOn));
        availableForReservationSince = null;
        updateDerivedStatus();
    }

    void extendDueDate(int days) {
        dueDate = dueDate.plusDays(days);
        extensionDaysThisRental += days;
    }

    void returnToLibrary(LocalDate date) {
        if (!borrowingHistory.isEmpty()) {
            BorrowRecord latest = borrowingHistory.get(borrowingHistory.size() - 1);
            if (latest.getReturnedOn() == null) {
                latest.markReturned(date);
            }
        }
        currentBorrower = null;
        dueDate = null;
        extensionDaysThisRental = 0;
        updateDerivedStatus();
    }

    void setManagedStatus(InstanceStatus newStatus) {
        status = newStatus;
        if (newStatus == InstanceStatus.AVAILABLE || newStatus == InstanceStatus.AVAILABLE_AND_RESERVED) {
            availableForReservationSince = reservationQueue.isEmpty() ? null : LocalDate.now();
        }
    }

    private void updateDerivedStatus() {
        if (currentBorrower != null) {
            status = reservationQueue.isEmpty()
                    ? InstanceStatus.BORROWED
                    : InstanceStatus.RESERVED_AND_BORROWED;
        } else if (status == InstanceStatus.SORTED_OUT
                || status == InstanceStatus.MISSING
                || status == InstanceStatus.ACQUISITION_PROCESS
                || status == InstanceStatus.READING_ROOM) {
            return;
        } else if (reservationQueue.isEmpty()) {
            status = InstanceStatus.AVAILABLE;
            availableForReservationSince = null;
        } else {
            status = InstanceStatus.AVAILABLE_AND_RESERVED;
            if (availableForReservationSince == null) {
                availableForReservationSince = LocalDate.now();
            }
        }
    }
}

    public LibrarySystem() {
        this(28, 84, 56, 7, 30, new java.math.BigDecimal("50.00"));
    }

    public LibrarySystem(int standardRentalDays, int maximumRentalDays,
                         int maximumExtensionDays, int reservationPickupDays,
                         int maximumBorrowedItems, java.math.BigDecimal feeLockThreshold) {
        this.standardRentalDays = standardRentalDays;
        this.maximumRentalDays = maximumRentalDays;
        this.maximumExtensionDays = maximumExtensionDays;

    public UserAccount createUser(UserAccount account) {

    public LibraryInstance addInstance(String shelfmark, String partialShelfmark,

        if (medium == null) {

        if (instances.putIfAbsent(shelfmark, instance) != null) {
            throw new IllegalArgumentException("Instance already exists: " + shelfmark);
        }

        LibraryInstance instance = instances.get(shelfmark);
        if (instance == null || instance.getCurrentBorrower() != null
                || !instance.getReservationQueue().isEmpty()) {
            return false;
        }
        instances.remove(shelfmark);

    public boolean changeInstanceStatus(String shelfmark, InstanceStatus status) {
        LibraryInstance instance = instances.get(shelfmark);
        if (instance == null || status == null || instance.getCurrentBorrower() != null) {

        instance.setManagedStatus(status);
        return true;

    public LibraryInstance getInstance(String shelfmark) {
        return instances.get(shelfmark);
    }

    public Collection<LibraryInstance> getAllInstances() {
        return Collections.unmodifiableCollection(instances.values());
    }

    public List<Medium> search(SearchCriteria criteria) {
        Objects.requireNonNull(criteria);
        List<Medium> results = new ArrayList<>();
        for (Medium medium : media.values()) {
            if (criteria.getTypes() != null && !criteria.getTypes().contains(medium.getType())) {
                continue;
            }
            boolean titleMatches = criteria.getTitleTerms().isEmpty()
                    || criteria.getTitleTerms().stream().anyMatch(medium::matchesText);
            if (!titleMatches) {
                continue;
            }
            if (criteria.getAttributeQuery() != null
                    && !medium.matchesText(criteria.getAttributeQuery())) {
                continue;
            }
            results.add(medium);
        }
        return results;
    }

    public List<LibraryInstance> getInstancesForMedium(String partialShelfmark) {
        return instances.values().stream()
                .filter(i -> i.getMedium().getPartialShelfmark().equals(partialShelfmark))
                .toList();
    }

    public boolean reserveInstance(String userNumber, String shelfmark) {
        UserAccount user = authenticate(userNumber);
        LibraryInstance instance = instances.get(shelfmark);
        if (user == null || instance == null || instance.getStatus() == InstanceStatus.MISSING
                || instance.getStatus() == InstanceStatus.SORTED_OUT
                || instance.getStatus() == InstanceStatus.ACQUISITION_PROCESS
                || instance.getStatus() == InstanceStatus.READING_ROOM
                || instance.isReservedBy(userNumber)) {
            return false;
        }
        if (!instance.addReservation(userNumber)) {
            return false;
        }
        user.addReserved(shelfmark);
        if (instance.getCurrentBorrower() == null
                && instance.isFirstInReservationQueue(userNumber)) {
            notifyUser(userNumber, "Reserved item is available: " + shelfmark);
        }
        return true;
    }

    public List<String> borrowInstances(String userNumber, List<String> shelfmarks, LocalDate date) {
        UserAccount user = authenticate(userNumber);
        if (user == null || shelfmarks == null) {
            return List.of();
        }
        List<String> receipts = new ArrayList<>();
        for (String shelfmark : shelfmarks) {
            LibraryInstance instance = instances.get(shelfmark);
            if (instance == null || user.getBorrowedShelfmarks().size() >= maximumBorrowedItems) {
                continue;
            }
            boolean available = instance.getStatus() == InstanceStatus.AVAILABLE
                    || instance.getStatus() == InstanceStatus.AVAILABLE_AND_RESERVED;
            boolean reservationAllowsBorrow = instance.getReservationQueue().isEmpty()
                    || instance.isFirstInReservationQueue(userNumber);
            if (!available || !reservationAllowsBorrow
                    || instance.getCurrentBorrower() != null) {
                continue;
            }

            if (instance.isFirstInReservationQueue(userNumber)) {
                instance.removeReservation(userNumber);
                user.removeReserved(shelfmark);
            }
            instance.borrow(userNumber, date, standardRentalDays);
            user.addBorrowed(shelfmark);
            receipts.add("Borrowed " + shelfmark + "; due " + date.plusDays(standardRentalDays));
        }
        return receipts;
    }

    public boolean extendRental(String userNumber, String shelfmark, int additionalDays,
                                LocalDate today) {
        UserAccount user = authenticate(userNumber);
        LibraryInstance instance = instances.get(shelfmark);
        if (user == null || instance == null || additionalDays <= 0
                || !user.getBorrowedShelfmarks().contains(shelfmark)
                || !userNumber.equals(instance.getCurrentBorrower())
                || !instance.getReservationQueue().isEmpty()
                || instance.getDueDate() == null) {
            return false;
        }
        int currentRentalDays = (int) java.time.temporal.ChronoUnit.DAYS.between(
                instance.getBorrowingHistory().get(instance.getBorrowingHistory().size() - 1)
                        .getBorrowedOn(), instance.getDueDate());
        if (instance.getExtensionDaysThisRental() + additionalDays > maximumExtensionDays
                || currentRentalDays + additionalDays > maximumRentalDays) {
            return false;
        }
        instance.extendDueDate(additionalDays);
        return true;
    }

    public boolean returnInstance(String shelfmark, LocalDate date) {
        LibraryInstance instance = instances.get(shelfmark);
        if (instance == null || instance.getCurrentBorrower() == null
                || (instance.getStatus() != InstanceStatus.BORROWED
                && instance.getStatus() != InstanceStatus.RESERVED_AND_BORROWED)) {
            return false;
        }
        UserAccount borrower = users.get(instance.getCurrentBorrower());
        if (borrower != null) {
            borrower.removeBorrowed(shelfmark);
        }
        instance.returnToLibrary(date);
        if (!instance.getReservationQueue().isEmpty()) {
            notifyUser(instance.getReservationQueue().get(0),
                    "Reserved item is available: " + shelfmark);
        }
        return true;
    }

    public void addFee(String userNumber, Fee fee) {
        UserAccount user = users.get(userNumber);
        if (user == null) {
            throw new IllegalArgumentException("Unknown user");
        }
        user.addFee(fee);
        refreshAccountLock(user, LocalDate.now());
    }

    public void refreshAccountLock(UserAccount user, LocalDate today) {
        boolean overdue = user.getBorrowedShelfmarks().stream()
                .map(instances::get)
                .filter(Objects::nonNull)
                .anyMatch(i -> i.getDueDate() != null && i.getDueDate().isBefore(today));
        if (user.getBorrowedShelfmarks().size() > maximumBorrowedItems
                || overdue || user.getUnpaidFeeTotal().compareTo(feeLockThreshold) > 0) {
            user.setLocked(true);
        }
    }

    public List<String> releaseExpiredReservations(LocalDate today) {
        List<String> released = new ArrayList<>();
        for (LibraryInstance instance : instances.values()) {
            LocalDate since = instance.getAvailableForReservationSince();
            if (instance.getCurrentBorrower() == null
                    && instance.getStatus() == InstanceStatus.AVAILABLE_AND_RESERVED
                    && since != null
                    && !today.isBefore(since.plusDays(reservationPickupDays))) {
                String expiredUser = instance.releaseExpiredFirstReservation();
                if (expiredUser != null) {
                    UserAccount account = users.get(expiredUser);
                    if (account != null) {
                        account.removeReserved(instance.getShelfmark());
                    }
                    released.add(instance.getShelfmark());
                    if (!instance.getReservationQueue().isEmpty()) {
                        String nextUser = instance.getReservationQueue().get(0);
                        notifyUser(nextUser, "Reserved item is available: " + instance.getShelfmark());
                        instance.setManagedStatus(InstanceStatus.AVAILABLE_AND_RESERVED);
                    }
                }
            }
        }
        return released;
    }

    public List<String> getOverdueShelfmarks(String userNumber, LocalDate today) {
        UserAccount user = users.get(userNumber);
        if (user == null) {
            return List.of();
        }
        return user.getBorrowedShelfmarks().stream()
                .filter(mark -> {
                    LibraryInstance instance = instances.get(mark);
                    return instance != null && instance.getDueDate() != null
                            && instance.getDueDate().isBefore(today);
                }).toList();
    }

    public List<String> getNotifications() {
        return Collections.unmodifiableList(notifications);
    }

    private void notifyUser(String userNumber, String message) {
        notifications.add(userNumber + ": " + message);
    }

    private UserAccount authenticate(String userNumber) {
        UserAccount user = users.get(userNumber);
        return user != null && user.isUsableOn(LocalDate.now()) ? user : null;
    }
}
