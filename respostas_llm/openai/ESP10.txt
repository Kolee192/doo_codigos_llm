import java.util.LinkedHashSet;
import java.util.Set;

public class User {
    private final String id;
    private String name;
    private String email;
    private String bio = "";
    private final Set<String> roles = new LinkedHashSet<>();

    public User(String id, String name, String email, String... roles) {
        this.id = id;
        this.name = name;
        this.email = email;
        if (roles != null) {
            for (String role : roles) {
                this.roles.add(role.toLowerCase());
            }
        }
    }

    public String getId() { return id; }
    public String getName() { return name; }
    public void setName(String name) { this.name = name; }
    public String getEmail() { return email; }
    public void setEmail(String email) { this.email = email; }
    public String getBio() { return bio; }
    public void setBio(String bio) { this.bio = bio; }
    public Set<String> getRoles() { return Set.copyOf(roles); }
    public boolean hasRole(String role) { return roles.contains(role.toLowerCase()); }
    public void addRole(String role) { roles.add(role.toLowerCase()); }
}

    private String id() { return UUID.randomUUID().toString(); }

    public User registerUser(String name, String email) {

            throw new IllegalArgumentException("Email is already registered");
        }
        User user = new User(id(), name, email, "attendee", "authenticated");

    }

    public void addUser(User user) { users.put(user.getId(), user); }
    public User getUser(String userId) { return require(users, userId, "User"); }
    public List<User> listProfiles() { return List.copyOf(users.values()); }

    public String trainingUrl(Training training) { return "/training/" + training.getSlug(); }

    public Registration registerForTraining(String userId, String trainingId, boolean acceptedCancellationTerms) {
        if (!acceptedCancellationTerms) throw new IllegalArgumentException("Cancellation terms must be accepted");
        LocalDateTime now = LocalDateTime.now();
        if (now.isBefore(registrationOpens) || now.isAfter(registrationCloses))
            throw new IllegalStateException("Registration is closed");
        Training training = getTraining(trainingId);
        if (training.getStart().toLocalDate().isBefore(LocalDate.now()))
            throw new IllegalStateException("Training has already taken place");
        Registration existing = training.registrationStore().get(userId);
        if (existing != null && existing.getStatus() != RegistrationStatus.CANCELLED
                && existing.getStatus() != RegistrationStatus.REFUNDED) {
            throw new IllegalStateException("Only one registration per user per training is allowed");
        }
        RegistrationStatus status = training.getRemainingSeats() > 0
                ? RegistrationStatus.REGISTERED : RegistrationStatus.WAITLISTED;
        Registration registration = new Registration(userId, trainingId, status);
        training.registrationStore().put(userId, registration);
        if (status == RegistrationStatus.WAITLISTED) training.waitlistStore().add(userId);
        notifications.add("Registration confirmed for " + userId + ": " + training.getTitle());
        return registration;
    }

    public void setCancellationDeadline(String trainingId, LocalDateTime deadline) {
        cancellationDeadlines.put(trainingId, deadline);
    }

    public void cancelTrainingRegistration(String userId, String trainingId) {
        Training training = getTraining(trainingId);
        Registration registration = training.registrationStore().get(userId);
        if (registration == null || registration.getStatus() != RegistrationStatus.REGISTERED)
            throw new IllegalStateException("No active registration");
        LocalDateTime deadline = cancellationDeadlines.get(trainingId);
        if (deadline != null && LocalDateTime.now().isAfter(deadline))
            throw new IllegalStateException("Cancellation deadline has passed");
        registration.setStatus(RegistrationStatus.CANCELLED);
        promoteWaitlisted(training);
    }

    private void promoteWaitlisted(Training training) {
        while (training.getRemainingSeats() > 0 && !training.waitlistStore().isEmpty()) {
            String next = training.waitlistStore().remove(0);
            Registration registration = training.registrationStore().get(next);
            if (registration != null && registration.getStatus() == RegistrationStatus.WAITLISTED) {
                registration.setStatus(RegistrationStatus.REGISTERED);
                notifications.add("A seat is available for " + next + ": " + training.getTitle());
                break;
            }
        }
    }

    public List<String> trainingAttendees(String trainerId, String trainingId) {
        Training training = getTraining(trainingId);
        if (!training.getTrainerIds().contains(trainerId)) throw new SecurityException("Not a trainer for this class");
        return training.getRegistrations().values().stream()
                .filter(r -> r.getStatus() == RegistrationStatus.REGISTERED
                        || r.getStatus() == RegistrationStatus.ATTENDED
                        || r.getStatus() == RegistrationStatus.NO_SHOW)
                .map(Registration::getUserId).map(this::getUser)
                .map(u -> u.getName() + " <" + u.getEmail() + ">").toList();
    }

    public void updateTrainingAttendance(String trainingId, String userId, RegistrationStatus status) {
        if (status != RegistrationStatus.ATTENDED && status != RegistrationStatus.NO_SHOW
                && status != RegistrationStatus.REFUNDED) throw new IllegalArgumentException("Invalid attendance status");
        Training training = getTraining(trainingId);
        Registration registration = training.registrationStore().get(userId);
        if (registration == null) throw new IllegalArgumentException("Registration not found");
        registration.setStatus(status);
        if (status == RegistrationStatus.REFUNDED) promoteWaitlisted(training);
    }

    public void registerForSummit(String userId, String summitId) {
        Summit summit = getSummit(summitId);
        if (summit.getAttendeeIds().contains(userId)) throw new IllegalStateException("Already registered");
        summit.addAttendee(userId);
    }

    public List<User> summitAttendees(String summitId) {
        return getSummit(summitId).getAttendeeIds().stream().map(this::getUser).toList();
    }

    public List<Summit> mySummits(String userId) {
        return summits.values().stream().filter(s -> s.getAttendeeIds().contains(userId))
                .sorted(Comparator.comparing(Summit::getStart)).toList();
    }

    public void addSummitAgendaItem(String summitId, String item) {
        getSummit(summitId).addAgendaItem(item);
    }

    public void sponsorSummit(String sponsorId, String summitId) {
        getSponsor(sponsorId);
        getSummit(summitId).addSponsor(sponsorId);
        assignSponsorToEvent(sponsorId, summitId);
    }

    public Session submitSession(String presenterId, String title, String abstractText) {
        Session session = new Session(id(), presenterId, title, abstractText);
        sessions.put(session.getId(), session);
        notifications.add("Session submitted: " + title);
        return session;
    }

    public void editSession(String userId, String sessionId, String title, String abstractText) {
        Session session = require(sessions, sessionId, "Session");
        if (!session.getPresenterId().equals(userId)) throw new SecurityException("Not the session presenter");
        session.edit(title, abstractText);
    }

    public void evaluateSession(String sessionId, SessionStatus status) {
        if (status == SessionStatus.PENDING) throw new IllegalArgumentException("Evaluation must be final");
        Session session = require(sessions, sessionId, "Session");
        session.setStatus(status);
        notifications.add("Session evaluated: " + session.getTitle() + " — " + status);
    }

    public void bulkApproveSessions(Collection<String> sessionIds) {
        for (String sessionId : sessionIds) evaluateSession(sessionId, SessionStatus.ACCEPTED);
    }

    public List<Session> acceptedSessions() {
        return sessions.values().stream().filter(s -> s.getStatus() == SessionStatus.ACCEPTED).toList();
    }

    public void scheduleSession(String sessionId, String room, LocalDateTime start, LocalDateTime end) {
        Session session = require(sessions, sessionId, "Session");
        if (session.getStatus() != SessionStatus.ACCEPTED) throw new IllegalStateException("Session is not accepted");
        session.schedule(room, start, end);
        schedule.put(sessionId, sessionId);
    }

    public List<Session> getSchedule() {
        return schedule.values().stream().map(sessions::get)
                .sorted(Comparator.comparing(Session::getStart, Comparator.nullsLast(Comparator.naturalOrder())))
                .toList();
    }

    public BlogPost createBlogPost(String authorId, String title, String content, boolean drupalPlanet) {
        BlogPost post = new BlogPost(id(), authorId, title, content, drupalPlanet);
        blogPosts.put(post.getId(), post);
        return post;
    }

    public void updateBlogPost(String postId, String title, String content, boolean drupalPlanet) {
        require(blogPosts, postId, "Blog post").update(title, content, drupalPlanet);
    }

    public List<BlogPost> readBlog() { return List.copyOf(blogPosts.values()); }
    public List<BlogPost> drupalPlanetPosts() {
        return blogPosts.values().stream().filter(BlogPost::isDrupalPlanet).toList();
    }

    public Job postJob(String sponsorId, String title, String description) {
        getSponsor(sponsorId);
        Job job = new Job(id(), sponsorId, title, description);
        jobs.put(job.getId(), job);
        return job;
    }

    public List<Job> listJobs() { return List.copyOf(jobs.values()); }
    public Sponsor jobSponsor(String jobId) { return getSponsor(require(jobs, jobId, "Job").getSponsorId()); }

    public List<String> getNotifications() { return List.copyOf(notifications); }

    private static <T> T require(Map<String, T> map, String key, String label) {
        T value = map.get(Objects.requireNonNull(key));
        if (value == null) throw new IllegalArgumentException(label + " not found: " + key);
        return value;
    }
