public class MedicalInfo {
    private List<String> allergies;
    private List<String> medications;
    private String dietaryRestrictions;
    private String doctorName;
    private String doctorPhone;
    private String specialInstructions;

    public MedicalInfo() {
        this.allergies = new ArrayList<>();
        this.medications = new ArrayList<>();
        this.dietaryRestrictions = "None";
        this.specialInstructions = "";
    }

    public MedicalInfo(List<String> allergies, List<String> medications, String dietaryRestrictions, String doctorName, String doctorPhone, String specialInstructions) {
        this.allergies = allergies != null ? new ArrayList<>(allergies) : new ArrayList<>();

        this.doctorName = doctorName;
        this.doctorPhone = doctorPhone;
        this.specialInstructions = specialInstructions != null ? specialInstructions : "";
    }

    public List<String> getAllergies() {
        return allergies;
    }

    public void addAllergy(String allergy) {
        this.allergies.add(allergy);
    }

    public List<String> getMedications() {
        return medications;
    }

    public void addMedication(String medication) {
        this.medications.add(medication);
    }

    public String getDietaryRestrictions() {
        return dietaryRestrictions;
    }

    public void setDietaryRestrictions(String dietaryRestrictions) {
        this.dietaryRestrictions = dietaryRestrictions;
    }

    public String getDoctorName() {
        return doctorName;
    }

    public void setDoctorName(String doctorName) {
        this.doctorName = doctorName;
    }

    public String getDoctorPhone() {
        return doctorPhone;
    }

    public void setDoctorPhone(String doctorPhone) {
        this.doctorPhone = doctorPhone;
    }

    public String getSpecialInstructions() {

    }

import java.time.LocalDateTime;

public class PaymentRecord {
    private String id;
    private String parentId;
    private double amountDue;
    private double amountPaid;
    private LocalDateTime lastPaymentDate;

    public PaymentRecord(String id, String parentId, double amountDue) {
        this.id = id;
        this.parentId = parentId;
        this.amountDue = amountDue;
        this.amountPaid = 0.0;
    }

    public String getId() {
        return id;
    }

    public String getParentId() {
        return parentId;
    }

    public double getAmountDue() {
        return amountDue;
    }

    public double getAmountPaid() {
        return amountPaid;
    }

    public boolean isFullyPaid() {
        return amountPaid >= amountDue;
    }

    public double getRemainingBalance() {
        return Math.max(0.0, amountDue - amountPaid);
    }

    public boolean scheduleActivity(String id, String title, String description, String groupId, String facilityId, LocalDateTime start, LocalDateTime end) {
        // Prevent conflicting schedules for facility
        boolean conflict = scheduledActivities.values().stream()
                .anyMatch(a -> a.overlapsWith(facilityId, start, end));
        if (conflict) {
            return false;
        }
        ScheduledActivity activity = new ScheduledActivity(id, title, description, groupId, facilityId, start, end);
        scheduledActivities.put(id, activity);
        return true;

    public void modifyScheduledActivity(String id, String title, String description, LocalDateTime start, LocalDateTime end) {
        ScheduledActivity activity = scheduledActivities.get(id);
        if (activity != null) {
            // Check for conflict excluding current activity
            boolean conflict = scheduledActivities.values().stream()
                    .filter(a -> !a.getId().equals(id))
                    .anyMatch(a -> a.overlapsWith(activity.getFacilityId(), start, end));
            if (!conflict) {
                activity.setTitle(title);
                activity.setDescription(description);
                activity.setStartTime(start);
                activity.setEndTime(end);
            }
        }
    }
