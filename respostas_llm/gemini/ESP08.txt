import java.util.Objects;

public class FinancialContact {
    private final String name;
    private final String phone;
    private final String email;
    private final String title;
    private final String address;

    public FinancialContact(String name, String phone, String email, String title, String address) {
        this.name = Objects.requireNonNull(name, "Name cannot be null");
        this.phone = phone != null ? phone : "";
        this.email = Objects.requireNonNull(email, "Email cannot be null");
        this.title = title != null ? title : "";
        this.address = address != null ? address : "";
    }

    public String getName() {
        return name;
    }

    public String getPhone() {
        return phone;
    }

    public String getEmail() {
        return email;
    }

    public String getTitle() {
        return title;
    }

    public String getAddress() {
        return address;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof FinancialContact that)) return false;
        return Objects.equals(email, that.email);
    }

    @Override
    public int hashCode() {
        return Objects.hash(email);
    }

    @Override
    public String toString() {
        return name + " (" + title + ") <" + email + "> Ph: " + phone + ", Addr: " + address;
    }
}

import java.util.Collections;
import java.util.HashSet;
import java.util.Set;

public class AuthorAuthority {
    private final String canonicalId; // e.g. ORCID: 0000-0002-1825-0097
    private String primaryName;
    private final Set<String> nameVariants;

    public AuthorAuthority(String canonicalId, String primaryName) {
        this.canonicalId = canonicalId;
        this.primaryName = primaryName;
        this.nameVariants = new HashSet<>();
        this.nameVariants.add(primaryName);
    }

    public String getCanonicalId() {
        return canonicalId;
    }

    public String getPrimaryName() {
        return primaryName;
    }

    public void setPrimaryName(String primaryName) {
        this.primaryName = primaryName;
        this.nameVariants.add(primaryName);
    }

    public void addVariant(String variant) {
        if (variant != null && !variant.isBlank()) {
            this.nameVariants.add(variant.trim());
        }
    }

    public Set<String> getNameVariants() {
        return Collections.unmodifiableSet(nameVariants);
    }

    public boolean matches(String query) {
        if (canonicalId.equalsIgnoreCase(query)) return true;
        return nameVariants.stream().anyMatch(v -> v.equalsIgnoreCase(query));
    }

    @Override
    public String toString() {
        return primaryName + " [" + canonicalId + "] (Aliases: " + nameVariants + ")";
    }
}

    }

        return Collections.unmodifiableList(replicationRecords);
    }

    public void addReplicationRecord(ReplicationRecord record) {
        this.replicationRecords.add(record);
    }

        StringBuilder sb = new StringBuilder();
        sb.append("--- Component: ").append(filename).append(" (ID: ").append(id).append(") ---\n");
        sb.append("Size: ").append(sizeBytes).append(" bytes | Format: ").append(format).append("\n");
        sb.append("Format Validity: [Valid: ").append(valid).append(", Well-Formed: ").append(wellFormed).append("]\n");
        sb.append("Scheduled Deletion Date: ").append(scheduledDeletionDate != null ? scheduledDeletionDate.toString() : "None").append("\n");
        sb.append("Replication Status: ");
        if (replicationRecords.isEmpty()) {
            sb.append("No replication records\n");
        } else {
            for (ReplicationRecord r : replicationRecords) {
                sb.append(r.toString()).append(" ");
            }
            sb.append("\n");
        }
        return sb.toString();
    }

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.UUID;
import java.util.stream.Collectors;

public class AuditService {
    private final List<AuditEvent> events = new ArrayList<>();

    public AuditEvent logEvent(AuditAction action, String targetId, String actorNetId, String comment) {
        AuditEvent event = new AuditEvent(UUID.randomUUID().toString(), action, targetId, actorNetId, comment);
        events.add(event);
        return event;
    }

    public List<AuditEvent> getEventsForTarget(String targetId) {
        return events.stream()
                .filter(e -> e.getTargetId().equals(targetId))
                .collect(Collectors.toList());
    }

    public List<AuditEvent> getAllEvents() {
        return Collections.unmodifiableList(events);
    }
}

    private boolean superuserMode;
    private String ipAddress;

    public User(String netId, String name, String email, String ipAddress) {
        this.netId = netId;
        this.name = name;
        this.email = email;
        this.roles = new HashSet<>();
        this.superuserMode = false;
        this.ipAddress = ipAddress;
    }

    public String getNetId() {
        return netId;
    }

    public String getName() {
        return name;
    }

    public String getEmail() {
        return email;
    }

        return ipAddress;
    }

    public void setIpAddress(String ipAddress) {
        this.ipAddress = ipAddress;
    }

    public void addRole(String role) {
        this.roles.add(role.toUpperCase());
    }

    public boolean hasRole(String role) {
        return this.roles.contains(role.toUpperCase()) || (superuserMode && "SUPERUSER".equalsIgnoreCase(role));
    }

    public boolean isSuperuserMode() {
        return superuserMode;
    }

    public void enterSuperuserScope() {
        if (this.roles.contains("SUPERUSER")) {
            this.superuserMode = true;
        } else {
            throw new SecurityException("User " + netId + " is not authorized as a superuser.");
        }
    }

    public void exitSuperuserScope() {
        this.superuserMode = false;
    }
}

        this.windowEndDate = windowEndDate;
    }

    public boolean isWindowOpen(LocalDate date) {
        if (windowStartDate != null && date.isBefore(windowStartDate)) return false;
        if (windowEndDate != null && date.isAfter(windowEndDate)) return false;
        return true;
    }

        this.memberNetIds.add(netId);
        this.submissionStatus.putIfAbsent(netId, false);

        this.memberNetIds.remove(netId);
        this.submissionStatus.remove(netId);
    }

    public void recordSubmission(String netId) {
        if (this.memberNetIds.contains(netId)) {
            this.submissionStatus.put(netId, true);
        }
    }

    public boolean hasSubmitted(String netId) {
        return this.submissionStatus.getOrDefault(netId, false);
    }

    public Set<String> getMemberNetIds() {
        return Collections.unmodifiableSet(memberNetIds);
    }

    public Map<String, Boolean> getSubmissionStatusReport() {
        return Collections.unmodifiableMap(submissionStatus);

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Collection;
import java.util.List;
import java.util.concurrent.atomic.AtomicReference;
import java.util.stream.Collectors;

public class SearchService {

    public enum DateFilterType {
        NONE,
        BEFORE,
        AFTER,
        BETWEEN
    }

    private final AtomicReference<List<Item>> indexRef = new AtomicReference<>(new ArrayList<>());

    public void updateIndex(Collection<Item> items) {
        // Zero-downtime shadow index update
        List<Item> newIndex = new ArrayList<>(items);
        indexRef.set(newIndex);
    }

    public List<Item> search(String query, DateFilterType filterType, LocalDate date1, LocalDate date2) {
        List<Item> currentSnapshot = indexRef.get();
        return currentSnapshot.stream()
                .filter(item -> !item.isDeleted() && !item.isHidden())
                .filter(item -> {
                    if (query == null || query.isBlank()) return true;
                    String q = query.toLowerCase();
                    return item.getTitle().toLowerCase().contains(q)
                            || item.getId().toLowerCase().contains(q)
                            || item.getAuthorAuthorityIds().stream().anyMatch(a -> a.toLowerCase().contains(q));
                })
                .filter(item -> {
                    LocalDate d = item.getDepositDate();
                    if (d == null) return false;
                    switch (filterType) {
                        case BEFORE:
                            return date1 != null && d.isBefore(date1);
                        case AFTER:
                            return date1 != null && d.isAfter(date1);
                        case BETWEEN:
                            return date1 != null && date2 != null && !d.isBefore(date1) && !d.isAfter(date2);
                        case NONE:
                        default:
                            return true;
                    }
                })
                .collect(Collectors.toList());
    }
}

    public boolean depositItem(Item item, String depositorNetId, String licenseAcceptedId) {
        Collection col = collections.get(item.getPrimaryCollectionId());
        if (col == null) {
            errorLog.logError("Collection not found for deposit: " + item.getPrimaryCollectionId());
            return false;
        }

        DepositForm form = col.getDepositForm();
        if (form != null && !form.getAllowedLicenses().isEmpty()) {
            boolean validLicenseAccepted = form.getAllowedLicenses().stream()
                    .anyMatch(l -> l.getId().equalsIgnoreCase(licenseAcceptedId));
            if (!validLicenseAccepted) {
                errorLog.logError("Deposit rejected: Required collection license was not accepted for item " + item.getId());
                return false;
            }
            item.setLicense(licenses.get(licenseAcceptedId));
        }

        // Validate format for components
        for (Component c : item.getComponents()) {
            if (!ACCEPTED_FORMATS.contains(c.getFormat().toUpperCase())) {
                c.setValid(false);
                errorLog.logError("Unrecognized format for component " + c.getFilename() + ": " + c.getFormat());
            }
        }

        items.put(item.getId(), item);
        col.addItemId(item.getId());
        queueReplication(col, item);

        auditService.logEvent(AuditAction.INGEST, item.getId(), depositorNetId, "Item deposited into collection " + col.getId());
        searchService.updateIndex(items.values());
        return true;

    public void queueReplication(Collection col, Item item) {
        ReplicationPolicy policy = col.getReplicationPolicy();
        if (!policy.isReplicateRemotely()) return;

        List<String> targetSites = new ArrayList<>();
        if (policy.getPrimaryRemoteSite() != null) targetSites.add(policy.getPrimaryRemoteSite());
        targetSites.addAll(policy.getSecondarySites());

        for (Component c : item.getComponents()) {
            for (String site : targetSites) {
                ReplicationRecord rec = new ReplicationRecord(site);
                rec.setStatus(ReplicationStatus.ENCRYPTED);
                String simulatedChecksum = computeSha256("ServerSecretKey:" + c.getFilename() + ":" + c.getSizeBytes());
                rec.setEncryptedChecksumSha256(simulatedChecksum);
                rec.setStatus(ReplicationStatus.COMPLETED);
                c.addReplicationRecord(rec);
            }
        }
        auditService.logEvent(AuditAction.INGEST, item.getId(), "SYSTEM", "Remote replication completed for " + targetSites);
    }

        AccessPermission perm = item.getCurrentAccessPermission();

            case RESTRICTED_CURATOR_ONLY:

    private static String computeSha256(String input) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(input.getBytes(java.nio.charset.StandardCharsets.UTF_8));
            StringBuilder hexString = new StringBuilder();
            for (byte b : hash) {
                String hex = Integer.toHexString(0xff & b);
                if (hex.length() == 1) hexString.append('0');
                hexString.append(hex);
            }
            return hexString.toString();
        } catch (Exception e) {
            return "simulated-sha256-hash-" + input.hashCode();
        }
    }

// File: BatchIngestService.java
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.UUID;

public class BatchIngestService {
    private final RepositoryService repositoryService;

    public BatchIngestService(RepositoryService repositoryService) {
        this.repositoryService = repositoryService;
    }

    public List<String> batchIngestProQuestEtds(String collectionId, List<BatchIngestRecord> etdRecords, FinancialContact fc) {
        List<String> createdIds = new ArrayList<>();
        for (BatchIngestRecord rec : etdRecords) {
            String itemId = "etd-" + UUID.randomUUID().toString().substring(0, 8);
            Item item = new Item(itemId, rec.getTitle(), collectionId, rec.getAuthorNetIdOrName(), fc, AccessPermission.PUBLIC);
            item.setEtd(true);

            if (rec.getOrcidId() != null) {
                item.addAuthorAuthorityId(rec.getOrcidId());
            }

            for (Component comp : rec.getComponents()) {
                item.addComponent(comp);
            }

            if (rec.getEmbargoReleaseDate() != null) {
                // ETD embargo: metadata open to all, files restricted to curators/repoadmins until release date
                item.setEmbargo(rec.getEmbargoReleaseDate(), AccessPermission.PUBLIC);
                item.setCurrentAccessPermission(AccessPermission.RESTRICTED_CURATOR_ONLY);
            }

            boolean success = repositoryService.depositItem(item, rec.getAuthorNetIdOrName(), rec.getLicenseId());
            if (success) {
                createdIds.add(itemId);
            }
        }
        return createdIds;
    }
}
