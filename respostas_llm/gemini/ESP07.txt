public class Field {
    private final String name;
    private final FieldType type;
    private final String description;

    public Field(String name, FieldType type, String description) {
        this.name = Objects.requireNonNull(name, "Field name cannot be null");
        this.type = Objects.requireNonNull(type, "Field type cannot be null");
        this.description = description == null ? "" : description;
    }

    public Field(String name, FieldType type) {
        this(name, type, "");
    }

    public String getName() {
        return name;
    }

    public FieldType getType() {
        return type;
    }

    public String getDescription() {
        return description;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Field field)) return false;
        return name.equalsIgnoreCase(field.name) && type == field.type;
    }

    @Override
    public int hashCode() {
        return Objects.hash(name.toLowerCase(), type);
    }

    @Override
    public String toString() {
        return name + " (" + type + ")";
    }
}

    private final List<Field> removedFields;
    private final List<String> modifiedFields;

    public SchemaDiff(List<Field> addedFields, List<Field> removedFields, List<String> modifiedFields) {
        this.addedFields = Collections.unmodifiableList(addedFields);
        this.removedFields = Collections.unmodifiableList(removedFields);

    }

    public List<Field> getRemovedFields() {
        return removedFields;
    }

    public List<String> getModifiedFields() {
        return modifiedFields;
    }

    public boolean hasChanges() {
        return !addedFields.isEmpty() || !removedFields.isEmpty() || !modifiedFields.isEmpty();
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        sb.append("SchemaDiff:\n");
        sb.append("  Added fields: ").append(addedFields).append("\n");
        sb.append("  Removed fields: ").append(removedFields).append("\n");
        sb.append("  Modified fields: ").append(modifiedFields);
        return sb.toString();
    }

package dataregistry;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Optional;

public class Schema {
    private final List<Field> fields;

    public Schema(List<Field> fields) {
        this.fields = fields != null ? new ArrayList<>(fields) : new ArrayList<>();
    }

    public Schema() {
        this(new ArrayList<>());
    }

    public List<Field> getFields() {
        return Collections.unmodifiableList(fields);
    }

    public Optional<Field> getField(String name) {
        return fields.stream()
                .filter(f -> f.getName().equalsIgnoreCase(name))
                .findFirst();
    }

    public SchemaDiff compareTo(Schema other) {
        if (other == null) {
            return new SchemaDiff(new ArrayList<>(fields), Collections.emptyList(), Collections.emptyList());
        }

        List<Field> added = new ArrayList<>();
        List<Field> removed = new ArrayList<>();
        List<String> modified = new ArrayList<>();

        for (Field f : fields) {
            Optional<Field> otherField = other.getField(f.getName());
            if (otherField.isEmpty()) {
                added.add(f);
            } else if (otherField.get().getType() != f.getType()) {
                modified.add(f.getName() + " changed type from " + otherField.get().getType() + " to " + f.getType());
            }
        }

        for (Field of : other.getFields()) {
            if (getField(of.getName()).isEmpty()) {
                removed.add(of);
            }
        }

        return new SchemaDiff(added, removed, modified);
    }
}

package dataregistry;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

public class DataPackageValidator {

    public ValidationResult validate(DataPackageDescriptor descriptor) {
        List<String> errors = new ArrayList<>();

        if (descriptor == null) {
            errors.add("Descriptor cannot be null");
            return ValidationResult.failure(errors);
        }

        if (descriptor.getName() == null || descriptor.getName().trim().isEmpty()) {
            errors.add("Package name is required");
        } else if (!descriptor.getName().matches("^[a-z0-9-_]+$")) {
            errors.add("Package name must only contain lowercase alphanumeric, hyphen or underscore characters");
        }

        if (descriptor.getResources() == null || descriptor.getResources().isEmpty()) {
            errors.add("At least one resource must be provided");
        } else {
            for (Resource res : descriptor.getResources()) {
                if (res.getName() == null || res.getName().isBlank()) {
                    errors.add("Resource name cannot be blank");
                }
                if (res.getSchema() == null || res.getSchema().getFields().isEmpty()) {
                    errors.add("Resource " + res.getName() + " must have a defined schema with fields");
                } else {
                    validateResourceData(res, errors);
                }
            }
        }

        return errors.isEmpty() ? ValidationResult.success() : ValidationResult.failure(errors);
    }

    private void validateResourceData(Resource res, List<String> errors) {
        Schema schema = res.getSchema();
        for (int i = 0; i < res.getDataRows().size(); i++) {
            Map<String, Object> row = res.getDataRows().get(i);
            for (Field field : schema.getFields()) {
                Object val = row.get(field.getName());
                if (val != null && !isTypeCompatible(val, field.getType())) {
                    errors.add(String.format("Row %d: value '%s' in field '%s' does not match declared type %s",
                            i + 1, val, field.getName(), field.getType()));
                }
            }
        }
    }

    private boolean isTypeCompatible(Object value, FieldType expectedType) {
        return switch (expectedType) {
            case STRING -> value instanceof String;
            case INTEGER -> value instanceof Integer || value instanceof Long;
            case NUMBER -> value instanceof Number;
            case BOOLEAN -> value instanceof Boolean;
            case DATE -> value instanceof String && ((String) value).matches("^\\d{4}-\\d{2}-\\d{2}$");
            case JSON -> value instanceof String || value instanceof Map || value instanceof List;
        };
    }
}

    }

    public boolean isDeleted() {
        return deleted;
    }

    public void setDeleted(boolean deleted) {
        this.deleted = deleted;
    }

    public int getDownloadCount() {
        return downloadCount;
    }

    public void incrementDownloadCount() {
        this.downloadCount++;
    }

    public Instant getCreatedAt() {
        return createdAt;
    }

    public Instant getUpdatedAt() {
        return updatedAt;
    }

    public Duration getAge() {
        return Duration.between(updatedAt, Instant.now());
    }

    public DataPackageDescriptor getCurrentDescriptor() {
        return currentDescriptor;
    }

    public void updateDescriptor(DataPackageDescriptor newDescriptor, boolean overwriteSnapshot) {
        this.currentDescriptor = newDescriptor;
        this.name = newDescriptor.getName();
        this.updatedAt = Instant.now();
        snapshot(newDescriptor.getVersion(), overwriteSnapshot);
    }

    public void snapshot(String tagOrVersion, boolean overwrite) {
        boolean remoteCaching = currentDescriptor.getResources().stream().anyMatch(Resource::isRemoteCached);
        if (!overwrite && snapshots.containsKey(tagOrVersion)) {
            throw new IllegalStateException("Tag/Version '" + tagOrVersion + "' already exists! Use force/overwrite to replace.");
        }
        snapshots.put(tagOrVersion, new DataPackageSnapshot(tagOrVersion, currentDescriptor, remoteCaching));
    }

    public boolean hasTag(String tag) {
        return snapshots.containsKey(tag);
    }

    public Optional<DataPackageSnapshot> getSnapshot(String tagOrVersion) {
        return Optional.ofNullable(snapshots.get(tagOrVersion));
    }

    public Map<String, DataPackageSnapshot> getAllSnapshots() {
        return Collections.unmodifiableMap(snapshots);
    }

    public void addSubscriber(String email) {
        subscriberEmails.add(email);
    }

    public Set<String> getSubscribers() {
        return Collections.unmodifiableSet(subscriberEmails);
    }
}

package dataregistry;

import java.util.Map;

public class DataPackageExporter {

    public String exportToZipArchiveBundle(DataPackageDescriptor descriptor) {
        StringBuilder zipBundle = new StringBuilder();
        zipBundle.append("=== ZIP ARCHIVE BUNDLE: ").append(descriptor.getName()).append(".zip ===\n");
        zipBundle.append("1. /datapackage.json (descriptor: name=").append(descriptor.getName())
                .append(", version=").append(descriptor.getVersion()).append(")\n");

        for (Resource res : descriptor.getResources()) {
            zipBundle.append("2. /data/").append(res.getName()).append(".csv (")
                    .append(res.getDataRows().size()).append(" records, remoteCached=")
                    .append(res.isRemoteCached()).append(")\n");
        }
        zipBundle.append("=== END ARCHIVE ===");
        return zipBundle.toString();
    }

    public String exportToSqliteDdlAndInsert(DataPackageDescriptor descriptor) {
        StringBuilder sql = new StringBuilder();
        sql.append("-- SQLite database export for DataPackage: ").append(descriptor.getName()).append("\n");

        for (Resource res : descriptor.getResources()) {
            String tableName = res.getName().replaceAll("[^a-zA-Z0-9_]", "_");
            sql.append("CREATE TABLE IF NOT EXISTS ").append(tableName).append(" (\n");

            List<Field> fields = res.getSchema().getFields();
            for (int i = 0; i < fields.size(); i++) {
                Field f = fields.get(i);
                String sqlType = switch (f.getType()) {
                    case INTEGER -> "INTEGER";
                    case NUMBER -> "REAL";
                    case BOOLEAN -> "INTEGER";
                    default -> "TEXT";
                };
                sql.append("    ").append(f.getName()).append(" ").append(sqlType);
                if (i < fields.size() - 1) sql.append(",");
                sql.append("\n");
            }
            sql.append(");\n");

            for (Map<String, Object> row : res.getDataRows()) {
                sql.append("INSERT INTO ").append(tableName).append(" VALUES (");
                for (int i = 0; i < fields.size(); i++) {
                    Field f = fields.get(i);
                    Object val = row.get(f.getName());
                    if (val == null) {
                        sql.append("NULL");
                    } else if (val instanceof Number || val instanceof Boolean) {
                        sql.append(val);
                    } else {

                    }

                }
                sql.append(");\n");
            }
        }
        return sql.toString();
    }

import java.util.*;

public class LocalProjectEnvironment {
    private final String projectName;
    private final Map<String, ProjectDependency> manifestDependencies = new LinkedHashMap<>();
    // Local storage of installed packages: packageName -> (version -> DataPackageSnapshot)
    private final Map<String, Map<String, DataPackageSnapshot>> installedPackages = new HashMap<>();

    public LocalProjectEnvironment(String projectName) {
        this.projectName = projectName;
    }

    public String getProjectName() {
        return projectName;
    }

    public void pinDependency(String packageName, String version) {
        manifestDependencies.put(packageName, new ProjectDependency(packageName, version));
    }

    public Map<String, ProjectDependency> getManifestDependencies() {
        return Collections.unmodifiableMap(manifestDependencies);
    }

    public void installPackageVersion(String packageName, DataPackageSnapshot snapshot) {
        installedPackages
                .computeIfAbsent(packageName, k -> new HashMap<>())
                .put(snapshot.getTagOrVersion(), snapshot);
    }

    public Optional<DataPackageSnapshot> getInstalledVersion(String packageName, String version) {
        Map<String, DataPackageSnapshot> versions = installedPackages.get(packageName);
        if (versions == null) return Optional.empty();
        return Optional.ofNullable(versions.get(version));
    }

    public List<String> listInstalledVersions(String packageName) {
        Map<String, DataPackageSnapshot> versions = installedPackages.get(packageName);
        if (versions == null) return Collections.emptyList();
        return new ArrayList<>(versions.keySet());
    }
}

    public DataPackage publishPackage(User actor, String publisherName, DataPackageDescriptor descriptor, Visibility visibility) {
        apiCallCount.incrementAndGet();
        PublisherOrganization org = publishersByName.get(publisherName.toLowerCase());
        if (org == null) throw new IllegalArgumentException("Publisher not found: " + publisherName);
        if (!org.hasPublishAccess(actor)) throw new SecurityException("User does not have publishing permissions for " + publisherName);

        // Validation upon publishing
        ValidationResult valResult = validator.validate(descriptor);
        if (!valResult.isValid()) {
            throw new IllegalArgumentException("Package validation failed: " + String.join(", ", valResult.getErrors()));
        }

        // Plan limitation check for private packages
        if (visibility == Visibility.PRIVATE) {
            long existingPrivates = packagesById.values().stream()
                    .filter(p -> p.getPublisherId().equals(org.getId()) && p.getVisibility() == Visibility.PRIVATE && !p.isDeleted())
                    .count();
            if (existingPrivates >= org.getPricingPlan().getMaxPrivatePackages()) {
                throw new IllegalStateException("Pricing plan quota exceeded: cannot create more private packages under " + org.getPricingPlan().getName());
            }
        }

        String packageKey = org.getName() + "/" + descriptor.getName();
        DataPackage existing = packagesById.get(packageKey);
        if (existing == null) {
            DataPackage created = new DataPackage("pkg_" + UUID.randomUUID().toString().substring(0, 8), org.getId(), descriptor, visibility);
            packagesById.put(packageKey, created);
            return created;
        } else {
            existing.updateDescriptor(descriptor, false);
            existing.setVisibility(visibility);
            existing.setPublished(true);
            existing.setDeleted(false);
            notificationService.notifySubscribers(existing, "Package updated to version " + descriptor.getVersion());
            return existing;
        }
    }

        pkg.snapshot(tag, overwrite);
    }

        packagesById.remove(publisherName.toLowerCase() + "/" + packageName.toLowerCase());
    }

    public Optional<DataPackage> getPackage(String publisherName, String packageName) {
        apiCallCount.incrementAndGet();
        DataPackage pkg = packagesById.get(publisherName.toLowerCase() + "/" + packageName.toLowerCase());
        if (pkg != null && !pkg.isDeleted()) {
            return Optional.of(pkg);
        }
        return Optional.empty();
    }

    public DataPackageSnapshot downloadPackage(String publisherName, String packageName, Optional<String> versionTag, User consumer) {
        apiCallCount.incrementAndGet();
        DataPackage pkg = getPackage(publisherName, packageName)
                .orElseThrow(() -> new IllegalArgumentException("Package not found or deleted"));

        PublisherOrganization org = publishersById.get(pkg.getPublisherId());
        if (pkg.getVisibility() == Visibility.PRIVATE) {
            if (consumer == null || !org.getMembers().containsKey(consumer)) {
                throw new SecurityException("Private package can only be accessed by publisher members");
            }
        }

        pkg.incrementDownloadCount();

        if (versionTag.isPresent()) {
            return pkg.getSnapshot(versionTag.get())
                    .orElseThrow(() -> new IllegalArgumentException("Version/Tag " + versionTag.get() + " not found"));
        } else {
            return pkg.getSnapshot(pkg.getCurrentDescriptor().getVersion())
                    .orElseGet(() -> new DataPackageSnapshot("latest", pkg.getCurrentDescriptor(), true));
        }
    }

    public List<DataPackage> searchPackages(String query, Optional<String> publisherFilter) {
        apiCallCount.incrementAndGet();
        String q = query.toLowerCase();
        return packagesById.values().stream()
                .filter(p -> !p.isDeleted() && p.isPublished() && p.getVisibility() == Visibility.PUBLIC)
                .filter(p -> publisherFilter.isEmpty() || publishersById.get(p.getPublisherId()).getName().equalsIgnoreCase(publisherFilter.get()))
                .filter(p -> p.getName().toLowerCase().contains(q) ||
                             p.getCurrentDescriptor().getTitle().toLowerCase().contains(q) ||
                             p.getCurrentDescriptor().getDescription().toLowerCase().contains(q))
                .sorted(Comparator.comparingInt(DataPackage::getDownloadCount).reversed())
                .collect(Collectors.toList());
    }

        return new SiteMetrics(usersById.size(), publishersById.size(), packagesById.size(), totalDownloads, apiCallCount.get());
    }

        System.out.println("\n--- 9. Lifecycle: Unpublish, Delete, Restore & Purge ---");
        registry.unpublishPackage(alice, "acme-data", "air-quality");
        System.out.println("Package unpublished. Available in public search? "
                + !registry.searchPackages("air-quality", Optional.empty()).isEmpty());

        registry.deletePackage(alice, "acme-data", "air-quality");
        System.out.println("Package soft deleted. Found in active lookup? "
                + registry.getPackage("acme-data", "air-quality").isPresent());

        // Restore via CLI
        cli.undelete(alice, "acme-data", "air-quality");
        System.out.println("Package undeleted/restored. Found in active lookup? "
                + registry.getPackage("acme-data", "air-quality").isPresent());

        // 10. Platform Administration, Site Configuration & Key Metrics
        System.out.println("\n--- 10. Administration & Platform Activity Metrics ---");
        SiteConfig siteConfig = registry.getSiteConfig();
        siteConfig.setSiteTitle("Open DataHub Enterprise Cloud");
        System.out.println("Updated Site Title: " + registry.getSiteConfig().getSiteTitle());

        SiteMetrics metrics = registry.getMetrics();
        System.out.println(metrics);
        System.out.println("\nDemo completed successfully.");
    }
