import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.Map;
import java.util.Set;

public class User {
    private final String username;
    private String passwordHash;
    private String fullName;
    private String email;
    private boolean isSystemAdmin;
    private final Map<String, Role> repositoryRoles = new HashMap<>();

    public User(String username, String password, String fullName, String email, boolean isSystemAdmin) {
        this.username = username;
        this.passwordHash = hashPassword(password);
        this.fullName = fullName;
        this.email = email;
        this.isSystemAdmin = isSystemAdmin;
    }

    public static String hashPassword(String password) {
        return Integer.toHexString(password.hashCode());
    }

    public boolean verifyPassword(String password) {
        return passwordHash.equals(hashPassword(password));
    }

    public void setPassword(String newPassword) {
        this.passwordHash = hashPassword(newPassword);
    }

    public String getUsername() {
        return username;
    }

    public String getFullName() {
        return fullName;
    }

    public void setFullName(String fullName) {
        this.fullName = fullName;
    }

    public String getEmail() {
        return email;
    }

    public void setEmail(String email) {
        this.email = email;
    }

    public boolean isSystemAdmin() {
        return isSystemAdmin;
    }

    public void setSystemAdmin(boolean systemAdmin) {
        isSystemAdmin = systemAdmin;
    }

    public void assignRole(String repositoryId, Role role) {
        repositoryRoles.put(repositoryId, role);
    }

    public void removeRole(String repositoryId) {
        repositoryRoles.remove(repositoryId);
    }

    public Role getRoleForRepository(String repositoryId) {
        if (isSystemAdmin) {
            return Role.SYSTEM_ADMIN;
        }
        return repositoryRoles.get(repositoryId);
    }

    public Map<String, Role> getRepositoryRoles() {
        return Collections.unmodifiableMap(repositoryRoles);
    }

    public Set<Permission> getPermissions(String repositoryId) {
        Set<Permission> permissions = new HashSet<>();
        if (isSystemAdmin) {
            Collections.addAll(permissions, Permission.values());
            return permissions;
        }

        Role role = repositoryRoles.get(repositoryId);
        if (role == null) {
            return permissions;
        }

        switch (role) {
            case REPOSITORY_MANAGER -> {
                permissions.add(Permission.VIEW_RECORD);
                permissions.add(Permission.CREATE_BASIC_RECORD);
                permissions.add(Permission.CREATE_ADVANCED_RECORD);
                permissions.add(Permission.EDIT_RECORD);
                permissions.add(Permission.DELETE_RECORD);
                permissions.add(Permission.MANAGE_REPOSITORY);
            }
            case ADVANCED_DATA_ENTRY -> {
                permissions.add(Permission.VIEW_RECORD);
                permissions.add(Permission.CREATE_BASIC_RECORD);
                permissions.add(Permission.CREATE_ADVANCED_RECORD);
                permissions.add(Permission.EDIT_RECORD);
            }
            case BASIC_DATA_ENTRY -> {
                permissions.add(Permission.VIEW_RECORD);
                permissions.add(Permission.CREATE_BASIC_RECORD);
                permissions.add(Permission.EDIT_RECORD);
            }
            case READ_ONLY -> {
                permissions.add(Permission.VIEW_RECORD);
            }
            default -> {}
        }
        return permissions;
    }
}

import java.util.HashMap;
import java.util.Map;
import java.util.Optional;

public class AuthService {
    private final Map<String, User> userStore = new HashMap<>();
    private final LdapDirectory ldapDirectory;

    public AuthService(LdapDirectory ldapDirectory) {
        this.ldapDirectory = ldapDirectory;
    }

    public void createUser(User actor, User newUser) {
        if (actor != null && !actor.isSystemAdmin()) {
            throw new SecurityException("Only System Administrators can create users.");
        }
        userStore.put(newUser.getUsername(), newUser);
    }

    public Optional<User> authenticateLocal(String username, String password) {
        User user = userStore.get(username);
        if (user != null && user.verifyPassword(password)) {
            return Optional.of(user);
        }
        return Optional.empty();
    }

    public Optional<User> authenticateLdap(String username, String password) {
        if (ldapDirectory != null && ldapDirectory.authenticate(username, password)) {
            User user = userStore.get(username);
            if (user != null) {
                return Optional.of(user);
            }
            User syncedUser = new User(username, password, username, username + "@ldap.org", false);
            userStore.put(username, syncedUser);
            return Optional.of(syncedUser);
        }
        return Optional.empty();
    }

    public void resetPassword(User actor, String targetUsername, String newPassword) {
        if (!actor.isSystemAdmin()) {
            throw new SecurityException("Only System Administrators can reset passwords.");
        }
        User target = userStore.get(targetUsername);
        if (target == null) {
            throw new IllegalArgumentException("User not found: " + targetUsername);
        }
        target.setPassword(newPassword);
    }

    public Optional<User> findByUsername(String username) {
        return Optional.ofNullable(userStore.get(username));
    }
}

import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.StandardCopyOption;
import java.util.HashMap;
import java.util.Map;
import java.util.Properties;

public class SystemConfiguration {
    private final Properties properties = new Properties();
    private final Map<String, String> helpTopicMappings = new HashMap<>();
    private String helpCenterUrl = "https://help.archivesspace.org";
    private String databasePath = "./demo_db/archivesspace.db";
    private String backupDirectory = "./demo_db_backups";

    public SystemConfiguration() {
        setDefaultHelpMappings();
    }

    private void setDefaultHelpMappings() {
        helpTopicMappings.put("resource", "/help/resources/overview");
        helpTopicMappings.put("accession", "/help/accessions/overview");
        helpTopicMappings.put("digital_object", "/help/digital-objects/overview");
        helpTopicMappings.put("agent", "/help/agents/overview");
        helpTopicMappings.put("subject", "/help/subjects/overview");
    }

    public void loadExternalConfiguration(File configFile) throws IOException {
        if (configFile != null && configFile.exists()) {
            try (FileInputStream fis = new FileInputStream(configFile)) {
                properties.load(fis);
            }
            if (properties.containsKey("help.center.url")) {
                this.helpCenterUrl = properties.getProperty("help.center.url");
            }
            if (properties.containsKey("db.path")) {
                this.databasePath = properties.getProperty("db.path");
            }
            if (properties.containsKey("db.backup.dir")) {
                this.backupDirectory = properties.getProperty("db.backup.dir");
            }
        }
    }

    public void setHelpCenterUrl(String url) {
        this.helpCenterUrl = url;
    }

    public String getHelpCenterUrl() {
        return helpCenterUrl;
    }

    public void setHelpTopicMapping(String topic, String relativePath) {
        helpTopicMappings.put(topic, relativePath);
    }

    public String getHelpUrlForTopic(String topic) {
        String path = helpTopicMappings.getOrDefault(topic, "/help/index");
        return helpCenterUrl.replaceAll("/$", "") + "/" + path.replaceAll("^/", "");
    }

    public void setDatabasePath(String path) {
        this.databasePath = path;
    }

    public String getDatabasePath() {
        return databasePath;
    }

    public void setBackupDirectory(String backupDirectory) {
        this.backupDirectory = backupDirectory;
    }

    public String getBackupDirectory() {
        return backupDirectory;
    }

    public File backupDatabase() throws IOException {
        File dbFile = new File(databasePath);
        File backupDir = new File(backupDirectory);
        if (!backupDir.exists()) {
            backupDir.mkdirs();
        }
        if (!dbFile.exists()) {
            if (dbFile.getParentFile() != null) {
                dbFile.getParentFile().mkdirs();
            }
            dbFile.createNewFile();
        }
        File backupTarget = new File(backupDir, "db_backup_" + System.currentTimeMillis() + ".bak");
        Files.copy(dbFile.toPath(), backupTarget.toPath(), StandardCopyOption.REPLACE_EXISTING);
        return backupTarget;
    }
}

import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class DynamicEnumManager {
    private final Map<String, List<String>> enumLists = new HashMap<>();

    public DynamicEnumManager() {
        loadDefaultLists();
    }

    public void loadDefaultLists() {
        List<String> levels = new ArrayList<>(List.of(
            "collection", "fonds", "class", "recordgrp", "series", "subseries", "file", "item", "other"
        ));
        enumLists.put("archival_record_level", levels);

        List<String> agentSources = new ArrayList<>(List.of(
            "local", "naf", "lcsh", "viaf", "ulan"
        ));
        enumLists.put("name_source", agentSources);

        List<String> subjectSources = new ArrayList<>(List.of(
            "lcsh", "mesh", "fast", "local", "aat"
        ));
        enumLists.put("subject_source", subjectSources);

        List<String> containerTypes = new ArrayList<>(List.of(
            "box", "folder", "volume", "reel", "carton", "drawer"
        ));
        enumLists.put("container_type", containerTypes);
    }

    public void loadValuesFromStorage(String enumName, List<String> values) {
        enumLists.put(enumName, new ArrayList<>(values));
    }

    public List<String> getValues(String enumName) {
        return enumLists.getOrDefault(enumName, Collections.emptyList());
    }

    public boolean containsValue(String enumName, String value) {
        List<String> list = enumLists.get(enumName);
        return list != null && list.stream().anyMatch(v -> v.equalsIgnoreCase(value));
    }

    public void addValue(String enumName, String value) {
        List<String> list = enumLists.computeIfAbsent(enumName, k -> new ArrayList<>());
        if (!containsValue(enumName, value)) {
            list.add(value);
        }
    }

    public int bulkUpdateValue(String enumName, String oldValue, String newValue) {
        List<String> list = enumLists.get(enumName);
        if (list == null) {
            return 0;
        }
        int count = 0;
        for (int i = 0; i < list.size(); i++) {
            if (list.get(i).equalsIgnoreCase(oldValue)) {
                list.set(i, newValue);
                count++;
            }
        }
        return count;
    }
}

import java.util.ArrayList;
import java.util.List;

public abstract class ArchivalRecord {
    private final String id;
    private final String repositoryId;
    private String title;
    private int lockVersion;
    private String sourceRecordId;
    private SourceSystem sourceSystem = SourceSystem.ARCHIVES_SPACE_NATIVE;

    private final List<DateSubrecord> dates = new ArrayList<>();
    private final List<ExtentSubrecord> extents = new ArrayList<>();
    private final List<String> subjectIds = new ArrayList<>();
    private final List<AgentLink> agentLinks = new ArrayList<>();
    private final List<Note> notes = new ArrayList<>();
    private final List<RightsStatement> rightsStatements = new ArrayList<>();
    private CollectionManagement collectionManagement;

    public ArchivalRecord(String id, String repositoryId, String title) {
        this.id = id;
        this.repositoryId = repositoryId;
        this.title = title;
        this.lockVersion = 0;
    }

    public String getId() { return id; }
    public String getRepositoryId() { return repositoryId; }
    public String getTitle() { return title; }
    public void setTitle(String title) { this.title = title; }
    public int getLockVersion() { return lockVersion; }
    public void incrementLockVersion() { this.lockVersion++; }
    public void setLockVersion(int lockVersion) { this.lockVersion = lockVersion; }

    public String getSourceRecordId() { return sourceRecordId; }
    public void setSourceRecordId(String sourceRecordId) { this.sourceRecordId = sourceRecordId; }
    public SourceSystem getSourceSystem() { return sourceSystem; }
    public void setSourceSystem(SourceSystem sourceSystem) { this.sourceSystem = sourceSystem; }

    public List<DateSubrecord> getDates() { return dates; }
    public void addDate(DateSubrecord date) { this.dates.add(date); }

    public List<ExtentSubrecord> getExtents() { return extents; }
    public void addExtent(ExtentSubrecord extent) { this.extents.add(extent); }

    public List<String> getSubjectIds() { return subjectIds; }
    public void linkSubject(String subjectId) {
        if (!subjectIds.contains(subjectId)) {
            subjectIds.add(subjectId);
        }
    }

    public List<AgentLink> getAgentLinks() { return agentLinks; }
    public void linkAgent(String agentId, AgentRole role) {
        agentLinks.add(new AgentLink(agentId, role));
    }

    public List<Note> getNotes() { return notes; }
    public void addNote(Note note) { notes.add(note); }

    public List<RightsStatement> getRightsStatements() { return rightsStatements; }
    public void addRightsStatement(RightsStatement stmt) { rightsStatements.add(stmt); }

    public CollectionManagement getCollectionManagement() { return collectionManagement; }
    public void setCollectionManagement(CollectionManagement cm) { this.collectionManagement = cm; }

    public String renderOrderedSections() {
        StringBuilder sb = new StringBuilder();
        sb.append("--- [Dates Section] ---\n");
        for (DateSubrecord d : dates) {
            sb.append("  * ").append(d).append("\n");
        }
        sb.append("--- [Extents Section] ---\n");
        for (ExtentSubrecord e : extents) {
            sb.append("  * ").append(e).append("\n");
        }
        sb.append("--- [Subjects Section] ---\n");
        for (String sId : subjectIds) {
            sb.append("  * Subject ID: ").append(sId).append("\n");
        }
        return sb.toString();
    }
}

import java.util.List;

public class HierarchyManager {

    public static boolean moveUp(List<ResourceComponent> siblings, ResourceComponent target) {
        int index = siblings.indexOf(target);
        if (index > 0) {
            siblings.remove(index);
            siblings.add(index - 1, target);
            return true;
        }
        return false;
    }

    public static boolean moveDown(List<ResourceComponent> siblings, ResourceComponent target) {
        int index = siblings.indexOf(target);
        if (index >= 0 && index < siblings.size() - 1) {
            siblings.remove(index);
            siblings.add(index + 1, target);
            return true;
        }
        return false;
    }

    public static boolean reparent(ResourceComponent target, ResourceComponent newParent, List<ResourceComponent> currentSiblings) {
        if (target == null || target == newParent) {
            return false;
        }
        currentSiblings.remove(target);
        if (newParent != null) {
            newParent.addChild(target);
        }
        return true;
    }

    public static boolean dragAndDrop(List<ResourceComponent> currentSiblings, ResourceComponent item, List<ResourceComponent> targetList, int targetIndex) {
        if (item == null || targetList == null) {
            return false;
        }
        currentSiblings.remove(item);
        if (targetIndex < 0) {
            targetList.add(0, item);
        } else if (targetIndex >= targetList.size()) {
            targetList.add(item);
        } else {
            targetList.add(targetIndex, item);
        }
        return true;
    }

    public static boolean moveDigitalObjectUp(List<DigitalObject> siblings, DigitalObject target) {
        int index = siblings.indexOf(target);
        if (index > 0) {
            siblings.remove(index);
            siblings.add(index - 1, target);
            return true;
        }
        return false;
    }

    public static boolean moveDigitalObjectDown(List<DigitalObject> siblings, DigitalObject target) {
        int index = siblings.indexOf(target);
        if (index >= 0 && index < siblings.size() - 1) {
            siblings.remove(index);
            siblings.add(index + 1, target);
            return true;
        }
        return false;
    }
}

    public void saveResource(User user, Resource resource, int expectedLockVersion) {

        Resource existing = resources.get(resource.getId());
        if (existing != null) {
            if (existing.getLockVersion() != expectedLockVersion) {
                throw new ConcurrencyException("Record has been modified by another user. Current version: "
                    + existing.getLockVersion() + ", your version: " + expectedLockVersion);
            }
            resource.setLockVersion(expectedLockVersion + 1);
        } else {
            resource.setLockVersion(0);
        }
        resources.put(resource.getId(), resource);
    }

        Resource res = resources.get(resourceId);

            resources.remove(resourceId);
        }
    }

        }
        return Optional.empty();
    }

    public List<ArchivalRecord> search(User user, String repoId, String query) {
        checkPermission(user, repoId, Permission.VIEW_RECORD);
        String q = query.toLowerCase();
        List<ArchivalRecord> results = new ArrayList<>();

        for (Resource res : resources.values()) {
            if (res.getRepositoryId().equals(repoId)) {
                if (res.getTitle().toLowerCase().contains(q) || res.getResourceIdentifier().toLowerCase().contains(q)) {
                    results.add(res);
                }
            }
        }
        for (Accession acc : accessions.values()) {
            if (acc.getRepositoryId().equals(repoId)) {
                if (acc.getTitle().toLowerCase().contains(q) || acc.getAccessionNumber().toLowerCase().contains(q)) {
                    results.add(acc);
                }
            }
        }
        for (DigitalObject dobj : digitalObjects.values()) {
            if (dobj.getRepositoryId().equals(repoId)) {
                if (dobj.getTitle().toLowerCase().contains(q) || dobj.getDigitalObjectId().toLowerCase().contains(q)) {
                    results.add(dobj);
                }
            }
        }
        return results;
    }

    private void checkPermission(User user, String repoId, Permission... permissions) {
        if (user.isSystemAdmin()) {
            return;
        }
        var userPerms = user.getPermissions(repoId);
        for (Permission p : permissions) {
            if (userPerms.contains(p)) {
                return;
            }
        }
        throw new SecurityException("User " + user.getUsername() + " does not have required permissions in repository " + repoId);
    }

import java.util.UUID;

public class AccessionCsvImporter implements Importer {

    @Override
    public ImportResult parse(String content, String repositoryId) {
        ImportResult result = new ImportResult();
        if (content == null || content.isBlank()) {
            result.addMessage(ImportMessage.Level.ERROR, "Empty CSV content", "Input file contains no content.");
            return result;
        }

        String[] lines = content.split("\\r?\\n");
        if (lines.length <= 1) {
            result.addMessage(ImportMessage.Level.WARNING, "No data rows in CSV", "CSV only contains header or is empty.");
            return result;
        }

        for (int i = 1; i < lines.length; i++) {
            String line = lines[i].trim();
            if (line.isEmpty()) continue;
            String[] tokens = line.split(",", -1);
            if (tokens.length < 3) {
                result.addMessage(ImportMessage.Level.ERROR, "Row " + (i + 1) + " invalid format", "Expected at least accessionNumber, title, accessionDate");
                continue;
            }
            String accNum = tokens[0].trim();
            String title = tokens[1].trim();
            String accDate = tokens[2].trim();
            String extNumber = tokens.length > 3 ? tokens[3].trim() : "1";
            String extType = tokens.length > 4 ? tokens[4].trim() : "linear_feet";

            Accession acc = new Accession("acc_" + UUID.randomUUID(), repositoryId, title, accNum, accDate);
            acc.addDate(new DateSubrecord("accession", "single", accDate, accDate, null));
            acc.addExtent(new ExtentSubrecord("whole", extNumber, extType, null));
            result.addRecord(acc);
            result.addMessage(ImportMessage.Level.INFO, "Imported accession " + accNum, null);
        }
        return result;
    }
}

import java.util.UUID;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class EadImporter implements Importer {

    @Override
    public ImportResult parse(String content, String repositoryId) {
        ImportResult result = new ImportResult();
        if (content == null || content.isBlank()) {
            result.addMessage(ImportMessage.Level.ERROR, "EAD import failed", "File is empty or null.");
            return result;
        }

        SourceSystem detectedSource = SourceSystem.ARCHIVES_SPACE_NATIVE;
        if (content.contains("generator=\"Archon\"") || content.contains("<archon:recordid>")) {
            detectedSource = SourceSystem.ARCHON;
        } else if (content.contains("generator=\"Archivists' Toolkit\"") || content.contains("xmlns:at=\"http://archiviststoolkit.org\"")) {
            detectedSource = SourceSystem.ARCHIVISTS_TOOLKIT;
        }

        String title = extractTag(content, "unittitle");
        if (title == null) {
            title = extractTag(content, "titleproper");
        }
        if (title == null) {
            result.addMessage(ImportMessage.Level.ERROR, "Missing title in EAD", "Neither <unittitle> nor <titleproper> could be found.");
            return result;
        }

        String unitid = extractTag(content, "unitid");
        if (unitid == null) {
            unitid = "COLL-" + System.currentTimeMillis();
            result.addMessage(ImportMessage.Level.WARNING, "No <unitid> found", "Generated default identifier: " + unitid);
        }

        String sourceRecordId = extractTag(content, "archon:recordid");
        if (sourceRecordId == null) {
            sourceRecordId = extractTag(content, "at:recordid");
        }
        if (sourceRecordId == null) {
            sourceRecordId = "imported-" + unitid;
        }

        Resource res = new Resource("res_" + UUID.randomUUID(), repositoryId, title, unitid, LevelOfDescription.of("collection"));
        res.setSourceSystem(detectedSource);
        res.setSourceRecordId(sourceRecordId);

        String unitdate = extractTag(content, "unitdate");
        if (unitdate != null) {
            res.addDate(new DateSubrecord("creation", "inclusive", unitdate, null, null));
        }

        String extent = extractTag(content, "extent");
        if (extent != null) {
            res.addExtent(new ExtentSubrecord("whole", extent, "items", null));
        }

        String bioghist = extractTag(content, "bioghist");
        if (bioghist != null) {
            res.addNote(new Note("note_" + UUID.randomUUID(), "bioghist", "Biographical / Historical Note", bioghist));
        }

        Matcher compMatcher = Pattern.compile("<c(?=\\s|>)(.*?)>(.*?)</c>", Pattern.DOTALL).matcher(content);
        int cIndex = 1;
        while (compMatcher.find()) {
            String cAttrs = compMatcher.group(1);
            String cBody = compMatcher.group(2);
            String cTitle = extractTag(cBody, "unittitle");
            String cUnitid = extractTag(cBody, "unitid");
            String cLevel = extractAttribute(cAttrs, "level");
            if (cLevel == null) cLevel = "file";
            if (cTitle == null) cTitle = "Component " + cIndex;

            ResourceComponent rc = new ResourceComponent("rc_" + UUID.randomUUID(), cUnitid != null ? cUnitid : "C-" + cIndex, cTitle, LevelOfDescription.of(cLevel));
            res.addChild(rc);
            cIndex++;
        }

        result.addRecord(res);
        result.addMessage(ImportMessage.Level.INFO, "EAD parsed successfully from " + detectedSource, null);
        return result;
    }

    private String extractTag(String text, String tag) {
        Pattern pattern = Pattern.compile("<" + tag + "[^>]*>(.*?)</" + tag + ">", Pattern.DOTALL | Pattern.CASE_INSENSITIVE);
        Matcher matcher = pattern.matcher(text);
        if (matcher.find()) {
            return matcher.group(1).trim();
        }
        return null;
    }

    private String extractAttribute(String attrs, String attrName) {
        Pattern pattern = Pattern.compile(attrName + "=\"([^\"]+)\"", Pattern.CASE_INSENSITIVE);
        Matcher matcher = pattern.matcher(attrs);
        if (matcher.find()) {
            return matcher.group(1).trim();
        }
        return null;
    }

import java.util.UUID;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class MarcXmlImporter implements Importer {
    private final boolean agentsAndSubjectsOnly;

    public MarcXmlImporter(boolean agentsAndSubjectsOnly) {
        this.agentsAndSubjectsOnly = agentsAndSubjectsOnly;
    }

    @Override
    public ImportResult parse(String content, String repositoryId) {
        ImportResult result = new ImportResult();
        if (content == null || content.isBlank()) {
            result.addMessage(ImportMessage.Level.ERROR, "MARCXML is empty", "Invalid XML content");
            return result;
        }

        Matcher agent100Matcher = Pattern.compile("<datafield[^>]*tag=\"100\"[^>]*>(.*?)</datafield>", Pattern.DOTALL).matcher(content);
        while (agent100Matcher.find()) {
            String fieldContent = agent100Matcher.group(1);
            String name = extractSubfield(fieldContent, "a");
            if (name != null) {
                Agent agent = new Agent("agent_" + UUID.randomUUID(), AgentType.PERSON, name, null, "naf");
                result.addAgent(agent);
                result.addMessage(ImportMessage.Level.INFO, "Imported Agent: " + name, null);
            }
        }

        Matcher subject650Matcher = Pattern.compile("<datafield[^>]*tag=\"650\"[^>]*>(.*?)</datafield>", Pattern.DOTALL).matcher(content);
        while (subject650Matcher.find()) {
            String fieldContent = subject650Matcher.group(1);
            String term = extractSubfield(fieldContent, "a");
            String stdId = extractSubfield(fieldContent, "0");
            if (term != null) {
                Subject subj = new Subject("subj_" + UUID.randomUUID(), repositoryId, term, stdId != null ? stdId : "lcsh/" + term.hashCode(), "lcsh");
                result.addSubject(subj);
                result.addMessage(ImportMessage.Level.INFO, "Imported Subject: " + term, null);
            }
        }

        if (!agentsAndSubjectsOnly) {
            String title = null;
            Matcher title245 = Pattern.compile("<datafield[^>]*tag=\"245\"[^>]*>(.*?)</datafield>", Pattern.DOTALL).matcher(content);
            if (title245.find()) {
                title = extractSubfield(title245.group(1), "a");
            }
            if (title == null) {
                result.addMessage(ImportMessage.Level.ERROR, "Resource title missing in MARCXML 245 field", "Field 245$a not found");
                return result;
            }

            Resource res = new Resource("res_" + UUID.randomUUID(), repositoryId, title, "MARC-" + System.currentTimeMillis(), LevelOfDescription.of("collection"));
            for (Agent a : result.getAgents()) {
                res.linkAgent(a.getId(), AgentRole.CREATOR);
            }
            for (Subject s : result.getSubjects()) {
                res.linkSubject(s.getId());
            }
            result.addRecord(res);
        }

        return result;
    }

    private String extractSubfield(String datafield, String code) {
        Pattern pattern = Pattern.compile("<subfield[^>]*code=\"" + code + "\">(.*?)</subfield>", Pattern.DOTALL);
        Matcher matcher = pattern.matcher(datafield);
        if (matcher.find()) {
            return matcher.group(1).trim();
        }
        return null;
    }
}

// File: EacCpfImporter.java
import java.util.UUID;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

public class EacCpfImporter implements Importer {

    @Override
    public ImportResult parse(String content, String repositoryId) {
        ImportResult result = new ImportResult();
        if (content == null || content.isBlank()) {
            result.addMessage(ImportMessage.Level.ERROR, "EAC-CPF empty", "No content");
            return result;
        }

        AgentType agentType = AgentType.PERSON;
        if (content.contains("<entityType>corporateBody</entityType>")) {
            agentType = AgentType.CORPORATE_ENTITY;
        } else if (content.contains("<entityType>family</entityType>")) {
            agentType = AgentType.FAMILY;
        }

        String primaryName = null;
        Pattern nameEntry = Pattern.compile("<nameEntry[^>]*>(.*?)</nameEntry>", Pattern.DOTALL);
        Matcher m = nameEntry.matcher(content);
        if (m.find()) {
            String entry = m.group(1);
            Pattern part = Pattern.compile("<part>(.*?)</part>");
            Matcher pm = part.matcher(entry);
            if (pm.find()) {
                primaryName = pm.group(1).trim();
            }
        }

        if (primaryName == null) {
            result.addMessage(ImportMessage.Level.ERROR, "Missing <part> in <nameEntry>", "Cannot extract agent name from EAC-CPF");
            return result;
        }

        Agent agent = new Agent("agent_" + UUID.randomUUID(), agentType, primaryName, null, "viaf");
        result.addAgent(agent);
        result.addMessage(ImportMessage.Level.INFO, "Imported EAC-CPF Agent: " + primaryName, null);
        return result;
    }
}

import java.util.List;

public class EadExporter {

    public String export(Resource resource) {
        StringBuilder sb = new StringBuilder();
        sb.append("<?xml version=\"1.0\" encoding=\"utf-8\"?>\n");
        sb.append("<ead>\n");
        sb.append("  <eadheader>\n");
        sb.append("    <eadid>").append(resource.getResourceIdentifier()).append("</eadid>\n");
        sb.append("    <filedesc>\n");
        sb.append("      <titlestmt>\n");
        sb.append("        <titleproper>").append(resource.getTitle()).append("</titleproper>\n");
        if (resource.getFindingAidData().getFindingAidAuthor() != null) {
            sb.append("        <author>").append(resource.getFindingAidData().getFindingAidAuthor()).append("</author>\n");
        }
        sb.append("      </titlestmt>\n");
        sb.append("    </filedesc>\n");
        sb.append("  </eadheader>\n");
        sb.append("  <archdesc level=\"").append(resource.getLevel().getLevel()).append("\">\n");
        sb.append("    <did>\n");
        sb.append("      <unittitle>").append(resource.getTitle()).append("</unittitle>\n");
        sb.append("      <unitid>").append(resource.getResourceIdentifier()).append("</unitid>\n");
        for (DateSubrecord d : resource.getDates()) {
            sb.append("      <unitdate>").append(d.getExpression()).append("</unitdate>\n");
        }
        for (ExtentSubrecord e : resource.getExtents()) {
            sb.append("      <extent>").append(e.getNumber()).append(" ").append(e.getExtentType()).append("</extent>\n");
        }
        sb.append("    </did>\n");

        for (Note note : resource.getNotes()) {
            sb.append("    <note type=\"").append(note.getType()).append("\">\n");
            sb.append("      <p>").append(note.getContent()).append("</p>\n");
            sb.append("    </note>\n");
        }

        if (!resource.getChildren().isEmpty()) {
            sb.append("    <dsc>\n");
            for (ResourceComponent rc : resource.getChildren()) {
                exportComponent(rc, sb, 6);
            }
            sb.append("    </dsc>\n");
        }

        sb.append("  </archdesc>\n");
        sb.append("</ead>\n");
        return sb.toString();
    }

    private void exportComponent(ResourceComponent comp, StringBuilder sb, int indentSpaces) {
        String indent = " ".repeat(indentSpaces);
        sb.append(indent).append("<c level=\"").append(comp.getLevel().getLevel()).append("\">\n");
        sb.append(indent).append("  <did>\n");
        sb.append(indent).append("    <unittitle>").append(comp.getTitle()).append("</unittitle>\n");
        if (comp.getComponentId() != null) {
            sb.append(indent).append("    <unitid>").append(comp.getComponentId()).append("</unitid>\n");
        }
        sb.append(indent).append("  </did>\n");
        for (ResourceComponent child : comp.getChildren()) {
            exportComponent(child, sb, indentSpaces + 2);
        }
        sb.append(indent).append("</c>\n");
    }
}

        sb.append("  <control>\n");

        sb.append("  </control>\n");
        sb.append("  <cpfDescription>\n");
        sb.append("    <identity>\n");
        String typeTag = switch (agent.getAgentType()) {
            case PERSON -> "person";
            case CORPORATE_ENTITY -> "corporateBody";
            case FAMILY -> "family";
        };
        sb.append("      <entityType>").append(typeTag).append("</entityType>\n");
        for (NameForm nf : agent.getNameForms()) {
            sb.append("      <nameEntry>\n");
            sb.append("        <part>").append(nf.getFullDisplay()).append("</part>\n");
            if (nf.getSource() != null) {
                sb.append("        <authorizedForm>").append(nf.getSource()).append("</authorizedForm>\n");
            }
            sb.append("      </nameEntry>\n");
        }
        sb.append("    </identity>\n");
        sb.append("  </cpfDescription>\n");
        sb.append("</eac-cpf>\n");
        return sb.toString();
    }
