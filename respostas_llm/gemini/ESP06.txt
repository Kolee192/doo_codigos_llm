import java.util.Objects;

public class Card implements Comparable<Card> {
    private final double value;
    private final String label;

    public Card(double value, String label) {
        this.value = value;
        this.label = label;
    }

    public Card(double value) {
        this(value, formatValue(value));
    }

    private static String formatValue(double val) {
        if (val == (long) val) {
            return String.format("%d", (long) val);
        } else {
            return String.format("%s", val);
        }
    }

    public double getValue() {
        return value;
    }

    public String getLabel() {
        return label;
    }

    @Override
    public int compareTo(Card other) {
        return Double.compare(this.value, other.value);
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Card card)) return false;
        return Double.compare(card.value, value) == 0;
    }

    @Override
    public int hashCode() {
        return Objects.hash(value);
    }

    @Override
    public String toString() {
        return label;
    }
}

import java.time.Duration;
import java.time.Instant;

public class CountdownTimer {
    private static final long DEFAULT_DURATION_SECONDS = 120; // 2 minutes
    private Instant startedAt;
    private boolean isRunning;

    public CountdownTimer() {
        this.isRunning = false;
    }

    public synchronized void start() {
        this.startedAt = Instant.now();
        this.isRunning = true;
    }

    public synchronized void reset() {
        this.startedAt = null;
        this.isRunning = false;
    }

    public synchronized boolean isRunning() {
        if (!isRunning) return false;
        if (getRemainingSeconds() <= 0) {
            isRunning = false;
            return false;
        }
        return true;
    }

    public synchronized long getRemainingSeconds() {
        if (!isRunning || startedAt == null) {
            return DEFAULT_DURATION_SECONDS;
        }
        long elapsed = Duration.between(startedAt, Instant.now()).toSeconds();
        long remaining = DEFAULT_DURATION_SECONDS - elapsed;
        return Math.max(0, remaining);
    }

    public synchronized String getFormattedRemaining() {
        long sec = getRemainingSeconds();
        long minutes = sec / 60;
        long seconds = sec % 60;
        return String.format("%02d:%02d", minutes, seconds);
    }
}

import java.time.Instant;
import java.util.*;

public class Round {
    private final int roundNumber;
    private final Map<String, EstimateVote> votes;
    private RoundStatus status;
    private Double acceptedEstimate;
    private final Instant startedAt;
    private Instant endedAt;

    public Round(int roundNumber) {
        this.roundNumber = roundNumber;
        this.votes = new LinkedHashMap<>();
        this.status = RoundStatus.VOTING;
        this.startedAt = Instant.now();
    }

    public int getRoundNumber() {
        return roundNumber;
    }

    public synchronized void recordVote(String estimatorId, Card card) {
        if (status != RoundStatus.VOTING) {
            throw new IllegalStateException("Voting has concluded for this round.");
        }
        EstimateVote vote = votes.get(estimatorId);
        if (vote != null) {
            vote.setCard(card);
        } else {
            votes.put(estimatorId, new EstimateVote(estimatorId, card));
        }
    }

    public synchronized boolean hasVoted(String estimatorId) {
        return votes.containsKey(estimatorId);
    }

    public synchronized Card getVote(String estimatorId) {
        EstimateVote vote = votes.get(estimatorId);
        return vote != null ? vote.getCard() : null;
    }

    public synchronized Map<String, EstimateVote> getVotes() {
        return Collections.unmodifiableMap(new LinkedHashMap<>(votes));
    }

    public synchronized int getVoteCount() {
        return votes.size();
    }

    public synchronized void reveal() {
        if (this.status == RoundStatus.VOTING) {
            this.status = RoundStatus.REVEALED;
        }
    }

    public synchronized boolean isRevealed() {
        return this.status == RoundStatus.REVEALED || this.status == RoundStatus.ACCEPTED;
    }

    public synchronized void acceptEstimate(double estimate) {
        this.acceptedEstimate = estimate;
        this.status = RoundStatus.ACCEPTED;
        this.endedAt = Instant.now();
    }

    public synchronized RoundStatus getStatus() {
        return status;
    }

    public synchronized Double getAcceptedEstimate() {
        return acceptedEstimate;
    }

    public synchronized OptionalDouble calculateAverage() {
        if (votes.isEmpty()) return OptionalDouble.empty();
        return votes.values().stream()
                .mapToDouble(v -> v.getCard().getValue())
                .average();
    }

    public synchronized boolean isUnanimous() {
        if (votes.isEmpty()) return false;
        double first = votes.values().iterator().next().getCard().getValue();
        for (EstimateVote vote : votes.values()) {
            if (Double.compare(vote.getCard().getValue(), first) != 0) {
                return false;
            }
        }
        return true;
    }

    public synchronized Optional<Double> getUnanimousValue() {
        if (isUnanimous()) {
            return Optional.of(votes.values().iterator().next().getCard().getValue());
        }
        return Optional.empty();
    }

    public Instant getStartedAt() {
        return startedAt;
    }

    public Instant getEndedAt() {
        return endedAt;
    }
}

import java.time.Instant;
import java.util.*;

public class Game {
    public static final int MAX_ESTIMATORS = 15;

    private final String id;
    private final String moderatorUsername;
    private String name;
    private String description;
    private final DeckType deckType;
    private final List<Estimator> estimators;
    private final List<Story> stories;
    private Story currentStory;
    private final CountdownTimer timer;
    private final Instant createdAt;

    public Game(String id, String moderatorUsername, String name, String description, DeckType deckType) {
        this.id = id;
        this.moderatorUsername = moderatorUsername;
        this.name = name;
        this.description = description;
        this.deckType = deckType != null ? deckType : DeckType.MODIFIED_FIBONACCI;
        this.estimators = new ArrayList<>();
        this.stories = new ArrayList<>();
        this.timer = new CountdownTimer();
        this.createdAt = Instant.now();
    }

    public String getId() {
        return id;
    }

    public String getModeratorUsername() {
        return moderatorUsername;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getDescription() {
        return description;
    }

    public void setDescription(String description) {
        this.description = description;
    }

    public DeckType getDeckType() {
        return deckType;
    }

    public Instant getCreatedAt() {
        return createdAt;
    }

    public CountdownTimer getTimer() {
        return timer;
    }

    public String getInvitationUrl(String baseUrl) {
        return baseUrl + "/games/" + id + "/join";
    }

    public synchronized Estimator join(String estimatorId, String name, String avatarUrl) {
        if (estimators.size() >= MAX_ESTIMATORS) {
            throw new IllegalStateException("Game is full. Maximum " + MAX_ESTIMATORS + " estimators allowed.");
        }
        for (Estimator e : estimators) {
            if (e.getId().equals(estimatorId)) {
                return e;
            }
        }
        char codeLetter = (char) ('A' + estimators.size());
        String anonCode = "Estimator " + codeLetter;
        Estimator estimator = new Estimator(estimatorId, name, avatarUrl, anonCode);
        estimators.add(estimator);
        return estimator;
    }

    public synchronized List<Estimator> getEstimators() {
        return Collections.unmodifiableList(new ArrayList<>(estimators));
    }

    public synchronized void addStory(Story story) {
        stories.add(story);
    }

    public synchronized boolean editStory(String storyId, String newTitle, String newDescription) {
        for (Story s : stories) {
            if (s.getId().equals(storyId)) {
                s.setTitle(newTitle);
                s.setDescription(newDescription);
                return true;
            }
        }
        return false;
    }

    public synchronized boolean deleteStory(String storyId) {
        if (currentStory != null && currentStory.getId().equals(storyId)) {
            currentStory = null;
        }
        return stories.removeIf(s -> s.getId().equals(storyId));
    }

    public synchronized List<Story> getStories() {
        return Collections.unmodifiableList(new ArrayList<>(stories));
    }

    public synchronized Story getStory(String storyId) {
        return stories.stream().filter(s -> s.getId().equals(storyId)).findFirst().orElse(null);
    }

    public synchronized Story selectStoryToEstimate(String storyId) {
        Story story = getStory(storyId);
        if (story == null) {
            throw new IllegalArgumentException("Story not found: " + storyId);
        }
        this.currentStory = story;
        story.startNewRound();
        this.timer.reset();
        return story;
    }

    public synchronized Story getCurrentStory() {
        return currentStory;
    }

    public synchronized boolean submitEstimate(String estimatorId, Card card) {
        if (currentStory == null) {
            throw new IllegalStateException("No story is currently selected for estimation.");
        }
        Round round = currentStory.getCurrentRound();
        if (round == null || round.isRevealed()) {
            throw new IllegalStateException("Current round is not active for voting.");
        }

        round.recordVote(estimatorId, card);

        // Check if all joined estimators have now voted
        if (!estimators.isEmpty() && round.getVoteCount() >= estimators.size()) {
            round.reveal();
            timer.reset(); // Two-minute timer resets itself as soon as all play an estimate
            return true;
        }
        return false;
    }

    public synchronized void forceShowEstimates() {
        if (currentStory != null && currentStory.getCurrentRound() != null) {
            currentStory.getCurrentRound().reveal();
        }
    }

    public synchronized void acceptAgreedEstimate(double estimate) {
        if (currentStory != null && currentStory.getCurrentRound() != null) {
            Round round = currentStory.getCurrentRound();
            round.acceptEstimate(estimate);
            currentStory.setFinalEstimate(estimate);
        }
    }

    public synchronized void acceptAverageEstimate() {
        if (currentStory != null && currentStory.getCurrentRound() != null) {
            Round round = currentStory.getCurrentRound();
            OptionalDouble avg = round.calculateAverage();
            if (avg.isPresent()) {
                acceptAgreedEstimate(avg.getAsDouble());
            } else {
                throw new IllegalStateException("No votes to average.");
            }
        }
    }

    public synchronized boolean hasAutoSuggestedEstimate() {
        if (currentStory != null && currentStory.getCurrentRound() != null) {
            Round round = currentStory.getCurrentRound();
            return round.isRevealed() && round.isUnanimous();
        }
        return false;
    }

    public synchronized Optional<Double> getAutoSuggestedEstimate() {
        if (hasAutoSuggestedEstimate()) {
            return currentStory.getCurrentRound().getUnanimousValue();
        }
        return Optional.empty();
    }
}

import java.util.*;
import java.util.concurrent.ConcurrentHashMap;

/**
 * Simulates indexed repository for optimal database performance and fast query lookups.
 */
public class DatabaseIndexRepository {
    // Primary indexes
    private final Map<String, UserAccount> usersById = new ConcurrentHashMap<>();
    private final Map<String, Game> gamesById = new ConcurrentHashMap<>();

    // Secondary / Foreign-Key indexes
    private final Map<String, UserAccount> usersByUsername = new ConcurrentHashMap<>();
    private final Map<String, UserAccount> usersByEmail = new ConcurrentHashMap<>();
    private final Map<String, List<String>> gameIdsByModerator = new ConcurrentHashMap<>();

    // User indexing operations
    public void indexUser(UserAccount account) {
        usersById.put(account.getId(), account);
        usersByUsername.put(account.getUsername().toLowerCase(), account);
        usersByEmail.put(account.getEmail().toLowerCase(), account);
    }

    public void removeUser(String userId) {
        UserAccount account = usersById.remove(userId);
        if (account != null) {
            usersByUsername.remove(account.getUsername().toLowerCase());
            usersByEmail.remove(account.getEmail().toLowerCase());
        }
    }

    public UserAccount findUserById(String id) {
        return usersById.get(id);
    }

    public UserAccount findUserByUsername(String username) {
        return usersByUsername.get(username.toLowerCase());
    }

    public UserAccount findUserByEmail(String email) {
        return usersByEmail.get(email.toLowerCase());
    }

    // Game indexing operations
    public void indexGame(Game game) {
        gamesById.put(game.getId(), game);
        gameIdsByModerator.computeIfAbsent(game.getModeratorUsername().toLowerCase(), k -> new ArrayList<>())
                .add(game.getId());
    }

    public void removeGame(String gameId) {
        Game game = gamesById.remove(gameId);
        if (game != null) {
            List<String> list = gameIdsByModerator.get(game.getModeratorUsername().toLowerCase());
            if (list != null) {
                list.remove(gameId);
            }
        }
    }

    public Game findGameById(String gameId) {
        return gamesById.get(gameId);
    }

    public List<Game> findGamesByModerator(String username) {
        List<String> ids = gameIdsByModerator.getOrDefault(username.toLowerCase(), Collections.emptyList());
        List<Game> result = new ArrayList<>();
        for (String id : ids) {
            Game g = gamesById.get(id);
            if (g != null) {
                result.add(g);
            }
        }
        return result;
    }
}

import java.util.List;
import java.util.Map;

public class TranscriptExporter {

    public static String toCsv(Game game) {
        StringBuilder sb = new StringBuilder();
        // UTF-8 CSV header
        sb.append("Story ID,Story Title,Story Description,Round,Estimator Name,Estimate,Final Estimate\n");
        for (Story story : game.getStories()) {
            String finalEst = story.getFinalEstimate() != null ? String.valueOf(story.getFinalEstimate()) : "N/A";
            List<Round> rounds = story.getRounds();
            if (rounds.isEmpty()) {
                sb.append(escapeCsv(story.getId())).append(",")
                        .append(escapeCsv(story.getTitle())).append(",")
                        .append(escapeCsv(story.getDescription())).append(",")
                        .append("0,,,").append(finalEst).append("\n");
            } else {
                for (Round round : rounds) {
                    Map<String, EstimateVote> votes = round.getVotes();
                    for (Estimator estimator : game.getEstimators()) {
                        EstimateVote vote = votes.get(estimator.getId());
                        String voteVal = (vote != null) ? vote.getCard().getLabel() : "No vote";
                        sb.append(escapeCsv(story.getId())).append(",")
                                .append(escapeCsv(story.getTitle())).append(",")
                                .append(escapeCsv(story.getDescription())).append(",")
                                .append(round.getRoundNumber()).append(",")
                                .append(escapeCsv(estimator.getName())).append(",")
                                .append(escapeCsv(voteVal)).append(",")
                                .append(finalEst).append("\n");
                    }
                }
            }
        }
        return sb.toString();
    }

    public static String toAnonymizedCsv(Game game) {
        StringBuilder sb = new StringBuilder();
        sb.append("Story ID,Round,Anonymous Estimator,Vote Value,Final Estimate\n");
        for (Story story : game.getStories()) {
            String finalEst = story.getFinalEstimate() != null ? String.valueOf(story.getFinalEstimate()) : "N/A";
            for (Round round : story.getRounds()) {
                for (Estimator estimator : game.getEstimators()) {
                    EstimateVote vote = round.getVotes().get(estimator.getId());
                    String voteVal = (vote != null) ? String.valueOf(vote.getCard().getValue()) : "none";
                    sb.append(escapeCsv(story.getId())).append(",")
                            .append(round.getRoundNumber()).append(",")
                            .append(escapeCsv(estimator.getAnonymousCode())).append(",")
                            .append(escapeCsv(voteVal)).append(",")
                            .append(finalEst).append("\n");
                }
            }
        }
        return sb.toString();
    }

        }

        sb.append("</body>\n</html>");

import java.time.ZoneId;
import java.util.*;
import java.util.concurrent.atomic.AtomicInteger;

/**
 * Facade coordinating planning poker operations, persistence, metrics, and security.
 */
public class PlanningPokerSystem {
    private final String baseUrl;
    private final DatabaseIndexRepository repository;
    private final SystemMetrics metrics;
    private final AtomicInteger idCounter = new AtomicInteger(100);

    public PlanningPokerSystem(String baseUrl) {
        this.baseUrl = baseUrl.endsWith("/") ? baseUrl.substring(0, baseUrl.length() - 1) : baseUrl;
        this.repository = new DatabaseIndexRepository();
        this.metrics = new SystemMetrics();
    }

    // Account Management
    public UserAccount createAccount(String username, String password, String fullName, String email, ZoneId timezone) {
        if (repository.findUserByUsername(username) != null) {
            throw new IllegalArgumentException("Username already exists: " + username);
        }
        if (repository.findUserByEmail(email) != null) {
            throw new IllegalArgumentException("Email already registered: " + email);
        }
        String id = "usr_" + idCounter.incrementAndGet();
        UserAccount account = new UserAccount(id, username, password, fullName, email, timezone);
        repository.indexUser(account);
        metrics.recordAccountCreated();
        return account;
    }

    public UserAccount login(String username, String password) {
        UserAccount account = repository.findUserByUsername(username);
        if (account == null || !account.getPassword().equals(password)) {
            throw new SecurityException("Invalid username or password.");
        }
        return account;
    }

    public void updateAccountDetails(String userId, String fullName, String email, ZoneId timezone) {
        UserAccount account = repository.findUserById(userId);
        if (account == null) {
            throw new NoSuchElementException("User not found: " + userId);
        }
        account.setFullName(fullName);
        account.setEmail(email);
        account.setTimezone(timezone);
        repository.indexUser(account);
    }

    public void deleteAccount(String userId) {
        UserAccount account = repository.findUserById(userId);
        if (account != null) {
            // Delete all games owned by this moderator
            List<Game> games = repository.findGamesByModerator(account.getUsername());
            for (Game g : games) {
                repository.removeGame(g.getId());
            }
            repository.removeUser(userId);
        }
    }

    public String sendPasswordReminder(String email) {
        UserAccount account = repository.findUserByEmail(email);
        if (account == null) {
            throw new NoSuchElementException("No account found for email: " + email);
        }
        // Generates reminder email notification content
        return "To: " + account.getEmail() + "\n" +
                "Subject: Password Reminder for Planning Poker\n\n" +
                "Hello " + account.getFullName() + ",\n" +
                "Your password for username '" + account.getUsername() + "' is: " + account.getPassword() + "\n" +
                "Please log in at: " + baseUrl + "/login\n";
    }

    // Game Management
    public Game createGame(String moderatorUsername, String name, String description, DeckType deckType) {
        UserAccount mod = repository.findUserByUsername(moderatorUsername);
        if (mod == null) {
            throw new IllegalArgumentException("Moderator user does not exist: " + moderatorUsername);
        }
        String id = "game_" + idCounter.incrementAndGet();
        Game game = new Game(id, moderatorUsername, name, description, deckType);
        repository.indexGame(game);
        metrics.recordGameCreated();
        return game;
    }

    public List<Game> browsePreviousGames(String moderatorUsername) {
        return repository.findGamesByModerator(moderatorUsername);
    }

    public void deleteGame(String gameId) {
        repository.removeGame(gameId);
    }

    public Game getGame(String gameId) {
        return repository.findGameById(gameId);
    }

    public String getJoinUrl(String gameId) {
        return baseUrl + "/games/" + gameId + "/join";
    }

    // Estimator actions
    public Estimator joinGame(String gameId, String estimatorName, String avatarUrl) {
        Game game = repository.findGameById(gameId);
        if (game == null) {
            throw new NoSuchElementException("Game not found: " + gameId);
        }
        String estimatorId = "est_" + idCounter.incrementAndGet();
        Estimator estimator = game.join(estimatorId, estimatorName, avatarUrl);
        metrics.recordEstimatorJoined();
        return estimator;
    }

    // Story Management
    public Story addStory(String gameId, String title, String description) {
        Game game = repository.findGameById(gameId);
        if (game == null) throw new NoSuchElementException("Game not found");
        String storyId = "story_" + idCounter.incrementAndGet();
        Story story = new Story(storyId, title, description);
        game.addStory(story);
        metrics.recordStoryCreated();
        return story;
    }

    public void importStoriesSpreadsheet(String gameId, String spreadsheetText) {
        if (spreadsheetText == null || spreadsheetText.isBlank()) return;
        String[] lines = spreadsheetText.split("\\r?\\n");
        for (String line : lines) {
            if (line.trim().isEmpty()) continue;
            // Support tab or comma-separated: Title [delimiter] Description
            String[] parts = line.contains("\t") ? line.split("\t", 2) : line.split(",", 2);
            String title = parts[0].trim();
            String desc = parts.length > 1 ? parts[1].trim() : "";
            addStory(gameId, title, desc);
        }
    }

    public Story selectItemForEstimation(String gameId, String storyId) {
        Game game = repository.findGameById(gameId);
        if (game == null) throw new NoSuchElementException("Game not found");
        Story story = game.selectStoryToEstimate(storyId);
        metrics.recordRoundPlayed();
        return story;
    }

    public void castEstimate(String gameId, String estimatorId, Card card) {
        Game game = repository.findGameById(gameId);
        if (game == null) throw new NoSuchElementException("Game not found");
        game.submitEstimate(estimatorId, card);
        metrics.recordVoteCast();
    }

    public SystemMetrics getMetrics() {
        return metrics;
    }

    public DatabaseIndexRepository getRepository() {
        return repository;
    }

    public String getBaseUrl() {
        return baseUrl;
    }
}
