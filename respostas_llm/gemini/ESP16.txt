        this.price = price;
    }

    public void setName(String name) {
        this.name = name;
    }

    }

    public double getPrice() {
        return price;
    }

    public void setPrice(double price) {
        this.price = price;
    }

    public boolean equals(Object o) {
        if (this == o) return true;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class EmailService {
    private final List<Email> sentEmails = new ArrayList<>();

    public void sendEmail(String recipient, String subject, String body) {
        Email email = new Email(recipient, subject, body);
        sentEmails.add(email);
    }

    public List<Email> getSentEmails() {
        return Collections.unmodifiableList(sentEmails);
    }

    public Email getLatestEmailFor(String recipient) {
        for (int i = sentEmails.size() - 1; i >= 0; i--) {
            if (sentEmails.get(i).getRecipient().equalsIgnoreCase(recipient)) {
                return sentEmails.get(i);
            }
        }
        return null;
    }
}

    }

        String shortDesc = restaurant.getDescription();
        if (shortDesc.length() > 60) {
            shortDesc = shortDesc.substring(0, 57) + "...";
        }
        String twoLineDesc = shortDesc + "\nFreshly prepared daily.";

        return "--------------------------------------------------\n" +
                "Name: " + restaurant.getName() + "\n" +
                "Telephone: " + restaurant.getPhone() + "\n" +
                "Food Type: " + restaurant.getRestaurantType() + "\n" +

                "Description:\n" + twoLineDesc + "\n" +
                "Website: " + restaurant.getWebsiteUrl() + "\n" +
                "Info Link: [Select Information Link]\n" +
                "--------------------------------------------------";

        return "Pin at (" + String.format("%.4f", restaurant.getLatitude()) + ", " +
                String.format("%.4f", restaurant.getLongitude()) + ") -> " +
                restaurant.getName() + " [Info Link: Select Pin]";
    }

    public boolean equals(Object o) {

        if (!(o instanceof RestaurantSearchResult that)) return false;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

public class SearchResultView {
    private ResultViewType viewType;
    private final List<RestaurantSearchResult> baseResults;
    private List<RestaurantSearchResult> filteredResults;
    private final int defaultZoomLevel = 12;
    private SortOption currentSortOption;
    private SortOrder currentSortOrder;

    public SearchResultView(ResultViewType viewType, List<RestaurantSearchResult> initialResults, boolean searchByPriceDefault) {
        this.viewType = viewType;
        this.baseResults = new ArrayList<>(initialResults.stream().limit(100).toList());
        this.filteredResults = new ArrayList<>(this.baseResults);
        this.currentSortOrder = SortOrder.ASCENDING;
        if (searchByPriceDefault) {
            applyPriceSearchDefaultSorting();
        } else {
            applyOtherSearchDefaultSorting();
        }
    }

    public ResultViewType getViewType() {
        return viewType;
    }

    public void setViewType(ResultViewType viewType) {
        this.viewType = viewType;
    }

    public int getDefaultZoomLevel() {
        return defaultZoomLevel;
    }

    public List<RestaurantSearchResult> getDisplayedResults() {
        return filteredResults;
    }

    public int getResultCount() {
        return filteredResults.size();
    }

    public void applyFilter(FilterCriteria filter) {
        filteredResults = new ArrayList<>(baseResults.stream()
                .filter(res -> filter.getMaxDistanceKm() == null || res.getDistanceKm() <= filter.getMaxDistanceKm())
                .filter(res -> filter.getMaxPrice() == null || res.getRestaurant().getAveragePrice() <= filter.getMaxPrice())
                .filter(res -> filter.getRestaurantType() == null || filter.getRestaurantType().isBlank() ||
                        res.getRestaurant().getRestaurantType().equalsIgnoreCase(filter.getRestaurantType()))
                .filter(res -> filter.getSpecificDish() == null || filter.getSpecificDish().isBlank() ||
                        res.getRestaurant().getMenu().stream().anyMatch(d -> d.getName().equalsIgnoreCase(filter.getSpecificDish())))
                .limit(100)
                .toList());

        if (currentSortOption != null) {
            applySorting(currentSortOption, currentSortOrder);
        }
    }

    public void applyPriceSearchDefaultSorting() {
        currentSortOption = SortOption.PRICE;
        currentSortOrder = SortOrder.ASCENDING;
        filteredResults.sort(
                Comparator.comparingDouble((RestaurantSearchResult r) -> r.getRestaurant().getAveragePrice())
                        .thenComparingDouble(RestaurantSearchResult::getDistanceKm)
                        .thenComparing(r -> r.getRestaurant().getRestaurantType(), String.CASE_INSENSITIVE_ORDER)
                        .thenComparing(RestaurantSearchResult::getMatchedDishName, String.CASE_INSENSITIVE_ORDER)
        );
    }

    public void applyOtherSearchDefaultSorting() {
        currentSortOption = SortOption.DISTANCE;
        currentSortOrder = SortOrder.ASCENDING;
        filteredResults.sort(
                Comparator.comparingDouble(RestaurantSearchResult::getDistanceKm)
                        .thenComparingDouble((RestaurantSearchResult r) -> r.getRestaurant().getAveragePrice())
                        .thenComparing(r -> r.getRestaurant().getRestaurantType(), String.CASE_INSENSITIVE_ORDER)
                        .thenComparing(RestaurantSearchResult::getMatchedDishName, String.CASE_INSENSITIVE_ORDER)
        );
    }

    public void toggleSort(SortOption option) {
        if (currentSortOption == option) {
            currentSortOrder = (currentSortOrder == SortOrder.ASCENDING) ? SortOrder.DESCENDING : SortOrder.ASCENDING;
        } else {
            currentSortOption = option;
            currentSortOrder = SortOrder.ASCENDING;
        }
        applySorting(currentSortOption, currentSortOrder);
    }

    private void applySorting(SortOption option, SortOrder order) {
        Comparator<RestaurantSearchResult> comparator = switch (option) {
            case PRICE -> Comparator.comparingDouble(r -> r.getRestaurant().getAveragePrice());
            case DISTANCE -> Comparator.comparingDouble(RestaurantSearchResult::getDistanceKm);
            case RESTAURANT_TYPE -> Comparator.comparing(r -> r.getRestaurant().getRestaurantType(), String.CASE_INSENSITIVE_ORDER);
            case SPECIFIC_DISH -> Comparator.comparing(RestaurantSearchResult::getMatchedDishName, String.CASE_INSENSITIVE_ORDER);
            case RESTAURANT_NAME -> Comparator.comparing(r -> r.getRestaurant().getName(), String.CASE_INSENSITIVE_ORDER);
        };

        if (order == SortOrder.DESCENDING) {
            comparator = comparator.reversed();
        }
        filteredResults.sort(comparator);
    }

    public String renderFilteringMenu() {
        return "=== FILTERING MENU ===\n" +
                "1. Increase/Decrease Max Distance\n" +
                "2. Increase/Decrease Max Price\n" +
                "3. Choose Restaurant Type\n" +
                "4. Choose Specific Dish\n" +
                "======================";
    }

    public String renderSortingHeader() {
        return "=== SORT OPTIONS === [PRICE] [DISTANCE] [RESTAURANT_TYPE] [SPECIFIC_DISH] [RESTAURANT_NAME] (Current: "
                + currentSortOption + " " + currentSortOrder + ")";
    }

    public String render() {
        StringBuilder sb = new StringBuilder();
        if (viewType == ResultViewType.MAP) {
            sb.append("=== MAP VIEW (Zoom: ").append(defaultZoomLevel).append("x) ===\n");
            sb.append("[Filter Button: Filter Results]\n");
            sb.append("Showing closest & relevant restaurants (Max 100):\n");
            for (RestaurantSearchResult r : filteredResults) {
                sb.append(r.toMapPinView()).append("\n");
            }
        } else {
            sb.append("=== LIST VIEW ===\n");
            sb.append(renderSortingHeader()).append("\n");
            sb.append("[Filter Button: Filter Results]\n");
            sb.append("Showing restaurants (Max 100):\n");
            for (RestaurantSearchResult r : filteredResults) {
                sb.append(r.toListElementView()).append("\n");
            }
        }
        return sb.toString();
    }
}

    private final Map<String, RestaurantOwner> owners = new HashMap<>();
    private final Map<String, Administrator> admins = new HashMap<>();
    private final Map<String, Restaurant> restaurants = new HashMap<>();
    private final Map<String, String> ownerRestaurantMap = new HashMap<>(); // ownerUsername -> restaurantId
    private final List<String> restaurantTypes = new ArrayList<>();
    private final List<Dish> dishCatalog = new ArrayList<>();

    public SystemDatabase() {
        // Default restaurant types
        restaurantTypes.add("Italian");
        restaurantTypes.add("Japanese");
        restaurantTypes.add("Swedish");
        restaurantTypes.add("Mexican");
        restaurantTypes.add("Fast Food");

        // Default admin
        admins.put("admin", new Administrator("admin", "admin123"));
    }

    public Map<String, User> getUsers() {
        return users;
    }

    public Map<String, RestaurantOwner> getOwners() {
        return owners;
    }

    public Map<String, Administrator> getAdmins() {
        return admins;
    }

    public Map<String, Restaurant> getRestaurants() {
        return restaurants;
    }

    public List<String> getRestaurantTypes() {
        return restaurantTypes;
    }

    public List<Dish> getDishCatalog() {
        return dishCatalog;
    }

    public void registerRestaurant(String ownerUsername, Restaurant restaurant) {
        restaurants.put(restaurant.getId(), restaurant);
        if (ownerUsername != null) {
            ownerRestaurantMap.put(ownerUsername, restaurant.getId());
        }
    }

    public Restaurant getRestaurantByOwner(String ownerUsername) {
        String restaurantId = ownerRestaurantMap.get(ownerUsername);
        if (restaurantId != null) {
            return restaurants.get(restaurantId);
        }
        return null;
    }

    public void deleteRestaurantByOwner(String ownerUsername) {
        String restaurantId = ownerRestaurantMap.remove(ownerUsername);
        if (restaurantId != null) {
            restaurants.remove(restaurantId);
        }
    }

    public Optional<User> findUserByUsername(String username) {
        return Optional.ofNullable(users.get(username));
    }

    public Optional<User> findUserByEmail(String email) {
        return users.values().stream()
                .filter(u -> u.getEmail().equalsIgnoreCase(email))
                .findFirst();
    }

    public Optional<RestaurantOwner> findOwnerByUsername(String username) {
        return Optional.ofNullable(owners.get(username));
    }

    public Optional<RestaurantOwner> findOwnerByEmail(String email) {
        return owners.values().stream()
                .filter(o -> o.getEmail().equalsIgnoreCase(email))
                .findFirst();
    }

    public double calculateDistance(double userLat, double userLon, double restLat, double restLon) {
        // Approximate planar distance in km
        double dLat = (restLat - userLat) * 111.0;
        double dLon = (restLon - userLon) * 111.0 * Math.cos(Math.toRadians(userLat));
        return Math.sqrt(dLat * dLat + dLon * dLon);
    }
}

    // FR6, FR7, FR8, FR12, FR13, FR15, FR16, FR17, FR18: Search
    public String search(SearchCriteria criteria) {
        if (currentUser == null) {
            return "Please log in to search.";
        }

        List<RestaurantSearchResult> matchingResults = new ArrayList<>();

        for (Restaurant r : database.getRestaurants().values()) {
            double distance = database.calculateDistance(userLatitude, userLongitude, r.getLatitude(), r.getLongitude());

            // Destination distance check (FR13, default min 0, max 10)
            if (criteria.getMinDistanceKm() != null && distance < criteria.getMinDistanceKm()) {
                continue;
            }
            if (criteria.getMaxDistanceKm() != null && distance > criteria.getMaxDistanceKm()) {
                continue;
            }

            // Price range check (FR12)
            if (criteria.getMinPrice() != null && r.getAveragePrice() < criteria.getMinPrice()) {
                continue;
            }
            if (criteria.getMaxPrice() != null && r.getAveragePrice() > criteria.getMaxPrice()) {
                continue;
            }

            // Restaurant type check (FR15)
            if (criteria.getRestaurantType() != null && !criteria.getRestaurantType().isBlank()) {
                if (!r.getRestaurantType().equalsIgnoreCase(criteria.getRestaurantType())) {
                    continue;
                }
            }

            // Specific dish check (FR16)
            String matchedDish = null;
            if (criteria.getSpecificDish() != null && !criteria.getSpecificDish().isBlank()) {
                Optional<Dish> dishOpt = r.getMenu().stream()
                        .filter(d -> d.getName().equalsIgnoreCase(criteria.getSpecificDish()))
                        .findFirst();
                if (dishOpt.isEmpty()) {
                    continue;
                } else {
                    matchedDish = dishOpt.get().getName();
                }
            }

            // Free-text search check (FR17)
            if (criteria.getFreeText() != null && !criteria.getFreeText().isBlank()) {
                String term = criteria.getFreeText().toLowerCase();
                boolean matchesName = r.getName().toLowerCase().contains(term);
                boolean matchesDesc = r.getDescription().toLowerCase().contains(term);
                boolean matchesAddress = r.getAddress().toLowerCase().contains(term);
                boolean matchesType = r.getRestaurantType().toLowerCase().contains(term);
                boolean matchesMenu = r.getMenu().stream()
                        .anyMatch(d -> d.getName().toLowerCase().contains(term) || d.getDescription().toLowerCase().contains(term));

                if (!matchesName && !matchesDesc && !matchesAddress && !matchesType && !matchesMenu) {
                    continue;
                }
            }

            matchingResults.add(new RestaurantSearchResult(r, distance, matchedDish));
        }

        // FR18: No match found - keep user on search page
        if (matchingResults.isEmpty()) {
            this.onSearchPage = true;
            this.currentResultView = null;
            return "No match found. You remain on the search page to conduct a new search.";
        }

        // Default view: FR12 says search by price defaults to list view. Other searches default to map view.
        ResultViewType defaultViewType;
        if (criteria.isSearchByPriceExplicit()) {
            defaultViewType = ResultViewType.LIST;
        } else {
            defaultViewType = ResultViewType.MAP;
        }

        this.currentResultView = new SearchResultView(defaultViewType, matchingResults, criteria.isSearchByPriceExplicit());
        this.onSearchPage = false;
        return "Search completed. Found " + currentResultView.getResultCount() + " results.";
    }
