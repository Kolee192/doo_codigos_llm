import java.util.Objects;

public class Temperature {
    private final double value;
    private final TemperatureScale scale;

    public Temperature(double value, TemperatureScale scale) {
        this.value = value;
        this.scale = Objects.requireNonNull(scale, "Temperature scale cannot be null");
    }

    public double getValue() {
        return value;
    }

    public TemperatureScale getScale() {
        return scale;
    }

    public Temperature toFahrenheit() {
        if (scale == TemperatureScale.FAHRENHEIT) {
            return this;
        }
        double fVal = (value * 9.0 / 5.0) + 32.0;
        return new Temperature(fVal, TemperatureScale.FAHRENHEIT);
    }

    public Temperature toCelsius() {
        if (scale == TemperatureScale.CELSIUS) {
            return this;
        }
        double cVal = (value - 32.0) * 5.0 / 9.0;
        return new Temperature(cVal, TemperatureScale.CELSIUS);
    }

    @Override
    public String toString() {
        return String.format("%.1f °%s", value, scale == TemperatureScale.FAHRENHEIT ? "F" : "C");
    }
}

import java.time.DayOfWeek;
import java.time.LocalDateTime;

public class Thermostat {
    public static final int MIN_TEMP_FAHRENHEIT = 60;
    public static final int MAX_TEMP_FAHRENHEIT = 80;

    private final String id;
    private final String room;
    private Temperature currentTemperature;
    private int targetTemperatureFahrenheit;
    private boolean manualOverrideActive;
    private LocalDateTime overrideUntil;

    // 7 days a week, 24 hourly settings per day (temperatures in °F)
    private final int[][] weeklySchedule;

    public Thermostat(String id, String room) {
        this.id = id;
        this.room = room;
        this.currentTemperature = new Temperature(70.0, TemperatureScale.FAHRENHEIT);
        this.targetTemperatureFahrenheit = 70;
        this.manualOverrideActive = false;
        this.overrideUntil = null;
        this.weeklySchedule = new int[7][24];
        for (int d = 0; d < 7; d++) {
            for (int h = 0; h < 24; h++) {
                weeklySchedule[d][h] = 70;
            }
        }
    }

    public String getId() {
        return id;
    }

    public String getRoom() {
        return room;
    }

    public Temperature getCurrentTemperature() {
        return currentTemperature;
    }

    public void setCurrentTemperature(Temperature currentTemperature) {
        this.currentTemperature = currentTemperature;
    }

    public int getTargetTemperatureFahrenheit() {
        return targetTemperatureFahrenheit;
    }

    public Temperature getTargetTemperature(TemperatureScale scale) {
        Temperature tempF = new Temperature(targetTemperatureFahrenheit, TemperatureScale.FAHRENHEIT);
        return scale == TemperatureScale.FAHRENHEIT ? tempF : tempF.toCelsius();
    }

    public void setTargetTemperatureFahrenheit(int targetTempF) {
        if (targetTempF < MIN_TEMP_FAHRENHEIT || targetTempF > MAX_TEMP_FAHRENHEIT) {
            throw new IllegalArgumentException(String.format(
                    "Thermostat temperature must be between %d °F and %d °F inclusive.",
                    MIN_TEMP_FAHRENHEIT, MAX_TEMP_FAHRENHEIT));
        }
        this.targetTemperatureFahrenheit = targetTempF;
    }

    public void setTargetTemperature(Temperature temperature) {
        Temperature fahrenheit = temperature.toFahrenheit();
        int roundedF = (int) Math.round(fahrenheit.getValue());
        setTargetTemperatureFahrenheit(roundedF);
    }

    public void setManualTarget(int targetTempF, LocalDateTime until) {
        setTargetTemperatureFahrenheit(targetTempF);
        this.manualOverrideActive = true;
        this.overrideUntil = until;
    }

    public void scheduleHour(DayOfWeek day, int hour, int targetTempF) {
        if (hour < 0 || hour >= 24) {
            throw new IllegalArgumentException("Hour must be between 0 and 23.");
        }
        if (targetTempF < MIN_TEMP_FAHRENHEIT || targetTempF > MAX_TEMP_FAHRENHEIT) {
            throw new IllegalArgumentException("Schedule temp must be between 60°F and 80°F.");
        }
        int dayIndex = day.getValue() - 1; // Monday=0, Sunday=6
        weeklySchedule[dayIndex][hour] = targetTempF;
    }

    public int getScheduledTemp(DayOfWeek day, int hour) {
        int dayIndex = day.getValue() - 1;
        return weeklySchedule[dayIndex][hour];
    }

    public void applyScheduleForTime(LocalDateTime dateTime) {
        if (manualOverrideActive) {
            if (overrideUntil != null && dateTime.isAfter(overrideUntil)) {
                manualOverrideActive = false;
                overrideUntil = null;
            } else {
                return; // Override still in effect
            }
        }
        int scheduled = getScheduledTemp(dateTime.getDayOfWeek(), dateTime.getHour());
        setTargetTemperatureFahrenheit(scheduled);
    }

    public boolean isManualOverrideActive() {
        return manualOverrideActive;
    }

    public LocalDateTime getOverrideUntil() {
        return overrideUntil;
    }

    @Override
    public String toString() {
        return String.format("Thermostat[%s in %s: Current=%s, Target=%d °F, Override=%b]",
                id, room, currentTemperature, targetTemperatureFahrenheit, manualOverrideActive);
    }
}

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class SecuritySystem {
    public static final int MAX_CONTACT_SENSORS = 50;

    private final List<ContactSensor> contactSensors;
    private final SoundAlarm soundAlarm;
    private final LightAlarmSubsystem lightAlarmSubsystem;
    private boolean armed;

    public SecuritySystem(SoundAlarm soundAlarm, LightAlarmSubsystem lightAlarmSubsystem) {
        this.contactSensors = new ArrayList<>();
        this.soundAlarm = soundAlarm;
        this.lightAlarmSubsystem = lightAlarmSubsystem;
        this.armed = true;
    }

    public void addContactSensor(ContactSensor sensor) {
        if (contactSensors.size() >= MAX_CONTACT_SENSORS) {
            throw new IllegalStateException("Maximum of " + MAX_CONTACT_SENSORS + " contact sensors reached.");
        }
        contactSensors.add(sensor);
    }

    public List<ContactSensor> getContactSensors() {
        return Collections.unmodifiableList(contactSensors);
    }

    public SoundAlarm getSoundAlarm() {
        return soundAlarm;
    }

    public LightAlarmSubsystem getLightAlarmSubsystem() {
        return lightAlarmSubsystem;
    }

    public boolean isArmed() {
        return armed;
    }

    public void setArmed(boolean armed) {
        this.armed = armed;
        if (!armed) {
            soundAlarm.deactivate();
            lightAlarmSubsystem.deactivate();
        }
    }

    public SecurityBreachEvent checkSensorAndTriggerAlarm(String sensorId, ContactState newState, LocalDateTime time) {
        ContactSensor target = contactSensors.stream()
                .filter(s -> s.getId().equals(sensorId))
                .findFirst()
                .orElse(null);

        if (target == null) {
            return null;
        }

        target.setState(newState);

        if (armed && newState == ContactState.OPEN) {
            soundAlarm.activate();
            lightAlarmSubsystem.activate();
            return new SecurityBreachEvent(time, target.getId(), target.getName(), target.getLocation());
        }
        return null;
    }

    public boolean isAlarmActive() {
        return soundAlarm.isSounding() || lightAlarmSubsystem.isFlashing();
    }
}

public class DHDatabase {
    private final List<SensorReading> sensorReadings = new ArrayList<>();
    private final List<SecurityBreachEvent> breachEvents = new ArrayList<>();
    private final List<SystemDowntimePeriod> downtimePeriods = new ArrayList<>();

    public synchronized void recordReading(SensorReading reading) {
        sensorReadings.add(reading);
    }

    public synchronized void recordBreach(SecurityBreachEvent event) {
        breachEvents.add(event);
    }

    public synchronized void recordDowntime(SystemDowntimePeriod downtime) {
        downtimePeriods.add(downtime);
    }

    public synchronized List<SensorReading> getReadingsBetween(LocalDateTime start, LocalDateTime end) {
        return sensorReadings.stream()
                .filter(r -> !r.getTimestamp().isBefore(start) && !r.getTimestamp().isAfter(end))
                .collect(Collectors.toList());
    }

    public synchronized List<SecurityBreachEvent> getBreachesBetween(LocalDateTime start, LocalDateTime end) {
        return breachEvents.stream()
                .filter(b -> !b.getTimestamp().isBefore(start) && !b.getTimestamp().isAfter(end))
                .collect(Collectors.toList());
    }

    public synchronized List<SystemDowntimePeriod> getDowntimesBetween(LocalDateTime start, LocalDateTime end) {
        return downtimePeriods.stream()
                .filter(d -> !d.getStartTime().isAfter(end) && !d.getEndTime().isBefore(start))
                .collect(Collectors.toList());
    }
}

public class DailyPlan {
    public static final int MAX_PERIODS = 4;
    private final int dayOfMonth;
    private final List<TimePeriodSetting> periodSettings;

    public DailyPlan(int dayOfMonth) {
        this.dayOfMonth = dayOfMonth;
        this.periodSettings = new ArrayList<>();
    }

    public int getDayOfMonth() {
        return dayOfMonth;
    }

    public void addPeriodSetting(TimePeriodSetting setting) {
        if (periodSettings.size() >= MAX_PERIODS) {
            throw new IllegalStateException("A daily plan can have up to " + MAX_PERIODS + " periods.");
        }
        periodSettings.add(setting);
    }

    public List<TimePeriodSetting> getPeriodSettings() {
        return Collections.unmodifiableList(periodSettings);
    }

    public TimePeriodSetting findSettingForTime(LocalTime time) {
        for (TimePeriodSetting setting : periodSettings) {
            if ((time.equals(setting.getStartTime()) || time.isAfter(setting.getStartTime()))
                    && time.isBefore(setting.getEndTime())) {
                return setting;
            }
        }
        return periodSettings.isEmpty() ? null : periodSettings.get(0);
    }
}

public class DHPlanner {
    private final Map<String, MonthPlan> monthPlans = new HashMap<>();

    private String key(int year, int month) {
        return year + "-" + String.format("%02d", month);
    }

    public void saveMonthPlan(MonthPlan plan) {
        monthPlans.put(key(plan.getYear(), plan.getMonth()), plan);
    }

    public MonthPlan getMonthPlan(int year, int month) {
        return monthPlans.get(key(year, month));
    }

    public TimePeriodSetting getActivePlanSetting(LocalDateTime dateTime) {
        MonthPlan mp = getMonthPlan(dateTime.getYear(), dateTime.getMonthValue());
        if (mp == null) {
            return null;
        }
        DailyPlan dp = mp.getDailyPlan(dateTime.getDayOfMonth());
        if (dp == null) {
            return null;
        }
        return dp.findSettingForTime(dateTime.toLocalTime());
    }
}

import java.time.LocalDateTime;
import java.time.YearMonth;
import java.util.*;
import java.util.stream.Collectors;

public class MonthlyReport {
    private final int year;
    private final int month;
    private final DHDatabase database;
    private final List<Thermostat> thermostats;
    private final List<Humidistat> humidistats;

    public MonthlyReport(int year, int month, DHDatabase database,
                         List<Thermostat> thermostats, List<Humidistat> humidistats) {
        this.year = year;
        this.month = month;
        this.database = database;
        this.thermostats = thermostats;
        this.humidistats = humidistats;
    }

    public static class DaySensorStats {
        public double average;
        public double min = Double.MAX_VALUE;
        public LocalDateTime minTime;
        public double max = -Double.MAX_VALUE;
        public LocalDateTime maxTime;
        public int count = 0;
    }

    public String generateReport() {
        LocalDateTime now = LocalDateTime.now();
        YearMonth requestedYm = YearMonth.of(year, month);
        YearMonth currentYm = YearMonth.from(now);

        long monthsDiff = (currentYm.getYear() - requestedYm.getYear()) * 12L + (currentYm.getMonthValue() - requestedYm.getMonthValue());
        if (monthsDiff < 0 || monthsDiff > 24) {
            throw new IllegalArgumentException("Report can only be provided for months in the past two years.");
        }

        LocalDateTime startOfMonth = requestedYm.atDay(1).atStartOfDay();
        LocalDateTime endOfMonth = requestedYm.atEndOfMonth().atTime(23, 59, 59);

        List<SensorReading> readings = database.getReadingsBetween(startOfMonth, endOfMonth);
        List<SecurityBreachEvent> breaches = database.getBreachesBetween(startOfMonth, endOfMonth);
        List<SystemDowntimePeriod> downtimes = database.getDowntimesBetween(startOfMonth, endOfMonth);

        StringBuilder sb = new StringBuilder();
        sb.append("=================================================================\n");
        sb.append(String.format("         DIGITAL HOME MONTHLY REPORT - %02d/%d\n", month, year));
        sb.append("=================================================================\n\n");

        sb.append("1. TEMPERATURE METRICS (Per Thermostat)\n");
        sb.append("-----------------------------------------------------------------\n");
        for (Thermostat t : thermostats) {
            sb.append(String.format("Thermostat %s (%s):\n", t.getId(), t.getRoom()));
            Map<Integer, DaySensorStats> dailyStats = computeDailyStats(readings, t.getId(), SensorReading.DeviceType.THERMOSTAT);
            if (dailyStats.isEmpty()) {
                sb.append("  No readings available for this month.\n");
            } else {
                dailyStats.keySet().stream().sorted().forEach(day -> {
                    DaySensorStats s = dailyStats.get(day);
                    sb.append(String.format("  Day %02d: Avg=%.1f°F | Min=%.1f°F (%s) | Max=%.1f°F (%s)\n",
                            day, s.average, s.min, s.minTime.toLocalTime(), s.max, s.maxTime.toLocalTime()));
                });
            }
            sb.append("\n");
        }

        sb.append("2. HUMIDITY METRICS (Per Humidistat)\n");
        sb.append("-----------------------------------------------------------------\n");
        for (Humidistat h : humidistats) {
            sb.append(String.format("Humidistat %s (%s):\n", h.getId(), h.getRoom()));
            Map<Integer, DaySensorStats> dailyStats = computeDailyStats(readings, h.getId(), SensorReading.DeviceType.HUMIDISTAT);
            if (dailyStats.isEmpty()) {
                sb.append("  No readings available for this month.\n");
            } else {
                dailyStats.keySet().stream().sorted().forEach(day -> {
                    DaySensorStats s = dailyStats.get(day);
                    sb.append(String.format("  Day %02d: Avg=%.1f%% | Min=%.1f%% (%s) | Max=%.1f%% (%s)\n",
                            day, s.average, s.min, s.minTime.toLocalTime(), s.max, s.maxTime.toLocalTime()));
                });
            }
            sb.append("\n");
        }

        sb.append("3. SECURITY BREACHES\n");
        sb.append("-----------------------------------------------------------------\n");
        if (breaches.isEmpty()) {
            sb.append("  No security breaches recorded.\n");
        } else {
            for (SecurityBreachEvent b : breaches) {
                sb.append(String.format("  [BREACH] Date/Time: %s | Sensor: %s (%s at %s)\n",
                        b.getTimestamp(), b.getSensorId(), b.getSensorName(), b.getLocation()));
            }
        }
        sb.append("\n");

        sb.append("4. SYSTEM OUTAGES / NON-OPERATION PERIODS\n");
        sb.append("-----------------------------------------------------------------\n");
        if (downtimes.isEmpty()) {
            sb.append("  DH System was operational continuously throughout this month.\n");
        } else {
            for (SystemDowntimePeriod dt : downtimes) {
                sb.append(String.format("  [OUTAGE] From: %s to %s | Reason: %s\n",
                        dt.getStartTime(), dt.getEndTime(), dt.getReason()));
            }
        }
        sb.append("=================================================================\n");
        return sb.toString();
    }

    private Map<Integer, DaySensorStats> computeDailyStats(List<SensorReading> readings, String deviceId, SensorReading.DeviceType type) {
        Map<Integer, List<SensorReading>> grouped = readings.stream()
                .filter(r -> r.getDeviceId().equals(deviceId) && r.getDeviceType() == type)
                .collect(Collectors.groupingBy(r -> r.getTimestamp().getDayOfMonth()));

        Map<Integer, DaySensorStats> result = new HashMap<>();
        for (Map.Entry<Integer, List<SensorReading>> entry : grouped.entrySet()) {
            DaySensorStats stats = new DaySensorStats();
            double sum = 0;
            for (SensorReading r : entry.getValue()) {
                sum += r.getValue();
                stats.count++;
                if (r.getValue() < stats.min) {
                    stats.min = r.getValue();
                    stats.minTime = r.getTimestamp();
                }
                if (r.getValue() > stats.max) {
                    stats.max = r.getValue();
                    stats.maxTime = r.getTimestamp();
                }
            }
            stats.average = sum / stats.count;
            result.put(entry.getKey(), stats);
        }
        return result;
    }
}

    public void applyPlanAtTime(LocalDateTime currentTime) {
        TimePeriodSetting setting = planner.getActivePlanSetting(currentTime);
        if (setting == null) {
            return;
        }

        // Apply temperature to thermostats not manually overridden
        for (Thermostat t : gateway.getThermostats()) {
            t.clearOverrideIfExpired ? (void) null : null;
            if (t.isManualOverrideActive()) {
                if (t.getOverrideUntil() != null && currentTime.isAfter(t.getOverrideUntil())) {
                    t.setTargetTemperatureFahrenheit(setting.getTargetTemperatureF());
                }
            } else {
                t.setTargetTemperatureFahrenheit(setting.getTargetTemperatureF());
            }
        }

        // Apply humidity
        for (Humidistat h : gateway.getHumidistats()) {
            if (h.isManualOverrideActive()) {
                if (h.getOverrideUntil() != null && currentTime.isAfter(h.getOverrideUntil())) {
                    h.setTargetHumidityPercent(setting.getTargetHumidityPercent());
                }
            } else {
                h.setTargetHumidityPercent(setting.getTargetHumidityPercent());
            }
        }

        // Apply switch states
        for (Map.Entry<String, SwitchState> entry : setting.getSwitchStates().entrySet()) {
            for (PowerSwitch ps : gateway.getPowerSwitches()) {
                if (ps.getId().equals(entry.getKey())) {
                    ps.clearOverrideIfExpired(currentTime);
                    if (!ps.isManualOverrideActive()) {
                        ps.setState(entry.getValue());
                    }
                }
            }
        }

        // Apply security armed state
        gateway.getSecuritySystem().setArmed(setting.isSecurityArmed());
    }

    // Reporting
    public String requestMonthlyReport(int year, int month) {
        MonthlyReport report = new MonthlyReport(year, month, gateway.getDatabase(),
                gateway.getThermostats(), gateway.getHumidistats());
        return report.generateReport();
    }
