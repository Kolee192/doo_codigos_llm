// File: HcpObjectStorage.java
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.HashMap;
import java.util.Map;

public class HcpObjectStorage implements StorageBackend {
    private final Map<String, byte[]> objectStore = new HashMap<>();

    @Override
    public void store(String fileId, byte[] data) {
        objectStore.put(fileId, data.clone());
    }

    @Override
    public byte[] retrieve(String fileId) {
        byte[] data = objectStore.get(fileId);
        return data == null ? null : data.clone();
    }

    @Override
    public boolean verifyIntegrity(String fileId, String expectedSha256) {
        byte[] data = objectStore.get(fileId);
        if (data == null) {
            return false;
        }
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(data);
            StringBuilder hexString = new StringBuilder();
            for (byte b : hash) {
                String hex = Integer.toHexString(0xff & b);
                if (hex.length() == 1) hexString.append('0');
                hexString.append(hex);
            }
            return hexString.toString().equalsIgnoreCase(expectedSha256);
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException("SHA-256 algorithm not available", e);
        }
    }

    @Override
    public long getCapacityAvailable() {
        return Long.MAX_VALUE;
    }
}

public class DoiService {
    private final String prefix = "10.15125/BATH-";

    public String mintDoi(String datasetId, int versionNumber) {
        return prefix + datasetId.toUpperCase() + ".V" + versionNumber;
    }

    public String generatePersistentUrl(String doi) {
        return "https://doi.org/" + doi;
    }

    public String formatApaCitation(Dataset dataset, DatasetVersion version) {
        String authors = String.join(", ", version.getMetadata().getAuthors());
        int year = version.getCreatedTimestamp().getYear();
        String title = version.getMetadata().getTitle();
        String doi = version.getDoi();
        return String.format("%s (%d). %s [Data set]. University of Bath Data Archive. %s",
                authors.isEmpty() ? "University of Bath" : authors,
                year,
                title,
                doi != null ? "https://doi.org/" + doi : version.getPersistentUrl());
    }

    public String formatBibtexCitation(Dataset dataset, DatasetVersion version) {
        String key = "bath_" + dataset.getDatasetId() + "_v" + version.getVersionNumber();
        String authors = String.join(" and ", version.getMetadata().getAuthors());
        return String.format("""
                @misc{%s,
                  author = {%s},
                  title = {{%s}},
                  year = {%d},
                  publisher = {University of Bath Research Data Archive},
                  doi = {%s},
                  url = {%s}
                }""",
                key,
                authors,
                version.getMetadata().getTitle(),
                version.getCreatedTimestamp().getYear(),
                version.getDoi(),
                version.getPersistentUrl());
    }
}

// File: MetadataValidator.java
import java.util.ArrayList;
import java.util.List;

public class MetadataValidator {
    public static class ValidationResult {
        private final boolean valid;
        private final List<String> errors;

        public ValidationResult(boolean valid, List<String> errors) {
            this.valid = valid;
            this.errors = errors;
        }

        public boolean isValid() {
            return valid;
        }

        public List<String> getErrors() {
            return errors;
        }
    }

    public ValidationResult validate(DatasetVersion version) {
        List<String> errors = new ArrayList<>();
        DatasetMetadata meta = version.getMetadata();

        if (meta == null) {
            errors.add("Metadata object is missing.");
            return new ValidationResult(false, errors);
        }

        if (meta.getTitle() == null || meta.getTitle().trim().isEmpty()) {
            errors.add("Mandatory field missing: Title is required.");
        }
        if (meta.getAuthors() == null || meta.getAuthors().isEmpty()) {
            errors.add("Mandatory field missing: At least one Author/Creator is required.");
        }
        if (meta.getDescription() == null || meta.getDescription().trim().length() < 10) {
            errors.add("Mandatory field missing: Description must be at least 10 characters.");
        }
        if (meta.getSubjectDiscipline() == null || meta.getSubjectDiscipline().trim().isEmpty()) {
            errors.add("Mandatory field missing: Subject discipline is required.");
        }
        if (version.getLicense() == null) {
            errors.add("Mandatory field missing: Valid license selection is required.");
        }

        return new ValidationResult(errors.isEmpty(), errors);
    }
}

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

public class AnalyticsService {
    private final List<UsageMetricRecord> metrics = new ArrayList<>();
    private final Map<String, Integer> citationCounts = new HashMap<>();

    public void logMetric(String datasetId, int versionNumber, UsageMetricRecord.Action action, String country, String sector) {
        metrics.add(new UsageMetricRecord(datasetId, versionNumber, action, country, sector));
    }

    public void recordCitation(String datasetId) {
        citationCounts.put(datasetId, citationCounts.getOrDefault(datasetId, 0) + 1);
        metrics.add(new UsageMetricRecord(datasetId, 1, UsageMetricRecord.Action.CITATION, "GLOBAL", "ACADEMIC"));
    }

    public int getCitationCount(String datasetId) {
        return citationCounts.getOrDefault(datasetId, 0);
    }

    public long getTotalDownloads(String datasetId) {
        return metrics.stream()
                .filter(m -> m.getDatasetId().equals(datasetId) && m.action() == UsageMetricRecord.Action.DOWNLOAD)
                .count();
    }

    public long getTotalViews(String datasetId) {
        return metrics.stream()
                .filter(m -> m.getDatasetId().equals(datasetId) && m.action() == UsageMetricRecord.Action.VIEW)
                .count();
    }

    public Map<String, Long> getDownloadsByCountry(String datasetId) {
        return metrics.stream()
                .filter(m -> m.getDatasetId().equals(datasetId) && m.action() == UsageMetricRecord.Action.DOWNLOAD)
                .collect(Collectors.groupingBy(UsageMetricRecord::getCountry, Collectors.counting()));
    }

    public Map<String, Long> getDownloadsBySector(String datasetId) {
        return metrics.stream()
                .filter(m -> m.getDatasetId().equals(datasetId) && m.action() == UsageMetricRecord.Action.DOWNLOAD)
                .collect(Collectors.groupingBy(UsageMetricRecord::getSector, Collectors.counting()));
    }
}

    public List<OaiRecord> harvestByFunder(List<Dataset> datasets, String funderName) {
        List<OaiRecord> harvested = new ArrayList<>();
        for (Dataset d : datasets) {
            if (d.getStatus() == DatasetStatus.PUBLISHED) {
                DatasetVersion v = d.getLatestVersion();
                if (v != null && funderName.equalsIgnoreCase(v.getMetadata().getFunderName())) {
                    harvested.add(new OaiRecord(
                            d.getDatasetId(),
                            v.getMetadata().getTitle(),
                            v.getMetadata().getAuthors(),
                            v.getMetadata().getSubjectDiscipline(),
                            v.getMetadata().getDescription(),

                            v.getDoi(),
                            v.getMetadata().getFunderName(),
                            v.getMetadata().getGrantNumber()

    public Dataset createDatasetDraft(String actingUserId, String onBehalfOfUserId, boolean externalRepoLink) {
        User actor = ldapService.authenticate(actingUserId)
                .orElseThrow(() -> new IllegalArgumentException("User not found in University LDAP: " + actingUserId));

        String targetDepositorId = actingUserId;
        if (onBehalfOfUserId != null && !onBehalfOfUserId.equals(actingUserId)) {
            if (!actor.canDepositOnBehalfOf(onBehalfOfUserId) && actor.getRole() != Role.FACILITY_MANAGER && actor.getRole() != Role.ADMINISTRATOR) {
                throw new SecurityException("User " + actingUserId + " is not authorized to deposit on behalf of " + onBehalfOfUserId);
            }
            targetDepositorId = onBehalfOfUserId;
        }

        String datasetId = "bath-data-" + UUID.randomUUID().toString().substring(0, 8);
        Dataset dataset = new Dataset(datasetId, targetDepositorId, externalRepoLink);
        datasets.put(datasetId, dataset);
        return dataset;
    }

    public DatasetMetadata prepareMetadataWithDefaults(String depositorId, String title, String description, String discipline) {
        DatasetMetadata meta = new DatasetMetadata(title, description, discipline);
        Map<String, String> ldapDefaults = ldapService.fetchUserProfileMetadata(depositorId);
        if (ldapDefaults.containsKey("creator.name")) {
            meta.addAuthor(ldapDefaults.get("creator.name") + " (" + ldapDefaults.get("creator.department") + ")");
        }
        DatasetMetadata previous = rememberedDepositorMetadata.get(depositorId);
        if (previous != null) {
            meta.setCrisProjectId(previous.getCrisProjectId());
            meta.setDmpId(previous.getDmpId());
            meta.setFunderName(previous.getFunderName());
            meta.setGrantNumber(previous.getGrantNumber());
        }
        return meta;
    }

    public StoredFile uploadFile(String datasetId, String fileName, String contentType, byte[] data, boolean isLive) {
        String fileId = "file-" + UUID.randomUUID().toString();
        String checksum = calculateSha256(data);
        storageBackend.store(fileId, data);
        return new StoredFile(fileId, fileName, contentType, data.length, checksum, isLive, data);
    }

    public DatasetVersion addVersionToDataset(String datasetId, String actingUserId, DatasetMetadata metadata,
                                              License license, DisposalPolicy disposalPolicy,
                                              AccessLevel accessLevel, LocalDate embargoEndDate,
                                              List<StoredFile> files, String changeSummary) {
        Dataset dataset = datasets.get(datasetId);
        if (dataset == null) {
            throw new IllegalArgumentException("Dataset not found: " + datasetId);
        }

        int nextVersionNumber = dataset.getVersions().size() + 1;
        DatasetVersion version = new DatasetVersion(nextVersionNumber, changeSummary, metadata, license, disposalPolicy, accessLevel, embargoEndDate);
        for (StoredFile file : files) {
            version.addFile(file);
        }

        dataset.addVersion(version);
        rememberedDepositorMetadata.put(dataset.getDepositorId(), metadata);
        if (metadata.getCrisProjectId() != null) {
            crisService.linkDatasetToCris(metadata.getCrisProjectId(), datasetId);
        }
        return version;
    }

    public void submitForReview(String datasetId) {
        Dataset dataset = datasets.get(datasetId);
        if (dataset == null) {
            throw new IllegalArgumentException("Dataset not found");
        }
        dataset.setStatus(DatasetStatus.SUBMITTED_FOR_REVIEW);
    }

    public boolean adminApproveAndPublish(String adminUserId, String datasetId) {
        User admin = ldapService.authenticate(adminUserId)
                .orElseThrow(() -> new IllegalArgumentException("Admin not found"));
        if (admin.getRole() != Role.ADMINISTRATOR) {
            throw new SecurityException("Only University of Bath Archive Administrators can approve datasets.");
        }

        Dataset dataset = datasets.get(datasetId);
        if (dataset == null) {
            throw new IllegalArgumentException("Dataset not found");
        }

        DatasetVersion latest = dataset.getLatestVersion();
        if (latest == null) {
            return false;
        }

        MetadataValidator.ValidationResult validation = metadataValidator.validate(latest);
        if (!validation.isValid()) {
            throw new IllegalStateException("Metadata validation failed: " + String.join(", ", validation.getErrors()));
        }

        String doi = doiService.mintDoi(dataset.getDatasetId(), latest.getVersionNumber());
        String purl = doiService.generatePersistentUrl(doi);
        latest.setDoi(doi);
        latest.setPersistentUrl(purl);

        dataset.setStatus(DatasetStatus.PUBLISHED);
        return true;
    }

    public FixityAuditReport auditFileIntegrity(String fileId, String expectedSha256, String fileName) {
        boolean match = storageBackend.verifyIntegrity(fileId, expectedSha256);
        return new FixityAuditReport(fileId, fileName, match, expectedSha256);
    }

    public boolean canAccessData(Dataset dataset, DatasetVersion version, User user, LocalDate currentDate, boolean onCampus) {
        if (user != null && (user.getRole() == Role.ADMINISTRATOR || user.getUserId().equals(dataset.getDepositorId()))) {
            return true;
        }

        if (version.getPrivilegedCollaborators().contains(user != null ? user.getUserId() : "")) {
            return true;
        }

        if (version.isEmbargoActive(currentDate)) {
            return false;
        }

        if (version.getAccessLevel() == AccessLevel.OPEN) {
            return true;
        }

        if (version.getAccessLevel() == AccessLevel.RESTRICTED) {
            return onCampus && user != null;
        }

        return false;
    }

    public List<Dataset> searchCatalog(String query, String subjectFilter) {
        String q = query == null ? "" : query.toLowerCase();
        String subj = subjectFilter == null ? "" : subjectFilter.toLowerCase();

        return datasets.values().stream()
                .filter(d -> d.getStatus() == DatasetStatus.PUBLISHED)
                .filter(d -> {
                    DatasetVersion v = d.getLatestVersion();
                    if (v == null) return false;
                    boolean matchesQuery = q.isEmpty() ||
                            v.getMetadata().getTitle().toLowerCase().contains(q) ||
                            v.getMetadata().getDescription().toLowerCase().contains(q) ||
                            v.getMetadata().getKeywords().stream().anyMatch(k -> k.toLowerCase().contains(q));
                    boolean matchesSubject = subj.isEmpty() ||
                            v.getMetadata().getSubjectDiscipline().toLowerCase().contains(subj);
                    return matchesQuery && matchesSubject;
                })
                .collect(Collectors.toList());
    }

    public List<Dataset> primoSearchEndpoint(String genericSearchTerm) {
        return searchCatalog(genericSearchTerm, null);
    }

    public Dataset swordDeposit(SwordDepositRequest request) {
        User actor = ldapService.authenticate(request.getDepositorId())
                .orElseThrow(() -> new IllegalArgumentException("Unknown SWORD user"));
        Dataset dataset = createDatasetDraft(actor.getUserId(), request.getOnBehalfOf(), false);

        DatasetMetadata meta = new DatasetMetadata(request.getTitle(), request.getAbstractText(), request.getDiscipline());
        meta.addAuthor(actor.getFullName());

        List<StoredFile> files = new ArrayList<>();
        if (request.getFileBytes() != null && request.getFileBytes().length > 0) {
            StoredFile file = uploadFile(dataset.getDatasetId(), request.getFileName(), "application/octet-stream", request.getFileBytes(), false);
            files.add(file);
        }

        License lic = "CC-BY-4.0".equalsIgnoreCase(request.getLicenseIdentifier()) ? License.CC_BY_4_0 : License.RESTRICTED_COLLABORATION;
        DisposalPolicy policy = new DisposalPolicy(10, LocalDate.now().plusYears(10), "Standard 10-year research funder retention");

        addVersionToDataset(dataset.getDatasetId(), actor.getUserId(), meta, lic, policy, AccessLevel.OPEN, null, files, "Deposited via SWORDv2 API");
        submitForReview(dataset.getDatasetId());
        return dataset;
    }

    public void approveDisposal(String adminUserId, String datasetId) {
        User admin = ldapService.authenticate(adminUserId).orElseThrow();
        if (admin.getRole() != Role.ADMINISTRATOR) {
            throw new SecurityException("Only admin can approve disposal");
        }
        Dataset dataset = datasets.get(datasetId);
        if (dataset != null && dataset.getLatestVersion() != null) {
            dataset.getLatestVersion().getDisposalPolicy().approveDisposal();
            dataset.setStatus(DatasetStatus.DISPOSED);
        }
    }

    public byte[] wholesaleExportArchive() {
        StringBuilder sb = new StringBuilder();
        sb.append("BATH-ARCHIVE-WHOLESALE-EXPORT-V1\n");
        for (Dataset d : datasets.values()) {
            DatasetVersion v = d.getLatestVersion();
            if (v != null) {
                sb.append("DATASET:").append(d.getDatasetId())
                  .append("|STATUS:").append(d.getStatus())
                  .append("|TITLE:").append(v.getMetadata().getTitle())
                  .append("|DOI:").append(v.getDoi())

                  .append("\n");
            }
        }

    }

    public int wholesaleImportExternalData(List<Dataset> externalDatasets) {

            imported++;
        }
        return imported;
    }

    public List<Dataset> queryEntireArchive(DatasetStatus statusFilter, String funderFilter) {
        return datasets.values().stream()
                .filter(d -> statusFilter == null || d.getStatus() == statusFilter)
                .filter(d -> {
                    if (funderFilter == null) return true;
                    DatasetVersion v = d.getLatestVersion();
                    return v != null && funderFilter.equalsIgnoreCase(v.getMetadata().getFunderName());
                })
                .collect(Collectors.toList());
    }

    private String calculateSha256(byte[] data) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(data);
            StringBuilder sb = new StringBuilder();
            for (byte b : hash) {
                String hex = Integer.toHexString(0xff & b);
                if (hex.length() == 1) sb.append('0');
                sb.append(hex);
            }
            return sb.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException(e);
        }
    }
