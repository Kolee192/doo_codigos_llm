        this.yearOfProduction = yearOfProduction;

        this.fleetLocation = fleetLocation;

        this.yearOfProduction = yearOfProduction;
    }

    public CarStatus getStatus() {
        return status;
    }

    public void setStatus(CarStatus status) {
        this.status = status;
    }

    public CarModel getCarModel() {
        return carModel;
    }

    public void setCarModel(CarModel carModel) {
        this.carModel = carModel;
    }

    public FleetLocation getFleetLocation() {
        return fleetLocation;
    }

    public void setFleetLocation(FleetLocation fleetLocation) {
        this.fleetLocation = fleetLocation;
    }

    public List<Rental> getRentals() {
        return Collections.unmodifiableList(rentals);
    }

    public void addRental(Rental rental) {
        if (rental != null && !rentals.contains(rental)) {
            rentals.add(rental);
            if (rental.getCar() != this) {
                rental.setCar(this);
            }
        }
    }

    public double computeAverageRentalPrice() {
        if (rentals.isEmpty()) {
            return 0.0;
        }
        return rentals.stream()
                .mapToDouble(Rental::getPrice)
                .average()
                .orElse(0.0);
    }

    @Override
    public String toString() {
        return (carModel != null ? carModel.getName() : "Unknown Model") + " [" + licensePlate + ", " + color + "]";
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Car car)) return false;
        return Objects.equals(licensePlate, car.licensePlate);
    }

    @Override
    public int hashCode() {
        return Objects.hash(licensePlate);
    }
}

public class Rental {
    private Car car;
    private double price;
    private RentalType rentalType;
    private CarPickup carPickup;
    private CarDropoff carDropoff;

    public Rental(Car car, double price, RentalType rentalType) {
        this.car = car;
        this.price = price;
        this.rentalType = rentalType;
        if (car != null) {
            car.addRental(this);
        }
    }

    public Rental(Car car, double price, RentalType rentalType, CarPickup carPickup, CarDropoff carDropoff) {
        this(car, price, rentalType);
        this.carPickup = carPickup;
        this.carDropoff = carDropoff;
    }

    public Car getCar() {
        return car;
    }

    public void setCar(Car car) {
        this.car = car;
    }

    public double getPrice() {
        return price;
    }

    public void setPrice(double price) {
        this.price = price;
    }

    public RentalType getRentalType() {
        return rentalType;
    }

    public void setRentalType(RentalType rentalType) {
        this.rentalType = rentalType;
    }

    public CarPickup getCarPickup() {
        return carPickup;
    }

    public void setCarPickup(CarPickup carPickup) {
        this.carPickup = carPickup;
    }

    public CarDropoff getCarDropoff() {
        return carDropoff;
    }

    public void setCarDropoff(CarDropoff carDropoff) {
        this.carDropoff = carDropoff;
    }

    public boolean isBookedForToday() {
        return isBookedFor(LocalDate.now());
    }

    public boolean isBookedFor(LocalDate date) {
        if (date == null || carPickup == null || carDropoff == null) {
            return false;
        }
        LocalDate start = carPickup.getDate();
        LocalDate end = carDropoff.getDate();
        if (start == null || end == null) {
            return false;
        }
        return !date.isBefore(start) && !date.isAfter(end);
    }

    @Override
    public String toString() {
        return "Rental of " + (car != null ? car.toString() : "N/A") + " - Price: " + price;
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Rental rental)) return false;
        return Double.compare(rental.price, price) == 0 &&
                Objects.equals(car, rental.car) &&
                Objects.equals(rentalType, rental.rentalType) &&
                Objects.equals(carPickup, rental.carPickup) &&
                Objects.equals(carDropoff, rental.carDropoff);
    }

    @Override
    public int hashCode() {
        return Objects.hash(car, price, rentalType, carPickup, carDropoff);
    }
}

        return Objects.equals(vatNumber, that.vatNumber) && Objects.equals(companyNumber, that.companyNumber);
    }

    @Override
    public int hashCode() {
        return Objects.hash(super.hashCode(), vatNumber, companyNumber);
    }
}
