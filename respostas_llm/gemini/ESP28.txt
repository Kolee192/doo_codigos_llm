import java.io.Serializable;
import java.util.Objects;

public class Tile implements Comparable<Tile>, Serializable {
    private static final long serialVersionUID = 1L;

    private final Suit suit;
    private final int rank; // 1-9 for bamboo/characters/dots; 1-4 for East/South/West/North; 1-3 for Red/Green/White
    private final String name;

    public Tile(Suit suit, int rank) {
        this.suit = suit;
        this.rank = rank;
        this.name = buildName(suit, rank);
    }

    private static String buildName(Suit suit, int rank) {
        return switch (suit) {
            case BAMBOO -> rank + " Bamboo";
            case CHARACTERS -> rank + " Character";
            case DOTS -> rank + " Dot";
            case WINDS -> switch (rank) {
                case 1 -> "East Wind";
                case 2 -> "South Wind";
                case 3 -> "West Wind";
                case 4 -> "North Wind";
                default -> "Unknown Wind";
            };
            case DRAGONS -> switch (rank) {
                case 1 -> "Red Dragon";
                case 2 -> "Green Dragon";
                case 3 -> "White Dragon";
                default -> "Unknown Dragon";
            };
        };
    }

    public Suit getSuit() {
        return suit;
    }

    public int getRank() {
        return rank;
    }

    public String getName() {
        return name;
    }

    public boolean isHonor() {
        return suit == Suit.WINDS || suit == Suit.DRAGONS;
    }

    @Override
    public int compareTo(Tile o) {
        if (this.suit != o.suit) {
            return this.suit.compareTo(o.suit);
        }
        return Integer.compare(this.rank, o.rank);
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Tile tile)) return false;
        return rank == tile.rank && suit == tile.suit;
    }

    @Override
    public int hashCode() {
        return Objects.hash(suit, rank);
    }

    @Override
    public String toString() {
        return "[" + name + "]";
    }
}

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class TileSet {
    public static List<Tile> createStandardTileSet() {
        List<Tile> tiles = new ArrayList<>(136);
        // Suited tiles: 4 sets of 1-9 for Bamboo, Characters, Dots
        for (int i = 0; i < 4; i++) {
            for (int rank = 1; rank <= 9; rank++) {
                tiles.add(new Tile(Suit.BAMBOO, rank));
                tiles.add(new Tile(Suit.CHARACTERS, rank));
                tiles.add(new Tile(Suit.DOTS, rank));
            }
            // Honor tiles: 4 copies of 4 Winds
            for (int w = 1; w <= 4; w++) {
                tiles.add(new Tile(Suit.WINDS, w));
            }
            // Honor tiles: 4 copies of 3 Dragons
            for (int d = 1; d <= 3; d++) {
                tiles.add(new Tile(Suit.DRAGONS, d));
            }
        }
        return tiles;
    }
}

public class PreferenceManager {
    public static void savePreferences(Preferences prefs, String filePath) {
        try (ObjectOutputStream oos = new ObjectOutputStream(new FileOutputStream(filePath))) {
            oos.writeObject(prefs);
        } catch (IOException e) {
            System.err.println("Failed to save preferences: " + e.getMessage());
        }
    }

    public static Preferences loadPreferences(String filePath) {
        File file = new File(filePath);
        if (!file.exists()) {
            return new Preferences();
        }
        try (ObjectInputStream ois = new ObjectInputStream(new FileInputStream(file))) {
            return (Preferences) ois.readObject();
        } catch (Exception e) {
            System.err.println("Failed to load preferences, using defaults: " + e.getMessage());
            return new Preferences();
        }
    }
}

import java.util.*;

public class MahjongRuleValidator {

    public static boolean canPung(List<Tile> hand, Tile discard) {
        long count = hand.stream().filter(t -> t.equals(discard)).count();
        return count >= 2;
    }

    public static boolean canKong(List<Tile> hand, Tile discard) {
        long count = hand.stream().filter(t -> t.equals(discard)).count();
        return count >= 3;
    }

    public static boolean canChow(List<Tile> hand, Tile discard) {
        if (discard.isHonor()) {
            return false;
        }
        Suit s = discard.getSuit();
        int r = discard.getRank();

        boolean hasMinus2 = containsSuitRank(hand, s, r - 2);
        boolean hasMinus1 = containsSuitRank(hand, s, r - 1);
        boolean hasPlus1 = containsSuitRank(hand, s, r + 1);
        boolean hasPlus2 = containsSuitRank(hand, s, r + 2);

        // Pattern 1: [r-2, r-1, r]
        if (hasMinus2 && hasMinus1) return true;
        // Pattern 2: [r-1, r, r+1]
        if (hasMinus1 && hasPlus1) return true;
        // Pattern 3: [r, r+1, r+2]
        if (hasPlus1 && hasPlus2) return true;

        return false;
    }

    public static List<List<Tile>> getChowCombinations(List<Tile> hand, Tile discard) {
        List<List<Tile>> combos = new ArrayList<>();
        if (discard.isHonor()) return combos;

        Suit s = discard.getSuit();
        int r = discard.getRank();

        if (containsSuitRank(hand, s, r - 2) && containsSuitRank(hand, s, r - 1)) {
            combos.add(List.of(findTile(hand, s, r - 2), findTile(hand, s, r - 1), discard));
        }
        if (containsSuitRank(hand, s, r - 1) && containsSuitRank(hand, s, r + 1)) {
            combos.add(List.of(findTile(hand, s, r - 1), discard, findTile(hand, s, r + 1)));
        }
        if (containsSuitRank(hand, s, r + 1) && containsSuitRank(hand, s, r + 2)) {
            combos.add(List.of(discard, findTile(hand, s, r + 1), findTile(hand, s, r + 2)));
        }
        return combos;
    }

    private static boolean containsSuitRank(List<Tile> hand, Suit suit, int rank) {
        return hand.stream().anyMatch(t -> t.getSuit() == suit && t.getRank() == rank);
    }

    private static Tile findTile(List<Tile> hand, Suit suit, int rank) {
        return hand.stream().filter(t -> t.getSuit() == suit && t.getRank() == rank).findFirst().orElse(null);
    }

    public static boolean checkMahjong(List<Tile> handTiles, List<Meld> melds, Tile winningTile, RuleVariation variation) {
        List<Tile> fullHand = new ArrayList<>(handTiles);
        if (winningTile != null) {
            fullHand.add(winningTile);
        }
        Collections.sort(fullHand);

        // Mahjong requires 14 tiles equivalent: 4 melds + 1 pair
        // (melds count towards the 4 melds)
        int neededMelds = 4 - melds.size();
        if (fullHand.size() != neededMelds * 3 + 2) {
            return false;
        }

        if (variation == RuleVariation.CLEARED_HAND) {
            if (!isClearedHand(fullHand, melds)) {
                return false;
            }
        }

        return canFormMeldsAndPair(fullHand, neededMelds);
    }

    private static boolean isClearedHand(List<Tile> hand, List<Meld> melds) {
        // Cleared Hand (Flush/Half-flush or no honors): only 1 suit (+ optional honors)
        Suit primarySuit = null;
        for (Tile t : hand) {
            if (!t.isHonor()) {
                if (primarySuit == null) {
                    primarySuit = t.getSuit();
                } else if (primarySuit != t.getSuit()) {
                    return false;
                }
            }
        }
        for (Meld m : melds) {
            for (Tile t : m.getTiles()) {
                if (!t.isHonor()) {
                    if (primarySuit == null) {
                        primarySuit = t.getSuit();
                    } else if (primarySuit != t.getSuit()) {
                        return false;
                    }
                }
            }
        }
        return true;
    }

    private static boolean canFormMeldsAndPair(List<Tile> tiles, int neededMelds) {
        if (tiles.size() == 2) {
            return tiles.get(0).equals(tiles.get(1));
        }

        // Try every possible pair
        Set<Tile> checkedPairs = new HashSet<>();
        for (int i = 0; i < tiles.size() - 1; i++) {
            Tile t1 = tiles.get(i);
            Tile t2 = tiles.get(i + 1);
            if (t1.equals(t2) && !checkedPairs.contains(t1)) {
                checkedPairs.add(t1);
                List<Tile> remainder = new ArrayList<>(tiles);
                remainder.remove(t1);
                remainder.remove(t2);
                if (canFormMelds(remainder, neededMelds)) {
                    return true;
                }
            }
        }
        return false;
    }

    private static boolean canFormMelds(List<Tile> tiles, int meldsRemaining) {
        if (meldsRemaining == 0 && tiles.isEmpty()) {
            return true;
        }
        if (tiles.size() < 3) return false;

        Tile first = tiles.get(0);

        // Try Pung: 3 identical tiles
        long identicalCount = tiles.stream().filter(t -> t.equals(first)).count();
        if (identicalCount >= 3) {
            List<Tile> remainder = new ArrayList<>(tiles);
            remainder.remove(first);
            remainder.remove(first);
            remainder.remove(first);
            if (canFormMelds(remainder, meldsRemaining - 1)) {
                return true;
            }
        }

        // Try Chow: only for suited tiles
        if (!first.isHonor()) {
            Suit s = first.getSuit();
            int r = first.getRank();
            Tile second = findTile(tiles, s, r + 1);
            Tile third = findTile(tiles, s, r + 2);
            if (second != null && third != null) {
                List<Tile> remainder = new ArrayList<>(tiles);
                remainder.remove(first);
                remainder.remove(second);
                remainder.remove(third);
                if (canFormMelds(remainder, meldsRemaining - 1)) {
                    return true;
                }
            }
        }

        return false;
    }

    public static boolean isFishing(List<Tile> hand, List<Meld> melds, RuleVariation variation) {
        // Fishing (Ready hand / Tenpai): hand needs 1 tile to form Mahjong
        List<Tile> sampleTiles = TileSet.createStandardTileSet();
        Set<Tile> tested = new HashSet<>();
        for (Tile t : sampleTiles) {
            if (tested.add(t)) {
                if (checkMahjong(hand, melds, t, variation)) {
                    return true;
                }
            }
        }
        return false;
    }

    public static int calculateScore(List<Tile> hand, List<Meld> melds, Wind roundWind, Wind seatWind) {
        // Base Chinese score calculation
        int score = 20; // Base points for winning hand
        for (Meld m : melds) {
            if (m.getType() == MeldType.PUNG) {
                score += m.isConcealed() ? 8 : 4;
            } else if (m.getType() == MeldType.KONG) {
                score += m.isConcealed() ? 32 : 16;
            }
        }
        return Math.max(score, 20);
    }
}

import java.util.*;

public class ComputerOpponentAI {

    public static Tile chooseDiscard(Player co, List<Tile> discardedTiles, RuleVariation variation) {
        List<Tile> hand = co.getHand();
        if (hand.isEmpty()) return null;

        COLevel level = co.getCoLevel();
        COStyle style = co.getCoStyle();

        // Level 1: Beginner
        if (level == COLevel.BEGINNER) {
            // Discard isolated honor tiles first, then random
            for (Tile t : hand) {
                if (t.isHonor() && Collections.frequency(hand, t) == 1) {
                    return t;
                }
            }
            return hand.get(0);
        }

        // Level 2 & 3: Intermediate & Advanced with Look-ahead / Strategy
        // Rate tiles by utility to preserve pairs, partial sequences (Chows), etc.
        Map<Tile, Double> tileScores = new HashMap<>();

        for (Tile t : hand) {
            double score = 0.0;
            int count = Collections.frequency(hand, t);

            // Pair or triplet protection
            if (count >= 2) {
                score += count * 30.0;
            }

            // Chow potential
            if (!t.isHonor()) {
                Suit s = t.getSuit();
                int r = t.getRank();
                boolean neighbor1 = hand.stream().anyMatch(h -> h.getSuit() == s && Math.abs(h.getRank() - r) == 1);
                boolean neighbor2 = hand.stream().anyMatch(h -> h.getSuit() == s && Math.abs(h.getRank() - r) == 2);
                if (neighbor1) score += 20.0;
                if (neighbor2) score += 10.0;
            }

            // Advanced look-ahead: consider discarded tiles visibility
            if (level == COLevel.ADVANCED) {
                long seenInDiscards = discardedTiles.stream().filter(d -> d.equals(t)).count();
                // Fewer unseen tiles remaining makes forming sets harder
                score -= seenInDiscards * 5.0;

                // Adjust based on style
                if (style == COStyle.DEFENSIVE) {
                    // Favor keeping safe honor tiles if already discarded by others
                    if (seenInDiscards > 0) score += 15.0;
                } else if (style == COStyle.AGGRESSIVE) {
                    // Aggressive pushes for high suited sequences
                    if (!t.isHonor() && (t.getRank() >= 3 && t.getRank() <= 7)) {
                        score += 8.0;
                    }
                }
            }

            tileScores.put(t, score);
        }

        // Discard tile with lowest score
        Tile bestDiscard = hand.get(0);
        double minScore = Double.MAX_VALUE;
        for (Tile t : hand) {
            double s = tileScores.getOrDefault(t, 0.0);
            if (s < minScore) {
                minScore = s;
                bestDiscard = t;
            }
        }
        return bestDiscard;
    }

    public static boolean shouldClaimDiscard(Player co, Tile discard, MeldType claimType) {
        if (co.getCoLevel() == COLevel.BEGINNER) {
            return claimType == MeldType.PUNG || claimType == MeldType.KONG;
        }
        // Intermediate & Advanced prefer Pung/Kong or Chows if aggressive/balanced
        if (co.getCoStyle() == COStyle.DEFENSIVE && claimType == MeldType.CHOW) {
            return false; // Defensive prefers closed hand
        }
        return true;
    }
}

        long humanCount = players.stream().filter(p -> !p.isComputer()).count();
        return (int) Math.max(0, totalHumanCapacity - humanCount);
    }

    public synchronized boolean addPlayer(Player player) {
        if (players.size() >= 4) return false;
        players.add(player);
        return true;
    }

    public synchronized void startGame() {
        // Complete remaining spots with COs up to 4 players
        int coIndex = 1;
        while (players.size() < 4) {
            Player co = new Player("CO_" + coIndex, "Computer " + coIndex, "Icon_CO.png", true);
            co.setCoLevel(defaultCOLevel);
            players.add(co);
            coIndex++;
        }

        // Randomise seating position
        Collections.shuffle(players);
        Wind[] winds = Wind.values();
        for (int i = 0; i < 4; i++) {
            players.get(i).setSeatWind(winds[i]);
        }

        // Randomise tile order
        wall.clear();
        wall.addAll(TileSet.createStandardTileSet());
        Collections.shuffle(wall);

        // Deal tiles: 13 each, East (dealer) starts with 14
        for (Player p : players) {
            p.getHand().clear();
            p.getMelds().clear();
            for (int i = 0; i < 13; i++) {
                p.addTile(wall.remove(0));
            }
        }
        // East player picks up 1st tile (turn 0)
        currentTurnIndex = 0;
        players.get(0).addTile(wall.remove(0));

        active = true;
        finished = false;
    }

    public synchronized Tile drawTile(Player player) {
        if (wall.isEmpty()) {
            return null; // Wall depleted -> Draw
        }
        Tile drawn = wall.remove(0);
        player.addTile(drawn);
        return drawn;
    }

    public synchronized void discardTile(Player player, Tile tile) {
        player.removeTile(tile);
        discards.add(tile);
        lastDiscardedTile = tile;
        lastDiscarderIndex = players.indexOf(player);

        // Check if player is fishing
        boolean fishing = MahjongRuleValidator.isFishing(player.getHand(), player.getMelds(), ruleVariation);
        player.setFishing(fishing);

        // Advance turn clockwise
        currentTurnIndex = (currentTurnIndex + 1) % 4;
    }

    public synchronized void performMeld(Player claimant, Meld meld) {
        claimant.addMeld(meld);
        for (Tile t : meld.getTiles()) {
            claimant.removeTile(t);
        }
        currentTurnIndex = players.indexOf(claimant);
    }

    public synchronized void replacePlayerWithCO(Player humanPlayer, COLevel level) {
        int index = players.indexOf(humanPlayer);
        if (index >= 0) {
            Player co = new Player("CO_sub_" + System.currentTimeMillis(), "CO (sub for " + humanPlayer.getName() + ")", "Icon_CO.png", true);
            co.setSeatWind(humanPlayer.getSeatWind());
            co.setCoLevel(level);
            co.getHand().addAll(humanPlayer.getHand());
            co.getMelds().addAll(humanPlayer.getMelds());
            co.addScore(humanPlayer.getScore());
            co.setFishing(humanPlayer.isFishing());
            players.set(index, co);
        }
    }

    public synchronized void endRound(Player winner) {
        if (winner != null) {
            int score = MahjongRuleValidator.calculateScore(winner.getHand(), winner.getMelds(), currentRoundWind, winner.getSeatWind());
            winner.addScore(score);
        }
        roundCount++;
        if (roundCount > 4) {
            finished = true;
            active = false;
        } else {
            // Next round wind
            currentRoundWind = Wind.values()[(currentRoundWind.ordinal() + 1) % 4];
        }
    }

    public synchronized void terminateGame() {
        active = false;
        finished = true;
    }

    public GameSnapshot createSnapshot(Player userPlayer) {
        return new GameSnapshot(gameId, currentTurnIndex, currentRoundWind, wall, userPlayer.getHand(), discards, scoreLimit);
    }

    public void restoreSnapshot(Player userPlayer, GameSnapshot snapshot) {
        this.currentTurnIndex = snapshot.getCurrentTurnIndex();
        this.currentRoundWind = snapshot.getRoundWind();
        this.wall.clear();
        this.wall.addAll(snapshot.getWall());
        this.discards.clear();
        this.discards.addAll(snapshot.getDiscards());
        userPlayer.getHand().clear();
        userPlayer.getHand().addAll(snapshot.getUserHand());
        userPlayer.sortHand();
    }
}

    }

    // User premature leave replacement with CO
    public void leaveGamePrematurely() {
        if (currentGame != null && !isSinglePlayerMode) {
            System.out.println("[Client " + clientId + "] Player left game prematurely. Creating substitute CO...");
            currentGame.replacePlayerWithCO(localPlayer, preferences.getDefaultCOLevel());
            this.currentGame = null;
        }
    }

    // End current game
    public void endGame() {
        if (currentGame != null) {
            currentGame.terminateGame();
            System.out.println("[Client " + clientId + "] Game terminated. You can now quit or start/join a new game.");
        }
    }
}
