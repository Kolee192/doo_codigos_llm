public enum DocumentFormat {
    PDF,
    PS,
    ODT;

    public static DocumentFormat fromFileName(String filename) {
        if (filename == null || !filename.contains(".")) {
            throw new IllegalArgumentException("Invalid file name: " + filename);
        }
        String ext = filename.substring(filename.lastIndexOf('.') + 1).toUpperCase();
        try {
            return DocumentFormat.valueOf(ext);
        } catch (IllegalArgumentException e) {
            throw new IllegalArgumentException("Unsupported document format: " + ext + ". Supported formats: PDF, PS, ODT");
        }
    }
}

import java.util.Collections;
import java.util.List;

public class PagedResult<T> {
    private final List<T> allItems;
    private final int pageSize;
    private int currentPage;

    public PagedResult(List<T> allItems, int pageSize) {
        this.allItems = allItems != null ? allItems : Collections.emptyList();
        this.pageSize = Math.max(pageSize, 1);
        this.currentPage = 0;
    }

    public int getTotalPages() {
        if (allItems.isEmpty()) {
            return 1;
        }
        return (int) Math.ceil((double) allItems.size() / pageSize);
    }

    public int getCurrentPage() {
        return currentPage;
    }

    public boolean hasNextPage() {
        return currentPage < getTotalPages() - 1;
    }

    public boolean hasPreviousPage() {
        return currentPage > 0;
    }

    public void nextPage() {
        if (hasNextPage()) {
            currentPage++;
        }
    }

    public void previousPage() {
        if (hasPreviousPage()) {
            currentPage--;
        }
    }

    public List<T> getCurrentPageItems() {
        if (allItems.isEmpty()) {
            return Collections.emptyList();
        }
        int fromIndex = currentPage * pageSize;
        int toIndex = Math.min(fromIndex + pageSize, allItems.size());
        if (fromIndex >= allItems.size()) {
            return Collections.emptyList();
        }
        return allItems.subList(fromIndex, toIndex);
    }

    public List<T> getAllItems() {
        return Collections.unmodifiableList(allItems);
    }

    public int getTotalResults() {
        return allItems.size();
    }
}

import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;

public class DocumentParser {

    public static class ParsedDocument {
        private String title;
        private List<Author> authors;
        private int publicationYear;
        private String abstractText;
        private String fullText;
        private List<Reference> references;

        public ParsedDocument(String title, List<Author> authors, int publicationYear,
                              String abstractText, String fullText, List<Reference> references) {
            this.title = title;
            this.authors = authors;
            this.publicationYear = publicationYear;
            this.abstractText = abstractText;
            this.fullText = fullText;
            this.references = references;
        }

        public String getTitle() { return title; }
        public List<Author> getAuthors() { return authors; }
        public int getPublicationYear() { return publicationYear; }
        public String getAbstractText() { return abstractText; }
        public String getFullText() { return fullText; }
        public List<Reference> getReferences() { return references; }
    }

    public ParsedDocument parse(String filename, byte[] documentData, String defaultAuthor) {
        DocumentFormat format = DocumentFormat.fromFileName(filename);
        String rawContent = new String(documentData, StandardCharsets.UTF_8);

        String title = filename.substring(0, filename.lastIndexOf('.')).replace("_", " ");
        String abstractText = "Abstract extracted from " + format + " file.";
        String fullText = rawContent;
        int year = 2023;

        // Try to parse structured content if available
        for (String line : rawContent.split("\n")) {
            String trimmed = line.trim();
            if (trimmed.startsWith("Title:")) {
                title = trimmed.substring("Title:".length()).trim();
            } else if (trimmed.startsWith("Abstract:")) {
                abstractText = trimmed.substring("Abstract:".length()).trim();
            } else if (trimmed.startsWith("Year:")) {
                try {
                    year = Integer.parseInt(trimmed.substring("Year:".length()).trim());
                } catch (NumberFormatException ignored) {}
            }
        }

        List<Author> authors = new ArrayList<>();
        authors.add(new Author("auth-" + defaultAuthor, defaultAuthor, "VUB Department of Computer Science", defaultAuthor + "@vub.be"));

        List<Reference> references = new ArrayList<>();
        return new ParsedDocument(title, authors, year, abstractText, fullText, references);
    }
}

import java.util.List;

public class ReferenceExporter {

    public static String exportToBib(List<Reference> references) {
        StringBuilder sb = new StringBuilder();
        for (Reference ref : references) {
            String authorStr = ref.getAuthors().isEmpty() ? "Unknown" : ref.getAuthors().get(0).getName();
            String citeKey = (authorStr.replaceAll("\\s+", "") + ref.getPublicationYear()).toLowerCase();
            sb.append("@article{").append(citeKey).append(",\n")
              .append("  title = {").append(ref.getTitle()).append("},\n")
              .append("  author = {").append(authorStr).append("},\n")
              .append("  year = {").append(ref.getPublicationYear()).append("},\n")
              .append("  abstract = {").append(ref.getAbstractText()).append("}\n")
              .append("}\n\n");
        }
        return sb.toString();
    }

    public static String exportToRis(List<Reference> references) {
        StringBuilder sb = new StringBuilder();
        for (Reference ref : references) {
            sb.append("TY  - JOUR\n");
            sb.append("TI  - ").append(ref.getTitle()).append("\n");
            for (Author a : ref.getAuthors()) {
                sb.append("AU  - ").append(a.getName()).append("\n");
            }
            sb.append("PY  - ").append(ref.getPublicationYear()).append("\n");
            sb.append("AB  - ").append(ref.getAbstractText()).append("\n");
            sb.append("ER  - \n\n");
        }
        return sb.toString();
    }
}

import java.util.List;
import java.util.Objects;
import java.util.stream.Collectors;

public class UserService {
    private final UserRepository userRepository;
    private final EmailService emailService;
    private final SystemConfiguration systemConfiguration;

    public UserService(UserRepository userRepository, EmailService emailService, SystemConfiguration systemConfiguration) {
        this.userRepository = userRepository;
        this.emailService = emailService;
        this.systemConfiguration = systemConfiguration;
    }

    public User login(String username, String password, UserSession session) {
        User user = userRepository.findByUsername(username)
                .orElseThrow(() -> new IllegalArgumentException("Invalid username or password."));
        if (!user.getPassword().equals(password)) {
            throw new IllegalArgumentException("Invalid username or password.");
        }
        session.setCurrentUser(user);
        return user;
    }

    public void register(String username, String password, String name, String email,
                         String university, String department) {
        if (userRepository.exists(username)) {
            throw new IllegalArgumentException("Username already exists in the system: " + username);
        }
        User newUser = new User(username, password, name, email, university, department, UserRole.MEMBER, "Default");
        userRepository.save(newUser);

        String adminEmail = systemConfiguration.get("admin.email");
        emailService.sendEmail(
                adminEmail,
                "New User Registration Request",
                "New user registered: " + username + " (" + name + ", " + email + ", " + university + ", " + department + ")"
        );
    }

    public void requestPasswordChange(String username, String newPassword) {
        User user = userRepository.findByUsername(username)
                .orElseThrow(() -> new IllegalArgumentException("No valid username found on the system: " + username));
        user.setPassword(newPassword);
        userRepository.save(user);

        String adminEmail = systemConfiguration.get("admin.email");
        emailService.sendEmail(
                adminEmail,
                "Password Change Notification",
                "User " + username + " has requested and updated their password."
        );
    }

    public void editPersonalData(UserSession session, String newPassword, String university, String department, String name) {
        if (!session.isLoggedIn()) {
            throw new IllegalStateException("Must be logged in to edit personal data.");
        }
        User current = session.getCurrentUser();
        if (newPassword != null && !newPassword.isBlank()) {
            current.setPassword(newPassword);
        }
        if (university != null && !university.isBlank()) {
            current.setUniversity(university);
        }
        if (department != null && !department.isBlank()) {
            current.setDepartment(department);
        }
        if (name != null && !name.isBlank()) {
            current.setName(name);
        }
        userRepository.save(current);
    }

    public List<User> searchUsers(UserSession session, UserSearchCriteria criteria) {
        ensureModeratorOrAdmin(session);
        return userRepository.findAll().stream()
                .filter(u -> criteria.getUsername() == null || u.getUsername().toLowerCase().contains(criteria.getUsername().toLowerCase()))
                .filter(u -> criteria.getName() == null || u.getName().toLowerCase().contains(criteria.getName().toLowerCase()))
                .filter(u -> criteria.getEmail() == null || u.getEmail().toLowerCase().contains(criteria.getEmail().toLowerCase()))
                .filter(u -> criteria.getRole() == null || u.getRole() == criteria.getRole())
                .collect(Collectors.toList());
    }

    public void editUserSettings(UserSession session, String username, UserRole newRole, String newGroupName, String newEmail) {

        groupRepository.save(group);

        return groupRepository.findAll().stream()
                .filter(g -> nameFilter == null || g.getName().toLowerCase().contains(nameFilter.toLowerCase()))
                .collect(Collectors.toList());
    }

        ensureAdmin(session);
        Group group = groupRepository.findByName(groupName)
                .orElseThrow(() -> new IllegalArgumentException("Group does not exist: " + groupName));
        if (!group.isEmpty()) {
            throw new IllegalStateException("Group is not empty: cannot delete group with existing members.");
        }
        groupRepository.delete(groupName);
    }

    public void editGroupSettings(UserSession session, String groupName, String newDescription) {
        ensureAdmin(session);
        Group group = groupRepository.findByName(groupName)
                .orElseThrow(() -> new IllegalArgumentException("Group does not exist: " + groupName));
        if (newDescription != null) {
            group.setDescription(newDescription);
        }
        groupRepository.save(group);
    }

    private void ensureAdmin(UserSession session) {
        if (!session.isLoggedIn() || session.getCurrentUser().getRole() != UserRole.ADMINISTRATOR) {
            throw new SecurityException("Access denied: Administrator privileges required.");
        }
    }

import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Set;
import java.util.UUID;
import java.util.stream.Collectors;

public class PublicationService {
    private final PublicationRepository publicationRepository;
    private final DocumentParser documentParser;

    public PublicationService(PublicationRepository publicationRepository, DocumentParser documentParser) {
        this.publicationRepository = publicationRepository;
        this.documentParser = documentParser;
    }

    public PagedResult<Publication> search(UserSession session, PublicationSearchCriteria criteria, int pageSize) {
        ensureCanAccessPublications(session);
        List<Publication> filtered = publicationRepository.findAll().stream()
                .filter(p -> criteria.getKeyword() == null ||
                        p.getTitle().toLowerCase().contains(criteria.getKeyword().toLowerCase()) ||
                        p.getAbstractText().toLowerCase().contains(criteria.getKeyword().toLowerCase()))
                .filter(p -> criteria.getAuthorName() == null ||
                        p.getAuthors().stream().anyMatch(a -> a.getName().toLowerCase().contains(criteria.getAuthorName().toLowerCase())))
                .filter(p -> criteria.getStartYear() == null || p.getPublicationYear() >= criteria.getStartYear())
                .filter(p -> criteria.getEndYear() == null || p.getPublicationYear() <= criteria.getEndYear())
                .filter(p -> criteria.getReferenceKeyword() == null ||
                        p.getReferences().stream().anyMatch(r -> r.getTitle().toLowerCase().contains(criteria.getReferenceKeyword().toLowerCase())))
                .collect(Collectors.toList());

        return new PagedResult<>(filtered, pageSize);
    }

    public Publication viewDetails(UserSession session, String publicationId) {
        ensureCanAccessPublications(session);
        return publicationRepository.findById(publicationId)
                .orElseThrow(() -> new IllegalArgumentException("Publication not found: " + publicationId));
    }

    public List<Publication> viewMultipleDetails(UserSession session, List<String> publicationIds) {
        ensureCanAccessPublications(session);
        List<Publication> result = new ArrayList<>();
        for (String id : publicationIds) {
            publicationRepository.findById(id).ifPresent(result::add);
        }
        return result;
    }

    public byte[] downloadPublication(UserSession session, String publicationId) {
        ensureCanAccessPublications(session);
        Publication p = viewDetails(session, publicationId);
        return p.getDocumentData();
    }

    public List<byte[]> downloadMultiplePublications(UserSession session, List<String> publicationIds) {
        ensureCanAccessPublications(session);
        List<byte[]> files = new ArrayList<>();
        for (String id : publicationIds) {
            files.add(downloadPublication(session, id));
        }
        return files;
    }

    public List<Reference> consultReferences(UserSession session, String publicationId) {
        ensureCanAccessPublications(session);
        Publication p = viewDetails(session, publicationId);
        return p.getReferences();
    }

    public List<Reference> consultReferencesForMultiple(UserSession session, List<String> publicationIds) {
        ensureCanAccessPublications(session);
        Set<Reference> references = new LinkedHashSet<>();
        for (String id : publicationIds) {
            references.addAll(consultReferences(session, id));
        }
        return new ArrayList<>(references);
    }

    public String displayFullText(UserSession session, String publicationId) {
        ensureCanAccessPublications(session);
        Publication p = viewDetails(session, publicationId);
        if (p.getFullText() == null || p.getFullText().isBlank()) {
            throw new IllegalStateException("Full text is not available in the system for this publication.");
        }
        return p.getFullText();
    }

    public Publication viewReferenceDetails(UserSession session, String publicationId, String referenceId) {
        ensureCanAccessPublications(session);
        Publication parent = viewDetails(session, publicationId);
        Reference reference = parent.getReferences().stream()
                .filter(r -> r.getId().equals(referenceId))
                .findFirst()
                .orElseThrow(() -> new IllegalArgumentException("Reference not found in publication: " + referenceId));

        if (reference.getTargetPublicationId() == null) {
            throw new IllegalArgumentException("Reference is not registered as a publication in the system.");
        }

        return publicationRepository.findById(reference.getTargetPublicationId())
                .orElseThrow(() -> new IllegalArgumentException("Referenced publication does not exist in the system."));
    }

    public String exportReferencesBib(UserSession session, String publicationId) {
        List<Reference> references = consultReferences(session, publicationId);
        return ReferenceExporter.exportToBib(references);
    }

    public String exportReferencesRis(UserSession session, String publicationId) {
        List<Reference> references = consultReferences(session, publicationId);
        return ReferenceExporter.exportToRis(references);
    }

    public Author viewAuthor(UserSession session, String publicationId, String authorName) {
        ensureCanAccessPublications(session);
        Publication p = viewDetails(session, publicationId);
        return p.getAuthors().stream()
                .filter(a -> a.getName().equalsIgnoreCase(authorName))
                .findFirst()
                .orElseThrow(() -> new IllegalArgumentException("Author not found on publication: " + authorName));
    }

    public List<Publication> getMyPublications(UserSession session) {
        ensurePublisherOrAbove(session);
        String username = session.getCurrentUser().getUsername();
        return publicationRepository.findAll().stream()
                .filter(p -> p.getOwnerUsername() != null && p.getOwnerUsername().equalsIgnoreCase(username))
                .collect(Collectors.toList());
    }

    public void editOwnPublication(UserSession session, String publicationId, String newTitle, String newAbstract, int newYear) {
        ensurePublisherOrAbove(session);
        Publication publication = publicationRepository.findById(publicationId)
                .orElseThrow(() -> new IllegalArgumentException("Publication not found: " + publicationId));

        boolean isOwner = session.getCurrentUser().getUsername().equalsIgnoreCase(publication.getOwnerUsername());
        boolean isAdminOrMod = session.getCurrentUser().getRole() == UserRole.ADMINISTRATOR ||
                              session.getCurrentUser().getRole() == UserRole.MODERATOR;

        if (!isOwner && !isAdminOrMod) {
            throw new SecurityException("Only owner or Moderator/Admin can edit this publication.");
        }

        if (newTitle != null) publication.setTitle(newTitle);
        if (newAbstract != null) publication.setAbstractText(newAbstract);
        if (newYear > 0) publication.setPublicationYear(newYear);

        publicationRepository.save(publication);
    }

    public DocumentParser.ParsedDocument inspectDocumentForUpload(UserSession session, String filename, byte[] documentData) {
        ensurePublisherOrAbove(session);
        return documentParser.parse(filename, documentData, session.getCurrentUser().getUsername());
    }

    public Publication confirmUploadPublication(UserSession session, String filename, byte[] documentData,
                                                String title, String abstractText, int publicationYear) {
        ensurePublisherOrAbove(session);
        DocumentFormat format = DocumentFormat.fromFileName(filename);
        DocumentParser.ParsedDocument parsed = documentParser.parse(filename, documentData, session.getCurrentUser().getUsername());

        String pubId = "PUB-" + UUID.randomUUID().toString().substring(0, 8);
        String finalTitle = (title != null && !title.isBlank()) ? title : parsed.getTitle();
        String finalAbstract = (abstractText != null && !abstractText.isBlank()) ? abstractText : parsed.getAbstractText();
        int finalYear = (publicationYear > 0) ? publicationYear : parsed.getPublicationYear();

        Publication publication = new Publication(
                pubId,
                finalTitle,
                parsed.getAuthors(),
                finalYear,
                finalAbstract,
                parsed.getFullText(),
                format,
                documentData,
                parsed.getReferences(),
                session.getCurrentUser().getUsername()
        );

        publicationRepository.save(publication);
        return publication;
    }

    private void ensureCanAccessPublications(UserSession session) {
        if (!session.isLoggedIn() && !session.canSearchWithoutLogin()) {
            throw new SecurityException("Access denied: Please login from the internet or connect via VUB-network.");
        }
    }

    private void ensurePublisherOrAbove(UserSession session) {
        if (!session.isLoggedIn() ||
                (session.getCurrentUser().getRole() != UserRole.PUBLISHER &&
                 session.getCurrentUser().getRole() != UserRole.MODERATOR &&
                 session.getCurrentUser().getRole() != UserRole.ADMINISTRATOR)) {
            throw new SecurityException("Access denied: Publisher, Moderator or Administrator role required.");
        }
    }
}
