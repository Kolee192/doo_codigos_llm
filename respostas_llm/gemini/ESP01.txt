public class Hearing {
    private final String hearingId;

    private LocalDateTime scheduledDateTime;
    private final String governingBody;
    private final List<String> requiredAffidavits;
    private final List<String> satisfiedAffidavits;
    private boolean applicantReady;
    private boolean staffReady;
    private String decision;
    private String appealOutcome;

    public Hearing(String hearingId, String applicationId, LocalDateTime scheduledDateTime, String governingBody) {
        this.hearingId = hearingId;
        this.applicationId = applicationId;
        this.scheduledDateTime = scheduledDateTime;
        this.governingBody = governingBody;
        this.requiredAffidavits = new ArrayList<>();
        this.satisfiedAffidavits = new ArrayList<>();
        this.applicantReady = false;
        this.staffReady = false;
    }

    public String getHearingId() {
        return hearingId;
    }

    public String getApplicationId() {
        return applicationId;
    }

    public LocalDateTime getScheduledDateTime() {
        return scheduledDateTime;
    }

    public void setScheduledDateTime(LocalDateTime scheduledDateTime) {
        this.scheduledDateTime = scheduledDateTime;
    }

    public String getGoverningBody() {
        return governingBody;
    }

    public void addRequiredAffidavit(String affidavitType) {
        this.requiredAffidavits.add(affidavitType);
    }

    public void satisfyAffidavit(String affidavitType) {
        if (!this.satisfiedAffidavits.contains(affidavitType)) {
            this.satisfiedAffidavits.add(affidavitType);
        }
    }

    public boolean areAffidavitsComplete() {
        return satisfiedAffidavits.containsAll(requiredAffidavits);
    }

    public boolean isApplicantReady() {
        return applicantReady;
    }

    public void setApplicantReady(boolean applicantReady) {
        this.applicantReady = applicantReady;
    }

    public boolean isStaffReady() {
        return staffReady;
    }

    public void setStaffReady(boolean staffReady) {
        this.staffReady = staffReady;
    }

    public String getDecision() {
        return decision;
    }

    public void setDecision(String decision) {
        this.decision = decision;
    }

    public String getAppealOutcome() {
        return appealOutcome;
    }

    public void setAppealOutcome(String appealOutcome) {
        this.appealOutcome = appealOutcome;
    }

    @Override
    public String toString() {
        return "Hearing{" + "hearingId='" + hearingId + '\'' + ", applicationId='" + applicationId + '\'' + ", scheduledDateTime=" + scheduledDateTime + ", governingBody='" + governingBody + '\'' + ", decision='" + decision + '\'' + ", appealOutcome='" + appealOutcome + '\'' + '}';

public class Inspection {

    private final String applicationId;
    private final String inspectionType;
    private String inspectorId;
    private LocalDateTime scheduledTime;
    private InspectionStatus status;
    private String inspectionResult;
    private String comments;

    public Inspection(String inspectionId, String applicationId, String inspectionType) {
        this.inspectionId = inspectionId;
        this.applicationId = applicationId;
        this.inspectionType = inspectionType;
        this.status = InspectionStatus.REQUESTED;
    }

    public String getInspectionId() {
        return inspectionId;
    }

    public String getApplicationId() {
        return applicationId;
    }

    public String getInspectionType() {
        return inspectionType;
    }

    public String getInspectorId() {
        return inspectorId;
    }

    public void setInspectorId(String inspectorId) {
        this.inspectorId = inspectorId;
        this.status = InspectionStatus.ASSIGNED;
    }

    public LocalDateTime getScheduledTime() {
        return scheduledTime;
    }

    public void setScheduledTime(LocalDateTime scheduledTime) {
        this.scheduledTime = scheduledTime;
        if (this.status == InspectionStatus.REQUESTED) {
            this.status = InspectionStatus.SCHEDULED;
        }
    }

    public InspectionStatus getStatus() {
        return status;
    }

    public void setStatus(InspectionStatus status) {
        this.status = status;
    }

    public String getInspectionResult() {
        return inspectionResult;
    }

    public String getComments() {
        return comments;
    }

    public void recordResults(boolean passed, String comments) {
        this.status = passed ? InspectionStatus.PASSED : InspectionStatus.FAILED;
        this.inspectionResult = passed ? "PASSED" : "FAILED";
        this.comments = comments;
    }

    @Override
    public String toString() {
        return "Inspection{" + "id='" + inspectionId + '\'' + ", type='" + inspectionType + '\'' + ", inspector='" + inspectorId + '\'' + ", status=" + status + ", result='" + inspectionResult + '\'' + '}';
    }

    // 1. Public User: Search for Information
    public List<String> searchPublicInformation(String keyword) {
        String query = keyword.toLowerCase();
        List<String> results = new ArrayList<>();

        for (Property p : properties.values()) {
            if (p.getParcelPin().toLowerCase().contains(query) || p.getAddress().toLowerCase().contains(query)) {
                results.add("Property: " + p.getParcelPin() + " (" + p.getAddress() + ")");
            }
        }
        for (CountyService s : services.values()) {
            if (s.getServiceName().toLowerCase().contains(query) || s.getDescription().toLowerCase().contains(query)) {
                results.add("Service: " + s.getServiceName() + " - " + s.getDescription());
            }
        }
        for (Ordinance o : ordinances.values()) {
            if (o.getTitle().toLowerCase().contains(query) || o.getDescription().toLowerCase().contains(query)) {
                results.add("Ordinance: " + o.getTitle() + " (" + o.getCodeSection() + ")");
            }
        }
        return results;
    }

    }

    // 41. Plan Review Staff: Review Code Modifications
    public CodeModificationRequest submitCodeModification(String applicationId, String codeSection, String justification) {
        CountyApplication app = applications.get(applicationId);
        if (app != null) {
            CodeModificationRequest req = new CodeModificationRequest(nextId("CMOD"), codeSection, justification);
            app.getCodeModifications().add(req);
            return req;
        }
        return null;
    }

    public void reviewCodeModification(String applicationId, String modificationId, boolean approved) {
        CountyApplication app = applications.get(applicationId);
        if (app != null) {
            for (CodeModificationRequest mod : app.getCodeModifications()) {
                if (mod.getModificationId().equals(modificationId)) {
                    mod.review(approved);
                }
            }
        }
    }

    // 42. Plan Review Staff: Track Completion of Required Plan Reviews
    public boolean areAllPlanReviewsApproved(String applicationId) {
        CountyApplication app = applications.get(applicationId);
        if (app == null || app.getPlanReviews().isEmpty()) {
            return false;
        }
        return app.getPlanReviews().stream().allMatch(p -> p.getStatus() == PlanReviewStatus.APPROVED);
    }

    // 43. Development Staff: Issue Development-related Permit
    public Permit issueDevelopmentPermit(String applicationId, String staffId) {
        CountyApplication app = applications.get(applicationId);
        if (app != null && areAllPlanReviewsApproved(applicationId) && app.isFeeFullyPaid() && !app.hasActiveHold()) {
            String permitNum = nextId("DEVPERMIT");
            Permit permit = new Permit(permitNum, applicationId, app.getParcelPin(), "BUILDING_DEVELOPMENT", staffId);
            permits.put(permitNum, permit);
            Property prop = properties.get(app.getParcelPin());
            if (prop != null) {
                prop.getPermitNumbers().add(permitNum);
            }
            return permit;
        }
        return null;
    }

    // 44. Staff: Manage Bonds
    public Bond postBond(String applicationId, double amount) {
        CountyApplication app = applications.get(applicationId);
        if (app != null) {
            Bond bond = new Bond(nextId("BND"), applicationId, amount);
            app.getBonds().add(bond);
            return bond;
        }
        return null;
    }

    public void releaseBond(String applicationId, String bondId) {
        CountyApplication app = applications.get(applicationId);
        if (app != null) {
            for (Bond b : app.getBonds()) {
                if (b.getBondId().equals(bondId)) {
                    b.setStatus(BondStatus.RELEASED);
                }
            }
        }
    }

    // 45. Applicant: Complete Building Development Project (Release bond & occupancy)
    public boolean completeDevelopmentProject(String applicationId) {
        CountyApplication app = applications.get(applicationId);
        if (app != null) {
            // Release all active bonds
            for (Bond b : app.getBonds()) {
                if (b.getStatus() == BondStatus.POSTED || b.getStatus() == BondStatus.HELD) {
                    b.setStatus(BondStatus.RELEASED);
                }
            }
            app.setOccupancyGranted(true);
            return true;
        }
        return false;
    }

    // 46. Customer: Request Inspection
    public Inspection requestInspection(String applicationId, String inspectionType) {
        Inspection inspection = new Inspection(nextId("INSP"), applicationId, inspectionType);
        inspections.put(inspection.getInspectionId(), inspection);
        return inspection;
    }

    // 47. Inspection Staff: Create Inspection
    public Inspection scheduleInspection(String inspectionId, LocalDateTime scheduledTime) {
        Inspection inspection = inspections.get(inspectionId);
        if (inspection != null) {
            inspection.setScheduledTime(scheduledTime);
        }
        return inspection;
    }

    // 48. Inspection Supervisor: Assign Inspections & Manage Workload
    public void assignInspection(String inspectionId, String inspectorId) {
        Inspection inspection = inspections.get(inspectionId);
        if (inspection != null) {
            inspection.setInspectorId(inspectorId);
        }
    }

    public Map<String, Long> getInspectorWorkload() {
        return inspections.values().stream()
                .filter(i -> i.getStatus() != InspectionStatus.PASSED && i.getInspectorId() != null)
                .collect(Collectors.groupingBy(Inspection::getInspectorId, Collectors.counting()));
    }

    // 49. Inspector: Conduct Inspection
    public void conductInspection(String inspectionId, boolean passed, String comments) {
        Inspection inspection = inspections.get(inspectionId);
        if (inspection != null) {
            inspection.recordResults(passed, comments);
        }
    }

    // 50. Enforcement Staff: Issue Notice of Violation
    public Violation issueNoticeOfViolation(String parcelPin, String responsibleParty, String description) {
        Violation violation = new Violation(nextId("VIO"), parcelPin, responsibleParty, description);
        violations.put(violation.getViolationId(), violation);
        return violation;
    }

    // 51. Staff: Search Property Information
    public Property searchProperty(String parcelPin) {
        return properties.get(parcelPin);
    }

    // 52. Staff: Geospatial Change Notification
    public void notifyGeospatialChange(String parcelPin, String description) {
        String alert = "[" + LocalDateTime.now() + "] Parcel " + parcelPin + " geospatial change: " + description;
        geospatialChangeAlerts.add(alert);
    }

    public List<String> getGeospatialChangeAlerts() {
        return new ArrayList<>(geospatialChangeAlerts);
    }

    // 53. County Staff: Custom Query / Report
    public List<CountyApplication> customApplicationQuery(Predicate<CountyApplication> filter) {
        return applications.values().stream().filter(filter).collect(Collectors.toList());
    }

    // 54. Staff: Create Condo Unit
    public CondoUnit createCondoUnit(String parentParcelPin, String childParcelPin, String unitNumber, String address) {
        Property parentProperty = properties.get(parentParcelPin);
        if (parentProperty != null) {
            CondoUnit unit = new CondoUnit(nextId("CND"), unitNumber, parentParcelPin, childParcelPin, address);
            parentProperty.getCondoUnits().add(unit);
            Property childProperty = new Property(childParcelPin, address, parentProperty.getCurrentOwner());
            properties.put(childParcelPin, childProperty);
            return unit;
        }
        return null;
    }

    // 55. Technical Staff: Update GIS Property Management (Split/Merged PIN)
    public void updateGisPropertyManagement(String oldParcelPin, String newParcelPin, String newGisLayer) {
        Property prop = properties.get(oldParcelPin);
        if (prop != null) {
            properties.remove(oldParcelPin);
            prop.setParcelPin(newParcelPin);
            prop.setGisLayerReference(newGisLayer);
            properties.put(newParcelPin, prop);
            notifyGeospatialChange(newParcelPin, "Split/Merged from " + oldParcelPin + " to GIS layer " + newGisLayer);
        }
    }

    // 56. Staff: Send or Post Citations and Fines
    public void postCitationAndFine(String violationId, double fineAmount) {
        Violation v = violations.get(violationId);
        if (v != null) {
            v.setFineAmount(fineAmount);
            v.setCitationPosted(true);
            v.setNoticeSent(true);
        }
    }

    // Helpers to bootstrap data
    public void addProperty(Property property) {
        properties.put(property.getParcelPin(), property);
    }

    public void addService(CountyService service) {
        services.put(service.getServiceId(), service);
    }

    public CountyApplication getApplication(String appId) {
        return applications.get(appId);
    }

    public Violation getViolation(String violationId) {
        return violations.get(violationId);
    }

    public Hearing getHearing(String hearingId) {
        return hearings.get(hearingId);
    }

    public Meeting getMeeting(String meetingId) {
        return meetings.get(meetingId);
    }
}

import java.time.LocalDate;

import java.util.List;

public class Main {
    public static void main(String[] args) {
        System.out.println("=== County Land Development & Permitting System Demo ===");
        CountyLandManagementSystem countySystem = new CountyLandManagementSystem();

        // 1. Initial Setup: Properties, Services, Ordinances
        Property prop1 = new Property("PIN-1001", "123 Main St", "John Landowner");
        countySystem.addProperty(prop1);

        CountyService rezoningService = new CountyService("SRV-REZONING", "Rezoning & Special Exception", "Zoning change request", 500.0);
        rezoningService.addRequirement("Site Survey");
        rezoningService.addRequirement("Traffic Study");
        countySystem.addService(rezoningService);

        CountyService buildingPermitService = new CountyService("SRV-BLDG", "Commercial Building Permit", "Building plan authorization", 1200.0);
        buildingPermitService.addRequirement("Architectural Plans");
        buildingPermitService.addRequirement("Structural Engineering Specs");
        countySystem.addService(buildingPermitService);

        countySystem.addOrUpdateOrdinance("ORD-2023-01", "Setback Requirements", "Minimum 25ft setback for commercial properties", "Sec. 14-B", LocalDate.of(2023, 1, 1));

        // 2. Public User Search
        System.out.println("\n--- 1. Public Search ---");
        List<String> searchResults = countySystem.searchPublicInformation("Main St");
        searchResults.forEach(System.out::println);

        // 3. User Registration
        System.out.println("\n--- 2. User Registration ---");
        User applicant = countySystem.registerUser("alice_dev", "Alice Developer", "alice@example.com", UserRole.APPLICANT);
        User planner = countySystem.registerUser("bob_planner", "Bob Smith", "bob@county.gov", UserRole.PLANNING_STAFF);
        User reviewer = countySystem.registerUser("carol_eng", "Carol Reviewer", "carol@county.gov", UserRole.PLAN_REVIEW_STAFF);
        User inspector = countySystem.registerUser("dave_insp", "Dave Inspector", "dave@county.gov", UserRole.INSPECTOR);
        System.out.println("Registered applicant: " + applicant);

        // 4. Pre-Application Assistance
        System.out.println("\n--- 3. Pre-Application Assistance ---");
        PreApplicationRequest preApp = countySystem.requestPreApplicationAssistance(applicant.getUserId(), "SRV-REZONING", "Discussing mixed-use development");
        System.out.println("Submitted pre-app assistance request: " + preApp.getRequestId());
        countySystem.respondToPreApplication(preApp.getRequestId(), "Preliminary review scheduled", LocalDateTime.now().plusDays(3));
        System.out.println("Staff responded: " + preApp.getResponseNotes());

        // 5. Submit Application, Docs, Pay Fee
        System.out.println("\n--- 4. Application Submission, Document, Fee ---");
        CountyApplication app = countySystem.submitApplication(applicant.getUserId(), "PIN-1001", "SRV-REZONING", "Mixed Use Residential/Retail");
        System.out.println("Application created: " + app.getApplicationId() + " Status: " + app.getStatus());

        countySystem.submitSupportingDocumentation(app.getApplicationId(), "Concept Site Plan", "PDF", applicant.getUserId());
        countySystem.payFee(app.getApplicationId(), 500.0);
        System.out.println("Applicant status check: " + countySystem.checkStatus(app.getApplicationId()));

        // 6. Completeness Review and Internal Note
        System.out.println("\n--- 5. Staff Completeness Review & Notes ---");
        countySystem.performCompletenessReview(app.getApplicationId(), true);
        countySystem.recordInternalNote(app.getApplicationId(), planner.getUserId(), "Documentation verified complete.");
        countySystem.assignApplicationForDetailedReview(app.getApplicationId(), planner.getUserId());
        System.out.println("After completeness: " + app.getStatus());

        // 7. Holds Management
        System.out.println("\n--- 6. Hold and Release ---");
        Hold hold = countySystem.applyHold(app.getApplicationId(), "Pending revised environmental assessment", planner.getUserId());
        System.out.println("Hold applied: " + hold.getHoldId() + ", Application status: " + app.getStatus());
        countySystem.removeHold(app.getApplicationId(), hold.getHoldId(), planner.getUserId());
        System.out.println("Hold released, status: " + app.getStatus());

        // 8. Planning Consistency & Staff Report
        System.out.println("\n--- 7. Planning Staff Review & Proffers ---");
        countySystem.evaluateComprehensivePlanConsistency(app.getApplicationId(), true);
        countySystem.performZoningStaffReview(app.getApplicationId(), true);
        Proffer proffer = countySystem.addDraftProffer(app.getApplicationId(), "Applicant shall dedicate 10ft easement for pedestrian walkway");
        countySystem.approveProffer(app.getApplicationId(), proffer.getProfferId(), true);
        countySystem.developStaffReport(app.getApplicationId(), "Favorable recommendation based on Master Plan compliance.");

        // 9. Hearings Management
        System.out.println("\n--- 8. Public Hearing Process ---");
        Hearing hearing = countySystem.assignHearingDate(app.getApplicationId(), LocalDateTime.now().plusWeeks(2), "Board of Zoning Appeals");
        countySystem.addRequiredAffidavit(hearing.getHearingId(), "Proof of Neighbor Notification");
        countySystem.satisfyHearingRequirement(hearing.getHearingId(), "Proof of Neighbor Notification");
        countySystem.prepareHearingExecution(hearing.getHearingId(), true, true);
        countySystem.captureHearingDecision(hearing.getHearingId(), "APPROVED");
        System.out.println("Hearing captured decision: " + hearing.getDecision() + ", App status: " + app.getStatus());

        // 10. Zoning Permit Issuance
        System.out.println("\n--- 9. Issue Zoning Permit ---");
        Permit zoningPermit = countySystem.issueZoningPermit(app.getApplicationId(), planner.getUserId());
        System.out.println("Zoning Permit Issued: " + zoningPermit);

        // 11. Plan Review and Workload
        System.out.println("\n--- 10. Plan Review Lifecycle ---");
        PlanReviewTask planTask = countySystem.assignPlanReview(app.getApplicationId(), "Civil Engineering Set v1", reviewer.getUserId());
        System.out.println("Plan review workload: " + countySystem.getPlanReviewerWorkload());
        countySystem.reviewPlan(app.getApplicationId(), planTask.getTaskId(), PlanReviewStatus.APPROVED, List.of("Maintain silt fencing"), null);
        System.out.println("Plan reviews all approved? " + countySystem.areAllPlanReviewsApproved(app.getApplicationId()));

        // 12. Bonds, Development Permit & Project Completion
        System.out.println("\n--- 11. Bonds, Development Permit & Occupancy ---");
        Bond bond = countySystem.postBond(app.getApplicationId(), 25000.0);
        Permit devPermit = countySystem.issueDevelopmentPermit(app.getApplicationId(), planner.getUserId());
        System.out.println("Development permit issued: " + devPermit);

        // Request and Conduct Inspection
        Inspection insp = countySystem.requestInspection(app.getApplicationId(), "Final Building Code Compliance");
        countySystem.assignInspection(insp.getInspectionId(), inspector.getUserId());
        countySystem.conductInspection(insp.getInspectionId(), true, "Construction adheres to all standards");
        System.out.println("Inspection conducted: " + insp);

        // Project Completion
        countySystem.completeDevelopmentProject(app.getApplicationId());
        countySystem.denoteFinalAction(app.getApplicationId(), true);
        System.out.println("Bond status: " + bond.getStatus() + ", Occupancy granted: " + app.isOccupancyGranted() + ", Final Action Status: " + app.getStatus());

        // 13. Enforcement, Notice of Violation & Citations
        System.out.println("\n--- 12. Enforcement & Citations ---");
        Violation violation = countySystem.issueNoticeOfViolation("PIN-1001", "John Landowner", "Unauthorized debris burning on site");
        countySystem.postCitationAndFine(violation.getViolationId(), 250.0);
        System.out.println("Violation processed: " + violation);

        // 14. Condo Unit Creation and GIS Updates
        System.out.println("\n--- 13. Condominiums & GIS Parcel Management ---");
        CondoUnit condo = countySystem.createCondoUnit("PIN-1001", "PIN-1001-C1", "Unit 101", "123 Main St, Suite 101");
        System.out.println("Created Condo: " + condo);
        countySystem.updateGisPropertyManagement("PIN-1001", "PIN-1001-M", "GIS_LAYER_PARCEL_SPLIT_2023");
        System.out.println("Geospatial alerts: " + countySystem.getGeospatialChangeAlerts());

        // 15. Custom Query / Reporting
        System.out.println("\n--- 14. Custom Query & Planning Report ---");
        List<CountyApplication> paidCompletedApps = countySystem.customApplicationQuery(
                a -> a.isFeeFullyPaid() && a.getStatus() == ApplicationStatus.FINAL_ACTION_COMPLETED
        );
        System.out.println("Custom query returned applications count: " + paidCompletedApps.size());
        System.out.println("Demo completed successfully.");
    }
