import java.util.*;

public class EmailService {
    private final List<String> sentEmails = new ArrayList<>();

    public void sendEmail(String from, String to, String subject, String body) {
        String logEntry = String.format("From: %s | To: %s | Subject: %s | Content: %s", from, to, subject, body);
        sentEmails.add(logEntry);
    }

    public List<String> getSentEmails() {
        return Collections.unmodifiableList(sentEmails);
    }

    public String getLastSentEmail() {
        if (sentEmails.isEmpty()) return null;
        return sentEmails.get(sentEmails.size() - 1);
    }
}

    private LocalDate stopPublishingDate;

    public NewsItem(String id, String title, String content, int priority, 
                    LocalDate startPublishingDate, LocalDate oldNewsDate, LocalDate stopPublishingDate) {
        this.id = id;
        this.title = title;
        this.content = content;
        this.priority = priority;

        this.stopPublishingDate = stopPublishingDate;
    }

    public String getId() { return id; }
    public String getTitle() { return title; }
    public void setTitle(String title) { this.title = title; }
    public String getContent() { return content; }
    public void setContent(String content) { this.content = content; }
    public int getPriority() { return priority; }
    public void setPriority(int priority) { this.priority = priority; }
    public LocalDate getStartPublishingDate() { return startPublishingDate; }
    public void setStartPublishingDate(LocalDate startPublishingDate) { this.startPublishingDate = startPublishingDate; }
    public LocalDate getOldNewsDate() { return oldNewsDate; }
    public void setOldNewsDate(LocalDate oldNewsDate) { this.oldNewsDate = oldNewsDate; }
    public LocalDate getStopPublishingDate() { return stopPublishingDate; }
    public void setStopPublishingDate(LocalDate stopPublishingDate) { this.stopPublishingDate = stopPublishingDate; }

    public boolean isCurrent(LocalDate today) {
        return (today.isEqual(startPublishingDate) || today.isAfter(startPublishingDate)) &&
               today.isBefore(oldNewsDate) &&
               today.isBefore(stopPublishingDate);
    }

    public boolean isOld(LocalDate today) {
        return (today.isEqual(oldNewsDate) || today.isAfter(oldNewsDate)) &&
               today.isBefore(stopPublishingDate);
    }
}

    public String viewProfile(String targetUserId, String viewerUserId) {
        User target = users.get(targetUserId);
        if (target == null) return "User not found";
        
        if (target.isProfilePrivate()) {
            return String.format("Profile: [Private] Name: %s", target.getFullName());
        }

        StringBuilder sb = new StringBuilder();
        sb.append(String.format("Profile of: %s %s\n", target.getFullName(), target.getCertificationBadge()));
        if (!target.isEmailPrivate()) {
            sb.append(String.format("Email: %s\n", target.getEmail()));
        } else {
            sb.append("Email: [Private]\n");
        }
        sb.append(String.format("Bio: %s\nLocation: %s\nCompany: %s\n", target.getBio(), target.getLocation(), target.getCompany()));
        if (target.getRole() == Role.PRACTITIONER || target.getCertifications().contains(CertificationType.CSP)) {
            sb.append(String.format("Practitioner Experience: %s\n", target.getExperienceDetails()));
        }
        if (target.getRole() == Role.TRAINER || target.getCertifications().contains(CertificationType.CST)) {
            sb.append(String.format("Trainer Philosophy: %s\n", target.getTrainerPhilosophy()));
            sb.append(String.format("Trainer Rating: %.2f / 5.0\n", getTrainerAverageRating(target.getId())));
            sb.append("Upcoming Classes:\n");
            for (CourseEvent ce : getUpcomingTrainerCourses(target.getId())) {
                sb.append(String.format(" - %s at %s (Starts: %s) [Link: /course/%s]\n", ce.getTitle(), ce.getLocation(), ce.getStartDate(), ce.getId()));
            }
        }
        return sb.toString();
    }

    public CertificationApplication applyForCertification(String userId, CertificationType certType, String summary) {

    public List<CertificationApplication> getPendingApplications(String adminUserId) {

        return applications.values().stream().filter(a -> a.getStatus() == ApplicationStatus.PENDING).collect(Collectors.toList());

    public void evaluateApplication(String adminUserId, String applicationId, boolean approve) {
        User admin = users.get(adminUserId);
        if (admin == null || admin.getRole() != Role.ADMIN) throw new SecurityException("Admin required");
        CertificationApplication app = applications.get(applicationId);
        if (app != null) {
            app.setStatus(approve ? ApplicationStatus.APPROVED : ApplicationStatus.REJECTED);
            if (approve) {
                User applicant = users.get(app.getApplicantUserId());
                if (applicant != null) {
                    applicant.addCertification(app.getDesiredCert());
                    if (app.getDesiredCert() == CertificationType.CSP) {
                        applicant.setRole(Role.PRACTITIONER);

    public CourseEvent createCourseOrEvent(String creatorUserId, CourseEvent item) {
        User creator = users.get(creatorUserId);
        if (creator == null) throw new IllegalArgumentException("User not found");

        if (creator.getRole() == Role.TRAINER) {
            // Charging listing fee for Other Course or Event
            if (item.getType() == ItemType.OTHER_COURSE || item.getType() == ItemType.EVENT) {
                item.setFeeCharged(this.otherListingFee);
            }
        } else if (creator.getRole() == Role.ADMIN) {
            // Admin creates for free (e.g. Scrum Gatherings)
            item.setFeeCharged(0.0);
        } else {
            throw new SecurityException("Only trainers or admins can create courses/events");
        }

        coursesEvents.put(item.getId(), item);
        return item;
    }

    public void updateCourseOrEvent(String userId, String itemId, String newTitle, String newDesc, LocalDate start, LocalDate end) {
        User u = users.get(userId);
        CourseEvent ce = coursesEvents.get(itemId);
        if (ce == null) return;
        if (u.getRole() == Role.ADMIN || u.getRole() == Role.EDITOR || ce.getTrainerId().equals(userId)) {
            ce.setTitle(newTitle);
            ce.setDescription(newDesc);
            ce.setStartDate(start);
            ce.setEndDate(end);
        } else {
            throw new SecurityException("Not authorized to update this course/event");
        }
    }

    public void deleteCourseOrEvent(String userId, String itemId) {
        User u = users.get(userId);
        CourseEvent ce = coursesEvents.get(itemId);
        if (ce == null) return;
        if (u.getRole() == Role.ADMIN || ce.getTrainerId().equals(userId)) {
            coursesEvents.remove(itemId);
        } else {
            throw new SecurityException("Not authorized to delete this course/event");
        }
    }

    public CourseEvent copyCourseOrEvent(String trainerUserId, String itemId, String newItemId) {

        CourseEvent copy = existing.copy(newItemId);

    // --- Survey and Trainer Ratings ---
    public boolean submitCourseSurvey(SurveyResponse survey) {
        // Prevent duplicate answers by same participant for same class
        boolean alreadySubmitted = surveys.stream()
                .anyMatch(s -> s.getCourseId().equals(survey.getCourseId()) &&
                               s.getParticipantEmail().equalsIgnoreCase(survey.getParticipantEmail()));
        if (alreadySubmitted) {
            return false;
        }
        surveys.add(survey);
        
        CourseEvent course = coursesEvents.get(survey.getCourseId());
        if (course != null) {
            User trainer = users.get(course.getTrainerId());
            if (trainer != null) {
                emailService.sendEmail("surveys@scrumalliance.org", trainer.getEmail(), 
                    "New Survey Result for " + course.getTitle(), 
                    String.format("Participant rated %d/5: %s", survey.getRating(), survey.getFeedback()));
            }
        }
        return true;
    }

    public double getCourseAverageRating(String courseId) {
        return surveys.stream()
                .filter(s -> s.getCourseId().equals(courseId))
                .mapToInt(SurveyResponse::getRating)
                .average().orElse(0.0);
    }

    public double getTrainerAverageRating(String trainerUserId) {
        List<String> trainerCourseIds = coursesEvents.values().stream()
                .filter(c -> c.getTrainerId().equals(trainerUserId))
                .map(CourseEvent::getId)
                .toList();

        return surveys.stream()
                .filter(s -> trainerCourseIds.contains(s.getCourseId()))
                .mapToInt(SurveyResponse::getRating)
                .average().orElse(0.0);
    }

    public Map<String, Double> adminGetTrainerRatingAverages(String adminUserId) {
        User admin = users.get(adminUserId);
        if (admin == null || admin.getRole() != Role.ADMIN) throw new SecurityException("Admin required");

        Map<String, Double> map = new HashMap<>();
        for (User u : users.values()) {
            if (u.getRole() == Role.TRAINER) {
                map.put(u.getFullName(), getTrainerAverageRating(u.getId()));
            }
        }
        return map;
    }

    public void adminApproveClassPaymentAndMoveToRegistry(String adminUserId, String courseId) {
        User admin = users.get(adminUserId);
        if (admin == null || admin.getRole() != Role.ADMIN) throw new SecurityException("Admin required");

        CourseEvent ce = coursesEvents.get(courseId);
        if (ce != null && ce.isPending()) {
            List<PendingClassAttendee> attendees = pendingClassRosters.getOrDefault(courseId, Collections.emptyList());
            for (PendingClassAttendee a : attendees) {
                registry.add(new RegistryEntry(a.getFullName(), a.getEmail(), a.getCertificationType(), courseId, LocalDate.now()));
                // Send automated welcoming email
                String emailBody = String.format(newCertEmailTemplate, a.getFullName(), a.getCertificationType());
                emailService.sendEmail("registry@scrumalliance.org", a.getEmail(), "Welcome to Scrum Alliance!", emailBody);
            }
            ce.setPending(false);
            pendingClassRosters.remove(courseId);
        }
    }
