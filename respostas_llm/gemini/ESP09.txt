// File: AccessPolicy.java
import java.util.*;

public class AccessPolicy {
    private boolean isPublic;
    private final Set<String> allowedUserIds;
    private final Set<String> allowedGroupIds;

    public AccessPolicy() {
        this.isPublic = false;
        this.allowedUserIds = new HashSet<>();
        this.allowedGroupIds = new HashSet<>();
    }

    public boolean isPublic() { return isPublic; }
    public void setPublic(boolean isPublic) { this.isPublic = isPublic; }

    public void grantUser(String userId) { allowedUserIds.add(userId); }
    public void revokeUser(String userId) { allowedUserIds.remove(userId); }

    public void grantGroup(String groupId) { allowedGroupIds.add(groupId); }
    public void revokeGroup(String groupId) { allowedGroupIds.remove(groupId); }

    public boolean canAccess(User user, String ownerId) {
        if (isPublic) return true;
        if (user.getId().equals(ownerId)) return true;
        if (user.hasRole(UserRole.LAB_ADMINISTRATOR) || user.hasRole(UserRole.SYSTEM_ADMINISTRATOR)) return true;
        if (allowedUserIds.contains(user.getId())) return true;
        for (String g : user.getGroupMemberships()) {
            if (allowedGroupIds.contains(g)) return true;
        }
        return false;
    }
}

    private final Map<String, Set<String>> allowedMetadataValues;

        allowedTags = new HashSet<>(Arrays.asList(
            "electrophysiology", "imaging", "behaviour", "genetics", "worm", "c_elegans",
            "optogenetics", "fMRI", "control", "drug_response", "cell_image", "workflow_output"
        ));
        allowedMetadataValues = new HashMap<>();
        allowedMetadataValues.put("cell_type", new HashSet<>(Arrays.asList("AWA", "AWC", "ASH", "BAG", "Neuron", "Glial")));
        allowedMetadataValues.put("microscope_type", new HashSet<>(Arrays.asList("Confocal", "Two-Photon", "Epifluorescence")));
        allowedMetadataValues.put("drug_name", new HashSet<>(Arrays.asList("Serotonin", "Dopamine", "Octopamine", "None")));
        allowedMetadataValues.put("video_encoding", new HashSet<>(Arrays.asList("CJ-WormTrack-MJPEG", "CJ-WormTrack-RAW", "H.264")));
    }

    public boolean isTagAllowed(String tag) {
        return allowedTags.contains(tag.toLowerCase());
    }

    public void registerTag(String tag) {
        allowedTags.add(tag.toLowerCase());
    }

    public boolean isMetadataValueAllowed(String key, String value) {
        if (!allowedMetadataValues.containsKey(key)) {
            return true; // Not restricted
        }
        return allowedMetadataValues.get(key).contains(value);
    }

    public void registerAllowedMetadataValue(String key, String value) {
        allowedMetadataValues.computeIfAbsent(key, k -> new HashSet<>()).add(value);
    }
}

    public void setName(String name) { this.name = name; }

    public void setDeleted(boolean deleted) { this.deleted = deleted; }
    public FileDataQuality getQuality() { return quality; }
    public void setQuality(FileDataQuality quality) { this.quality = quality; }

    public void setBespokeMetadata(String key, String value) {
        this.bespokeMetadata.put(key, value);
    }

    public void addVersion(byte[] data, long size) {
        if (nonModifiable && !versions.isEmpty()) {
            throw new IllegalStateException("File is marked as non-modifiable / protected from modification.");
        }
        int nextVer = versions.size() + 1;
        String checksum = "chk-" + Integer.toHexString(Arrays.hashCode(data)) + "-" + nextVer;
        versions.add(new FileVersion(nextVer, data, size, checksum));
    }

    public List<FileVersion> getVersions() {
        return Collections.unmodifiableList(versions);
    }

    public FileVersion getLatestVersion() {
        if (versions.isEmpty()) return null;
        return versions.get(versions.size() - 1);
    }

    public FileVersion getVersion(int ver) {
        for (FileVersion v : versions) {
            if (v.getVersionNumber() == ver) return v;
        }
        return null;
    }

    public void revertToVersion(int ver) {
        if (nonModifiable) {
            throw new IllegalStateException("File is non-modifiable, cannot revert.");
        }
        FileVersion target = getVersion(ver);
        if (target != null) {
            addVersion(target.getContent(), target.getSizeBytes());
        }
    }
}

    private final String id;
    private final String logBookId;
    private final String sectionId;
    private String title;
    private String htmlContent;
    private final String creatorId;
    private final Instant createdAt;
    private final String cryptographicTimestampProof;
    private final Set<String> attachedFileIds;
    private final AccessPolicy accessPolicy;
    private final Map<String, String> bespokeMetadata;

    public LogBookPage(String id, String logBookId, String sectionId, String title, String htmlContent, String creatorId) {
        this.id = id;
        this.logBookId = logBookId;
        this.sectionId = sectionId;
        this.title = title;
        this.htmlContent = htmlContent;
        this.creatorId = creatorId;
        this.createdAt = Instant.now();
        this.attachedFileIds = new HashSet<>();
        this.accessPolicy = new AccessPolicy();
        this.bespokeMetadata = new HashMap<>();
        this.cryptographicTimestampProof = generateProof(id, creatorId, createdAt);
    }

    private static String generateProof(String pageId, String creator, Instant instant) {
        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            String raw = pageId + ":" + creator + ":" + instant.toString() + ":PROVENANCE_KEY";
            byte[] hash = md.digest(raw.getBytes());
            StringBuilder sb = new StringBuilder();
            for (byte b : hash) sb.append(String.format("%02x", b));
            return sb.toString();
        } catch (Exception e) {
            return "PROOF-" + instant.toEpochMilli();
        }
    }

    public String getId() { return id; }
    public String getLogBookId() { return logBookId; }
    public String getSectionId() { return sectionId; }
    public String getTitle() { return title; }
    public void setTitle(String title) { this.title = title; }
    public String getHtmlContent() { return htmlContent; }
    public void setHtmlContent(String htmlContent) { this.htmlContent = htmlContent; }
    public String getCreatorId() { return creatorId; }
    public Instant getCreatedAt() { return createdAt; }
    public String getCryptographicTimestampProof() { return cryptographicTimestampProof; }

    public AccessPolicy getAccessPolicy() { return accessPolicy; }
    public Map<String, String> getBespokeMetadata() { return Collections.unmodifiableMap(bespokeMetadata); }

import java.util.*;

public class LogBookNavigator {
    private final LogBook logBook;
    private final Map<String, LogBookPage> pagesMap;
    private final List<String> orderedPageSequence;
    private int currentIndex;

    public LogBookNavigator(LogBook logBook, Map<String, LogBookPage> pagesMap) {
        this.logBook = logBook;
        this.pagesMap = pagesMap;
        this.orderedPageSequence = new ArrayList<>();
        for (LogBookSection section : logBook.getAllSections()) {
            orderedPageSequence.addAll(section.getPageIds());
        }
        this.currentIndex = 0;
    }

    public boolean hasNext() {
        return currentIndex < orderedPageSequence.size() - 1;
    }

    public boolean hasPrevious() {
        return currentIndex > 0;
    }

    public LogBookPage next() {
        if (hasNext()) {
            currentIndex++;
            return getCurrentPage();
        }
        return null;
    }

    public LogBookPage previous() {
        if (hasPrevious()) {
            currentIndex--;
            return getCurrentPage();
        }
        return null;
    }

    public LogBookPage getCurrentPage() {
        if (orderedPageSequence.isEmpty()) return null;
        return pagesMap.get(orderedPageSequence.get(currentIndex));
    }
}

    public ExperimentalProtocol(String protocolId, int version, String title, String description, String parentProtocolId) {
        this.protocolId = protocolId;
        this.version = version;
        this.title = title;
        this.description = description;
        this.parentProtocolId = parentProtocolId;
        this.createdAt = Instant.now();
        this.stages = new ArrayList<>();
    }

    public String getProtocolId() { return protocolId; }
    public int getVersion() { return version; }
    public String getTitle() { return title; }
    public String getDescription() { return description; }
    public String getParentProtocolId() { return parentProtocolId; }
    public Instant getCreatedAt() { return createdAt; }
    public List<Stage> getStages() { return Collections.unmodifiableList(stages); }

    public void addStage(Stage stage) { stages.add(stage); }

    public ExperimentalProtocol branch(String newProtocolId, String newTitle) {
        ExperimentalProtocol branched = new ExperimentalProtocol(newProtocolId, 1, newTitle, this.description, this.protocolId);
        for (Stage s : this.stages) {
            Stage copy = new Stage(s.getName(), s.getInstructions());
            for (Map.Entry<String, String> e : s.getMetadata().entrySet()) {
                copy.setMetadata(e.getKey(), e.getValue());
            }
            branched.addStage(copy);
        }
        return branched;
    }
}

    private final Map<String, String> cellFormulas;

    public CollaborativeSpreadsheet(String sheetId, String title) {
        this.sheetId = sheetId;
        this.title = title;
        this.cellValues = new HashMap<>();
        this.cellFormulas = new HashMap<>();
    }

    public String getSheetId() { return sheetId; }
    public String getTitle() { return title; }

    public void setCell(String cellRef, double val) {
        cellValues.put(cellRef.toUpperCase(), val);
    }

    public void setSumFormula(String targetCell, List<String> sourceCells) {
        cellFormulas.put(targetCell.toUpperCase(), "SUM(" + String.join(",", sourceCells) + ")");
        double sum = 0.0;
        for (String c : sourceCells) {
            sum += cellValues.getOrDefault(c.toUpperCase(), 0.0);
        }
        cellValues.put(targetCell.toUpperCase(), sum);
    }

    public double getCellValue(String cellRef) {
        return cellValues.getOrDefault(cellRef.toUpperCase(), 0.0);
    }
}

    // Searching
    public List<FileSearchResult> searchFiles(User user, SearchFilter filter, Set<String> limitToLogBookIds, Set<String> limitToSectionIds) {
        // Collect candidate page files if restricted by logbook / section
        Set<String> scopedFileIds = null;
        Map<String, String> fileToPageMap = new HashMap<>();

        for (LogBookPage page : pages.values()) {
            boolean includePage = true;
            if (limitToLogBookIds != null && !limitToLogBookIds.isEmpty() && !limitToLogBookIds.contains(page.getLogBookId())) {
                includePage = false;
            }
            if (limitToSectionIds != null && !limitToSectionIds.isEmpty() && !limitToSectionIds.contains(page.getSectionId())) {
                includePage = false;
            }

            for (String fid : page.getAttachedFileIds()) {
                fileToPageMap.put(fid, page.getId());
            }

            if (limitToLogBookIds != null || limitToSectionIds != null) {
                if (scopedFileIds == null) scopedFileIds = new HashSet<>();
                if (includePage) {
                    scopedFileIds.addAll(page.getAttachedFileIds());
                }
            }
        }

        List<FileSearchResult> results = new ArrayList<>();
        for (NeuroFile file : files.values()) {
            if (file.isDeleted()) continue;
            if (!file.getAccessPolicy().canAccess(user, file.getOwnerId())) continue;
            if (scopedFileIds != null && !scopedFileIds.contains(file.getId())) continue;

            if (filter != null) {
                if (filter.isExcludeBadQuality() && file.getQuality() == FileDataQuality.BAD) continue;
                if (filter.getFileType() != null && !file.getFileType().equalsIgnoreCase(filter.getFileType())) continue;
                if (filter.getFormat() != null && !file.getFormat().equalsIgnoreCase(filter.getFormat())) continue;
                if (filter.getTag() != null && !file.getTags().contains(filter.getTag().toLowerCase())) continue;
                if (filter.getMetadataKey() != null) {
                    String actualVal = file.getBespokeMetadata().get(filter.getMetadataKey());
                    if (actualVal == null || (filter.getMetadataValue() != null && !actualVal.equalsIgnoreCase(filter.getMetadataValue()))) {
                        continue;
                    }
                }
            }
            results.add(new FileSearchResult(file, fileToPageMap.get(file.getId())));
        }
        return results;
    }
