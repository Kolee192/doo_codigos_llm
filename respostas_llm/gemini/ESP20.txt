    }

    }

        roles.add(role);
    }

    public void removeRole(Role role) {
        roles.remove(role);
    }

        return roles.contains(role);

// File: MapExtent.java
public class MapExtent {
    private double minLongitude;
    private double minLatitude;
    private double maxLongitude;
    private double maxLatitude;

    public MapExtent(double minLongitude, double minLatitude, double maxLongitude, double maxLatitude) {
        this.minLongitude = minLongitude;
        this.minLatitude = minLatitude;
        this.maxLongitude = maxLongitude;
        this.maxLatitude = maxLatitude;
    }

    public double getMinLongitude() {
        return minLongitude;
    }

    public double getMinLatitude() {
        return minLatitude;
    }

    public double getMaxLongitude() {
        return maxLongitude;
    }

    public double getMaxLatitude() {
        return maxLatitude;
    }

    public void setExtent(double minLon, double minLat, double maxLon, double maxLat) {
        this.minLongitude = minLon;
        this.minLatitude = minLat;
        this.maxLongitude = maxLon;
        this.maxLatitude = maxLat;
    }

    public void pan(double deltaLon, double deltaLat) {
        this.minLongitude += deltaLon;
        this.maxLongitude += deltaLon;
        this.minLatitude += deltaLat;
        this.maxLatitude += deltaLat;
    }

    public void zoom(double factor) {
        double centerLon = (minLongitude + maxLongitude) / 2.0;
        double centerLat = (minLatitude + maxLatitude) / 2.0;
        double halfLonSpan = ((maxLongitude - minLongitude) * factor) / 2.0;
        double halfLatSpan = ((maxLatitude - minLatitude) * factor) / 2.0;
        this.minLongitude = centerLon - halfLonSpan;
        this.maxLongitude = centerLon + halfLonSpan;
        this.minLatitude = centerLat - halfLatSpan;
        this.maxLatitude = centerLat + halfLatSpan;
    }

    @Override
    public String toString() {
        return String.format("MapExtent[Lon: (%.2f to %.2f), Lat: (%.2f to %.2f)]",
                minLongitude, maxLongitude, minLatitude, maxLatitude);
    }
}

import java.time.LocalDate;

public class FilterCriterion {
    private final CriterionType type;
    private final String value;
    private final Double minNumeric;
    private final Double maxNumeric;
    private final LocalDate minDate;
    private final LocalDate maxDate;

    public FilterCriterion(CriterionType type, String value) {
        this(type, value, null, null, null, null);
    }

    public FilterCriterion(CriterionType type, Double minNumeric, Double maxNumeric) {
        this(type, null, minNumeric, maxNumeric, null, null);
    }

    public FilterCriterion(CriterionType type, LocalDate minDate, LocalDate maxDate) {
        this(type, null, null, null, minDate, maxDate);
    }

    private FilterCriterion(CriterionType type, String value, Double minNumeric, Double maxNumeric,
                            LocalDate minDate, LocalDate maxDate) {
        this.type = type;
        this.value = value;
        this.minNumeric = minNumeric;
        this.maxNumeric = maxNumeric;
        this.minDate = minDate;
        this.maxDate = maxDate;
    }

    public CriterionType getType() {
        return type;
    }

    public String getValue() {
        return value;
    }

    public Double getMinNumeric() {
        return minNumeric;
    }

    public Double getMaxNumeric() {
        return maxNumeric;
    }

    public LocalDate getMinDate() {
        return minDate;
    }

    public LocalDate getMaxDate() {
        return maxDate;
    }

    public boolean test(OccurrenceRecord record) {
        switch (type) {
            case TAXONOMIC:
                return value != null && record.getScientificName().toLowerCase().contains(value.toLowerCase());
            case DATASET:
                return value != null && record.getDatasetId().equalsIgnoreCase(value);
            case DEPTH:
                if (minNumeric != null && record.getDepth() < minNumeric) return false;
                if (maxNumeric != null && record.getDepth() > maxNumeric) return false;
                return true;
            case ALTITUDE:
                if (minNumeric != null && record.getAltitude() < minNumeric) return false;
                if (maxNumeric != null && record.getAltitude() > maxNumeric) return false;
                return true;
            case BASIS_OF_RECORD:
                return value != null && record.getBasisOfRecord().equalsIgnoreCase(value);
            case GEOGRAPHIC:
                return value != null && record.getCountryOrGeographic().toLowerCase().contains(value.toLowerCase());
            case DATE:
                if (minDate != null && record.getEventDate().isBefore(minDate)) return false;
                if (maxDate != null && record.getEventDate().isAfter(maxDate)) return false;
                return true;
            default:
                return true;
        }
    }

    @Override
    public String toString() {
        return "FilterCriterion{" + "type=" + type + ", value='" + value + '\'' +
                ", min=" + minNumeric + ", max=" + maxNumeric +
                ", minDate=" + minDate + ", maxDate=" + maxDate + '}';
    }
}

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class OccurrenceFilter {
    private final String name;
    private final List<FilterCriterion> criteria = new ArrayList<>();

    public OccurrenceFilter(String name) {
        this.name = name;
    }

    public String getName() {
        return name;
    }

    public void addCriterion(FilterCriterion criterion) {
        criteria.add(criterion);
    }

    public boolean removeCriterion(CriterionType type) {
        return criteria.removeIf(c -> c.getType() == type);
    }

    public List<FilterCriterion> getCriteria() {
        return Collections.unmodifiableList(criteria);
    }

    public boolean matches(OccurrenceRecord record) {
        for (FilterCriterion criterion : criteria) {
            if (!criterion.test(record)) {
                return false;
            }
        }
        return true;
    }
}

    public byte[] exportDistributionMapAsImage(List<OccurrenceRecord> records) {
        dataManagerService.recordUsage();
        String svg = "<svg xmlns='http://www.w3.org/2000/svg' width='800' height='400'>"
                + "<rect width='100%' height='100%' fill='#eef2f5'/>"
                + "<text x='20' y='30'>Occurrence Map (" + records.size() + " points)</text>";
        StringBuilder points = new StringBuilder();
        for (OccurrenceRecord r : records) {
            double x = ((r.getLongitude() + 180.0) / 360.0) * 800.0;
            double y = ((90.0 - r.getLatitude()) / 180.0) * 400.0;
            points.append(String.format("<circle cx='%.1f' cy='%.1f' r='4' fill='red' />", x, y));
        }
        svg = svg + points.toString() + "</svg>";
        return svg.getBytes(StandardCharsets.UTF_8);
    }

    public List<String> fullTextSearch(String term) {
        dataManagerService.recordUsage();
        List<String> results = new ArrayList<>();
        String lower = term.toLowerCase();

        dataManagerService.getOccurrenceResources().values().stream()
                .filter(r -> r.matchesKeyword(lower))
                .forEach(r -> results.add("Occurrence Resource: " + r.getTitle() + " (" + r.getId() + ")"));

        dataManagerService.getChecklistResources().values().stream()
                .filter(r -> r.matchesKeyword(lower))
                .forEach(r -> results.add("Checklist Resource: " + r.getTitle() + " (" + r.getId() + ")"));

        dataManagerService.getMultimediaResources().values().stream()
                .filter(r -> r.matchesKeyword(lower))
                .forEach(r -> results.add("Multimedia Resource: " + r.getTitle() + " (" + r.getId() + ")"));

        dataManagerService.getBibliographicResources().values().stream()
                .filter(r -> r.matchesKeyword(lower))
                .forEach(r -> results.add("Bibliographic Resource: " + r.getTitle() + " (" + r.getId() + ")"));

        dataManagerService.getInstitutionResources().values().stream()
                .filter(r -> r.matchesKeyword(lower))
                .forEach(r -> results.add("Institution Resource: " + r.getName() + " (" + r.getId() + ")"));

        dataManagerService.getExpertResources().values().stream()
                .filter(r -> r.matchesKeyword(lower))
                .forEach(r -> results.add("Expert Resource: " + r.getFullName() + " (" + r.getId() + ")"));

        portalManagerService.getNewsEntries().values().stream()
                .filter(n -> n.getTitle().toLowerCase().contains(lower) || n.getContent().toLowerCase().contains(lower))
                .forEach(n -> results.add("Portal News: " + n.getTitle()));

        portalManagerService.getBlogPosts().values().stream()
                .filter(b -> b.getTitle().toLowerCase().contains(lower) || b.getContent().toLowerCase().contains(lower))
                .forEach(b -> results.add("Blog Post: " + b.getTitle()));

        portalManagerService.getPortalPages().values().stream()
                .filter(p -> p.getTitle().toLowerCase().contains(lower) || p.getContent().toLowerCase().contains(lower))
                .forEach(p -> results.add("Portal Page: " + p.getTitle()));

        return results;
    }
