import java.util.Objects;

public class ColorRGB {
    private final int red;
    private final int green;
    private final int blue;

    public ColorRGB(int red, int green, int blue) {
        this.red = clamp(red);
        this.green = clamp(green);
        this.blue = clamp(blue);
    }

    public static ColorRGB fromHex(String hex) {
        if (hex == null) {
            throw new IllegalArgumentException("Hex string cannot be null");
        }
        String cleanHex = hex.trim().replace("#", "");
        if (cleanHex.length() != 6) {
            throw new IllegalArgumentException("Hex color must be 6 hex characters");
        }
        int r = Integer.parseInt(cleanHex.substring(0, 2), 16);
        int g = Integer.parseInt(cleanHex.substring(2, 4), 16);
        int b = Integer.parseInt(cleanHex.substring(4, 6), 16);
        return new ColorRGB(r, g, b);
    }

    private static int clamp(int val) {
        return Math.max(0, Math.min(255, val));
    }

    public int getRed() {
        return red;
    }

    public int getGreen() {
        return green;
    }

    public int getBlue() {
        return blue;
    }

    public double distanceTo(ColorRGB other) {
        if (other == null) {
            return Double.MAX_VALUE;
        }
        long dr = (long) this.red - other.red;
        long dg = (long) this.green - other.green;
        long db = (long) this.blue - other.blue;
        return Math.sqrt(dr * dr + dg * dg + db * db);
    }

    public String toHex() {
        return String.format("#%02X%02X%02X", red, green, blue);
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof ColorRGB colorRGB)) return false;
        return red == colorRGB.red && green == colorRGB.green && blue == colorRGB.blue;
    }

    @Override
    public int hashCode() {
        return Objects.hash(red, green, blue);
    }

    @Override
    public String toString() {
        return String.format("RGB(%d, %d, %d) [%s]", red, green, blue, toHex());
    }
}

import java.util.*;
import java.util.concurrent.ConcurrentHashMap;

public class PaintDatabase {
    private final Map<String, Paint> paints = new ConcurrentHashMap<>();

    private String makeKey(String collection, String paintNumber) {
        return collection.trim().toUpperCase() + "::" + paintNumber.trim().toUpperCase();
    }

    public synchronized void addPaint(Paint paint) {
        if (paint == null) throw new IllegalArgumentException("Paint cannot be null");
        String key = makeKey(paint.getCollection(), paint.getPaintNumber());
        paints.put(key, paint);
    }

    public synchronized boolean updatePaint(Paint paint) {
        if (paint == null) return false;
        String key = makeKey(paint.getCollection(), paint.getPaintNumber());
        if (paints.containsKey(key)) {
            paints.put(key, paint);
            return true;
        }
        return false;
    }

    public synchronized boolean deletePaint(String collection, String paintNumber) {
        if (collection == null || paintNumber == null) return false;
        String key = makeKey(collection, paintNumber);
        return paints.remove(key) != null;
    }

    public Optional<Paint> findByNumber(String collection, String paintNumber) {
        if (collection == null || paintNumber == null) return Optional.empty();
        return Optional.ofNullable(paints.get(makeKey(collection, paintNumber)));
    }

    public List<Paint> getAllPaints() {
        return new ArrayList<>(paints.values());
    }

    public List<Paint> getByCollection(String collection) {
        if (collection == null) return Collections.emptyList();
        List<Paint> result = new ArrayList<>();
        for (Paint p : paints.values()) {
            if (p.getCollection().equalsIgnoreCase(collection.trim())) {
                result.add(p);
            }
        }
        return result;
    }

    public Set<String> getCollections() {
        Set<String> collections = new HashSet<>();
        for (Paint p : paints.values()) {
            collections.add(p.getCollection());
        }
        return collections;
    }
}

import java.util.*;

public class ColorSearchEngine {
    private final PaintDatabase database;

    public ColorSearchEngine(PaintDatabase database) {
        this.database = Objects.requireNonNull(database, "PaintDatabase cannot be null");
    }

    public List<Paint> search(String query, String collectionFilter, int limit) {
        try {
            List<Paint> pool = (collectionFilter == null || collectionFilter.isBlank())
                    ? database.getAllPaints()
                    : database.getByCollection(collectionFilter);

            if (query == null || query.isBlank()) {
                return pool.stream().limit(Math.max(1, limit)).toList();
            }

            String q = query.trim();

            if (q.startsWith("#") || (q.length() == 6 && q.matches("^[0-9a-fA-F]{6}$"))) {
                try {
                    ColorRGB target = ColorRGB.fromHex(q);
                    return findNearest(target, pool, limit);
                } catch (Exception ignored) {
                    // fall back to text matching
                }
            }

            List<Paint> exactNumberMatches = new ArrayList<>();
            List<Paint> nameMatches = new ArrayList<>();

            for (Paint p : pool) {
                if (p.getPaintNumber().equalsIgnoreCase(q)) {
                    exactNumberMatches.add(p);
                } else if (p.getPaintName().toLowerCase().contains(q.toLowerCase())) {
                    nameMatches.add(p);
                }
            }

            List<Paint> combined = new ArrayList<>(exactNumberMatches);
            for (Paint p : nameMatches) {
                if (!combined.contains(p)) {
                    combined.add(p);
                }
            }

            if (combined.isEmpty()) {
                return Collections.emptyList();
            }

            return combined.subList(0, Math.min(combined.size(), Math.max(1, limit)));
        } catch (Exception e) {
            ErrorReportingService.getInstance().reportError("ColorSearchEngine.search", e);
            return Collections.emptyList();
        }
    }

    public List<Paint> findNearest(ColorRGB targetColor, List<Paint> candidates, int limit) {
        try {
            if (candidates == null || candidates.isEmpty() || targetColor == null) {
                return Collections.emptyList();
            }
            List<Paint> sorted = new ArrayList<>(candidates);
            sorted.sort(Comparator.comparingDouble(p -> p.getColor().distanceTo(targetColor)));
            return sorted.subList(0, Math.min(sorted.size(), Math.max(1, limit)));
        } catch (Exception e) {
            ErrorReportingService.getInstance().reportError("ColorSearchEngine.findNearest", e);
            return Collections.emptyList();
        }
    }

    public PaintDatabase getDatabase() {
        return database;
    }
}

import java.util.Optional;

public class ColorTranslator {
    private final ColorSearchEngine searchEngine;

    public ColorTranslator(ColorSearchEngine searchEngine) {
        this.searchEngine = searchEngine;
    }

    public Optional<Paint> translate(String oldPaintNumber, String oldCollection, String targetCollection) {
        try {
            Optional<Paint> oldPaintOpt = searchEngine.getDatabase().findByNumber(oldCollection, oldPaintNumber);
            if (oldPaintOpt.isEmpty()) {
                return Optional.empty();
            }

            Paint oldPaint = oldPaintOpt.get();
            List<Paint> targetPaints = searchEngine.getDatabase().getByCollection(targetCollection);
            if (targetPaints.isEmpty()) {
                return Optional.empty();
            }

            List<Paint> nearest = searchEngine.findNearest(oldPaint.getColor(), targetPaints, 1);
            return nearest.isEmpty() ? Optional.empty() : Optional.of(nearest.get(0));
        } catch (Exception e) {
            ErrorReportingService.getInstance().reportError("ColorTranslator.translate", e);
            return Optional.empty();
        }
    }
}

public class UploadedImage {
    private final String filename;
    private final int width;
    private final int height;
    private final ColorRGB[][] pixels;

    public UploadedImage(String filename, int width, int height) {
        this.filename = filename != null ? filename : "image.png";
        this.width = Math.max(1, width);
        this.height = Math.max(1, height);
        this.pixels = new ColorRGB[this.height][this.width];
        for (int y = 0; y < this.height; y++) {
            for (int x = 0; x < this.width; x++) {
                pixels[y][x] = new ColorRGB(255, 255, 255);
            }
        }
    }

    public void setPixel(int x, int y, ColorRGB color) {
        if (x >= 0 && x < width && y >= 0 && y < height) {
            pixels[y][x] = color != null ? color : new ColorRGB(0, 0, 0);
        }
    }

    public ColorRGB getPixel(int x, int y) {
        int clampedX = Math.max(0, Math.min(width - 1, x));
        int clampedY = Math.max(0, Math.min(height - 1, y));
        return pixels[clampedY][clampedX];
    }

    public ColorRGB getBlendedRegionColor(int startX, int startY, int regionWidth, int regionHeight) {
        int x1 = Math.max(0, startX);
        int y1 = Math.max(0, startY);
        int x2 = Math.min(width, startX + regionWidth);
        int y2 = Math.min(height, startY + regionHeight);

        long rSum = 0;
        long gSum = 0;
        long bSum = 0;
        int count = 0;

        for (int y = y1; y < y2; y++) {
            for (int x = x1; x < x2; x++) {
                ColorRGB c = pixels[y][x];
                rSum += c.getRed();
                gSum += c.getGreen();
                bSum += c.getBlue();
                count++;
            }
        }

        if (count == 0) {
            return getPixel(startX, startY);
        }
        return new ColorRGB((int)(rSum / count), (int)(gSum / count), (int)(bSum / count));
    }

    public String getFilename() {
        return filename;
    }

    public int getWidth() {
        return width;
    }

    public int getHeight() {
        return height;
    }
}

import java.time.Instant;
import java.util.Map;
import java.util.Optional;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public class SessionManager {
    private final Map<String, UserSession> sessions = new ConcurrentHashMap<>();

    public UserSession createSession(User user) {
        String sessionId = UUID.randomUUID().toString();
        UserSession session = new UserSession(sessionId, user);
        sessions.put(sessionId, session);
        return session;
    }

    public Optional<UserSession> getSession(String sessionId) {
        if (sessionId == null) return Optional.empty();
        UserSession session = sessions.get(sessionId);
        if (session != null) {
            if (session.isExpired(Instant.now())) {
                sessions.remove(sessionId);
                return Optional.empty();
            }
            session.touch();
            return Optional.of(session);
        }
        return Optional.empty();
    }

    public int cleanExpiredSessions(Instant referenceInstant) {
        int removedCount = 0;
        for (Map.Entry<String, UserSession> entry : sessions.entrySet()) {
            if (entry.getValue().isExpired(referenceInstant)) {
                sessions.remove(entry.getKey());
                removedCount++;
            }
        }
        return removedCount;
    }

    public int getActiveSessionCount() {
        return sessions.size();
    }
}

public class AdminService {
    private final PaintDatabase database;
    private final Map<String, User> userAccounts = new ConcurrentHashMap<>();

    public AdminService(PaintDatabase database) {
        this.database = database;
    }

    public boolean canAdd(User user) {
        return user != null && user.isAdministrative() && user.getAdminLevel().isPresent();
    }

    public boolean canUpdate(User user) {
        return user != null && user.isAdministrative() &&
                user.getAdminLevel().map(l -> l.getRank() >= AdminLevel.LEVEL_2.getRank()).orElse(false);
    }

    public boolean canDelete(User user) {
        return user != null && user.isAdministrative() &&
                user.getAdminLevel().map(l -> l.getRank() >= AdminLevel.LEVEL_3.getRank()).orElse(false);
    }

    public boolean canCreateAdminUser(User creator, AdminLevel targetLevel) {
        if (creator == null || !creator.isAdministrative() || creator.getAdminLevel().isEmpty()) {
            return false;
        }
        int creatorRank = creator.getAdminLevel().get().getRank();
        return creatorRank >= targetLevel.getRank();
    }

    public boolean addPaint(User admin, Paint paint) {
        if (!canAdd(admin)) {
            ErrorReportingService.getInstance().reportError(
                    "AdminService.addPaint",
                    new SecurityException("User " + (admin != null ? admin.getUsername() : "null") + " has insufficient permission to add paint"));
            return false;
        }
        database.addPaint(paint);
        return true;
    }

    public boolean updatePaint(User admin, Paint paint) {
        if (!canUpdate(admin)) {
            ErrorReportingService.getInstance().reportError(
                    "AdminService.updatePaint",
                    new SecurityException("User " + (admin != null ? admin.getUsername() : "null") + " has insufficient permission to update paint"));
            return false;
        }
        return database.updatePaint(paint);
    }

    public boolean deletePaint(User admin, String collection, String paintNumber) {
        if (!canDelete(admin)) {
            ErrorReportingService.getInstance().reportError(
                    "AdminService.deletePaint",
                    new SecurityException("User " + (admin != null ? admin.getUsername() : "null") + " has insufficient permission to delete paint"));
            return false;
        }
        return database.deletePaint(collection, paintNumber);
    }

    public boolean createAdminUser(User creator, String newUserId, String newUsername, AdminLevel level) {
        if (!canCreateAdminUser(creator, level)) {
            ErrorReportingService.getInstance().reportError(
                    "AdminService.createAdminUser",
                    new SecurityException("User " + (creator != null ? creator.getUsername() : "null") + " cannot create admin user of rank " + level.getRank()));
            return false;
        }
        User newUser = new User(newUserId, newUsername, true, level);
        userAccounts.put(newUserId, newUser);
        return true;
    }

    public void registerUser(User user) {
        if (user != null) {
            userAccounts.put(user.getUserId(), user);
        }
    }

    public User getUser(String userId) {
        return userAccounts.get(userId);
    }
}

        return new ColorRGB(red, green, blue);
    }

    private ColorRGB hsbToRgb(int hue, float saturation, float brightness) {
        float r = 0, g = 0, b = 0;
        if (saturation == 0) {
            r = g = b = brightness;
        } else {
            float h = (hue - (float) Math.floor(hue / 360.0f) * 360.0f) / 60.0f;
            int i = (int) h;
            float f = h - i;
            float p = brightness * (1.0f - saturation);
            float q = brightness * (1.0f - saturation * f);
            float t = brightness * (1.0f - (saturation * (1.0f - f)));
            switch (i) {
                case 0 -> { r = brightness; g = t; b = p; }
                case 1 -> { r = q; g = brightness; b = p; }
                case 2 -> { r = p; g = brightness; b = t; }
                case 3 -> { r = p; g = q; b = brightness; }
                case 4 -> { r = t; g = p; b = brightness; }
                case 5 -> { r = brightness; g = p; b = q; }
            }
        }
        return new ColorRGB(Math.round(r * 255), Math.round(g * 255), Math.round(b * 255));
    }
}
