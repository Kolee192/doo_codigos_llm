public class WysiwygContent {
    private String htmlContent;
    private String plainTextContent;

    public WysiwygContent(String htmlContent) {
        this.htmlContent = htmlContent;
        this.plainTextContent = stripHtml(htmlContent);
    }

    private String stripHtml(String html) {
        if (html == null) return "";
        return html.replaceAll("<[^>]*>", "").trim();
    }

    public String getHtmlContent() {
        return htmlContent;
    }

    public void setHtmlContent(String htmlContent) {
        this.htmlContent = htmlContent;
        this.plainTextContent = stripHtml(htmlContent);
    }

    public String getPlainTextContent() {
        return plainTextContent;
    }

    @Override
    public String toString() {
        return plainTextContent;
    }
}

public class PaymentGateway {
    public static class ChargeResult {
        private final boolean success;
        private final String chargeId;
        private final String errorMessage;

        public ChargeResult(boolean success, String chargeId, String errorMessage) {
            this.success = success;
            this.chargeId = chargeId;
            this.errorMessage = errorMessage;
        }

        public boolean isSuccess() {
            return success;
        }

        public String getChargeId() {
            return chargeId;
        }

        public String getErrorMessage() {
            return errorMessage;
        }
    }

    public ChargeResult chargeCard(String cardNumber, String expDate, String cvc, double amount) {
        if (cardNumber == null || cardNumber.trim().length() < 12) {
            return new ChargeResult(false, null, "Invalid credit card number.");
        }
        if (amount <= 0) {
            return new ChargeResult(false, null, "Amount must be greater than zero.");
        }
        String chargeId = "ch_stripe_" + UUID.randomUUID().toString().substring(0, 8);
        return new ChargeResult(true, chargeId, null);
    }

    public boolean refundCharge(String chargeId) {
        return chargeId != null && chargeId.startsWith("ch_stripe_");
    }
}

    public TrainingRegistration registerForTraining(String userId, String trainingId, boolean acceptCancellationTerms) {
        if (!acceptCancellationTerms) {
            throw new IllegalArgumentException("You must accept the cancellation terms to register.");
        }

        Training training = trainings.get(trainingId);
        if (training == null) {
            throw new IllegalArgumentException("Training not found.");
        }

        if (training.getCancellationDeadline() != null && LocalDateTime.now().isAfter(training.getCancellationDeadline())) {
            throw new IllegalStateException("Registration is closed. The deadline was " + training.getCancellationDeadline());
        }

        for (Training t : trainings.values()) {
            if (t.getDate().equals(training.getDate())) {
                for (TrainingRegistration reg : t.getRegistrations()) {
                    if (reg.getUserId().equals(userId) &&
                            (reg.getStatus() == RegistrationStatus.REGISTERED || reg.getStatus() == RegistrationStatus.ATTENDED)) {
                        throw new IllegalStateException("Limit to one training registration per day per authenticated user.");
                    }
                }
            }
        }

        RegistrationStatus status = (training.getRemainingRegistrations() > 0) ?
                RegistrationStatus.REGISTERED : RegistrationStatus.WAITLISTED;

        String regId = "treg-" + UUID.randomUUID().toString().substring(0, 8);
        TrainingRegistration registration = new TrainingRegistration(regId, trainingId, userId, acceptCancellationTerms, status);
        training.addRegistration(registration);

        User user = userService.getUser(userId);
        if (user != null) {
            user.addRole(UserRole.TRAINEE);
            user.addToSchedule(training.getId());
        }
        return registration;
    }

    public boolean cancelRegistration(String registrationId) {

            for (TrainingRegistration reg : t.getRegistrations()) {
                if (reg.getId().equals(registrationId)) {
                    if (t.getCancellationDeadline() != null && LocalDateTime.now().isAfter(t.getCancellationDeadline())) {
                        throw new IllegalStateException("Cannot cancel: cancellation deadline has passed.");
                    }
                    reg.setStatus(RegistrationStatus.CANCELLED);
                    for (TrainingRegistration waitlistReg : t.getRegistrations()) {
                        if (waitlistReg.getStatus() == RegistrationStatus.WAITLISTED) {

                            }
                            break;
                        }
                    }
                    return true;
                }
            }
        }
        return false;
    }

    public Summit createSummit(String title, WysiwygContent description, String coordinatorUserId,
                               String venueId, String roomName, LocalDate date, LocalTime start, LocalTime end, int capacity) {
        String id = "smt-" + UUID.randomUUID().toString().substring(0, 8);
        Summit summit = new Summit(id, title, description, coordinatorUserId, venueId, roomName, date, start, end, capacity);
        summits.put(id, summit);
        User coord = userService.getUser(coordinatorUserId);
        if (coord != null) {
            coord.addRole(UserRole.SUMMIT_COORDINATOR);
        }

    public void updateAgenda(String summitId, String coordinatorUserId, WysiwygContent agenda) {
        Summit summit = summits.get(summitId);
        if (summit != null) {
            if (!summit.getCoordinatorUserId().equals(coordinatorUserId)) {
                User u = userService.getUser(coordinatorUserId);
                if (u == null || !u.hasRole(UserRole.ADMINISTRATOR)) {
                    throw new SecurityException("Only the designated summit coordinator can update the agenda.");
                }
            }

        }

    public boolean registerUser(String userId, String summitId) {
        Summit summit = summits.get(summitId);
        if (summit == null) {
            throw new IllegalArgumentException("Summit not found.");
        }
        for (Summit s : summits.values()) {
            if (s.getDate().equals(summit.getDate()) && s.getRegisteredUserIds().contains(userId)) {
                throw new IllegalStateException("Double booking detected: already registered for another summit on " + summit.getDate());

        }
        boolean registered = summit.registerUser(userId);
        if (registered) {
            User u = userService.getUser(userId);
            if (u != null) {
                u.addToSchedule(summit.getId());
            }
        }
        return registered;
    }

                .filter(s -> s.getRegisteredUserIds().contains(userId))
                .collect(Collectors.toList());
    }

    public void sponsorSummit(String summitId, String sponsorId) {

        if (summit != null) {
            summit.addSponsor(sponsorId);
            sponsorshipService.assignSponsorToActivity(sponsorId, summitId);
        }
    }

    public List<Sponsor> getSummitSponsors(String summitId) {
        Summit s = summits.get(summitId);
        if (s == null) return Collections.emptyList();
        return s.getSponsorIds().stream()
                .map(sponsorshipService::getSponsor)
                .filter(sp -> sp != null)
                .collect(Collectors.toList());
    }

    public List<User> getSummitAttendees(String summitId) {
        Summit s = summits.get(summitId);
        if (s == null) return Collections.emptyList();
        return s.getRegisteredUserIds().stream()
                .map(userService::getUser)
                .filter(u -> u != null)
                .collect(Collectors.toList());
    }

    public void emailSummitAttendees(String summitId, String subject, String message) {
        for (User attendee : getSummitAttendees(summitId)) {
            notificationService.sendEmail(attendee.getEmail(), subject, message);
        }
    }

    public String getDirectionsToSummit(String summitId) {
        Summit s = summits.get(summitId);
        if (s == null) return "Summit not found";
        return venueService.getVenueMapAndDirections(s.getVenueId()) + " -> Room: " + s.getRoomName();
    }

    public List<Summit> getAllSummits() {
        return new ArrayList<>(summits.values());
    }

    public Summit getSummit(String id) {
        return summits.get(id);
    }

import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import java.util.stream.Collectors;

public class SessionService {
    private final Map<String, Session> sessions = new HashMap<>();
    private final UserService userService;
    private final NotificationService notificationService;

    public SessionService(UserService userService, NotificationService notificationService) {
        this.userService = userService;
        this.notificationService = notificationService;
    }

    public Session submitSession(String speakerUserId, String title, WysiwygContent description) {
        User speaker = userService.getUser(speakerUserId);
        if (speaker == null) {
            throw new IllegalArgumentException("Speaker user not found.");
        }
        String id = "ses-" + UUID.randomUUID().toString().substring(0, 8);
        Session session = new Session(id, title, description, speakerUserId);
        sessions.put(id, session);

        notificationService.sendEmail(speaker.getEmail(), "Session Submission Received: " + title,
                "Thank you for submitting '" + title + "'. It will be reviewed by the BADCamp organizers.");
        return session;
    }

    public void editSession(String sessionId, String speakerUserId, String newTitle, WysiwygContent newDesc) {
        Session session = sessions.get(sessionId);
        if (session == null) {
            throw new IllegalArgumentException("Session not found.");
        }
        if (!session.getSpeakerUserId().equals(speakerUserId)) {
            User u = userService.getUser(speakerUserId);
            if (u == null || !u.hasRole(UserRole.ADMINISTRATOR)) {
                throw new SecurityException("Unauthorized to edit this session.");
            }
        }
        if (newTitle != null) session.setTitle(newTitle);
        if (newDesc != null) session.setDescription(newDesc);
    }

    public void evaluateSession(String sessionId, SessionStatus newStatus) {
        Session session = sessions.get(sessionId);
        if (session != null) {
            session.setStatus(newStatus);
            User speaker = userService.getUser(session.getSpeakerUserId());
            if (speaker != null) {
                notificationService.sendEmail(speaker.getEmail(), "Session Evaluation Update: " + session.getTitle(),
                        "Your session '" + session.getTitle() + "' has been marked as " + newStatus + ".");
            }
        }
    }

    public void bulkApproveSessions(List<String> sessionIds) {
        for (String id : sessionIds) {
            evaluateSession(id, SessionStatus.ACCEPTED);
        }
    }

    public void assignRoomAndTimeSlot(String sessionId, String room, String day, String timeSlot) {
        Session session = sessions.get(sessionId);
        if (session != null) {
            session.setRoom(room);
            session.setDay(day);
            session.setTimeSlot(timeSlot);
        }
    }

    public List<Session> getApprovedSessions() {
        return sessions.values().stream()
                .filter(s -> s.getStatus() == SessionStatus.ACCEPTED)
                .collect(Collectors.toList());
    }

    public List<Session> getSessionsForSpeaker(String speakerUserId) {
        return sessions.values().stream()
                .filter(s -> s.getSpeakerUserId().equals(speakerUserId))
                .collect(Collectors.toList());
    }

    public List<Session> getSessionsForSponsorAttendees(String sponsorId, SponsorshipService sponsorshipService) {
        Sponsor sponsor = sponsorshipService.getSponsor(sponsorId);
        if (sponsor == null) return Collections.emptyList();
        return sessions.values().stream()
                .filter(s -> sponsor.getLinkedAttendeeUserIds().contains(s.getSpeakerUserId()))
                .collect(Collectors.toList());
    }

    public List<Session> getAllSessions() {
        return new ArrayList<>(sessions.values());
    }

    public Session getSession(String id) {
        return sessions.get(id);
    }
}

    public List<String> getUserSchedule(String userId) {
        User user = userService.getUser(userId);
        if (user == null) return List.of();
        List<String> schedule = new ArrayList<>();
        for (String id : user.getScheduleItemIds()) {
            Training t = trainingService.getTraining(id);
            if (t != null) {
                schedule.add("[Training] " + t.getTitle() + " on " + t.getDate() + " (" + t.getStartTime() + " - " + t.getEndTime() + ") @ " + t.getRoomName());
                continue;
            }
            Summit s = summitService.getSummit(id);
            if (s != null) {
                schedule.add("[Summit] " + s.getTitle() + " on " + s.getDate() + " (" + s.getStartTime() + " - " + s.getEndTime() + ") @ " + s.getRoomName());
                continue;
            }
            Session ses = sessionService.getSession(id);
            if (ses != null) {
                schedule.add("[Session] " + ses.getTitle() + " (" + ses.getDay() + " " + ses.getTimeSlot() + ") @ " + ses.getRoom());
            }
        }
        return schedule;
    }
