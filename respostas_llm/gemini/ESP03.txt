        if (dimensions != null) this.dimensions.addAll(dimensions);
        if (measures != null) this.measures.addAll(measures);

        dimensions.removeIf(d -> d.getName().equalsIgnoreCase(dimension.getName()));

        dimensions.removeIf(d -> d.getName().equalsIgnoreCase(name));
    }

        measures.removeIf(m -> m.getName().equalsIgnoreCase(measure.getName()));
        measures.add(Objects.requireNonNull(measure));
    }

    public void removeMeasure(String name) {
        measures.removeIf(m -> m.getName().equalsIgnoreCase(name));
    }

        return Collections.unmodifiableList(dimensions);

    public List<Measure> getMeasures() {
        return Collections.unmodifiableList(measures);

        return val != null ? val.toString() : null;

        if (val instanceof Number n) {
            return n.doubleValue();
        } else if (val instanceof String s) {

                return Double.parseDouble(s);
            } catch (NumberFormatException ignored) {}
        }
        return 0.0;

            return n.intValue();
        } else if (val instanceof String s) {
            try {
                return Integer.parseInt(s);
            } catch (NumberFormatException ignored) {}
        }
        return 0;
    }

    public Map<String, Object> getAll() {
        return new HashMap<>(fields);
    }

    @Override
    public String toString() {
        return fields.toString();
    }

// File: CsvValidator.java
import java.util.ArrayList;
import java.util.List;

public class CsvValidator {
    public static CsvValidationResult validateCsv(String rawCsvContent) {
        List<String> errors = new ArrayList<>();
        if (rawCsvContent == null || rawCsvContent.trim().isEmpty()) {
            errors.add("CSV content is empty.");
            return new CsvValidationResult(false, errors);
        }

        String[] lines = rawCsvContent.split("\\r?\\n");
        if (lines.length < 2) {
            errors.add("CSV must contain at least a header and one row of data.");
            return new CsvValidationResult(false, errors);
        }

        String[] headers = lines[0].split(",");
        int expectedColumnCount = headers.length;

        for (int i = 0; i < headers.length; i++) {
            if (headers[i].trim().isEmpty()) {
                errors.add("Header at column index " + (i + 1) + " is blank.");
            }
        }

        for (int rowIdx = 1; rowIdx < lines.length; rowIdx++) {
            String line = lines[rowIdx].trim();
            if (line.isEmpty()) continue;
            String[] tokens = line.split(",", -1);
            if (tokens.length != expectedColumnCount) {
                errors.add("Row " + (rowIdx + 1) + " has " + tokens.length + " columns; expected " + expectedColumnCount);
            }
        }

        return new CsvValidationResult(errors.isEmpty(), errors);
    }
}

public class GeoRequirementChecker {
    public static GeoMapRequirements checkMapRequirements(Dataset dataset) {
        List<String> missing = new ArrayList<>();
        boolean hasGeo = dataset.getModel().getDimensions().stream()
                .anyMatch(d -> d.getType() == DimensionType.GEO_HASC);

        if (!hasGeo) {
            missing.add("DataModel must have a GEO_HASC dimension (Hierarchical Administrative Subdivision Codes).");
        }

        boolean hasMeasure = !dataset.getModel().getMeasures().isEmpty();
        if (!hasMeasure) {
            missing.add("DataModel must contain at least one numeric measure to map as choropleth or bubble.");
        }

        boolean eligible = missing.isEmpty();
        String desc = eligible
                ? "Dataset fulfills map visual requirements. HASC code dimension linked."
                : "Dataset requires a GEO_HASC dimension (e.g., US.CA, DE.NW) and at least one measure.";

        return new GeoMapRequirements(eligible, missing, desc);
    }
}

import java.util.ArrayList;
import java.util.List;

public class DataPackager {

    public static List<DataRow> importCsv(String rawCsv) {
        List<DataRow> rows = new ArrayList<>();
        String[] lines = rawCsv.split("\\r?\\n");
        if (lines.length <= 1) return rows;

        String[] headers = lines[0].split(",");
        for (int i = 0; i < headers.length; i++) {
            headers[i] = headers[i].trim();
        }

        for (int i = 1; i < lines.length; i++) {
            String line = lines[i].trim();
            if (line.isEmpty()) continue;
            String[] tokens = line.split(",", -1);
            DataRow row = new DataRow();
            for (int col = 0; col < headers.length; col++) {
                String val = col < tokens.length ? tokens[col].trim() : "";
                row.set(headers[col], val);
            }
            rows.add(row);
        }
        return rows;
    }

    public static List<DataRow> importJson(String rawJson) {
        // Simple JSON parser for array of flat objects: [{"k":"v"}, ...]
        List<DataRow> rows = new ArrayList<>();
        String trimmed = rawJson.trim();
        if (trimmed.startsWith("[") && trimmed.endsWith("]")) {
            String content = trimmed.substring(1, trimmed.length() - 1).trim();
            String[] objects = content.split("(?<=\\}),\\s*(?=\\{)");
            for (String objStr : objects) {
                objStr = objStr.replace("{", "").replace("}", "").trim();
                if (objStr.isEmpty()) continue;
                DataRow row = new DataRow();
                String[] pairs = objStr.split(",");
                for (String pair : pairs) {
                    String[] kv = pair.split(":", 2);
                    if (kv.length == 2) {
                        String k = kv[0].replaceAll("[\"\\s]", "");
                        String v = kv[1].replaceAll("[\"\\s]", "");
                        row.set(k, v);
                    }
                }
                rows.add(row);
            }
        }
        return rows;
    }

    public static List<DataRow> importExcelMock(String simulatedExcelRows) {
        // Simulates Excel reading via TSV/CSV-like sheet representation
        return importCsv(simulatedExcelRows);
    }

    public static List<DataRow> importGoogleSpreadsheetMock(String sheetUrl, String sheetContent) {
        // In real system, calls Google Sheets API; parses tabular response
        return importCsv(sheetContent);
    }

    public static List<DataRow> importFiscalDataPackageDescriptor(String descriptorJson, String csvData) {
        // Reads Fiscal Data Package specification format and loads rows
        return importCsv(csvData);
    }
}

        ));

        geoDatabase.put(loc.getHascCode().toUpperCase(), loc);
    }

        return Optional.ofNullable(geoDatabase.get(hascCode.toUpperCase()));
    }

    public List<GeoLocation> getBorderingRegions(String hascCode) {
        List<GeoLocation> neighbors = new ArrayList<>();
        Optional<GeoLocation> locOpt = findByHasc(hascCode);
        if (locOpt.isPresent()) {
            for (String borderHasc : locOpt.get().getBorderingHascCodes()) {
                findByHasc(borderHasc).ifPresent(neighbors::add);
            }
        }
        return neighbors;
    }

import java.util.HashMap;
import java.util.Map;

public class CurrencyConverter {
    private final Map<String, Double> ratesToUsd = new HashMap<>();

    public CurrencyConverter() {
        ratesToUsd.put("USD", 1.0);
        ratesToUsd.put("EUR", 1.08); // 1 EUR = 1.08 USD
        ratesToUsd.put("GBP", 1.25); // 1 GBP = 1.25 USD
        ratesToUsd.put("JPY", 0.0064);
        ratesToUsd.put("CHF", 1.10);
    }

    public void setExchangeRate(String currency, double rateToUsd) {
        ratesToUsd.put(currency.toUpperCase(), rateToUsd);
    }

    public double convert(double amount, String fromCurrency, String toCurrency) {
        if (fromCurrency.equalsIgnoreCase(toCurrency)) {
            return amount;
        }
        Double fromRate = ratesToUsd.get(fromCurrency.toUpperCase());
        Double toRate = ratesToUsd.get(toCurrency.toUpperCase());
        if (fromRate == null || toRate == null) {
            throw new IllegalArgumentException("Unsupported currency: " + fromCurrency + " or " + toCurrency);
        }
        double amountInUsd = amount * fromRate;
        return amountInUsd / toRate;
    }
}

        this.rows = rows != null ? rows : new ArrayList<>();
        this.totalMeasureSum = totalMeasureSum;
    }

    public List<CubeQueryResultRow> getRows() {
        return rows;
    }

    public double getTotalMeasureSum() {
        return totalMeasureSum;
    }

    public String toCsv() {
        if (rows.isEmpty()) {
            return "No data";
        }
        List<String> dimKeys = new ArrayList<>(rows.get(0).getDimensions().keySet());
        StringBuilder sb = new StringBuilder();
        for (String k : dimKeys) {
            sb.append(k).append(",");
        }
        sb.append(rows.get(0).getMeasureName());
        if (rows.get(0).getCurrency() != null) {
            sb.append(",currency");
        }
        sb.append("\n");

        for (CubeQueryResultRow row : rows) {
            for (String k : dimKeys) {
                sb.append(row.getDimensions().get(k)).append(",");
            }
            sb.append(row.getValue());
            if (row.getCurrency() != null) {
                sb.append(",").append(row.getCurrency());
            }
            sb.append("\n");
        }
        return sb.toString();
    }
}

import java.util.*;
import java.util.stream.Collectors;

public class CubeEngine {
    private final CurrencyConverter currencyConverter;
    private final GeoSpatialService geoSpatialService;

    public CubeEngine(CurrencyConverter currencyConverter, GeoSpatialService geoSpatialService) {
        this.currencyConverter = currencyConverter;
        this.geoSpatialService = geoSpatialService;
    }

    public CubeQueryResult executeQuery(Dataset dataset, CubeQuery query) {
        List<DataRow> filteredRows = new ArrayList<>(dataset.getRecords());

        // Apply filters
        for (Map.Entry<String, Object> entry : query.getFilters().entrySet()) {
            filteredRows = filteredRows.stream()
                    .filter(r -> Objects.equals(String.valueOf(r.get(entry.getKey())), String.valueOf(entry.getValue())))
                    .collect(Collectors.toList());
        }

        String measureName = query.getMeasure();
        if (measureName == null && !dataset.getModel().getMeasures().isEmpty()) {
            measureName = dataset.getModel().getMeasures().get(0).getName();
        }

        final String activeMeasure = measureName;
        String baseCurrency = dataset.getModel().findMeasure(activeMeasure)
                .map(Measure::getCurrency)
                .orElse("USD");

        double totalOverall = 0.0;
        for (DataRow r : filteredRows) {
            totalOverall += r.getDouble(activeMeasure);
        }

        // Grouping
        Map<Map<String, Object>, List<DataRow>> groups = new LinkedHashMap<>();
        for (DataRow r : filteredRows) {
            Map<String, Object> key = new LinkedHashMap<>();
            for (String groupDim : query.getGroupByDimensions()) {
                key.put(groupDim, r.get(groupDim));
            }
            groups.computeIfAbsent(key, k -> new ArrayList<>()).add(r);
        }

        List<CubeQueryResultRow> resultRows = new ArrayList<>();
        String targetCurrency = query.getTargetCurrency() != null ? query.getTargetCurrency() : baseCurrency;

        for (Map.Entry<Map<String, Object>, List<DataRow>> entry : groups.entrySet()) {
            Map<String, Object> groupKey = entry.getKey();
            List<DataRow> groupData = entry.getValue();

            double aggValue = 0.0;
            switch (query.getAggregation()) {
                case SUM -> aggValue = groupData.stream().mapToDouble(r -> r.getDouble(activeMeasure)).sum();
                case COUNT -> aggValue = groupData.size();
                case AVG -> aggValue = groupData.stream().mapToDouble(r -> r.getDouble(activeMeasure)).average().orElse(0.0);
                case MIN -> aggValue = groupData.stream().mapToDouble(r -> r.getDouble(activeMeasure)).min().orElse(0.0);
                case MAX -> aggValue = groupData.stream().mapToDouble(r -> r.getDouble(activeMeasure)).max().orElse(0.0);
            }

            // Currency conversion if needed
            if (query.getAggregation() != AggregationType.COUNT && !baseCurrency.equalsIgnoreCase(targetCurrency)) {
                aggValue = currencyConverter.convert(aggValue, baseCurrency, targetCurrency);
            }

            // Normalization
            if (query.getNormalization() == NormalizationType.PERCENTAGE_OF_TOTAL && totalOverall > 0) {
                aggValue = (aggValue / totalOverall) * 100.0;
            } else if (query.getNormalization() != NormalizationType.NONE) {
                // If grouped by HASC or contains GEO
                String hasc = null;
                for (Object v : groupKey.values()) {
                    if (v != null && geoSpatialService.findByHasc(v.toString()).isPresent()) {
                        hasc = v.toString();
                        break;
                    }
                }
                if (hasc != null) {
                    GeoLocation loc = geoSpatialService.findByHasc(hasc).get();
                    switch (query.getNormalization()) {
                        case POPULATION -> aggValue = loc.getPopulation() > 0 ? aggValue / loc.getPopulation() : aggValue;
                        case AREA -> aggValue = loc.getAreaKm2() > 0 ? aggValue / loc.getAreaKm2() : aggValue;
                        case GDP -> aggValue = loc.getGdp() > 0 ? aggValue / loc.getGdp() : aggValue;
                        case GINI -> aggValue = loc.getGiniIndex() > 0 ? aggValue / loc.getGiniIndex() : aggValue;
                        default -> {}
                    }
                }
            }

            resultRows.add(new CubeQueryResultRow(groupKey, aggValue, activeMeasure, targetCurrency));
        }

        // Sorting
        if (query.getSortByDimensionOrMeasure() != null) {
            String sortField = query.getSortByDimensionOrMeasure();
            boolean isAsc = query.getSortDirection() == SortDirection.ASC;
            resultRows.sort((r1, r2) -> {
                int cmp;
                if (sortField.equalsIgnoreCase(activeMeasure)) {
                    cmp = Double.compare(r1.getValue(), r2.getValue());
                } else {
                    Object val1 = r1.getDimensions().get(sortField);
                    Object val2 = r2.getDimensions().get(sortField);
                    cmp = String.valueOf(val1).compareTo(String.valueOf(val2));
                }
                return isAsc ? cmp : -cmp;
            });
        }

        return new CubeQueryResult(resultRows, totalOverall);
    }

    public Integer detectMostRecentYear(Dataset dataset) {
        Optional<Dimension> yearDim = dataset.getModel().getDimensions().stream()
                .filter(d -> d.getType() == DimensionType.YEAR)
                .findFirst();
        if (yearDim.isEmpty()) return null;

        String name = yearDim.get().getName();
        return dataset.getRecords().stream()
                .map(r -> r.getInteger(name))
                .filter(Objects::nonNull)
                .max(Integer::compareTo)
                .orElse(null);
    }
}

    public List<Dataset> queryDatasetsByMetadata(String metaKey, String metaValue) {
        return datasetRepository.findPublicDatasets().stream()
                .filter(d -> metaValue.equalsIgnoreCase(d.getMetadata().get(metaKey)))
                .toList();
    }

    public List<DataRow> queryAcrossDatasetsByData(String column, Object value) {
        List<DataRow> combined = new ArrayList<>();
        for (Dataset d : datasetRepository.findPublicDatasets()) {
            for (DataRow r : d.getRecords()) {
                if (Objects.equals(String.valueOf(r.get(column)), String.valueOf(value))) {
                    combined.add(r);
                }
            }
        }
        return combined;
    }

        return geoSpatialService.getBorderingRegions(hascCode);

// File: PythonClientSimulator.java
import java.util.List;

/**
 * Simulates the OpenSpending Python client (e.g., 'os-client')
 * allows command line / programmatic bulk upload and management.
 */
public class PythonClientSimulator {
    private final OpenSpendingApi api;
    private final User authenticatedUser;

    public PythonClientSimulator(OpenSpendingApi api, User authenticatedUser) {
        this.api = api;
        this.authenticatedUser = authenticatedUser;
    }

    public Dataset bulkAddDataset(String datasetId, String title, String description,
                                  DataModel model, DataSourceType sourceType,
                                  String sourcePath, List<DataRow> rows, boolean isHidden) {
        DataSource ds = new DataSource(sourceType, sourcePath, true);
        DatasetVisibility vis = isHidden ? DatasetVisibility.HIDDEN : DatasetVisibility.PUBLIC;
        Dataset dataset = new Dataset(datasetId, title, description, authenticatedUser.getId(), vis, model, ds);
        dataset.setRecords(rows);
        api.publishDataset(authenticatedUser, dataset);
        return dataset;
    }

    public String dumpCubeCsv(String datasetId, CubeQuery query) {
        Dataset ds = api.getDatasetRepository().findById(datasetId)
                .orElseThrow(() -> new IllegalArgumentException("Dataset not found"));
        CubeQueryResult res = api.getCubeEngine().executeQuery(ds, query);
        return res.toCsv();
    }
}
