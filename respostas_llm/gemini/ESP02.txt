import java.time.DayOfWeek;
import java.time.LocalTime;
import java.util.EnumMap;
import java.util.Map;

public class OperatingHours {
    public static class TimeSlot {
        private final LocalTime openTime;
        private final LocalTime closeTime;

        public TimeSlot(LocalTime openTime, LocalTime closeTime) {
            this.openTime = openTime;
            this.closeTime = closeTime;
        }

        public LocalTime getOpenTime() {
            return openTime;
        }

        public LocalTime getCloseTime() {
            return closeTime;
        }

        public boolean covers(LocalTime time) {
            return !time.isBefore(openTime) && !time.isAfter(closeTime);
        }

        @Override
        public String toString() {
            return openTime + " - " + closeTime;
        }
    }

    private final Map<DayOfWeek, TimeSlot> schedule = new EnumMap<>(DayOfWeek.class);

    public void setHours(DayOfWeek day, LocalTime open, LocalTime close) {
        schedule.put(day, new TimeSlot(open, close));
    }

    public TimeSlot getHours(DayOfWeek day) {
        return schedule.get(day);
    }

    public boolean isOpen(DayOfWeek day, LocalTime time) {
        TimeSlot slot = schedule.get(day);
        return slot != null && slot.covers(time);
    }

    public Map<DayOfWeek, TimeSlot> getAllHours() {
        return new EnumMap<>(schedule);
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        for (DayOfWeek day : DayOfWeek.values()) {
            TimeSlot slot = schedule.get(day);
            sb.append(day).append(": ").append(slot != null ? slot.toString() : "Closed").append("; ");
        }
        return sb.toString();
    }
}

// File: SecurityUtils.java
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.Base64;

public final class SecurityUtils {
    private SecurityUtils() {}

    public static String hash(String input) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] encoded = digest.digest(input.getBytes(StandardCharsets.UTF_8));
            return Base64.getEncoder().encodeToString(encoded);
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException("SHA-256 algorithm not found", e);
        }
    }

    public static String encryptData(String rawData, String secretKey) {
        byte[] rawBytes = rawData.getBytes(StandardCharsets.UTF_8);
        byte[] keyBytes = secretKey.getBytes(StandardCharsets.UTF_8);
        byte[] xorBytes = new byte[rawBytes.length];
        for (int i = 0; i < rawBytes.length; i++) {
            xorBytes[i] = (byte) (rawBytes[i] ^ keyBytes[i % keyBytes.length]);
        }
        return Base64.getEncoder().encodeToString(xorBytes);
    }

    public static String decryptData(String encryptedBase64, String secretKey) {
        byte[] decoded = Base64.getDecoder().decode(encryptedBase64);
        byte[] keyBytes = secretKey.getBytes(StandardCharsets.UTF_8);
        byte[] rawBytes = new byte[decoded.length];
        for (int i = 0; i < decoded.length; i++) {
            rawBytes[i] = (byte) (decoded[i] ^ keyBytes[i % keyBytes.length]);
        }
        return new String(rawBytes, StandardCharsets.UTF_8);
    }
}

import java.time.DayOfWeek;
import java.time.LocalTime;
import java.util.EnumMap;
import java.util.Map;

public class UserSchedule {
    public static class AvailabilitySlot {
        private final LocalTime availableFrom;
        private final LocalTime availableTo;

        public AvailabilitySlot(LocalTime availableFrom, LocalTime availableTo) {
            this.availableFrom = availableFrom;
            this.availableTo = availableTo;
        }

        public LocalTime getAvailableFrom() {
            return availableFrom;
        }

        public LocalTime getAvailableTo() {
            return availableTo;
        }

        public boolean overlaps(LocalTime open, LocalTime close) {
            return !(availableTo.isBefore(open) || availableFrom.isAfter(close));
        }

        @Override
        public String toString() {
            return availableFrom + " to " + availableTo;
        }
    }

    private final Map<DayOfWeek, AvailabilitySlot> weekSchedule = new EnumMap<>(DayOfWeek.class);

    public void setAvailability(DayOfWeek day, LocalTime from, LocalTime to) {
        weekSchedule.put(day, new AvailabilitySlot(from, to));
    }

    public AvailabilitySlot getAvailability(DayOfWeek day) {
        return weekSchedule.get(day);
    }

    public Map<DayOfWeek, AvailabilitySlot> getWeekSchedule() {
        return new EnumMap<>(weekSchedule);
    }

    public boolean fitsFacility(RecyclingFacility facility) {
        OperatingHours hours = facility.getOperatingHours();
        if (hours == null) return false;
        for (Map.Entry<DayOfWeek, AvailabilitySlot> entry : weekSchedule.entrySet()) {
            OperatingHours.TimeSlot slot = hours.getHours(entry.getKey());
            if (slot != null && entry.getValue().overlaps(slot.getOpenTime(), slot.getCloseTime())) {
                return true;
            }
        }
        return false;
    }
}

public class BillingStatement {

    private final LocalDate issueDate;
    private final List<TransactionRecord> transactions;
    private final double totalActivityFees;

    public BillingStatement(String invoiceNumber, LocalDate issueDate, List<TransactionRecord> transactions) {
        this.invoiceNumber = invoiceNumber;
        this.issueDate = issueDate;
        this.transactions = transactions;
        this.totalActivityFees = transactions.stream().mapToDouble(TransactionRecord::getActivityFee).sum();
    }

        return invoiceNumber;
    }

    public LocalDate getIssueDate() {
        return issueDate;
    }

    public List<TransactionRecord> getTransactions() {

    }

    public double getTotalActivityFees() {
        return totalActivityFees;
    }

    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        sb.append("=== BILLING STATEMENT ===").append("\n");
        sb.append("Invoice #: ").append(invoiceNumber).append(" | Date: ").append(issueDate).append("\n");
        sb.append("Activity Count: ").append(transactions.size()).append("\n");
        for (TransactionRecord record : transactions) {
            sb.append("  - Tx: ").append(record.getId())
              .append(" | ").append(record.getWasteType())
              .append(" | Fee: $").append(String.format("%.2f", record.getActivityFee()))
              .append("\n");
        }
        sb.append("Total Activity Fees Payable: $").append(String.format("%.2f", totalActivityFees));
