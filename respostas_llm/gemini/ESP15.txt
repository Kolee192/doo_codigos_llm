import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.*;

public class FileAsset {
    private final String fileId;
    private final String filename;
    private final String mimeType;
    private final long sizeBytes;
    private final boolean streamingMedia;
    private final String doi;
    private final Map<String, byte[]> storageReplicas = new HashMap<>(); // Node -> Content
    private String verifiedChecksum;
    private final Map<String, String> technicalMetadata = new HashMap<>();

    public FileAsset(String fileId, String filename, String mimeType, long sizeBytes, byte[] content, boolean streamingMedia, String doi) {
        this.fileId = fileId;
        this.filename = filename;
        this.mimeType = mimeType;
        this.sizeBytes = sizeBytes;
        this.streamingMedia = streamingMedia;
        this.doi = doi;
        this.verifiedChecksum = calculateChecksum(content);
        // Redundant replication across distinct storage locations
        storageReplicas.put("Primary-Cornell-IR", Arrays.copyOf(content, content.length));
        storageReplicas.put("Replica-Lockss-Preservation", Arrays.copyOf(content, content.length));
        storageReplicas.put("Replica-Geographic-Secondary", Arrays.copyOf(content, content.length));
    }

    private String calculateChecksum(byte[] data) {
        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            byte[] hash = md.digest(data);
            StringBuilder hexString = new StringBuilder();
            for (byte b : hash) {
                String hex = Integer.toHexString(0xff & b);
                if (hex.length() == 1) hexString.append('0');
                hexString.append(hex);
            }
            return hexString.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException("SHA-256 not available", e);
        }
    }

    public boolean verifyIntegrityAndFix() {
        boolean allGood = true;
        byte[] goldenMaster = null;
        for (Map.Entry<String, byte[]> entry : storageReplicas.entrySet()) {
            String check = calculateChecksum(entry.getValue());
            if (check.equals(verifiedChecksum)) {
                goldenMaster = entry.getValue();
            } else {
                allGood = false;
            }
        }
        if (!allGood && goldenMaster != null) {
            // Restore corrupted replicas from golden master
            for (String key : new ArrayList<>(storageReplicas.keySet())) {
                storageReplicas.put(key, Arrays.copyOf(goldenMaster, goldenMaster.length));
            }
            return false; // had corruption, but restored
        }
        return allGood;
    }

    public void simulateCorruptionOnNode(String nodeName) {
        if (storageReplicas.containsKey(nodeName)) {
            byte[] corrupted = new byte[] {0, 1, 2, 3};
            storageReplicas.put(nodeName, corrupted);
        }
    }

    public byte[] retrieveContent() {
        return storageReplicas.get("Primary-Cornell-IR");
    }

    public String getFileId() { return fileId; }
    public String getFilename() { return filename; }
    public String getMimeType() { return mimeType; }
    public long getSizeBytes() { return sizeBytes; }
    public boolean isStreamingMedia() { return streamingMedia; }
    public String getDoi() { return doi; }
    public String getVerifiedChecksum() { return verifiedChecksum; }
    public Map<String, String> getTechnicalMetadata() { return technicalMetadata; }
    public void addTechnicalMetadata(String key, String value) { technicalMetadata.put(key, value); }
}

import java.util.*;

public class NotificationService {
    // Subscriber Email -> Subscribed Item / Collection / Author IDs
    private final Map<String, Set<String>> patronSubscriptions = new HashMap<>();
    private final List<String> sentEmailLog = new ArrayList<>();

    public void subscribe(String email, String subscriptionTargetId) {
        patronSubscriptions.computeIfAbsent(email, k -> new HashSet<>()).add(subscriptionTargetId);
    }

    public void unsubscribe(String email, String subscriptionTargetId) {
        if (patronSubscriptions.containsKey(email)) {
            patronSubscriptions.get(email).remove(subscriptionTargetId);
        }
    }

    public Set<String> getSubscriptionsForEmail(String email) {
        return patronSubscriptions.getOrDefault(email, Collections.emptySet());
    }

    public List<String> getSubscribersForTarget(String targetId) {
        List<String> subscribers = new ArrayList<>();
        for (Map.Entry<String, Set<String>> entry : patronSubscriptions.entrySet()) {
            if (entry.getValue().contains(targetId)) {
                subscribers.add(entry.getKey());
            }
        }
        return subscribers;
    }

    public void notifySubscribers(String targetId, String itemTitle, String message) {
        List<String> subscribers = getSubscribersForTarget(targetId);
        for (String email : subscribers) {
            String record = "EMAIL TO [" + email + "] Regarding [" + itemTitle + "]: " + message;
            sentEmailLog.add(record);
        }
    }

    public void notifyAuthorDownload(String authorEmail, String itemTitle, String downloaderInfo) {
        String notice = "DOWNLOAD NOTICE TO [" + authorEmail + "]: Your item '" + itemTitle + "' was downloaded by " + downloaderInfo;
        sentEmailLog.add(notice);
    }

    public List<String> getSentEmailLog() {
        return Collections.unmodifiableList(sentEmailLog);
    }
}

    // Security & Auth
    public void registerUser(User user) { users.put(user.getUserId(), user); }
    public User getUser(String userId) { return users.get(userId); }
    public User login(String username, String password) {
        return users.values().stream()
                .filter(u -> u.getUsername().equalsIgnoreCase(username) && u.getPassword().equals(password) && u.isActive())
                .findFirst().orElse(null);
    }
    public User loginViaShibboleth(String eppn) {
        return users.values().stream()
                .filter(u -> u.getEmail().equalsIgnoreCase(eppn) && u.isCornellAffiliated())
                .findFirst().orElse(null);
    }

    // Collection Management
    public RepositoryCollection createCollection(String id, String name, String description, String ownerId) {
        RepositoryCollection c = new RepositoryCollection(id, name, description, ownerId);
        collections.put(id, c);
        return c;
    }
    public RepositoryCollection getCollection(String id) { return collections.get(id); }
    public void renameCollection(String id, String newName) {
        RepositoryCollection c = collections.get(id);
        if (c != null) c.setName(newName);
    }
    public void moveCollection(String id, String newParentCollectionId) {
        RepositoryCollection c = collections.get(id);
        if (c != null) c.setParentCollectionId(newParentCollectionId);
    }

    // Item Management
    public RepositoryItem depositItem(RepositoryItem item, String targetCollectionId, User depositor) {
        items.put(item.getItemId(), item);
        if (targetCollectionId != null && collections.containsKey(targetCollectionId)) {
            collections.get(targetCollectionId).addItem(item.getItemId());
            item.assignCollection(targetCollectionId);
        }
        item.addRevision("Initial Deposit", "Item deposited into repository by " + depositor.getFullName(), depositor);
        syncService.syncWithSymplecticElements(item);
        syncService.syncWithActivityInsight(item);

        if (item.isSubscribable()) {
            notificationService.notifySubscribers(item.getItemId(), item.getPrimaryTitle(), "New content deposited: " + item.getPrimaryTitle());
            if (targetCollectionId != null) {
                notificationService.notifySubscribers(targetCollectionId, item.getPrimaryTitle(), "New content added to collection " + targetCollectionId);
            }
        }
        return item;
    }

    public void mapItemToCollection(String itemId, String targetCollectionId) {
        RepositoryItem it = items.get(itemId);
        RepositoryCollection col = collections.get(targetCollectionId);
        if (it != null && col != null) {
            col.addItem(itemId);
            it.assignCollection(targetCollectionId);
        }
    }

    public void markItemErrorHidden(String itemId, String reason, User admin) {
        RepositoryItem item = items.get(itemId);
        if (item != null) {
            item.setStatus(ItemStatus.FLAGGED_ERROR_HIDDEN);
            item.setSubscribable(false);
            item.addRevision("Marked Error / Hidden", reason, admin);
        }
    }

    public void unpublishItemByAgency(String itemId, String reason, User admin) {
        RepositoryItem item = items.get(itemId);
        if (item != null) {
            item.setStatus(ItemStatus.UNPUBLISHED_BY_AGENCY);
            item.addRevision("Unpublished per agency request", reason, admin);
        }
    }

    // Access control evaluation
    public boolean canUserAccessItem(User user, RepositoryItem item) {
        if (item.getStatus() == ItemStatus.FLAGGED_ERROR_HIDDEN || item.getStatus() == ItemStatus.UNPUBLISHED_BY_AGENCY) {
            return user != null && (user.hasRole(UserRole.DB_IR_ADMINISTRATOR) || user.hasRole(UserRole.LIBRARY_STAFF));
        }
        if (item.isEmbargoActive()) {
            return user != null && (user.hasRole(UserRole.DB_IR_ADMINISTRATOR) || user.hasRole(UserRole.LIBRARY_STAFF) || item.getAuthors().contains(user.getFullName()));
        }
        if (item.getAccessLevel() == AccessLevel.PUBLIC) return true;
        if (item.getAccessLevel() == AccessLevel.CAMPUS_ONLY) {
            return user != null && user.isCornellAffiliated();
        }
        if (item.getAccessLevel() == AccessLevel.INTERNAL_GROUP) {
            return user != null;
        }
        return false;
    }

    // View & Download Actions
    public RepositoryItem viewItem(String itemId, User user) {
        RepositoryItem it = items.get(itemId);
        if (it != null && canUserAccessItem(user, it)) {
            it.recordView();
            return it;
        }
        return null;
    }

    public FileAsset downloadFile(String itemId, String fileId, User user) {
        RepositoryItem it = viewItem(itemId, user);
        if (it != null) {
            for (ItemVersion ver : it.getVersions()) {
                for (FileAsset fa : ver.getFiles()) {
                    if (fa.getFileId().equals(fileId)) {
                        it.recordDownload();
                        // Notify author if faculty
                        for (User u : users.values()) {
                            if (it.getAuthors().contains(u.getFullName()) && u.hasRole(UserRole.FACULTY)) {
                                notificationService.notifyAuthorDownload(u.getEmail(), it.getPrimaryTitle(), (user != null ? user.getFullName() : "Anonymous patron"));
                            }
                        }
                        return fa;
                    }
                }
            }
        }
        return null;
    }

    // Search and Browsing
    public List<RepositoryItem> searchKeyword(String query, String collectionFilter, String subjectFilter) {
        String q = query == null ? "" : query.toLowerCase();
        return items.values().stream()
                .filter(it -> it.getStatus() == ItemStatus.PUBLISHED || it.getStatus() == ItemStatus.HISTORICAL_ARCHIVED)
                .filter(it -> collectionFilter == null || it.getCollectionIds().contains(collectionFilter))
                .filter(it -> subjectFilter == null || subjectFilter.equalsIgnoreCase(it.getSubjectArea()))
                .filter(it -> it.getPrimaryTitle().toLowerCase().contains(q)
                        || (it.getDescription() != null && it.getDescription().toLowerCase().contains(q))
                        || it.getKeywords().stream().anyMatch(k -> k.toLowerCase().contains(q))
                        || it.getHistoricalNames().stream().anyMatch(hn -> hn.toLowerCase().contains(q))
                        || it.getAuthors().stream().anyMatch(a -> a.toLowerCase().contains(q)))
                .collect(Collectors.toList());
    }

    public List<RepositoryItem> searchByAuthorName(String authorQuery) {
        String aq = authorQuery.toLowerCase();
        return items.values().stream()
                .filter(it -> it.getAuthors().stream().anyMatch(a -> a.toLowerCase().contains(aq)))
                .sorted(Comparator.comparing(RepositoryItem::getDateCreated))
                .collect(Collectors.toList());
    }

    public List<RepositoryItem> searchDissertationsByCommitteeMember(String facultyName) {
        String fn = facultyName.toLowerCase();
        return items.values().stream()
                .filter(it -> it.getDocumentType() == DocumentType.DISSERTATION_THESIS)
                .filter(it -> it.getThesisCommitteeMembers().stream().anyMatch(m -> m.toLowerCase().contains(fn)))
                .collect(Collectors.toList());
    }

    public List<RepositoryItem> getItemsInCollectionSorted(String collectionId) {
        RepositoryCollection c = collections.get(collectionId);
        if (c == null) return Collections.emptyList();
        List<RepositoryItem> list = c.getItemIds().stream().map(items::get).filter(Objects::nonNull).collect(Collectors.toList());
        switch (c.getSortOrder()) {
            case TITLE_ASC -> list.sort(Comparator.comparing(RepositoryItem::getPrimaryTitle));
            case DATE_ASC -> list.sort(Comparator.comparing(RepositoryItem::getDateCreated));
            case DATE_DESC -> list.sort(Comparator.comparing(RepositoryItem::getDateCreated).reversed());
            case AUTHOR_ASC -> list.sort(Comparator.comparing(i -> i.getAuthors().isEmpty() ? "" : i.getAuthors().get(0)));
            case CUSTOM_ORDER -> list.sort(Comparator.comparingInt(i -> c.getCustomRank(i.getItemId())));
        }
        return list;
    }

    // OAI-PMH Harvest Simulation
    public String exportOaiPmhMetadataXml(String sinceDateIso) {
        StringBuilder sb = new StringBuilder();
        sb.append("<OAI-PMH xmlns=\"http://www.openarchives.org/OAI/2.0/\">\n");
        sb.append("  <responseDate>").append(LocalDate.now()).append("</responseDate>\n");
        sb.append("  <ListRecords>\n");
        for (RepositoryItem it : items.values()) {
            sb.append("    <record>\n");
            sb.append("      <header><identifier>oai:cornell.edu:").append(it.getItemId()).append("</identifier></header>\n");
            sb.append("      <metadata>\n");
            sb.append("        <dc:title>").append(it.getPrimaryTitle()).append("</dc:title>\n");
            sb.append("        <dc:creator>").append(String.join("; ", it.getAuthors())).append("</dc:creator>\n");
            sb.append("        <dc:subject>").append(it.getSubjectArea()).append("</dc:subject>\n");
            sb.append("        <dc:rights>").append(it.getLicenseType().getDisplayName()).append("</dc:rights>\n");
            sb.append("        <dc:identifier>").append(it.getPersistentHandleUrl()).append("</dc:identifier>\n");
            sb.append("      </metadata>\n");
            sb.append("    </record>\n");
        }
        sb.append("  </ListRecords>\n");
        sb.append("</OAI-PMH>");
        return sb.toString();
    }

    // Google Discovery Metadata Generator
    public String generateGoogleScholarMetaTags(String itemId) {
        RepositoryItem it = items.get(itemId);
        if (it == null) return "";
        StringBuilder sb = new StringBuilder();
        sb.append("<meta name=\"citation_title\" content=\"").append(it.getPrimaryTitle()).append("\">\n");
        for (String a : it.getAuthors()) {
            sb.append("<meta name=\"citation_author\" content=\"").append(a).append("\">\n");
        }
        sb.append("<meta name=\"citation_publication_date\" content=\"").append(it.getDateCreated().toLocalDate()).append("\">\n");
        sb.append("<meta name=\"citation_pdf_url\" content=\"").append(it.getPersistentHandleUrl()).append("/download.pdf\">\n");
        return sb.toString();
    }

    // Release Calendar
    public void addCalendarEvent(ReleaseCalendarEvent evt) { releaseCalendar.add(evt); }
    public List<ReleaseCalendarEvent> getReleaseCalendar() { return Collections.unmodifiableList(releaseCalendar); }
    public String exportCalendarIcs() {
        StringBuilder sb = new StringBuilder("BEGIN:VCALENDAR\nVERSION:2.0\nPRODID:-//Cornell IR//Calendar//EN\n");
        for (ReleaseCalendarEvent e : releaseCalendar) {
            sb.append(e.toIcsFormat());
        }
        sb.append("END:VCALENDAR\n");
        return sb.toString();
    }

    // Succession / Asset Management
