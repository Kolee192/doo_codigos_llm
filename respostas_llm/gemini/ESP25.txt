    private final String systemId;
    private final Set<String> permissions = new HashSet<>();

    public StaffUser(String id, String username, String password, String swipeCardId, Role role, String branchId, String systemId) {
        this.id = id;
        this.username = username;
        this.password = password;
        this.swipeCardId = swipeCardId;
        this.role = role;
        this.branchId = branchId;
        this.systemId = systemId;
        assignDefaultPermissions();

        permissions.add("RUN_REPORT");
        if (role == Role.LOCAL_SYS_ADMIN || role == Role.LIBRARY_MANAGER || role == Role.LIBRARY_DIRECTOR || role == Role.GLOBAL_SYS_ADMIN) {
            permissions.add("CREATE_REPORT");
            permissions.add("CLONE_REPORT");
            permissions.add("MANAGE_FOLDERS");
            permissions.add("ACCESS_ALL_TABLES");
        }
        if (role == Role.GLOBAL_SYS_ADMIN || role == Role.LOCAL_SYS_ADMIN) {
            permissions.add("PURGE_RECORDS");
            permissions.add("TRANSFER_BATCH");
        }
    }

    }

    }

    public String getSwipeCardId() {

import java.util.HashMap;
import java.util.Map;

public class AuthenticationService {
    private final Map<String, StaffUser> usersByUsername = new HashMap<>();
    private final Map<String, StaffUser> usersByCard = new HashMap<>();

    public void registerStaff(StaffUser staff) {
        usersByUsername.put(staff.getUsername(), staff);
        if (staff.getSwipeCardId() != null) {
            usersByCard.put(staff.getSwipeCardId(), staff);
        }
    }

    public StaffUser loginWithCredentials(String username, String password) {
        StaffUser user = usersByUsername.get(username);
        if (user != null && user.getPassword().equals(password)) {
            return user;
        }
        return null;
    }

    public StaffUser loginWithCardSwipe(String swipeCardId) {
        return usersByCard.get(swipeCardId);
    }
}

// File: FinancialLedger.java
import java.time.LocalDate;
import java.util.*;

public class FinancialLedger {
    private final List<Charge> charges = new ArrayList<>();
    private final List<Payment> payments = new ArrayList<>();
    private final List<AuditEntry> auditEntries = new ArrayList<>();
    private final Map<String, Double> waivedByBranch = new HashMap<>();

    public void addCharge(Charge charge) {
        charges.add(charge);
    }

    public void applyPayment(Payment payment, String chargeId) {
        payments.add(payment);
        for (Charge charge : charges) {
            if (charge.getId().equals(chargeId)) {
                double applied = Math.min(payment.getAmount(), charge.getBalanceRemaining());
                charge.reduceBalance(applied);
                auditEntries.add(new AuditEntry(
                        UUID.randomUUID().toString(),
                        payment.getId(),
                        chargeId,
                        payment.getPatronId(),
                        payment.getBranchId(),
                        applied,
                        "PAYMENT",
                        payment.getDate()
                ));
                break;
            }
        }
    }

    public void waiveCharge(String chargeId, String staffBranchId, double amount, LocalDate date) {
        for (Charge charge : charges) {
            if (charge.getId().equals(chargeId)) {
                double applied = Math.min(amount, charge.getBalanceRemaining());
                charge.reduceBalance(applied);
                waivedByBranch.put(staffBranchId, waivedByBranch.getOrDefault(staffBranchId, 0.0) + applied);
                auditEntries.add(new AuditEntry(
                        UUID.randomUUID().toString(),
                        "WAIVER-" + UUID.randomUUID().toString().substring(0, 5),
                        chargeId,
                        charge.getPatronId(),
                        staffBranchId,
                        applied,
                        "WAIVER",
                        date
                ));
                break;
            }
        }
    }

    public double getPatronBalance(String patronId) {
        return charges.stream()
                .filter(c -> c.getPatronId().equals(patronId))
                .mapToDouble(Charge::getBalanceRemaining)
                .sum();
    }

    public List<Charge> getCharges() {
        return charges;
    }

    public List<Payment> getPayments() {
        return payments;
    }

    public List<AuditEntry> getAuditEntries() {
        return auditEntries;
    }

    public double getWaivedAmountForBranch(String branchId) {
        return waivedByBranch.getOrDefault(branchId, 0.0);
    }
}

// File: QueryCriteria.java
import java.util.regex.Pattern;

public class QueryCriteria {
    private final String field;
    private final String operator; // EQUALS, REGEX, GREATER_THAN, LESS_THAN, IN_PICKLIST
    private final String value;

    public QueryCriteria(String field, String operator, String value) {
        this.field = field;
        this.operator = operator;
        this.value = value;
    }

    public String getField() {
        return field;
    }

    public String getOperator() {
        return operator;
    }

    public String getValue() {
        return value;
    }

    public boolean matches(String fieldValue) {
        if (fieldValue == null) return false;
        return switch (operator) {
            case "EQUALS" -> fieldValue.equalsIgnoreCase(value);
            case "REGEX" -> Pattern.compile(value).matcher(fieldValue).find();
            case "GREATER_THAN" -> Double.parseDouble(fieldValue) > Double.parseDouble(value);
            case "LESS_THAN" -> Double.parseDouble(fieldValue) < Double.parseDouble(value);
            case "IN_PICKLIST" -> {
                String[] items = value.split(",");
                for (String it : items) {
                    if (it.trim().equalsIgnoreCase(fieldValue)) yield true;
                }
                yield false;
            }
            default -> false;
        };
    }

    @Override
    public String toString() {
        return field + " " + operator + " '" + value + "'";
    }
}

        QueuedReport job = new QueuedReport(UUID.randomUUID().toString(), templateName, staffId, position);
        queue.add(job);
        return job;
    }

    public synchronized int getQueuePosition(String reportJobId) {
        for (QueuedReport job : queue) {
            if (job.getReportJobId().equals(reportJobId)) {
                return job.getQueuePosition();
            }
        }
        return -1;
    }

    public synchronized void processNext() {
        if (!queue.isEmpty()) {
            queue.remove(0);
            for (int i = 0; i < queue.size(); i++) {
                queue.get(i).setQueuePosition(i + 1);
            }
        }
    }
}

// File: QueryExecutionEngine.java
import java.util.*;

public class QueryExecutionEngine {
    public QueryResult executeQuery(StaffUser user, QueryDefinition query, LibraryRepository repository) {
        // Enforce field/table access permissions
        if (!user.hasPermission("ACCESS_ALL_TABLES") && query.getTargetRecordType().equalsIgnoreCase("PATRON")) {
            throw new SecurityException("Staff member " + user.getUsername() + " not authorized to query Patron records directly.");
        }

        List<Map<String, String>> results = new ArrayList<>();
        if ("ITEM".equalsIgnoreCase(query.getTargetRecordType())) {
            for (ItemRecord item : repository.getItems().values()) {
                if (evalItem(item, query)) {
                    Map<String, String> row = new LinkedHashMap<>();
                    for (String field : query.getSelectedFields()) {
                        row.put(field, getItemValue(item, field));
                    }
                    results.add(row);
                }
            }
        } else if ("PATRON".equalsIgnoreCase(query.getTargetRecordType())) {
            for (Patron patron : repository.getPatrons().values()) {
                if (evalPatron(patron, query)) {
                    Map<String, String> row = new LinkedHashMap<>();
                    for (String field : query.getSelectedFields()) {
                        row.put(field, getPatronValue(patron, field));
                    }
                    results.add(row);
                }
            }
        }
        return new QueryResult(query, results);
    }

    private boolean evalItem(ItemRecord item, QueryDefinition query) {
        if (query.getCriteria().isEmpty()) return true;
        boolean isAnd = "AND".equalsIgnoreCase(query.getBooleanLogic());
        for (QueryCriteria crit : query.getCriteria()) {
            boolean matches = crit.matches(getItemValue(item, crit.getField()));
            if (isAnd && !matches) return false;
            if (!isAnd && matches) return true;
        }
        return isAnd;
    }

    private boolean evalPatron(Patron patron, QueryDefinition query) {
        if (query.getCriteria().isEmpty()) return true;
        boolean isAnd = "AND".equalsIgnoreCase(query.getBooleanLogic());
        for (QueryCriteria crit : query.getCriteria()) {
            boolean matches = crit.matches(getPatronValue(patron, crit.getField()));
            if (isAnd && !matches) return false;
            if (!isAnd && matches) return true;
        }
        return isAnd;
    }

    private String getItemValue(ItemRecord item, String field) {
        return switch (field.toLowerCase()) {
            case "id" -> item.getId();
            case "barcode" -> item.getBarcode();
            case "callnumber" -> item.getCallNumber();
            case "status" -> item.getStatus().name();
            case "price" -> String.valueOf(item.getPrice());
            case "branch" -> item.getCurrentBranchId();
            case "itemtype" -> item.getItemType();
            case "genre" -> item.getGenre();
            case "format" -> item.getFormat();
            default -> "";
        };
    }

    private String getPatronValue(Patron p, String field) {
        return switch (field.toLowerCase()) {
            case "id" -> p.getId();
            case "name" -> p.getName();
            case "county" -> p.getCounty();
            case "agerange" -> p.getAgeRange();
            case "zipcode" -> p.getZipCode();
            case "homebranch" -> p.getHomeBranchId();
            case "patrontype" -> p.getPatronType();
            case "language" -> p.getPreferredLanguage();
            default -> "";
        };
    }
}

                categoryItems.put(cat, categoryItems.getOrDefault(cat, 0) + 1);
                categoryCircs.put(cat, categoryCircs.getOrDefault(cat, 0) + item.getCheckoutCount());
                totalCircs += item.getCheckoutCount();
            }
        }

        for (String cat : categoryItems.keySet()) {
            Map<String, Double> breakdown = new HashMap<>();
            double itemPct = totalItems == 0 ? 0.0 : ((double) categoryItems.get(cat) / totalItems) * 100.0;
            double circPct = totalCircs == 0 ? 0.0 : ((double) categoryCircs.getOrDefault(cat, 0) / totalCircs) * 100.0;
            double shelfPct = (branch != null) ? branch.getShelfSpaceAllocation().getOrDefault(cat, 0.0) : 0.0;

            breakdown.put("percentCollection", itemPct);
            breakdown.put("percentCirculation", circPct);
            breakdown.put("percentShelfSpace", shelfPct);
            report.put(cat, breakdown);
        }
        return report;
    }

    // PINES-021: Item Purging
    public List<ItemRecord> identifyItemsToPurge(int missingMonthsThreshold) {
        List<ItemRecord> candidates = new ArrayList<>();
        for (ItemRecord item : repository.getItems().values()) {
            if (item.getStatus() == ItemStatus.DISCARD_WEED ||
               (item.getStatus() == ItemStatus.MISSING && item.getMonthsInStatus() >= missingMonthsThreshold)) {
                candidates.add(item);
            }
        }
        return candidates;
    }

    public void purgeItem(String itemId, StaffUser staff) {
        ItemRecord item = repository.getItems().remove(itemId);
        if (item != null) {
            BibRecord bib = repository.getBibs().get(item.getBibId());
            if (bib != null) {
                bib.removeItemId(itemId);
                if (bib.getItemIds().isEmpty()) {
                    repository.getBibs().remove(bib.getId());
                    deletionTracker.recordDeletion("BIB", bib.getId(), item.getCurrentBranchId(), item.getSystemId(), LocalDate.now());
                }
            }
            deletionTracker.recordDeletion("ITEM", itemId, item.getCurrentBranchId(), item.getSystemId(), LocalDate.now());
        }
    }

    // PINES-024: Inactive Patrons Report
    public List<Patron> getInactivePatrons(int days, LocalDate today, String systemId) {
        List<Patron> inactive = new ArrayList<>();
        for (Patron p : repository.getPatrons().values()) {
            Branch b = repository.getBranches().get(p.getHomeBranchId());
            if (systemId == null || (b != null && b.getSystemId().equals(systemId))) {
                if (p.getLastActivityDate() == null || ChronoUnit.DAYS.between(p.getLastActivityDate(), today) > days) {
                    inactive.add(p);
                }
            }
        }
        return inactive;
    }

    // PINES-027, PINES-028, PINES-029: Detailed Counts
    public Map<String, Long> getCheckInBreakdown() {
        Map<String, Long> map = new HashMap<>();
        for (TransactionRecord tx : repository.getTransactions()) {
            if ("CHECKIN".equals(tx.getTransactionType()) && tx.getCheckInType() != null) {
                map.put(tx.getCheckInType().name(), map.getOrDefault(tx.getCheckInType().name(), 0L) + 1);
            }
        }
        return map;
    }

    public Map<String, Long> getCheckOutBreakdown() {
        Map<String, Long> map = new HashMap<>();
        for (TransactionRecord tx : repository.getTransactions()) {
            if ("CHECKOUT".equals(tx.getTransactionType()) && tx.getCheckOutType() != null) {
                map.put(tx.getCheckOutType().name(), map.getOrDefault(tx.getCheckOutType().name(), 0L) + 1);
            }
        }
        return map;
    }

    public Map<String, Long> getHoldSourceBreakdown() {
        Map<String, Long> map = new HashMap<>();
        for (HoldRequest hold : repository.getHolds()) {
            map.put(hold.getSource().name(), map.getOrDefault(hold.getSource().name(), 0L) + 1);
        }
        return map;
    }

    public Map<String, Long> getHoldStatusBreakdown() {
        Map<String, Long> map = new HashMap<>();
        for (HoldRequest hold : repository.getHolds()) {
            map.put(hold.getStatus().name(), map.getOrDefault(hold.getStatus().name(), 0L) + 1);
        }
        return map;
    }

    // PINES-031: Value of items report
    public double calculateCollectionValue(String filterStatus, String filterFormat, String branchId) {
        double total = 0.0;
        for (ItemRecord item : repository.getItems().values()) {
            boolean statusMatch = (filterStatus == null || item.getStatus().name().equalsIgnoreCase(filterStatus));
            boolean formatMatch = (filterFormat == null || item.getFormat().equalsIgnoreCase(filterFormat));
            boolean branchMatch = (branchId == null || item.getCurrentBranchId().equalsIgnoreCase(branchId));

            if (statusMatch && formatMatch && branchMatch) {
                total += item.getPrice();
            }
        }
        return total;
    }

    // PINES-034: Financial reports
    public List<Patron> getPatronsWithBalanceExceeding(double minBalance) {
        List<Patron> results = new ArrayList<>();
        for (Patron p : repository.getPatrons().values()) {
            if (ledger.getPatronBalance(p.getId()) > minBalance) {
                results.add(p);
            }
        }
        return results;
    }

    public Map<ChargeType, Double> getChargesAccruedByType() {
        Map<ChargeType, Double> map = new HashMap<>();
        for (Charge charge : ledger.getCharges()) {
            map.put(charge.getChargeType(), map.getOrDefault(charge.getChargeType(), 0.0) + charge.getAmount());
        }
        return map;
    }

    public Map<PaymentMethod, Double> getPaymentsByMethod() {
        Map<PaymentMethod, Double> map = new HashMap<>();
        for (Payment payment : ledger.getPayments()) {
            map.put(payment.getMethod(), map.getOrDefault(payment.getMethod(), 0.0) + payment.getAmount());
        }
        return map;
    }
